# 5. PM4Py role-discovery and similarity code

```python
import pandas as pd
import pm4py
from pm4py.objects.log.util import dataframe_utils
from pm4py.objects.conversion.log import converter as log_converter

def discover_roles_from_event_log(event_log_path):
    # Load and format dataframe
    df = pd.read_csv(event_log_path)
    df = pm4py.format_dataframe(df,
                               case_id='case:concept:name',
                               activity_key='concept:name',
                               timestamp_key='time:timestamp')

    # Convert to event log
    log = log_converter.apply(df)

    # Discover organizational roles using default method
    org_roles = pm4py.discover_organizational_roles(log)
    print("Discovered Organizational Roles:")
    print(org_roles)

    # Discover activity-based resource similarity
    similarity_df = pm4py.discover_activity_based_resource_similarity(log)
    print("\nActivity-Based Resource Similarity:")
    print(similarity_df.head())

    # Discover handover of work network
    handover_net = pm4py.discover_handover_of_work_network(log)
    pm4py.view_handover_of_work_network(handover_net)

    # Discover working together network
    working_together_net = pm4py.discover_working_together_network(log)
    pm4py.view_working_together_network(working_together_net)

    # Discover subcontracting network
    subcontracting_net = pm4py.discover_subcontracting_network(log)
    pm4py.view_subcontracting_network(subcontracting_net)

    # Join role discovery with case complexity and substitution indicators
    # First get case attributes
    case_attributes = pm4py.get_case_attributes(log)

    # Get variants to analyze case complexity by resource
    variants = pm4py.get_variants(log)

    # Create resource-complexity dataframe
    resource_complexity = []
    for resource in df['org:resource'].unique():
        resource_cases = df[df['org:resource'] == resource]['case:concept:name'].unique()
        complexity_scores = []
        for case in resource_cases:
            case_df = df[df['case:concept:name'] == case]
            if 'case_complexity' in case_df.columns:
                complexity_scores.append(case_df['case_complexity'].iloc[0])
        if complexity_scores:
            resource_complexity.append({
                'resource': resource,
                'median_complexity': pd.Series(complexity_scores).median(),
                'mean_complexity': pd.Series(complexity_scores).mean(),
                'case_count': len(complexity_scores)
            })

    resource_complexity_df = pd.DataFrame(resource_complexity)

    # Get substitution indicators
    substitution_cases = df[df['temporary_substitution'] == True]['case:concept:name'].unique()
    substitution_resources = df[df['case:concept:name'].isin(substitution_cases)]['org:resource'].unique()

    # Join with role discovery
    # Note: This is illustrative - actual implementation would need to map roles to resources
    role_resource_mapping = {
        'Front Office': ['R01_Amira', 'R02_Ben', 'R11_Kai', 'R12_Lina'],
        'Technical': ['R03_Chen', 'R04_Dina'],
        'Security': ['R05_Erik'],
        'Commercial': ['R06_Fatima', 'R07_Giorgio'],
        'Supervisor': ['R08_Hana'],
        'Quality': ['R09_Ivan', 'R10_Julia'],
        'Coordinator': ['R13_Mateo'],
        'Automation': ['R14_Nora']
    }

    # Create enhanced role dataframe
    enhanced_roles = []
    for role, resources in role_resource_mapping.items():
        for resource in resources:
            complexity_row = resource_complexity_df[resource_complexity_df['resource'] == resource]
            is_substitution_resource = resource in substitution_resources

            enhanced_roles.append({
                'role': role,
                'resource': resource,
                'median_complexity': complexity_row['median_complexity'].values[0] if not complexity_row.empty else None,
                'case_count': complexity_row['case_count'].values[0] if not complexity_row.empty else 0,
                'is_substitution_resource': is_substitution_resource
            })

    enhanced_roles_df = pd.DataFrame(enhanced_roles)
    print("\nEnhanced Role Information:")
    print(enhanced_roles_df)

    # Cluster log to identify potential role groups
    # Using activity profiles for clustering
    from sklearn.cluster import KMeans
    from sklearn.preprocessing import StandardScaler

    # Create activity profile matrix
    activity_matrix = df.groupby(['org:resource', 'concept:name']).size().unstack(fill_value=0)
    activity_matrix = activity_matrix.div(activity_matrix.sum(axis=1), axis=0)  # Normalize

    # Scale and cluster
    scaler = StandardScaler()
    scaled_matrix = scaler.fit_transform(activity_matrix)

    # Determine optimal clusters (illustrative - would use elbow method in practice)
    kmeans = KMeans(n_clusters=8, random_state=42)
    clusters = kmeans.fit_predict(scaled_matrix)

    activity_matrix['cluster'] = clusters
    print("\nActivity-Based Clustering Results:")
    print(activity_matrix.sort_values('cluster'))

    # Caution: Avoid over-interpreting clusters as formal roles
    print("\nCAUTION: Activity-based clusters may not correspond to formal roles.")
    print("Consider case complexity, substitution patterns, and organizational authority.")
    print("For example, R07_Giorgio clusters with front-office agents but has commercial responsibilities.")

    return {
        'organizational_roles': org_roles,
        'resource_similarity': similarity_df,
        'enhanced_roles': enhanced_roles_df,
        'activity_clusters': activity_matrix
    }
```

**Using the Discovered Structure for Management:**

1. **Staffing and Workforce Planning:**
   - *Specialist Coverage*: Ensure adequate coverage for rare but critical roles (Security Specialist) through cross-training or on-call arrangements. The data shows Erik's absence would create significant risk.
   - *Shadow Role Development*: Formalize Giorgio's commercial shadow role by creating a "Senior Front Office Agent with Commercial Skills" position to reduce handoffs for complex cases.
   - *Temporary Workforce*: Implement quality checks for temporary agents' work, particularly on A2 Validate Entitlement, to reduce reopens (17 cases traced to incomplete validation).

2. **Training and Skill Development:**
   - *Complexity-Based Training*: Develop training programs that match case complexity profiles. For example, train front-office agents on medium-complexity diagnostics (Amira handles more than Ben) while keeping high-complexity cases with technical analysts.
   - *Cross-Functional Skills*: Build on existing shadow roles by creating formal cross-training programs. For example, technical analysts could learn basic quality check skills to reduce handoffs to quality analysts.
   - *Escalation Pathways*: Train the Incident Coordinator role on when to escalate versus when to filter cases, balancing false escalations against critical case delays.

3. **Risk Control:**
   - *Critical Role Monitoring*: Implement monitoring for critical roles (Security Specialist, Supervisor) to detect potential bottlenecks or absences before they impact service levels.
   - *Substitution Quality*: Track performance metrics during substitutions (like the 3874 minute pricing exception resolution time) to identify where formal cross-training is needed.
   - *Automation Boundary*: Monitor the transition from automated to human processes to identify cases where the automation boundary might be incorrectly filtering cases.

4. **Process Resilience:**
   - *Bridging Role Formalization*: Formalize the Incident Coordinator role to improve cross-functional coordination, particularly for reopens and escalations.
   - *Quality Gatekeeping*: Strengthen the quality analyst role as a final check before case closure, given their ability to catch issues that lead to reopens.
   - *Temporary Workforce Integration*: Develop better integration processes for temporary agents, including more comprehensive validation training to reduce rework.

**Cautions About Changing Staffing Based Only on Mined Roles:**

1. **Activity Frequency  Importance:**
   - *Risk*: Staffing decisions based solely on activity frequency might under-resource critical but low-volume roles (like Security Specialist) while over-resourcing high-volume but low-complexity roles.
   - *Example*: The automation bot handles 63% of registrations but these are all low-complexity cases. Reducing human front-office staff based on this volume would be a mistake as the remaining cases require more skill.
   - *Solution*: Always consider case complexity and business impact alongside activity volume when making staffing decisions.

2. **Shadow Roles  Formal Competence:**
   - *Risk*: Assuming that shadow roles (like Giorgio's commercial activities) represent full competence can lead to quality issues when formal specialists are absent.
   - *Example*: During Fatima's absence, pricing exception resolution times doubled when handled by Giorgio, indicating that while he can perform the activities, he lacks full expertise.
   - *Solution*: Use substitution performance metrics to determine whether shadow roles should be formalized or whether additional training is needed before relying on them for coverage.

3. **Additional Cautions:**
   - *Temporary Substitutions  Permanent Capability*: Temporary substitutions show that work can be done by others, but not necessarily at the same quality or efficiency level. Don't assume permanent capability based on temporary coverage.
   - *Similar Activity Profiles  Similar Authority*: Resources with similar activity profiles (like Chen and Dina) may have different levels of authority, experience, or risk exposure. Don't assume they can substitute for each other without considering these factors.
   - *Automation  Human Replacement*: High-volume automated activities (like Nora's registrations) don't necessarily mean the human roles can be reduced, as the remaining cases may require more complex human judgment.
