def _is_acyclic(df):
    nodes = set(df["ocel:oid"]) | set(df["ocel:oid_2"])
    edges = list(zip(df["ocel:oid"], df["ocel:oid_2"]))
    indeg = {n: 0 for n in nodes}
    for a, b in edges: indeg[b] += 1
    frontier = [n for n in nodes if indeg[n] == 0]
    seen = 0
    while frontier:
        n = frontier.pop(); seen += 1
        for a, b in edges:
            if a == n:
                indeg[b] -= 1
                if indeg[b] == 0: frontier.append(b)
    return seen == len(nodes)
```

```python
def build_analytic_relations(relations, objects, o2o):
    type_of = dict(zip(objects["ocel:oid"], objects["ocel:type"]))
    out = []
    for eid, grp in relations.groupby("ocel:eid", sort=True):
        activity = grp["ocel:activity"].iloc[0]
        ts_val = grp["ocel:timestamp"].iloc[0]
        known = set(grp["ocel:oid"])
        seen = set(known)
        frontier = [(oid, [oid], 0) for oid in grp["ocel:oid"]]   # keep order stable
        while frontier:
            oid, path, depth = frontier.pop(0)
            if depth >= MAX_PROPAGATION_DEPTH: continue
            for (src_type, qual), dst_type in PROPAGATE_VIA.items():
                if type_of[oid] != src_type: continue
                for nb in sorted(o2o_neighbours(oid, qual)):
                    if type_of[nb] != dst_type: continue
                    frontier.append((nb, path + [f"-[{qual}]->", nb], depth+1))
                    if nb in seen: continue
                    seen.add(nb)
                    out.append({
                        "ocel:eid": eid,
                        "ocel:activity": activity,
                        "ocel:timestamp": ts_val,
                        "ocel:oid": nb,
                        "ocel:type": type_of[nb],
                        "ocel:qualifier": analytic_qualifier(activity, type_of[nb]),
                        "ocel:provenance": "inferred(O2O): " + " ".join(path + [f"-[{qual}]->", nb]),
                    })
    cols = [...]
    return pd.DataFrame(out, columns=cols)

def trace_shapes(log) -> Dict[str, List[str]]:
    """object id -> activity sequence; accepts EventLog or DataFrame output."""
    out: Dict[str, List[str]] = {}
    if isinstance(log, pd.DataFrame):
        id_col = "ocel:oid" if "ocel:oid" in log.columns else "case:concept:name"
        act_col = "concept:name" if "concept:name" in log.columns else "ocel:activity"
        ts_col = "time:timestamp" if "time:timestamp" in log.columns else "ocel:timestamp"
        for oid, group in log.groupby(id_col, sort=True):
            out[str(oid)] = list(group.sort_values(ts_col, kind="mergesort")[act_col])
        return out
    for trace in log:
        oid = trace.attributes.get("ocel:oid", trace.attributes.get("concept:name", "?"))
        out[str(oid)] = [
            ev["concept:name"] if "concept:name" in ev else ev["ocel:activity"] for ev in trace

```python
# =====================================================================
#  TIER 2 (quarantined): O2O-propagated ANALYTICAL relations
#  Not merged into `relations`.  Used only behind MERGE_ANALYTIC = False.
# =====================================================================
MERGE_ANALYTIC = False

ACTIVITY_PARTICIPLE = {
    "Create Order": "created", "Add Item": "added", "Reserve Stock": "reserved",
    "Pack Package": "packed", "Dispatch Package": "dispatched",
    "Issue Invoice": "invoiced", "Record Payment": "paid",
    "Confirm Delivery": "confirmed", "Close Order": "closed",
}

# Declared propagation policy: from an object of type T, an O2O edge with
# qualifier q *extends operational scope* to objects of type U.  Note what is
# NOT allowed: Invoice -> Payment (siblings) and Order -> Item (nothing is
# invented upwards), which keeps the closure small and non-sideways.
PROPAGATE_VIA: Dict[Tuple[str, str], str] = {
    ("Delivery", "carries_package"): "Package",
    ("Package",  "packs_item"):      "Item",
    ("Item",     "contains_item"):   "Order",
    ("Invoice",  "bills_order"):     "Order",
    ("Payment",  "partially_settles"):    "Invoice",
    ("Payment",  "completes_settlement"): "Invoice",
}
MAX_PROPAGATION_DEPTH = 3


def o2o_neighbours(oid: str, qualifier: str) -> Set[str]:
    mask = o2o["ocel:qualifier"] == qualifier
    out: Set[str] = set()
    for a, b in o2o.loc[mask, ["ocel:oid", "ocel:oid_2"]].itertuples(index=False):
        if a == oid:
            out.add(b)
        elif b == oid:
            out.add(a)
    return out


def build_analytic_relations(relations: pd.DataFrame) -> pd.DataFrame:
    rows = []
    for eid, group in relations.groupby("ocel:eid", sort=True):
        activity = group["ocel:activity"].iloc[0]
        stamp = group["ocel:timestamp"].iloc[0]
        seen = set(group["ocel:oid"])
        frontier = [(oid, [oid], 0) for oid in group["ocel:oid"]]
        while frontier:
            oid, path, depth = frontier.pop(0)
            if depth >= MAX_PROPAGATION_DEPTH:
                continue
            for (src_type, qual), dst_type in PROPAGATE_VIA.items():
                if type_of[oid] != src_type:
                    continue
                for nb in sorted(o2o_neighbours(oid, qual)):
                    if type_of[nb] != dst_type:
                        continue
                    new_path = path + [f"-[{qual}]->", nb]
                    frontier.append((nb, new_path, depth + 1))
                    if nb in seen:
                        continue
                    seen.add(nb)
                    rows.append({
                        "ocel:eid": eid,
                        "ocel:activity": activity,
                        "ocel:timestamp": stamp,
                        "ocel:oid": nb,
                        "ocel:type": type_of[nb],
                        "ocel:qualifier": "analytic:%s_%s" % (ACTIVITY_PARTICIPLE[activity], type_of[nb].lower()),
                        "ocel:provenance": "inferred(O2O): " + " ".join(new_path),
                    })
    return pd.DataFrame(rows, columns=list(relations.columns))
```

def _is_acyclic(frame: pd.DataFrame) -> bool:
    nodes = set(frame["ocel:oid"]) | set(frame["ocel:oid_2"])
    edges = list(zip(frame["ocel:oid"], frame["ocel:oid_2"]))
    indegree = {n: 0 for n in nodes}
    for _, target in edges:
        indegree[target] += 1
    frontier = [n for n in nodes if indegree[n] == 0]
    visited = 0
    while frontier:
        node = frontier.pop()
        visited += 1
        for source, target in edges:
            if source == node:
                indegree[target] -= 1
                if indegree[target] == 0:
                    frontier.append(target)
    return visited == len(nodes)

Container / holder / creditor is always `oid`; member / held / settled is always `oid_2`. Consequences: the O2O graph is acyclic and reads `Delivery  Package  Item  Order  Invoice  Payment` (an hourglass with `Item` as shared neck). Six qualifiers, 12 rows: `contains_item` (OrderItem ×3), `packs_item` (PackageItem ×3), `carries_package` (DeliveryPackage ×2), `bills_order` (InvoiceOrder ×2), `partially_settles` (PaymentInvoice ×1), `completes_settlement` (PaymentInvoice ×1).

**A5 – Do logistics/financial events relate directly to Orders and Items? (Tier 1 = no).** `relations` contains **source-faithful** E2O links only — exactly the links the narration asserts. Therefore `Dispatch Package`, `Confirm Delivery` and `Record Payment` touch only their Package/Delivery/Payment/Invoice objects; they reach Orders/Items only through O2O (`D1 -carries_package-> P1 -packs_item-> I1 <-contains_item- O100`). By contrast `Reserve Stock`, `Pack Package` and `Issue Invoice` *do* carry direct Order/Item links because the narration states them explicitly ("operationally shared by O100 and O200", "relates to both Orders", "bills both Orders").
*Trade-off:* minimal relations keep observations and inferences apart and never invent causality, but they leave the Order/Item views "hole-y" (no dispatch, no delivery confirmation, no payment). Rich relations make Order traces look complete but (i) copy every shared event into N case traces, and (ii) silently mix inferred links with recorded ones. The chosen policy takes the honest, sparse core and offers the rich view as a **quarantined Tier 2**.

**A6 – Keeping analytical propagation from masquerading as fact.** (a) Propagated links live in a *separate* frame `relations_analytic`, never silently concatenated; (b) every row carries `ocel:provenance` = `"source"` vs `"inferred(O2O): D1 -[carries_package]-> P1 -[packs_item]-> I1"` (the exact O2O path is the audit trail); (c) analytic qualifiers obey the same R1 grammar but always carry the `analytic:` prefix (`analytic:dispatched_order`, `analytic:paid_order`, `analytic:confirmed_package`); (d) propagation is driven by a declared policy table `PROPAGATE_VIA` with a depth cap of 3, and it can never travel *sideways* (Invoice  sibling Payments is not an allowed step); (e) merging is behind an explicit `MERGE_ANALYTIC = False` flag, and the whole analysis pipeline runs on Tier 1.

**A7 – Further semantic assumptions.** `V1`'s billed total is 250 monetary units (= 100 + 150), the minimal reading of "M1 and M2 are partial payments settling V1". Hence `M1 partially_settles V1` and `M2 completes_settlement V1`: M2 is *partial in amount* but is the payment that *completes* the settlement. (If the total were unknown, both would have to be `partially_settles` — see §6/§7.) Amounts are **event attributes** (`ocel:amount`, `ocel:currency = "MU"`), the total is an **object attribute** of `V1` (`ocel:total_amount`); no amount is ever an object. `Add Item` adds exactly one item; `Pack Package` names exactly one package; activity strings are exactly as narrated (no renaming for readability).

**A8 – Version-dependent PM4Py assumptions (isolated).** (i) Constructor: `OCEL(events=events, objects=objects, relations=relations, o2o=o2o)` from `pm4py.objects.ocel.obj`; older builds want the four frames indexed by `ocel:eid`/`ocel:oid`, or want object attributes in `objects["ocel:ovmap"]` — both adjustments are shown as one-line variants inside a single boxed comment and nowhere else. (ii) `pm4py.discover_dfg` returns `(dfg, start, end)` on current builds and `(dfg, start)` on some — wrapped in `_discover_dfg`. (iii) `pm4py.ocel_flattening` returns an `EventLog`; `trace_shapes` also accepts the DataFrame flavour. No data semantics change anywhere.

def _is_acyclic(frame: pd.DataFrame) -> bool:
    """Kahn-style acyclicity check for the small O2O graph."""
    nodes = set(frame["ocel:oid"]) | set(frame["ocel:oid_2"])
    edges = list(zip(frame["ocel:oid"], frame["ocel:oid_2"]))
    indegree = {n: 0 for n in nodes}
    for _, target in edges:
        indegree[target] += 1
    frontier = [n for n in nodes if indegree[n] == 0]
    visited = 0
    while frontier:
        node = frontier.pop()
        visited += 1
        for source, target in edges:
            if source == node:
                indegree[target] -= 1
                if indegree[target] == 0:
                    frontier.append(target)
    return visited == len(nodes)


def _fmt(value):
    """Deterministic, printable form of log-skeleton constraint sets."""
    if isinstance(value, (set, frozenset)):
        return sorted((_fmt(v) for v in value), key=repr)
    return value

    return out

PROPAGATE_VIA: Dict[Tuple[str, str], str] = {
    ("Delivery", "carries_package"):  "Package",
    ("Package",  "packs_item"):       "Item",
    ("Item",     "contains_item"):    "Order",
    ("Invoice",  "bills_order"):      "Order",
    ("Payment",  "partially_settles"):    "Invoice",
    ("Payment",  "completes_settlement"): "Invoice",
}
MAX_PROPAGATION_DEPTH = 3


def o2o_neighbours(oid: str, qualifier: str) -> Set[str]:
    mask = o2o["ocel:qualifier"] == qualifier
    out: Set[str] = set()
    for a, b in o2o.loc[mask, ["ocel:oid", "ocel:oid_2"]].itertuples(index=False):
        if a == oid:
            out.add(b)
        elif b == oid:
            out.add(a)
    return out


def build_analytic_relations(rel: pd.DataFrame) -> pd.DataFrame:
    rows = []
    for eid, group in rel.groupby("ocel:eid", sort=True):
        activity = group["ocel:activity"].iloc[0]
        stamp = group["ocel:timestamp"].iloc[0]
        seen = set(group["ocel:oid"])
        frontier = [(oid, [oid], 0) for oid in group["ocel:oid"]]
        while frontier:
            oid, path, depth = frontier.pop(0)
            if depth >= MAX_PROPAGATION_DEPTH:
                continue
            for (src_type, qual), dst_type in PROPAGATE_VIA.items():
                if type_of[oid] != src_type:
                    continue
                for nb in sorted(o2o_neighbours(oid, qual)):
                    if type_of[nb] != dst_type:
                        continue
                    new_path = path + [f"-[{qual}]->", nb]
                    frontier.append((nb, new_path, depth + 1))
                    if nb in seen:
                        continue
                    seen.add(nb)
                    rows.append({
                        "ocel:eid": eid,
                        "ocel:activity": activity,
                        "ocel:timestamp": stamp,
                        "ocel:oid": nb,
                        "ocel:type": type_of[nb],
                        "ocel:qualifier": "analytic:%s_%s"
                                          % (ACTIVITY_PARTICIPLE[activity], type_of[nb].lower()),
                        "ocel:provenance": "inferred(O2O): " + " ".join(new_path),
                    })
    return pd.DataFrame(rows, columns=list(rel.columns))

## 4. Expected flattened behavior

### Order (2 traces: O100, O200)

- **Trace shape.** `O100 = Create Order  Add Item  Add Item  Reserve Stock  Pack Package  Pack Package  Issue Invoice  Close Order` (8). `O200 = Create Order  Add Item  Reserve Stock  Pack Package  Issue Invoice  Close Order` (6). 14 instances but only 11 distinct `ocel:eid`s.
- **Copied shared events.** `e06`, `e07`, `e11` each occur twice — once per order trace. Consequence: the flattened log reports `Reserve Stock = 2` (physically 1), `Pack Package = 3` (physically 2), `Issue Invoice = 2` (physically 1).
- **Omitted lifecycle aspects.** All logistics and money: `Dispatch Package`, `Confirm Delivery`, `Record Payment` (6 of 17 events are invisible). So the order view knows nothing about dispatch, delivery or settlement.
- **DFG edges.** `Create OrderAdd Item (2)`, `Add ItemAdd Item (1)`, `Add ItemReserve Stock (2)`, `Reserve StockPack Package (2)`, `Pack PackagePack Package (1)`, `Pack PackageIssue Invoice (2)`, `Issue InvoiceClose Order (2)`; start `{Create Order: 2}`, end `{Close Order: 2}`. Skeleton: equivalence classes `{Create Order, Reserve Stock, Issue Invoice, Close Order}` (exactly once per trace) and `{Add Item, Pack Package}` (same count per trace — 2 in O100, 1 in O200); `Create Order` always before everything, `Close Order` always after everything; `never_together = `; 7 `directly_follows` constraints.
- **Misleading interpretations.** (i) The `Pack PackagePack Package` edge is an interleaving artifact of one order going through two packages, not a rework loop. (ii) `e07` is *counted twice*, so every latency statistic double-weights the 09:00 packing. (iii) `Issue InvoiceClose Order` suggests invoicing triggers closing; actually closing follows delivery confirmation and the final payment (invisible here) — the causal story is exactly the part that got cut. (iv) Each trace looks like a self-contained order fulfilment although `P1` physically mixes both orders: the flat log shows convergence as if it were per-order work.

### Package (2 traces: P1, P2)

- **Trace shape.** `P1 = Pack Package  Dispatch Package`; `P2 = Pack Package  Dispatch Package` (2 each, 4 instances = 4 distinct events).
- **Copied shared events.** None — every event touches exactly one package.
- **Omitted lifecycle aspects.** Item addition, stock reservation, delivery confirmation (that lives on `Delivery`), invoicing and payment; and above all *which orders* the package serves.
- **DFG edges.** `Pack PackageDispatch Package (2)`; start `{Pack Package: 2}`, end `{Dispatch Package: 2}`. Skeleton: equivalence `{Pack Package, Dispatch Package}`, `Pack Package` always before `Dispatch Package`, one `directly_follows` constraint, `never_together = `.
- **Misleading interpretations.** `P1` and `P2` look like perfect twins, yet `P1` is the cross-order package (I1 of O100 + I3 of O200) and `P2` is single-order. The 2-event trace makes the package look trivially fast, hiding that the items had already waited in reservation.

### Delivery (2 traces: D1, D2)

- **Trace shape.** `D1 = Dispatch Package  Confirm Delivery` (10:0013:00 = 3h00); `D2 = Dispatch Package  Confirm Delivery` (10:15next-day 09:00 = 22h45).
- **Copied shared events.** None.
- **Omitted lifecycle aspects.** Packing, item/order identity, invoice and payment; also the fact that `D1` and `D2` carry packages for *overlapping* order sets.
- **DFG edges.** `Dispatch PackageConfirm Delivery (2)`; start `{Dispatch Package: 2}`, end `{Confirm Delivery: 2}`; skeleton analogous to Package (`{Dispatch Package, Confirm Delivery}` equivalent, one direct-follows constraint).
- **Misleading interpretations.** The 3h vs 22h45 spread looks like a delivery-service failure on `D2`; it is almost entirely the overnight/next-morning confirmation convention (a working-calendar artifact), not evidence of a problem. Also `Dispatch Package` here is *the package's* event, borrowed by the delivery — the same event type plays two roles (starting a delivery and finishing a package).

### Invoice (1 trace: V1)

- **Trace shape.** `V1 = Issue Invoice  Record Payment  Record Payment` (3 instances = 3 distinct events).
- **Copied shared events.** None *inside* this flattening (only one invoice), but `e11` is copied into the two Order traces — the invoice is the aggregation point of two orders, and flattening turns that aggregation into "one case".
- **Omitted lifecycle aspects.** Which orders and items are billed (the entire reason the invoice exists), which payment settles which order, and the delivery side. The 100/150 split survives only as the event attribute `ocel:amount`.
- **DFG edges.** `Issue InvoiceRecord Payment (1)`, `Record PaymentRecord Payment (1)`; start `{Issue Invoice: 1}`, end `{Record Payment: 1}`. Skeleton: equivalence classes `{Issue Invoice}` (count 1) and `{Record Payment}` (count 2), `Issue Invoice` always before `Record Payment`, `never_together = `, 2 `directly_follows` constraints.
- **Misleading interpretations.** The `Record PaymentRecord Payment` edge looks like a **rework/repair loop** ("payment recorded again"); it is really *two distinct objects* `M1` (100, `partially_settles`) and `M2` (150, `completes_settlement`) acting on the same invoice. With n = 1 trace every statistic is anecdotal. Also nothing in the flat invoice log tells you that `V1` bills two orders, so "invoice cycle time" silently averages two orders' work.

### Why flattening on Payment is particularly weak

Each payment object participates in **exactly one** event (`M1  e12`, `M2  e15`). Flattening therefore yields two singleton traces `[Record Payment]` and `[Record Payment]`: an **empty DFG**, no start/end distinction, a log skeleton whose only content is "Record Payment occurs once", no ordering information whatsoever. Yet the two facts that matter — the 100/150 split and the `partially_settles` / `completes_settlement` distinction — live in an **event attribute** and an **O2O qualifier**, both of which DFG/skeleton discovery ignore. Worse, the ordering `M1` before `M2` is a fact of the *Invoice*, not of the payments: the payment view cannot even express "two partial payments in sequence". `Payment` is a leaf object type here: it is created by its single event and never used again, so any per-object process view degenerates to a constant.

## 5. Expected object-centric structures

**OC-DFG.** One graph, nine activity nodes and six object-type lanes, where each activity node is shared by all lanes it touches. Per-type content equals the flattened DFGs of §4 plus the Item lane (`Add Item  Reserve Stock  Pack Package`). Start/end per lane: Order `{Create Order}{Close Order}`, Item `{Add Item}{Pack Package}`, Package `{Pack Package}{Dispatch Package}`, Delivery `{Dispatch Package}{Confirm Delivery}`, Invoice `{Issue Invoice}{Record Payment}`, Payment `{Record Payment}{Record Payment}`. The structural insight is where lanes hand over: **`Pack Package` is the junction of three lanes** (Order, Item, Package) and **`Issue Invoice` the junction of two** (Order, Invoice), while `Dispatch Package` hands Package  Delivery. In a flattened log these three activities must be copied; in the OC-DFG each is drawn once, so the many-to-many structure is visible instead of multiplied: one node explains how `O100`'s `I1` and `O200`'s `I3` end up in the single token `P1`, and how `O100`+`O200` end up in the single token `V1`. The `Record Payment  Record Payment` cycle appears once on the Invoice lane, generated by **two payment tokens** `M1`, `M2` rather than by a repeated execution of one token.

**OTG (object-type / object-taxonomy graph).** The O2O skeleton renders as an hourglass: `Delivery  Package  Item  Order  Invoice  Payment`, i.e. `D1  P1  {I1, I3}` and `D2  P2  {I2}` on the left, `{O100: I1, I2}` and `{O200: I3}` + `V1  {O100, O200}` + `{M1, M2}  V1` on the right. The critical structural fact is that this is **not a case tree**: `P1` hangs over `I1` and `I3`, which belong to `O100` and `O200` respectively, so the "case decomposition" is a DAG with a genuine merge at `P1` and at `V1`. `Item` is the shared neck (both `Package` and `Order` claim items), `V1` is the shared creditor of `O100` and `O200`, and `M1`/`M2` are two leaves hanging off `V1`. Any algorithm that insists on a tree/forest must either split `P1`/`V1` (destroying the shared object) or merge `O100`/`O200` (destroying the case).

**ET–OT (event-type  object-type) bipartite incidence.** Nine activity nodes against six type nodes:

| activity | #types | object types |
|---|---|---|
| Create Order, Confirm Delivery, Close Order | 1 | Order / Delivery / Order |
| Add Item, Reserve Stock | 2 | Item + Order |
| Pack Package | **3** | Package + Item + Order |
| Dispatch Package | 2 | Package + Delivery |
| Issue Invoice | 2 | Invoice + Order |
| Record Payment | 2 | Payment + Invoice |

The graph is almost bipartite-separable into an "order/item" cluster and a "logistics" cluster and a "money" cluster; the *only* nodes joining clusters are `Pack Package` (order/item  package) and `Issue Invoice` (order  money), with `Dispatch Package` as the package  delivery hinge. Event counts by arity: 6 events touch 1 type, 9 touch 2 types, 2 touch 3 types. So the coupling in this process is concentrated in 3 of 17 events — precisely the three events (`e06`, `e07`, `e11`) that are shared between `O100` and `O200`.

**Object-centric Petri net.** Expect one object-type place per type plus one transition per activity. Structural readings (deliberately *not* unique): (i) `Pack Package` is a transition with arcs to/from the `Order`, `Item` and `Package` places — it can be read as an **atomic synchronisation** that consumes `I1` and `I3` (from two different orders) and produces one `P1` token, or as **three independent per-type steps in a product**, in which case nothing forces `O100` and `O200` to be packed together. PM4Py's markings are *sets of objects per type*, so both readings are defensible and the model does not decide between them; a blocking-synchronisation reading also implies that a missing `I3` would stall `O100`'s packing, which the data neither confirms nor denies. (ii) The `Invoice` place must support **two firings of `Record Payment` against one invoice token** (`M1` then `M2`), i.e. a place whose tokens are object sets and a transition that can fire twice for the same `V1` token but with *different* `Payment` objects — a standard AC-net cannot express this without duplicating `V1`. (iii) `Payment` is a "one-shot" type: `Record Payment` creates `M1`/`M2` and nothing ever consumes them again, so those places are dead ends and their transitions have no downstream constraints. (iv) The divergence of `O100` into two packages/deliveries is a plain sequential split at `Pack Package`/`Dispatch Package`, while the convergence into `P1` and `V1` is the interesting, non-tree behaviour.

**Object-interactions summary.** Expected magnitudes: 12 objects, 17 events, 35 E2O rows, mean **2.06 objects per event** (min 1 at `e01`, `e04`, `e13`, `e14`, `e16`, `e17`; max 5 at `e07`). Pairwise co-participation is extremely concentrated: `O100 & O200` share 3 events (`e06`, `e07`, `e11`), `O100 & I1` share 3 (`e02`, `e06`, `e07`), `O200 & I3` share 3 (`e05`, `e06`, `e07`), `I1 & I3` share 2 (`e06`, `e07`), `O100 & I2` share 2 (`e03`, `e08`); every other pair shares  1. Two readings matter: `I1` and `I3` are more entangled with each other than `I1` is with `I2` (its "sibling" in `O100`) — the packing decision destroys the order boundary; and `M1` & `M2` share **zero** events even though they are the two halves of one settlement of `V1`, so the summary alone cannot express their relationship (only the O2O qualifiers can). The "interaction mass" sits on `Pack Package`, `Reserve Stock` and `Issue Invoice` — the same three events identified above.

## 6. Modeling alternatives

All alternatives keep the same 17 events, 12 objects and 12 O2O rows; only the E2O policy changes.

**1. Source-minimal relations (adopted, Tier 1).** Only links the narration asserts. *Effect:* Order traces are 8/6 events and stop at `Close Order`; the flattened Order log has 14 instances; `Reserve Stock`, `Issue Invoice` appear twice although executed once. Cleanest separation of fact and inference, at the price of a hole-y order view and 6 invisible events.

**2. Full operational involvement.** Every event is linked to every object it physically concerns, e.g. `e09  P1 (dispatched_package), D1 (assigned_delivery), I1 (dispatched_item), I3, O100 (dispatched_order), O200`, and `e12/e15  V1, M1/M2, O100, O200` (allocation-agnostic). *Effect:* `O100`'s trace grows to 14 events and `O200`'s to 10 (24 flattened instances of 17 events). The Order DFG now contains `Dispatch PackageDispatch Package` and `Confirm DeliveryConfirm Delivery` edges that are pure interleaving of two independent logistics threads (P1/D1 and P2/D2 leave 15 minutes apart), and `Record Payment` joins the order lane with **two** payment events that cannot be allocated to a single order. Throughput per order becomes measurable, but "process discovery" starts reporting loops and parallelism that nobody ever performed.

**3. O2O-propagated analytical relations (Tier 2, computed in the code).** Walk the declared `PROPAGATE_VIA` policy with depth  3 and mark every inferred row `analytic:<participle>_<type>` + provenance path. *Effect:* exactly **18** inferred rows (`e09` +4, `e10` +2, `e12` +2, `e13` +5, `e14` +3, `e15` +2  35 + 18 = 53). The order view gains `analytic:dispatched_order`, `analytic:confirmed_order`, `analytic:paid_order` and therefore looks as complete as alternative 2, but every added row is self-documenting and reversible, and the OC-DFG gains clearly-labelled lanes instead of silently polluted ones. Flattening still duplicates shared events — propagation does not fix that.

**4. One primary object per event (event-centric minimalism).** Each event keeps exactly one E2O row: `Create/CloseOrder`, `Add/ReserveItem`, `PackPackage`, `DispatchDelivery`, `IssueInvoice`, `RecordPayment`, `ConfirmDelivery`. *Effect:* every trace has length 1–3, most DFGs are trivial or empty, and **all convergence disappears**: `P1` mixing two orders and `V1` billing two orders become invisible. Perfect for per-object throughput, useless for process understanding.

**5. Order-allocated payments.** Split `M1`/`M2` across `O100`/`O200` (by amount, by item value, or 1:1) with `partially_settles_order`. *Effect:* per-order cash-flow traces and a plausible "Close Order after full payment" causal story appear — but the allocation is invented unless the source records it. `completes_settlement` on `M2` then also becomes order-specific and can contradict itself (which order did `M2` complete?).

**6. Item-level fulfilment only.** Drop the Order links from `Reserve Stock`/`Pack Package`. *Effect:* the Order view degenerates to `Create  Add  Close`, and the fact that `O100` and `O200` are operationally entangled vanishes entirely from discovery output — the single most important property of this process would be lost.

## 7. Failure modes

1. **Relation inflation.** Adding every transitively related object to every event (alternative 2/3 pushed too far) inflates `relations` from 35 towards 60+ rows and gives one physical action weight `n` in all statistics. Symptom: the flattened logs sum to far more than 17 event instances and every average latency is dominated by the shared events.
2. **Missing direct E2O links.** If `e06`/`e07`/`e11` did not carry their `demanding_order`/`source_order`/`billed_order` rows, the Order flattening would collapse to `Create  Add*  Close` and O100/O200 would look perfectly independent — the *opposite* of the business story. The model would be "clean" and wrong.
3. **Treating O2O propagation as observation.** Writing "D1 delivers O100" or "M1 pays for O100" into `relations` as if the source had recorded it conflates a business rule with a measurement; downstream one then reads invented causality ("the payment triggered the close"). Fixed by Tier-2 quarantine + `ocel:provenance` + `analytic:` prefixes.
4. **Duplicate shared events after flattening.** `e06`, `e07`, `e11` appear in both order traces (14 instances vs 11 distinct `ocel:eid`s); `Reserve Stock` is counted twice although executed once. Any per-activity frequency, waiting-time or conformance measure computed on the flattened log silently double counts; checking "flattened instance count == 17" then fails confusingly.
5. **Ambiguous payment allocation.** 100 + 150 = 250 settles `V1`, but nothing says how much of `M1`/`M2` belongs to `O100` vs `O200`. Claiming `O100` paid 100 is
