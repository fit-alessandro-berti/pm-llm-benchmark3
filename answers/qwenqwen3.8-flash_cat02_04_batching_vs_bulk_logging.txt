    starts = group["start"].dropna()
    ends = group["complete"].dropna()

    end_counts = ends.value_counts()
    return pd.Series({

def overlap_stats(group):
    intervals = group[["start", "complete"]].dropna()
    intervals = intervals[intervals["complete"] > intervals["start"]]
    n = len(intervals)
    if n < 2:
        return pd.Series({"overlap_rate": np.nan, "max_concurrent": int(n)})
    pairs = 0
    overlaps = 0
    rows = intervals.to_dict("records")
    for i in range(n):
        for j in range(i + 1, n):
            pairs += 1
            s1, e1 = rows[i]["start"], rows[i]["complete"]
            s2, e2 = rows[j]["start"], rows[j]["complete"]
            if max(s1, s2) < min(e1, e2):
                overlaps += 1
    events = []
    for r in rows:
        events.append((r["start"], 1))
        events.append((r["complete"], -1))
    events.sort(key=lambda x: (x[0], x[1]))
    cur = 0
    mx = 0
    for _, typ in events:
        cur += typ
        mx = max(mx, cur)
    return pd.Series({"overlap_rate": overlaps / pairs if pairs else np.nan, "max_concurrent": int(mx)})

def queue_stats(group):
    first_start = group["start"].min()
    queue_before_first_start = 0
    if pd.notna(first_start):
        queue_before_first_start = int((group["queue_arrival"] <= first_start).sum())
    events = []
    for _, r in group.iterrows():
        if pd.notna(r["queue_arrival"]):
            events.append((r["queue_arrival"], 1))
        if pd.notna(r["start"]):
            events.append((r["start"], -1))
    events.sort(key=lambda x: (x[0], x[1]))
    backlog = 0
    max_backlog = 0
    for _, typ in events:
        if typ == 1:
            backlog += 1
        else:
            backlog = max(0, backlog - 1)
        max_backlog = max(max_backlog, backlog)
    return pd.Series({
        "queue_before_first_start": queue_before_first_start,
        "max_backlog": int(max_backlog),
    })

def boundary_fraction(ts, minutes):
    if len(ts) == 0:
        return np.nan
    vals = []
    for t in ts:
        if pd.isna(t):
            continue
        vals.append((int(t.minute) % int(minutes) == 0) and (int(t.second) == 0) and (int(t.microsecond) == 0))
    return float(np.mean(vals)) if vals else np.nan

G6-like data can create both inflated waiting and near-zero service because the queue arrival timestamp may be real, while the start and complete timestamps are both generated by a bulk submission form.     if x is None:
        return pd.NaT
    s = str(x).strip()
    if s == "" or s.lower() == "nan":
        return pd.NaT
    return pd.Timestamp(f"{DATE.date()} {s}")

def parse_hms(x):
    if x is None:
        return pd.NaT
    s = str(x).strip()
    if s == "" or s.lower() == "nan":
        return pd.NaT
    return pd.Timestamp(f"{DATE.date()} {s}")

def equality_stats(group):
    starts = group["start"].dropna()
    ends = group["complete"].dropna()
    start_counts = starts.value_counts()
    end_counts = ends.value_counts()
    return pd.Series({
        "all_start_equal": bool(starts.nunique() == 1) if len(starts) > 0 else False,
        "all_complete_equal": bool(ends.nunique() == 1) if len(ends) > 0 else False,
        "max_start_group": int(start_counts.max()) if len(start_counts) > 0 else 0,
        "max_complete_group": int(end_counts.max()) if len(end_counts) > 0 else 0,
    })

    intervals = group[["start", "complete"]].dropna()
    intervals = intervals[intervals["complete"] > intervals["start"]]
    n = len(intervals)
    if n < 2:
        return pd.Series({"overlap_rate": np.nan, "max_concurrent": int(n)})

    pairs = 0
    overlaps = 0
    rows = intervals.to_dict("records")
    for i in range(n):
        for j in range(i + 1, n):
            pairs += 1
            s1, e1 = rows[i]["start"], rows[i]["complete"]
            s2, e2 = rows[j]["start"], rows[j]["complete"]
            if max(s1, s2) < min(e1, e2):
                overlaps += 1

    events = []
    for r in rows:
        events.append((r["start"], 1))
        events.append((r["complete"], -1))
    events.sort(key=lambda x: (x[0], x[1]))

    cur = 0
    mx = 0
    for _, typ in events:
        cur += typ
        mx = max(mx, cur)

    return pd.Series({
        "overlap_rate": overlaps / pairs if pairs else np.nan,
        "max_concurrent": int(mx),
    })

def queue_stats(group):
    first_start = group["start"].min()
    queue_before_first_start = 0
    if pd.notna(first_start):
        queue_before_first_start = int((group["queue_arrival"] <= first_start).sum())

    events = []
    for _, r in group.iterrows():
        if pd.notna(r["queue_arrival"]):
            events.append((r["queue_arrival"], 1))
        if pd.notna(r["start"]):
            events.append((r["start"], -1))

    events.sort(key=lambda x: (x[0], x[1]))

    backlog = 0
    max_backlog = 0
    for _, typ in events:
        if typ == 1:
            backlog += 1
        else:
            backlog = max(0, backlog - 1)
        max_backlog = max(max_backlog, backlog)

    return pd.Series({
        "queue_before_first_start": queue_before_first_start,
        "max_backlog": int(max_backlog),
    })

def boundary_fraction(ts, minutes):
    if len(ts) == 0:
        return np.nan
    vals = []
    for t in ts:
        if pd.isna(t):
            continue
        vals.append(
            (int(t.minute) % int(minutes) == 0)
            and (int(t.second) == 0)
            and (int(t.microsecond) == 0)
        )
    return float(np.mean(vals)) if vals else np.nan

def granularity_stats(group):
    starts = group["start"].dropna()
    ends = group["complete"].dropna()
    precision = group["precision_min"].iloc[0]
    return pd.Series({
        "precision_min": float(precision) if pd.notna(precision) else np.nan,
        "start_spread_min": float((starts.max() - starts.min()).total_seconds() / 60.0) if len(starts) > 1 else 0.0,
        "end_spread_min": float((ends.max() - ends.min()).total_seconds() / 60.0) if len(ends) > 1 else 0.0,
        "start_15min_boundary_frac": boundary_fraction(starts, 15),
        "complete_15min_boundary_frac": boundary_fraction(ends, 15),
        "recorded_at_distinct_seconds": int(group["recorded_at"].dropna().nunique()),
    })

def cluster_diag(group):
    eq = equality_stats(group)
    ov = overlap_stats(group)
    qs = queue_stats(group)
    gr = granularity_stats(group)

    out = pd.Series({
        "resource": "/".join(sorted(group["resource"].unique())),
        "n": len(group),
        "mean_wait_min_raw": float(group["wait_min_raw"].mean()),
        "mean_wait_min_nonneg": float(group["wait_min_nonneg"].mean()),
        "median_service_min": float(group["service_min"].median()),
        "mean_rec_lag_min": float(group["rec_lag_min"].mean()),
    })

    out = pd.concat([out, eq, ov, qs, gr])

    flags = []
    if eq["all_start_equal"] and eq["all_complete_equal"]:
        flags.append("start_end_equal")
    elif eq["all_start_equal"]:
        flags.append("start_equal")
    elif eq["all_complete_equal"]:
        flags.append("end_equal")

    if group["zero_or_negative_service"].any():
        flags.append("zero_or_negative_service")
    if group["start_before_queue"].any():
        flags.append("start_before_queue")
    if group["coarse_precision"].any():
        flags.append("coarse_15min")
    if group["shared_resource"].any():
        flags.append("shared_resource")
    if group["bulk_source"].any():
        flags.append("bulk_source")

    out["suspect_flags"] = ";".join(flags) if flags else "none"
    return out

def normalize_batch_result(res, param_label):
    rows = []

    def add_cases(batch_id, cases):
        if cases is None:
            return
        if isinstance(cases, pd.Series):
            cases = cases.tolist()
        elif isinstance(cases, np.ndarray):
            cases = cases.tolist()
        if isinstance(cases, (list, set, tuple)):
            for cc in cases:
                if pd.notna(cc):
                    rows.append({"param": param_label, "batch_id": batch_id, "case": cc})

    def process(obj, batch_id=None):
        if obj is None:
            return

        if isinstance(obj, pd.DataFrame):
            case_col = next(
                (c for c in ["case:concept:name", "case", "case_id", "Case ID", "caseId"] if c in obj.columns),
                None,
            )
            batch_col = next(
                (c for c in ["batch_id", "batch", "Batch ID", "batchId"] if c in obj.columns),
                None,
            )

            if case_col is not None and batch_col is not None:
                for _, r in obj.iterrows():
                    add_cases(r[batch_col], [r[case_col]])
            elif case_col is not None:
                add_cases(batch_id if batch_id is not None else 0, obj[case_col].dropna().tolist())
            else:
                for idx, r in obj.iterrows():
                    for c in obj.columns:
                        if isinstance(r[c], (list, set, tuple, pd.Series, np.ndarray)):
                            add_cases(idx, r[c])
                            break

        elif isinstance(obj, dict):
            if "batches" in obj:
                process(obj["batches"], batch_id)
                return

            case_keys = ["cases", "case_ids", "caseIds", "cases_list"]
            cases = None
            for key in case_keys:
                if key in obj:
                    cases = obj[key]
                    break

            if cases is not None:
                bid = obj.get("batch_id", obj.get("batch", batch_id))
                add_cases(bid, cases)
            else:
                for k, v in obj.items():
                    process(v, k)

        elif isinstance(obj, (list, set, tuple)):
            if len(obj) > 0 and all(isinstance(x, (str, int, np.integer)) for x in obj):
                add_cases(batch_id, obj)
            else:
                for i, item in enumerate(obj):
                    process(item, i)

        else:
            cases = None
            if hasattr(obj, "cases"):
                cases = getattr(obj, "cases")
            elif hasattr(obj, "case_ids"):
                cases = getattr(obj, "case_ids")
            if cases is not None:
                add_cases(batch_id, cases)

    process(res)

    out = pd.DataFrame(rows)
    if not out.empty:
        out["case"] = out["case"].astype(str)
        out = out.merge(df[["case", "cluster", "activity", "resource"]], on="case", how="left")
    return out

# 5. Robustness and alternative explanations

**Resource IDs trusted versus treated as shared accounts.**  
If resource IDs are trusted as one worker or one station, then G1, G2, G3, G5, and G7 all show overlapping intervals under a single resource. For a strictly serial resource, that implies either multitasking, capacity greater than one, or invalid lifecycle timestamps. If resource IDs are treated as shared or capacity-aware, the same pattern becomes more plausible as concurrent work: G1 may be a four-slot fixture, G5 may be a three-analyst pool, and G7 may be a station with unknown capacity. The most consequential change is G5: treating `analyst_pool` as one worker yields impossible utilization, while treating it as a pool makes concurrency plausible but still uncertain because four intervals overlap and the stated pool capacity is at most three.

**Timestamps exact versus interval-censored.**  
If timestamps are treated as exact, PM4Py batch discovery can over-detect operational batches from logging artifacts. For example, G6 has exact start/end equality at 17:00, but that equality is likely bulk submission, not physical simultaneous closure. If timestamps are interval-censored by stated precision, conclusions change materially. For 1-min precision, a true timestamp may lie within a one-minute interval, or within a half-minute interval if rounded to nearest; the prompt does not specify accuracy, only granularity. For 15-min precision in G7, the interval-censoring window is so wide that start and end equality cannot support strong physical concurrency claims. A conservative bound is that the true event lies within plus or minus the stated precision, or within a one-precision-width interval if truncation is assumed. Because the legacy MES rounding rule is undocumented, G7 should be treated with bounds rather than point estimates.

**Cases where the same temporal pattern supports multiple physical explanations.**

1. **G1 and G7: identical starts and identical completes.**  
   This can mean genuine simultaneous batch processing, as in a fixture holding four items, or it can mean serial work rounded into a common timestamp window, as in legacy MES quarter-hour boundaries. The data alone cannot distinguish these without fixture IDs, per-item sensors, or raw timestamps.

2. **G2 and G5: overlapping intervals under one resource.**  
   In G2, overlap can mean concurrent inspection work or a “begin group” start artifact. In G5, overlap can mean multiple analysts working concurrently or a shared-account artifact. The same overlap signature is compatible with real concurrency, logging artifacts, and resource identity failure.

3. **G3 and G6: identical completes, especially with G6 start equals complete.**  
   In G3, identical 11:30 completes can mean actual simultaneous completion or bulk form submission after finishing a folder. In G6, identical 17:00 start and complete can mean simultaneous ticket closure, but the manual bulk form context makes end-of-shift data entry more likely. The same end-equality pattern supports both physical batching and logging batching.

# 6. Data-collection redesign

Prioritized recommendations, where P0 is highest priority:

| priority | redesign element | recommended fields or controls | why it matters |
|---:|---|---|---|
| P0 | Execution timestamps | Record actual per-case `work_start`, `work_pause`, `work_resume`, and `work_complete` at the workstation, separate from form submission. | Prevents G2, G3, and G6 artifacts where UI or submission time masquerades as physical lifecycle time. |
| P0 | Resource and badge identity | Capture badge ID, session ID, VDI session, or operator ID; if a shared login is unavoidable, log the actual assigned worker separately. | Prevents G5 from mixing multiple workers under one resource ID and enables per-worker utilization and concurrency checks. |
| P0 | Group, batch, fixture, and folder IDs | Add `batch_id`, `fixture_id`, `folder_id`, `submission_id`, `parent_case_id`, and `item_position_on_fixture`. | Distinguishes operational batches from logging batches and enables capacity-aware analysis for G1, G2, G3, and G6. |
| P0 | Workstation capacity and state | Record `resource_capacity`, `available_capacity`, `blocked_by`, `safety_hold`, `setup`, `downtime`, and `queue_limit`. | Explains G8-like delayed starts and prevents misreading capacity loss as service-time inflation. |
| P1 | Source clock synchronization | Use NTP or equivalent, store `clock_source`, `clock_offset`, `timezone`, and `monotonic_sequence_no`; correlate events across systems using synchronized clocks. | Prevents G8 safety-check and repair-queue clock mismatch from biasing waiting time. |
| P1 | Recording timestamps and insertion metadata | Keep `recorded_at`, `sequence_no`, `insertion_batch_id`, and `recorded_at_minus_execution_lag`. | Separates database insertion from execution, as needed for G1, G6, and G7. |
| P2 | Pause, resume, and interruption events | Log `pause_reason`, `resume_reason`, `interrupt_start`, `interrupt_end`, and whether the case remains in service. | Prevents multitasking or blocked work from appearing as continuous service or false concurrency. |
| P2 | Timestamp precision and accuracy metadata | Store documented `precision`, estimated `accuracy`, rounding rule, truncation rule, and raw unrounded timestamp when available. | Enables interval-censoring for G7 and more reliable uncertainty bounds for all minute-resolution sources. |

Additional validation rules should be enforced at ingestion: `start >= queue_arrival` unless a clock-skew flag is present; `complete >= start` unless a bulk-submission flag is present; `max_concurrent_cases <= resource_capacity` unless a shared-account or capacity flag is present; and all batched timestamps should carry an explicit batch or submission ID.
