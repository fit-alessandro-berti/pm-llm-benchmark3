```python
class BudgetExceeded(Exception):
    pass

MAX_VISIBLE = 14          # max visible events per trace
MAX_OCC_PER_LABEL = 3     # bounds loop iterations (Correct Defect <= 3 etc.)
MAX_STATES = 200_000

def explore_language(net, im, fm):
    """
    Enumerate the visible-trace language of `net` under bounds:
      - silent transitions (label None) are hidden but may fire,
      - a trace counts as COMPLETED only if the EXACT final marking `fm`
        is reachable from the post-state (after silent firings),
      - traces are bounded by visible length and per-label occurrence counts.
    Returns dict(accepted=set of tuples, complete_prefixes=..., states=int,
                 budget_exhausted=bool).
    """
    accepted = set()
    init, _ = silent_closure(net, im)
    visited = set()
    queue = deque()
    for k, m in init.items():
        queue.append((m, ()))
    if mkey(fm) in init:
        accepted.add(())
    states = 0
    budget = False
    while queue:
        m, trace = queue.popleft()
        states += 1
        if states > MAX_STATES:
            budget = True
            break
        for t in enabled(net, m):
            if t.label is None or not can_fire(m, t):
                continue
            lab = t.label
            if trace.count(lab) >= MAX_OCC_PER_LABEL:
                continue
            if len(trace) >= MAX_VISIBLE:
                continue
            ntrace = trace + (lab,)
            clo, _ = silent_closure(net, fire(m, t))
            if mkey(fm) in clo:
                accepted.add(ntrace)
            for k2, m2 in clo.items():
                st = (k2, ntrace)
                if st not in visited:
                    visited.add(st)
                    queue.append((m2, ntrace))
    return {"accepted": accepted, "states": states, "budget_exhausted": budget}
```

```python
register = Transition(label="Register Change")

prepare_evidence = Transition(label="Prepare Evidence")

skip_legal = SilentTransition()

def markings_non_empty(im, fm, tag):
    """A conversion result is not an accepting Petri net without usable markings."""
```

```python
from collections import deque
import json, os
import pm4py
from pm4py.objects.petri_net.obj import Marking

# ---------------------------------------------------------------- parameters
MAX_VISIBLE = 16       # visible events per trace
MAX_OCC_PER_LABEL = 3  # caps loop iterations (<= 3 per activity)
MAX_STATES = 200_000   # exploration states per net
SILENT_CAP = 5_000     # markings per silent closure

class BudgetExceeded(Exception):
    pass

# ---------------------------------------------------------------- net engine
def mkey(marking):
    return tuple(sorted((p.name, c) for p, c in marking.items()))

def can_fire(marking, t):
    return all(marking.get(a.source, 0) >= a.weight for a in t.in_arcs)

def fire(marking, t):
    m = Marking(marking)
    for a in t.in_arcs:
        m[a.source] -= a.weight
        if m[a.source] == 0:
            del m[a.source]
    for a in t.out_arcs:
        m[a.target] += a.weight
    return m

def silent_closure(net, marking):
    """All markings reachable from `marking` using ONLY silent transitions
    (label is None), including `marking` itself. Exact final-marking equality
    is enforced by comparing full markings, never place subsets."""
    seen = {mkey(marking): marking}
    stack = [marking]
    while stack:
        m = stack.pop()
        for t in pm4py.get_enabled_transitions(net, m):
            if t.label is None and can_fire(m, t):
                m2 = fire(m, t)
                k = mkey(m2)
                if k not in seen:
                    seen[k] = m2
                    stack.append(m2)
                    if len(seen) > SILENT_CAP:
                        raise BudgetExceeded("silent closure cap exceeded")
    return seen

def explore_visible_language(net, im, fm):
    """Bounded visible-trace exploration.

    - transitions with label None are hidden (fired, never recorded);
    - a trace is ACCEPTED only if the exact final marking `fm` is reachable
      after its last visible event (silent firings allowed, marking equality
      required);
    - bounded by visible length and per-label occurrence counts.
    Returns accepted traces, budget flags and fired-label statistics.
    """
    accepted = set()
    fired_labels = set()
    budget_exhausted = False
    init = silent_closure(net, im)
    queue = deque((m, ()) for m in init.values())
    visited = {(k, ()) for k in init}
    if mkey(fm) in init:
        accepted.add(())
    states = 0
    while queue:
        m, trace = queue.popleft()
        states += 1
        if states > MAX_STATES:
            budget_exhausted = True
            break
        for t in pm4py.get_enabled_transitions(net, m):
            if t.label is None or not can_fire(m, t):
                continue
            lab = t.label
            if len(trace) >= MAX_VISIBLE or trace.count(lab) >= MAX_OCC_PER_LABEL:
                continue
            fired_labels.add(lab)
            ntrace = trace + (lab,)
            clo = silent_closure(net, fire(m, t))
            if mkey(fm) in clo:
                accepted.add(ntrace)
            for k2, m2 in clo.items():
                if (k2, ntrace) not in visited:
                    visited.add((k2, ntrace))
                    queue.append((m2, ntrace))
    return {"accepted": accepted, "states": states,
            "fired_visible_labels": fired_labels,
            "budget_exhausted": budget_exhausted}
```

```python
def accepts_visible_trace(net, im, fm, trace):
    """Exact membership test for one visible trace (subset construction over
    marking sets; terminates for bounded nets). Used for witness tests so
    that witness verdicts never depend on the exploration bounds."""
    try:
        current = silent_closure(net, im)
        for lab in trace:
            nxt = {}
            for m in current.values():
                for t in pm4py.get_enabled_transitions(net, m):
                    if t.label == lab and can_fire(m, t):
                        for k2, m2 in silent_closure(net, fire(m, t)).items():
                            nxt.setdefault(k2, m2)
            current = nxt
            if not current:
                return False
        return mkey(fm) in current
    except BudgetExceeded:
        return None  # inconclusive
```

Now Section 6 (visual misreading) — draft points (need to address: parallel gateways, exclusive gateways, loop visualization, silent routing nodes, apparent ordering of layout; propose annotations/documentation):

1. Parallel gateways: The net has two "asymmetric" synchronizations: verification's AND-join (Assess+Prepare) and release's AND-join (XOR-output + loop-output). In the BPMN rendering these appear as parallel gateways whose matching splits are at different "depths". Readers habitually pair each AND-join with the nearest enclosing AND-split and assume branch-local semantics: they will read "verification depends only on the Prepare branch" (if verification's join is drawn inside the lower branch) or "release's join closes the big register-split block", implying Legal and verification are sibling branches of one parallel block — the cross-dependency AssessVerification (an arc that crosses branch boundaries) is invisible unless arcs are traced. BPMN has no notation for "this join waits for a token from a different branch"; the diagram is behaviorally right and still routinely misread.
2. Exclusive gateways: XOR(Legal, ) becomes an exclusive split with one labeled outgoing flow (Legal Review) and one unlabeled flow (the skip). Readers may (a) not see the skip flow at all and believe Legal Review is mandatory; (b) ask which condition guards the choice — the model is free-choice/nondeterministic, but BPMN readers expect data conditions; adding invented conditions would change the model's meaning. Also the XOR-merge after Legal (rejoining the skip flow) may be drawn as a bare gateway, doubling confusion.
3. Loop visualization: the LOOP(do, redo) becomes a cycle: after Inspect Trial, a gateway either exits (silent) or routes to Correct Defect and back to Configure Trial. Misreadings: (i) the loop looks skippable (min-1-iteration is not visible); (ii) Correct Defect looks like it could be the final step before exit (it cannot — exit is only offered after the do-part); (iii) the cycle may be mistaken for an "infinite/while(true)" loop; (iv) readers may think the exit is conditional on some business rule; it is unconditional (offered after every iteration).
4. Silent routing nodes: the conversion creates silent transitions for: skip, loop-exit, loop-back, AND-joins, and (depending on the converter) parallel duplication after Register. Rendered as unlabeled gateways, they are indistinguishable at a glance; some renderers label them with generated ids (e.g., "skip_15") that look like activities or look like noise. The number of such nodes differs between the source net and the round-trip net — reviewers diffing the pictures will see "different models".
5. Layout: the BPMN layout is computed by a graph-layout algorithm; vertical/horizontal order (e.g., Prepare Evidence drawn above Legal Review, or Legal drawn between Assess and verification) suggests sequence or containment that has no semantics. Edge crossings and long edges reinforce false dependencies (e.g., the Assessverification dependency may be drawn as a long edge skirting the diagram, easily missed, or as a short edge that looks like branch-local flow). The POWL picture (a DAG) and the BPMN picture (a flow diagram) of the same model look structurally different; stakeholders comparing them will conclude one is wrong.

Section 9 — residual uncertainty (final):

6. def bpmn_inventory(bpmn_graph):
```python
    try:
        from pm4py.objects.bpmn.obj import BPMN
        tasks, starts, ends, gateways = [], 0, 0, 0
        for n in bpmn_graph.get_nodes():
            if isinstance(n, BPMN.Task):
                tasks.append(n.get_name())
            elif isinstance(n, BPMN.StartEvent):
                starts += 1
            elif isinstance(n, BPMN.EndEvent):
                ends += 1
            elif isinstance(n, (BPMN.ParallelGateway, BPMN.ExclusiveGateway, BPMN.InclusiveGateway)):
                gateways += 1
        return {"tasks": sorted(tasks), "start_events": starts,
                "end_events": ends, "gateways": gateways}
    except Exception as exc:
        return {"error": f"introspection unavailable ({exc.__class__.__name__}: {exc})"}
```
