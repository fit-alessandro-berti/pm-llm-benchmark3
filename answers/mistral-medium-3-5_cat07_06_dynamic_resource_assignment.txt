Here is the structured response to the dynamic resource-assignment policy design problem:

---

# 1. Objectives and constraints

### **Optimization Objectives**
- **Short-term throughput**: Minimize waiting times for urgent/high-priority cases while maintaining acceptable cycle times for normal/low-priority cases.
- **Skill development**: Align ~20% of normal/low-priority assignments with documented training priorities (e.g., R1_Amal  Diagnosis, R5_Eva  Security).
- **Workload fairness**: Distribute new assignments proportionally to capacity (hours remaining, current open cases) and avoid overloading high-overtime-risk resources.
- **Continuity of ownership**: Minimize reassignments, especially for high-complexity cases or those with prior handover costs.
- **SLA risk mitigation**: Prioritize cases with <6h SLA remaining or high priority (urgent/VIP).
- **Resilience**: Avoid creating bottlenecks (e.g., overloading Security/Approval specialists) and ensure backup coverage for critical skills.

### **Hard Constraints (Safety/Compliance)**
1. **Skill thresholds**:
   - Security Review: Skill 3 **or** Skill 1–2 **with supervision** by R7_Gita/R8_Hugo.
   - Final Approval: Skill 2 (Skill 1 may prepare but not finalize).
   - No assignments to Skill=0 for the required activity.
2. **High-priority cases**: Must be assigned to Skill 2 unless supervised (per constraint 2).
3. **Overtime risk**: Avoid assigning to resources with `overtime_risk="high"` (R2_Bruno, R4_David, R9_Iris) unless SLA is urgent and no alternative exists.
4. **Batch constraints**: Quality Check can only be batched if priority=normal/low **and** SLA >24h.
5. **Reassignment limits**: No resource may receive >4 new assignments in this batch.

### **Soft Constraints (Trade-offs)**
- **Continuity vs. skill match**: Prefer keeping current owner if handover cost is "high" **and** skill 2 (even if not specialist).
- **Training vs. SLA risk**: Training assignments (Skill=1) allowed for normal/low-priority cases if SLA >12h.
- **Fairness vs. urgency**: Balance workload fairness with SLA deadlines (e.g., assign urgent cases even if it slightly unbalances workload).
- **Resilience vs. specialization**: Avoid overloading specialists (e.g., R7_Gita/R8_Hugo for Security) to prevent bottlenecks.

# 2. Assignment policy

### **Algorithm Pseudocode**
```python
def assign_case(case, resources, max_new_assignments=4):
    # Step 1: Filter eligible resources
    eligible = []
    for resource in resources:
        # Hard constraints
        if resource.skill[case.required_skill] == 0:
            continue
        if case.priority == "urgent" and resource.skill[case.required_skill] < 2 and not has_supervisor(resource, case):
            continue
        if resource.overtime_risk == "high" and not (case.sla_remaining < 6 and no_alternative(eligible)):
            continue
        if resource.new_assignments >= max_new_assignments:
            continue
        if case.activity == "Security Review" and resource.skill["Security"] < 3 and not supervised_by_risk_team(resource):
            continue
        if case.activity == "Final Approval" and resource.skill["Approval"] < 2:
            continue
        if case.activity == "Quality Check" and case.priority in ["high", "urgent"]:
            continue  # Cannot batch high-priority Quality Checks

        eligible.append(resource)

    # Step 2: Score each eligible resource
    scores = []
    for resource in eligible:
        score = 0

        # SLA urgency (higher weight)
        sla_score = max(0, 10 - case.sla_remaining) * 2  # Penalize tight deadlines
        if case.priority == "urgent":
            sla_score += 5
        score += sla_score

        # Skill match (higher weight)
        skill = resource.skill[case.required_skill]
        if skill == 3:
            score += 10
        elif skill == 2:
            score += 7
        elif skill == 1:
            score += 3  # Only if supervised or low priority

        # Continuity (if current owner is eligible)
        if resource == case.current_owner:
            if case.handover_cost == "high":
                score += 8  # Strong continuity bonus
            elif case.handover_cost == "medium":
                score += 5
            else:
                score += 2

        # Workload fairness
        workload_penalty = resource.current_open_cases / resource.hours_remaining
        score -= workload_penalty * 2  # Penalize high workload

        # Training alignment (for normal/low priority)
        if case.priority in ["normal", "low"] and resource.training_priority == case.required_skill:
            score += 4  # Bonus for training

        # Resilience (avoid overloading specialists)
        if resource.skill[case.required_skill] == 3 and len(eligible) > 1:
            score -= 2  # Slight penalty to distribute specialist work

        scores.append((resource, score))

    # Step 3: Sort by score (descending) and pick top candidate
    scores.sort(key=lambda x: x[1], reverse=True)
    if not scores:
        return None  # No eligible resource (escalate)

    best_resource = scores[0][0]

    # Step 4: Exception handling
    # If best_resource is overtime-risk="high" and SLA >12h, try next candidate
    if best_resource.overtime_risk == "high" and case.sla_remaining > 12:
        for resource, _ in scores[1:]:
            if resource.overtime_risk != "high":
                best_resource = resource
                break

    return best_resource

def has_supervisor(resource, case):
    # Check if a supervisor (R7_Gita/R8_Hugo for Security) is available
    if case.required_skill == "Security":
        return any(r in ["R7_Gita", "R8_Hugo"] and r.hours_remaining > 0 for r in resources)
    return False

def supervised_by_risk_team(resource):
    return resource.team == "Risk" and resource.skill["Security"] >= 2
```

### **Scoring Components**
| Component               | Weight | Rationale                                                                 |
|-------------------------|--------|---------------------------------------------------------------------------|
| SLA urgency             | High   | Urgent cases and tight deadlines take priority.                          |
| Skill match             | High   | Specialist (3) > Competent (2) > Trainee (1, with supervision).          |
| Continuity              | High   | Avoid handover costs, especially for high-complexity cases.             |
| Workload fairness       | Medium | Penalize resources with high `open_cases/hours_remaining` ratio.        |
| Training alignment      | Low    | Bonus for normal/low-priority cases matching training priorities.       |
| Resilience              | Low    | Slight penalty for overloading specialists to avoid bottlenecks.       |

# 4. Comparison with baseline policies

| Metric                     | Proposed Policy                          | First-Available Routing                     | Specialist-Only Routing                     |
|----------------------------|------------------------------------------|---------------------------------------------|---------------------------------------------|
| **Throughput**             | Balanced; urgent cases prioritized.      | High (minimizes waiting time).              | Lower (queues at specialists).              |
| **SLA Risk**               | Low (urgent cases assigned to skilled).   | Medium (may assign to low-skill if available). | Low (but delays for non-urgent cases).      |
| **Error/Rework Risk**      | Low (skill 2 for high-priority; continuity). | High (skill mismatches likely).          | Very low (specialists only).                |
| **Workload Fairness**      | Good (considers hours/open cases).       | Poor (ignores specialization).              | Poor (overloads specialists).               |
| **Skill Development**      | High (20% training-aligned).             | Low (random assignments).                   | Low (only specialists work).                |
| **Continuity**             | High (retains owners where possible).    | Low (frequent reassignments).               | Medium (may reassign to specialists).       |
| **Resilience**             | High (avoids bottlenecks).               | Medium (no bottleneck avoidance).           | Low (specialists become single points of failure). |
| **Overtime Exposure**      | Controlled (avoids high-risk resources). | High (may overload available resources).    | Medium (specialists may be overloaded).     |

### **Trade-off Analysis**
- **Proposed Policy**:
  - **Strengths**: Balances all objectives; explainable; auditable.
  - **Weaknesses**: Slightly slower than first-available for throughput; more complex than specialist-only.
- **First-Available**:
  - **Strengths**: Simple; maximizes throughput.
  - **Weaknesses**: High rework/error risk; poor fairness; ignores continuity.
- **Specialist-Only**:
  - **Strengths**: Minimizes errors; high quality.
  - **Weaknesses**: Bottlenecks; poor fairness; slow for non-urgent cases.

# 5. Explanation to workers

**How Assignments Are Made – A Transparent Guide**

We assign cases based on **four priorities**, in this order:

1. **Urgency and Deadlines**
   - Cases with tight SLAs (e.g., <6 hours) or marked "urgent"/"VIP" go first.
   - Example: *C001 (Security Review, 3h SLA)* must go to a Security specialist (R7_Gita or R8_Hugo) immediately.

2. **Skill Match and Safety**
   - Work is assigned to the most skilled available person for the task.
   - **Exceptions**:
     - If you’re still learning a skill (e.g., R1_Amal for Diagnosis), you may get normal/low-priority cases to build experience—**but only if the deadline allows**.
     - For Security/Approval, trainees must be supervised by R7_Gita/R8_Hugo or a skill-3 colleague.

3. **Continuity Matters**
   - If you’re already working on a case, we’ll try to keep it with you—**especially if it’s complex or has been handed off before**.
   - Example: *C007 (Rework Correction)* stays with R3_Celine to avoid another costly handover.

4. **Fair Workload**
   - We spread new cases based on:
     - How many cases you’re already handling.
     - How many hours you have left today.
     - Overtime risk (we avoid piling work on R2_Bruno, R4_David, or R9_Iris unless it’s truly urgent).

**Why You Might Get a Case Outside Your Specialty**
- **Training**: If you’re prioritizing a skill (e.g., R5_Eva  Security), you may get supervised cases like *C015*.
- **Backup**: If a specialist is unavailable (e.g., R7_Gita is busy), we’ll pair you with one for critical tasks.
- **Fairness**: If you have fewer cases/hours than others, you might get a routine task (e.g., *C013* to R3_Celine).

**What We Avoid**
- Overloading anyone with >4 new cases at once.
- Assigning high-priority work to trainees without supervision.
- Breaking continuity for high-complexity cases unless absolutely necessary.

**Your Role**
- Flag if a case seems mismatched—we’ll adjust if safety or SLA is at risk.
- For supervised cases, coordinate with your assigned mentor (e.g., R7_Gita for Security).

# 7. PM4Py implementation sketch

```python
import pm4py
import pandas as pd
from pm4py.objects.log.util import dataframe_utils
from pm4py.objects.conversion.log import converter as log_converter

# Load event log (assuming CSV with the given columns)
df = pd.read_csv("event_log.csv")
df = pm4py.format_dataframe(df, case_id='case:concept:name', activity_key='concept:name', timestamp_key='time:timestamp')

# --- AUDIT: Handover Network ---
handover_df = pm4py.discover_handover_of_work_network(df, parameters={pm4py.objects.log.util.constants.PARAMETER_CONSTANT_CASE_ID_KEY: 'case:concept:name'})
pm4py.save_vis_sna(handover_df, "handover_network.html")  # Visualize handover patterns

# --- AUDIT: Workload Distribution ---
workload = df.groupby('org:resource').size().reset_index(name='case_count')
print("Workload per resource:\n", workload.sort_values('case_count', ascending=False))

# --- AUDIT: Skill Mismatch ---
# Join with skill matrix (example for Eligibility Review)
skill_matrix = pd.read_csv("skill_matrix.csv")
eligibility_assignments = df[df['concept:name'] == 'Eligibility Review']
mismatches = eligibility_assignments.merge(
    skill_matrix[['Resource', 'Eligibility']],
    left_on='org:resource',
    right_on='Resource'
)
mismatches = mismatches[mismatches['Eligibility'] < 1]  # Skill <1 (0 or missing)
print(f"Skill mismatches for Eligibility: {len(mismatches)}")

# --- AUDIT: SLA Misses ---
df = pm4py.insert_case_service_waiting_time(df)  # Add waiting times
sla_misses = df[df['time:timestamp'] > df['sla_deadline']]
print(f"SLA misses: {len(sla_misses.groupby('case:concept:name').ngroups())} cases")

# --- AUDIT: Training Alignment ---
training_priority = {
    'R1_Amal': 'Diagnosis',
    'R2_Bruno': 'Quality',
    # ... (populate from skill matrix)
}
normal_low_cases = df[df['case_priority'].isin(['normal', 'low'])]
training_aligned = normal_low_cases[
    normal_low_cases.apply(
        lambda row: training_priority.get(row['org:resource'], None) == row['required_skill'],
        axis=1
    )
]
print(f"Training-aligned assignments: {len(training_aligned)} / {len(normal_low_cases)}")

# --- AUDIT: Reassignment Impact ---
reassigned_cases = df[df['reassignment_count'] > 0]
rework_after_reassignment = reassigned_cases[
    reassigned_cases['concept:name'].isin(['Rework Correction', 'Appeal'])
]
print(f"Rework after reassignment: {len(rework_after_reassignment)}")

# --- AUDIT: Bottlenecks (Performance DFG) ---
dfg = pm4py.discover_performance_dfg(df)
print("Performance DFG (bottlenecks):\n", dfg)

# --- AUDIT: Resource Similarity (for supervision pairs) ---
similarity = pm4py.discover_activity_based_resource_similarity(df)
print("Resource similarity (for supervision):\n", similarity)

# --- POLICY VALIDATION: Check Hard Constraints ---
# Example: Security Review must have skill >=3 or supervised
security_cases = df[df['concept:name'] == 'Security Review']
security_assignments = security_cases.merge(
    skill_matrix[['Resource', 'Security']],
    left_on='org:resource',
    right_on='Resource'
)
violations = security_assignments[
    (security_assignments['Security'] < 3) &
    (~security_assignments['org:resource'].isin(['R7_Gita', 'R8_Hugo']))  # Not supervised
]
print(f"Security Review violations: {len(violations)}")

# --- POLICY VALIDATION: Overtime Risk ---
overtime_resources = ['R2_Bruno', 'R4_David', 'R9_Iris']
overtime_assignments = df[df['org:resource'].isin(overtime_resources)]
print(f"Assignments to overtime-risk resources: {len(overtime_assignments)}")
```
