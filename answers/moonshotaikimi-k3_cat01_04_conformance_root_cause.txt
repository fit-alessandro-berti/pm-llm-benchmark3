Disentangling a conformance collapse: artifacts vs. real change.
PM4Py reproduction built from the supplied variant-frequency aggregates.

The aggregates have no timestamps/resources, so this script reconstructs

from pm4py.objects.log.obj import EventLog, Trace, Event

# ----------------------------------------------------------------------

NORMATIVE = ("->( 'Receive', 'Classify', "

    ("Receive", "Risk Screen", "Queue Routed", "Enhanced Review", "Decision", "Notify"): 25,

    ("Receive", "Risk Screen", "Queue Routed", "Enhanced Review", "Compliance Check", "Supervisor Sign-off", "Decision", "Notify"): 150,
    ("Receive", "Risk Screen", "Queue Routed", "Enhanced Review", "Compliance Check", "Decision", "Notify"): 90,
    ("Receive", "Risk Screen", "Queue Routed", "Standard Review", "Supervisor Sign-off", "Decision", "Notify"): 60,
    ("Receive", "Risk Screen", "Queue Routed", "Standard Review", "Notify", "Decision"): 45,
    ("Receive", "Risk Screen", "Queue Routed", "Enhanced Review", "Decision", "Notify"): 30,
    ("Receive", "Risk Screen", "Queue Routed", "Manual Override", "Standard Review", "Decision", "Notify"): 35,
}

EXPECTED_ACTIVITIES = {a for v in list(APR_MAY)+list(JUN_JUL)+list(AUG_SEP) for a in v}

def render(variants, prefix, style):
    lines, i = [], 0
    for acts, n in variants.items():
        for _ in range(n):
            i += 1
            if style == "comma":
                lines.append(f"{prefix}{i}," + ",".join(acts))
            else:  # colon
                lines.append(f"{prefix}{i}: " + ",".join(acts))
    return "\n".join(lines)

def log_is_sane(log):
    try:
        if len(log) == 0:
            return False
        acts = {e["concept:name"] for t in log for e in t}
        return acts <= EXPECTED_ACTIVITIES and all(t[0]["concept:name"] == "Receive" for t in log)
    except Exception:
        return False

def build_log(variants, prefix):
    """Expand variant frequencies into an EventLog.

    Primary path: pm4py.parse_event_log_string (one trace per line).
    Two line layouts are tried because the accepted layout differs across
    PM4Py versions; a sanity check decides. Fallback: DataFrame route.
    """
    for style in ("comma", "colon"):
        try:
            log = pm4py.parse_event_log_string(render(variants, prefix, style))
            if log_is_sane(log):
                return log
        except Exception:
            pass
    import pandas as pd
    rows, i = [], 0
    for acts, n in variants.items():
        for _ in range(n):
            i += 1
            for k, a in enumerate(acts):
                rows.append({"case:concept:name": f"{prefix}{i}",
                             "concept:name": a,
                             "time:timestamp": pd.Timestamp("2026-04-01") + pd.Timedelta(hours=k)})
    df = pm4py.format_dataframe(pd.DataFrame(rows),
                                case_id="case:concept:name",
                                activity_key="concept:name",
                                timestamp_key="time:timestamp")
    return pm4py.convert_to_event_log(df)

logs = {"Apr-May": build_log(APR_MAY, "AM"),
        "Jun-Jul": build_log(JUN_JUL, "JJ"),
        "Aug-Sep": build_log(AUG_SEP, "AS")}

# 5. Alignment diagnostics + fitness helpers

from collections import Counter

def moves_summary(alignments):
    on_log, on_model = Counter(), Counter()
    for al in alignments:
        for a, b in al["alignment"]:
            log_side = a not in (None, ">>")
            model_side = b not in (None, ">>")
            if log_side and not model_side:
                on_log[a] += 1
            elif model_side and not log_side:
                on_model[b] += 1
    return on_log, on_model

def report(log, net, im, fm, tag):
    al = pm4py.conformance_diagnostics_alignments(log, net, im, fm)
    fit = pm4py.fitness_alignments(log, net, im, fm)
    try:
        prec = pm4py.precision_alignments(log, net, im, fm)
    except Exception:
        prec = float("nan")
    on_log, on_model = moves_summary(al)
    pct = fit.get("percentage_of_fitting_traces", float("nan"))
    avg = fit.get("averageFitness", fit.get("log_fitness", float("nan")))
    print(f"[{tag}] fitting={pct:.1f}%  avgFitness={avg:.3f}  precision={prec:.3f}")
    print(f"    moves on log:   {dict(on_log.most_common(5))}")
    print(f"    moves on model: {dict(on_model.most_common(5))}")
    return {"fitting": pct, "fitness": avg, "precision": prec}
```

# ----------------------------------------------------------------------
# 6. Transparent normalization pipeline
# ----------------------------------------------------------------------
LABEL_MAP = {"Risk Screen": "Classify"}          # UI release 4.2 rename
TECHNICAL_EVENTS = {"Queue Routed"}              # gateway technical event

def normalize(log, map_labels=True, drop_technical=True, restore_source_order=False):
    """Returns a NEW log; never mutates the input.

    restore_source_order simulates re-ordering by the source-system sequence
    number: with ingestion-time stamps removed, a spurious 'Notify, Decision'
    adjacency is restored to 'Decision, Notify'.
    """
    out = EventLog(attributes=dict(getattr(log, "attributes", {})))
    for trace in log:
        acts = []
        for ev in trace:
            a = ev["concept:name"]
            if map_labels:
                a = LABEL_MAP.get(a, a)
            if drop_technical and a in TECHNICAL_EVENTS:
                continue
            acts.append(a)
        if restore_source_order:
            i = 0
            while i < len(acts) - 1:
                if acts[i] == "Notify" and acts[i + 1] == "Decision":
                    acts[i], acts[i + 1] = acts[i + 1], acts[i]
                i += 1
        t = Trace(attributes=dict(trace.attributes))
        for a in acts:
            t.append(Event({"concept:name": a}))
        out.append(t)
    return out
```

Model comparison and data-aware stratification | Candidate tree with optional sign-off; fitness vs precision (and generalization) comparison; stratification by claim value; sign-off coverage check for > EUR 50k cases | `pm4py.parse_process_tree`, `pm4py.convert_to_petri_net`, `pm4py.fitness_alignments`, `pm4py.precision_alignments`, `pm4py.generalization_tbr`, dataframe-based case-attribute stratification | Claim-amount attribute joined to cases | Adopt a versioned model update with a documented data guard, or reject it as overpermissive; list high-value cases lacking sign-off | Optional-activity model overgeneralizes; attribute missingness; Simpson's paradox in aggregates |
| 4. Ground-truthing and behavioral attribution | Manual audit of the 55 missing-check traces; override-semantics audit; reviewer-level Pareto of residual moves; pre/post memo comparison | Filtered sublogs (variant/case filters), alignments on the audited sublog, per-reviewer DFGs via `pm4py.convert_to_dataframe` filtering | Source-system screenshots/records, reviewer assignments, audit outcomes | Classify residual as event loss vs real bypass; trigger remediation or close; feed confirmed semantics back into the model | Audit sampling bias; screenshots not authoritative; Hawthorne effect; small samples |

Disentangling a conformance collapse: artifacts vs. real change.
PM4Py reproduction built from the supplied variant-frequency aggregates.

NOTE: the aggregates carry no timestamps/resources, so this script reconstructs
synthetic logs preserving variant frequencies and activity order only; it will
not reproduce the vendor's exact numbers.

from pm4py.objects.log.obj import EventLog, Trace, Event

# 1+2. Normative process tree -> accepting Petri net
# ----------------------------------------------------------------------

             "X( *( 'Standard Review', 'Request Info' ), "
             "->( 'Enhanced Review', 'Compliance Check' ) ), "
             "'Decision', 'Notify' )")

    ("Receive", "Risk Screen", "Queue Routed", "Standard Review", "Notify", "Decision"): 70,
    ("Receive", "Risk Screen", "Standard Review", "Decision", "Notify"): 90,
    ("Receive", "Classify", "Queue Routed", "Standard Review", "Decision", "Notify"): 30,
    ("Receive", "Risk Screen", "Queue Routed", "Enhanced Review", "Decision", "Notify"): 25,
    ("Receive", "Risk Screen", "Queue Routed", "Manual Override", "Standard Review", "Decision", "Notify"): 20,

AUG_SEP = {
    ("Receive", "Risk Screen", "Queue Routed", "Standard Review", "Decision", "Notify"): 210,
    ("Receive", "Risk Screen", "Queue Routed", "Enhanced Review", "Compliance Check", "Supervisor Sign-off", "Decision", "Notify"): 150,
    ("Receive", "Risk Screen", "Queue Routed", "Enhanced Review", "Compliance Check", "Decision", "Notify"): 90,
    ("Receive", "Risk Screen", "Queue Routed", "Standard Review", "Supervisor Sign-off", "Decision", "Notify"): 60,
    ("Receive", "Risk Screen", "Queue Routed", "Standard Review", "Notify", "Decision"): 45,
    ("Receive", "Risk Screen", "Queue Routed", "Enhanced Review", "Decision", "Notify"): 30,
    ("Receive", "Risk Screen", "Queue Routed", "Manual Override", "Standard Review", "Decision", "Notify"): 35,
}
EXPECTED = {a for table in (APR_MAY, JUN_JUL, AUG_SEP) for v in table for a in v}

def render(variants, prefix, style):
    lines, i = [], 0
    for acts, n in variants.items():
        for _ in range(n):
            i += 1
            sep = "," if style == "comma" else ": "
            lines.append(f"{prefix}{i}{sep}" + ",".join(acts))
    return "\n".join(lines)

def sane(log):
    try:
        acts = {e["concept:name"] for t in log for e in t}
        return len(log) > 0 and acts <= EXPECTED and \
               all(t[0]["concept:name"] == "Receive" for t in log)
    except Exception:
        return False

def build_log(variants, prefix):
    """Expand variant frequencies into an EventLog.

    Primary path: pm4py.parse_event_log_string (one trace per line). Two line
    layouts are tried because the accepted layout differs across PM4Py
    versions; a sanity check decides. Fallback: DataFrame route.
    """
    for style in ("comma", "colon"):
        try:
            log = pm4py.parse_event_log_string(render(variants, prefix, style))
            if sane(log):
                return log
        except Exception:
            pass
    import pandas as pd
    rows, i = [], 0
    for acts, n in variants.items():
        for _ in range(n):
            i += 1
            for k, a in enumerate(acts):
                rows.append({"case:concept:name": f"{prefix}{i}",
                             "concept:name": a,
                             "time:timestamp": pd.Timestamp("2026-04-01")
                                               + pd.Timedelta(hours=k)})
    df = pm4py.format_dataframe(pd.DataFrame(rows),
                                case_id="case:concept:name",
                                activity_key="concept:name",
                                timestamp_key="time:timestamp")
    return pm4py.convert_to_event_log(df)

logs = {"Apr-May": build_log(APR_MAY, "AM"),
        "Jun-Jul": build_log(JUN_JUL, "JJ"),
        "Aug-Sep": build_log(AUG_SEP, "AS")}

# 4. Frequency DFGs per period
# ----------------------------------------------------------------------
def show_dfg(log, name, top=8):
    dfg, sa, ea = pm4py.discover_dfg(log)
    print(f"\n[DFG] {name}: {len(log)} traces, starts={sa}, ends={ea}")
    for (a, b), c in sorted(dfg.items(), key=lambda kv: -kv[1])[:top]:
        print(f"    {a} -> {b} : {c}")

for name, lg in logs.items():
    show_dfg(lg, name)

# ----------------------------------------------------------------------
# 5. Alignment diagnostics + fitness/precision helper
# ----------------------------------------------------------------------
from collections import Counter

def moves_summary(alignments):
    on_log, on_model = Counter(), Counter()
    for al in alignments:
        for a, b in al["alignment"]:
            log_side, model_side = a not in (None, ">>"), b not in (None, ">>")
            if log_side and not model_side:
                on_log[a] += 1
            elif model_side and not log_side:
                on_model[b] += 1
    return on_log, on_model

def report(log, net, im, fm, tag):
    al = pm4py.conformance_diagnostics_alignments(log, net, im, fm)
    fit = pm4py.fitness_alignments(log, net, im, fm)
    try:
        prec = pm4py.precision_alignments(log, net, im, fm)
    except Exception:
        prec = float("nan")
    on_log, on_model = moves_summary(al)
    pct = fit.get("percentage_of_fitting_traces", float("nan"))
    avg = fit.get("averageFitness", fit.get("log_fitness", float("nan")))
    print(f"[{tag}] fitting={pct:.1f}%  avgFitness={avg:.3f}  precision={prec:.3f}")
    print(f"    moves on log:   {dict(on_log.most_common(5))}")
    print(f"    moves on model: {dict(on_model.most_common(5))}")

# ----------------------------------------------------------------------
# 6. Transparent normalization pipeline (never mutates the input log)
# ----------------------------------------------------------------------
LABEL_MAP = {"Risk Screen": "Classify"}   # UI release 4.2 rename
TECHNICAL_EVENTS = {"Queue Routed"}       # gateway technical event

def normalize(log, map_labels=True, drop_technical=True,
              restore_source_order=False):
    """restore_source_order simulates ordering by the source-system sequence
    number: a spurious 'Notify, Decision' adjacency becomes 'Decision, Notify'.
    """
    out = EventLog()
    for trace in log:
        acts = []
        for ev in trace:
            a = ev["concept:name"]
            if map_labels:
                a = LABEL_MAP.get(a, a)
            if drop_technical and a in TECHNICAL_EVENTS:
                continue
            acts.append(a)
        if restore_source_order:
            for i in range(len(acts) - 1):
                if acts[i] == "Notify" and acts[i + 1] == "Decision":
                    acts[i], acts[i + 1] = acts[i + 1], acts[i]
        t = Trace(attributes=dict(trace.attributes))
        for a in acts:
            t.append(Event({"concept:name": a}))
        out.append(t)
    return out

# ----------------------------------------------------------------------

report(logs["Apr-May"], norm_net, norm_im, norm_fm, "Apr-May | raw")

    for stage, kw in stages:

               f"{period} | {stage}")

show_dfg(normalize(logs["Jun-Jul"], map_labels=True, drop_technical=True,
                   restore_source_order=True), "Jun-Jul fully normalized")

# 8. Candidate model: optional Supervisor Sign-off before Decision
# ----------------------------------------------------------------------
CANDIDATE = ("->( 'Receive', 'Classify', "
             "X( *( 'Standard Review', 'Request Info' ), "
             "->( 'Enhanced Review', 'Compliance Check' ) ), "
             "X( 'Supervisor Sign-off', tau ), "
             "'Decision', 'Notify' )")

def build_candidate_tree():
    try:
        return pm4py.parse_process_tree(CANDIDATE)
    except Exception:
        # Programmatic fallback: optional activity = XOR with a silent (tau) leaf
        from pm4py.objects.process_tree.obj import ProcessTree, Operator
        def leaf(lbl):
            t = ProcessTree(); t.label = lbl; return t
        def node(op, *ch):
            t = ProcessTree(operator=op)
            for c in ch:
                c.parent = t
            t.children = list(ch)
            return t
        return node(Operator.SEQUENCE, leaf("Receive"), leaf("Classify"),
                    node(Operator.XOR,
                         node(Operator.LOOP, leaf("Standard Review"), leaf("Request Info")),
                         node(Operator.SEQUENCE, leaf("Enhanced Review"), leaf("Compliance Check"))),
                    node(Operator.XOR, leaf("Supervisor Sign-off"), ProcessTree()),  # empty = tau
                    leaf("Decision"), leaf("Notify"))

cand_net, cand_im, cand_fm = pm4py.convert_to_petri_net(build_candidate_tree())

aug_sep_norm = normalize(logs["Aug-Sep"], map_labels=True, drop_technical=True,
                         restore_source_order=True)
print("\n=== Aug-Sep (normalized): normative vs candidate model ===")
report(aug_sep_norm, norm_net, norm_im, norm_fm, "normative")
report(aug_sep_norm, cand_net, cand_im, cand_fm, "candidate +optional sign-off")

## 7. Provisional conclusion

**Primary explanation.** The June–July fitness collapse is predominantly a measurement-artifact stack: a cosmetic label rename (`Risk Screen` for `Classify`), injection of a technical event (`Queue Routed`), and ingestion-time re-stamping that inverted `Decision`/`Notify` — each independently dated, each recovering a distinct slice of fitness when repaired (32588189 percent). From August, a fourth mechanism dominates: the mandated `Supervisor Sign-off` made the normative model genuinely obsolete, and a sign-off-aware model recovers most remaining fitness (6686 percent). Confidence: high (~85–90) that these four mechanisms explain the bulk of the *measured* drop.

**Real compliance problem.** Unresolved, not exonerated. Fifty-five enhanced-review traces lack a recorded `Compliance Check`; `Manual Override` appears to grow; the July memo shows someone was worried. These residuals survive every preprocessing stage and may be event loss, semantic drift, or true bypasses. Confidence that *some* real residual deviation exists: moderate (~55). Confidence in *systematic employee noncompliance*: low (~30) — there is no audit evidence, outcomes are stable, and blaming staff for moves on log would repeat the original error.

**Immediate actions.**
- **IT:** re-extract `Decision`/`Notify` using source sequence numbers; namespace and default-exclude technical events; backfill resource attributes; publish the `Manual Override` logging change; pin event-schema versions.
- **Process owners:** version the normative model and add `Supervisor Sign-off`; specify the EUR 50,000 guard as a data-aware constraint (a control-flow tree cannot express it); decide whether `Manual Override` is a sanctioned activity.
- **Compliance:** audit the 55 missing-check traces and a sample of overrides; stratify sign-off coverage by claim value; treat the memo as a hypothesis, not evidence.

**Must not be claimed yet:** that employees bypassed controls; that the process is healthy merely because outcomes are stable; that 86 percent fitness under the candidate model equals compliance — it may be overpermissive and cannot check the value condition; that `Queue Routed` is meaningless before its intermittency is explained; or that normalization is neutral — each cleaning step can hide real deviations.

## 8. Model-governance recommendation

1. **Model versioning with validity intervals.** Every normative model carries a version, an effective-from date, and an owner; every policy or regulatory change triggers a mandatory model-review ticket before its effective date; conformance reports always cite the model version used.
2. **Event-semantic contracts.** Activity names are a versioned contract between source systems and process-mining consumers; UI renames must not alter emitted event labels without a registered mapping; contract tests run in CI so that a screen rename can never silently become a log rename.
3. **Technical-event policy.** System-generated events are namespaced (e.g., `sys:Queue Routed`), registered in an event catalog with an explicit semantics field (user action vs system event), and default-excluded from conformance checking unless deliberately modeled; no new event type reaches production logs unregistered.
4. **Timestamp provenance.** Every event carries source timestamp, source sequence number, and ingestion timestamp; conformance ordering uses source time/sequence only; substituting ingestion time requires documented justification; pipelines run monotonicity and reversal checks per release.
5. **Data-aware constraints.** Value- or resource-dependent rules (the EUR 50,000 sign-off, authorizations) are encoded as data-aware checks alongside the control-flow model; every control-flow-only model is annotated with the conditions it cannot express, so fitness gains from permissive control-flow extensions are flagged as potential overgeneralization.
6. **Post-release validation.** After any UI, gateway, or ETL release, a conformance canary replays a frozen baseline sample and compares pre/post fitness with staged normalization; deviations beyond a threshold block or flag the release, and pipeline changes dual-run old and new extraction for one full cycle before cutover.
