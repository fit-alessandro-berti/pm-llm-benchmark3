```python
import io
import pandas as pd

CSV_DATA = """case_id,trace,complexity,group,shift,arrival_load,compliance_required,wait_before_review_h,review_service_h,rework_count,cycle_time_h,outcome_30d
C01,Register>Classify>Review>Close,Low,Orion,Night,High,No,3.3,3.1,0,7.2,No appeal
... (all 30 rows)
"""

MIN_SUPPORT = 3

def warn(label, n, threshold=MIN_SUPPORT):
    if n < threshold:
        print(f"  [WARNING] '{label}' has n={n} (<{threshold}): poor support, interpret with caution.")

df = pd.read_csv(io.StringIO(CSV_DATA))
print(f"Loaded {len(df)} completed cases (abandoned/open cases absent by design).\n")

# ---- 2. Crude comparison ----
crude = df.groupby("group")["cycle_time_h"].agg(cases="count", mean="mean", median="median")
print("Crude cycle time by group:\n", crude.round(2))
print(f"Crude Orion-Nova mean gap: {crude.loc['Orion','mean']-crude.loc['Nova','mean']:+.2f} h\n")

# ---- 2b. Stratified by complexity ----
strat = df.groupby(["complexity", "group"]).agg(
    n=("case_id", "count"),
    mean_cycle=("cycle_time_h", "mean"),
    mean_wait=("wait_before_review_h", "mean"),
    mean_service=("review_service_h", "mean"),
).round(2)
print("Stratified by complexity:\n", strat, "\n")
for (cx, g), sub in df.groupby(["complexity", "group"]):
    warn(f"complexity={cx}, group={g}", len(sub))

# ---- 3. Complexity-standardized contrast (pooled case-mix weights) ----
weights = df["complexity"].value_counts(normalize=True)
print("\nCommon weighting distribution (pooled complexity mix):")
print(weights.round(3).to_string())
std_means = {}
for g, sub in df.groupby("group"):
    m = sub.groupby("complexity")["cycle_time_h"].mean()
    std_means[g] = float(sum(weights[c] * m[c] for c in weights.index))
print(f"Standardized means: { {k: round(v,2) for k,v in std_means.items()} }")
print(f"Standardized Orion-Nova contrast: {std_means['Orion']-std_means['Nova']:+.2f} h")
print("(Association adjusted only for recorded complexity; NOT a causal estimate.)\n")

# ---- 4. Overlap: complexity x shift x group ----
overlap = (df.groupby(["complexity", "shift", "group"]).size().unstack("group", fill_value=0).reindex(columns=["Nova", "Orion"], fill_value=0))
print("Overlap counts (complexity x shift):\n", overlap, "\n")
for (cx, sh), row in overlap.iterrows():
    for g in ["Nova", "Orion"]:
        if row[g] == 0:
            print(f"  [WARNING] POSITIVITY: no {g} cases in complexity={cx}, shift={sh}.")
        else:
            warn(f"complexity={cx}, shift={sh}, group={g}", int(row[g]))

# ---- 5a. Trace variants ----
variants = df.groupby(["group", "trace"]).size().rename("n").reset_index()
print("\nVariants by group:\n", variants.to_string(index=False))
shared = set(df.loc[df.group=="Orion","trace"]) & set(df.loc[df.group=="Nova","trace"])
print("Shared variants:", shared if shared else "none", "\n")

# ---- 5b. Directly-follows via PM4Py (topology only; no fabricated timestamps) ----
def to_log_string(sub):
    return "\n".join(f"{r.case_id};{act.strip()}" for r in sub.itertuples() for act in r.trace.split(">"))

try:
    import pm4py
    for g in ["Orion", "Nova"]:
        log = pm4py.parse_event_log_string(to_log_string(df[df.group == g]), sep=";", case_id_key="case:concept:name", activity_key="concept:name")
        dfg, start, end = pm4py.discover_dfg(log)
        print(f"DFG directly-follows frequencies - {g}:")
        for (a, b), f in sorted(dfg.items()):
            print(f"  {a} -> {b}: {f}")
        print(f"  start={start}, end={end}\n")
except ImportError:
    print("pm4py unavailable; manual directly-follows cross-check:")
    for g in ["Orion", "Nova"]:
        dfreq = {}
        for t in df.loc[df.group == g, "trace"]:
            acts = [a.strip() for a in t.split(">")]
            for a, b in zip(acts, acts[1:]):
                dfreq[(a, b)] = dfreq.get((a, b), 0) + 1
        print(g, dfreq)

# ---- sanity: cycle ~ wait + service ----
df["residual_h"] = (df["cycle_time_h"] - df["wait_before_review_h"] - df["review_service_h"]).round(2)
print("cycle - (wait+service) residual summary (other activities):")
print(df.groupby("group")["residual_h"].describe().round(2))
```

## 8. Residual uncertainty

1. **Unobserved case risk in routing.** Within-stratum reversals could still be confounded by difficulty signals beyond the recorded complexity class. Most valuable addition: the classifier's pre-routing difficulty/risk score (or adjuster notes) recorded at `Classify`, enabling within-stratum adjustment and calibrated sensitivity analysis.
2. **Positivity gaps.** No Orion cases exist for Day Medium/High, so those contrasts require extrapolation. Most valuable addition: a small randomized (or rule-change) routing pilot that creates genuine overlap in the empty complexity×shift cells.
3. **Completed-case selection.** Abandonment and still-open cases may depend on duration, risk, and group, biasing every contrast. Most valuable addition: a full case census including abandoned/open cases with censoring timestamps and reasons.
4. **Unobserved queue dynamics.** `wait_before_review_h` bundles classification queueing and compliance waiting; backlog at decision points is unknown, and shared queues create interference. Most valuable addition: event-level timestamps plus a queue-length/backlog snapshot at each `Classify` event.
5. **Coarse, post-process quality outcome.** Two appeal events cannot support a quality ranking, and appeals reflect post-process factors. Most valuable addition: structured outcome detail (appeal reason, payment accuracy, litigation) with 90-day follow-up linked to case ID.
