# 4. Network and conformance analyses (PM4Py)

```python
# 4.1 Filter four-eyes violations using pm4py.filter_four_eyes_principle
from pm4py import filter_four_eyes_principle, format_dataframe
df = format_dataframe(event_log)  # assumes dataframe with case:concept:name, org:resource, concept:name, lifecycle:transition

# Define the two normative rules
rules = {
    "CO+AP": ["Create Purchase Order", "Approve Purchase Order"],
    "Bank+Pay": ["Change Vendor Bank Details", "Payment Approval"]
}

violations = {}
for rule_name, act_pair in rules.items():
    viol = filter_four_eyes_principle(df, activities=act_pair, 
                                      same_resource=True, 
                                      case_id="case:concept:name", 
                                      resource_col="org:resource")
    violations[rule_name] = viol
# Output: dataframe of violating cases, resources, and timestamps

# 4.2 Discover handover, workingtogether, and subcontracting networks
from pm4py import discover_handover_of_work_network, discover_working_together_network, discover_subcontracting_network

handover_net = discover_handover_of_work_network(df, 
                                                case_id="case:concept:name", 
                                                resource_col="org:resource")
working_tog_net = discover_working_together_network(df, 
                                                    case_id="case:concept:name", 
                                                    resource_col="org:resource")
subcontract_net = discover_subcontracting_network(df, 
                                                  case_id="case:concept:name", 
                                                  resource_col="org:resource")

# 4.3 Compare emergency vs nonemergency cases
emergency_cases = df[df["emergency_flag"] == True]
non_emergency = df[df["emergency_flag"] == False]

# 4.4 Timing analysis with pathperformance / servicewaitingtime
from pm4py import filter_paths_performance, get_service_time, get_variants

# Service time between Create PO and Approve PO
svc_time = get_service_time(df, "Create Purchase Order", "Approve Purchase Order",
                            case_id="case:concept:name", activity_col="concept:name",
                            timestamp_col="time:timestamp")

# Filter fast paths (e.g., < 30 min)
fast_paths = filter_paths_performance(df, 
                                       activity_name="Create Purchase Order",
                                       performance_threshold=30,  # minutes
                                       case_id="case:concept:name")

# 4.5 Declarative rule / logskeleton check for retrospective justification
from pm4py import discover_declare, conformance_declare, discover_log_skeleton

# Example rule: "Retrospective Justification must occur within 3 working days by a different resource"
# Encode as a DECLARE pattern (simplified)

    "Eventually(ResponsibleRole(Justification), ImmediatelyAfter(ResponsibleRole(Approve)))",
    "Not(And(SameResource(Justification, Approve), WithinTime(Justification, Approve, 3d)))"
]

log_skeleton = discover_log_skeleton(df, case_id="case:concept:name")
# Conformance check to see how many cases respect the rule

# 4.6 Isolate manual_upload and legacy_erp sourcesystem effects

                                          same_resource=True, case_id="case:concept:name", resource_col="org:resource")
```
