# 5. PM4Py role-discovery and similarity code

```python
import numpy as np
import pandas as pd
import pm4py


def analyze_organizational_roles(raw_df: pd.DataFrame):
    # 1. Load and format the dataframe to standard PM4Py structure
    df = pm4py.format_dataframe(
        raw_df,
        case_id="case:concept:name",
        activity_key="concept:name",
        timestamp_key="time:timestamp",
        timest_format="ISO8601",
    )

    # Insert service and waiting times to evaluate substitution performance penalties
    df = pm4py.insert_case_service_waiting_time(df)

    # 2. Separate routine operations from temporary substitution episodes
    # This prevents temporary cross-coverage (e.g., Giorgio in W3-4, Mateo in W5)
    # from distorting baseline role discovery.
    df_routine = pm4py.filter_event_attribute_values(
        df,
        attribute_key="temporary_substitution",
        values=[False, "False", 0],
        level="event",
        retain=True,
    )

    # 3. Discover baseline organizational roles and activity-based resource similarity
    roles_full = pm4py.discover_organizational_roles(df)
    roles_routine = pm4py.discover_organizational_roles(df_routine)

    act_similarity = pm4py.discover_activity_based_resource_similarity(
        df_routine
    )

    # 4. Compute social networks to distinguish execution from brokerage and authority
    # Handover of work reveals routing hubs (e.g., R13_Mateo, R14_Nora)
    how_network = pm4py.discover_handover_of_work_network(df)
    # Working together reveals cross-functional case collaboration
    wt_network = pm4py.discover_working_together_network(df)
    # Subcontracting reveals re-verification / oversight (e.g., R08_Hana -> R13_Mateo -> R08_Hana)
    sub_network = pm4py.discover_subcontracting_network(df)

    # 5. Enrich resource profiles with case complexity, substitution ratios, and authority signals
    # Map each resource to the discovered routine role ID
    resource_to_role = {}
    for idx, role in enumerate(roles_routine):
        # PM4Py role objects/tuples contain activities and resource frequency dicts
        res_dict = role[1] if isinstance(role, tuple) else role.resources
        for res in res_dict.keys():
            resource_to_role[res] = f"Role_{idx+1}"

    # Aggregate case-level complexity per resource (avoiding event-count inflation)
    case_res_df = df.groupby(
        ["case:concept:name", "org:resource"], as_index=False
    ).agg(
        case_complexity=("case_complexity", "first"),
        max_escalation=("escalation_level", "max"),
        had_substitution=("temporary_substitution", "max"),
    )

    res_complexity_summary = (
        case_res_df.groupby("org:resource")
        .agg(
            median_complexity=("case_complexity", "median"),
            p90_complexity=("case_complexity", lambda x: np.percentile(x, 90)),
            total_cases=("case:concept:name", "count"),
            sub_case_rate=("had_substitution", "mean"),
        )
        .reset_index()
    )

    # 6. Guard against conflating activity frequency with formal authority:
    # Check if approvals (A9) or security reviews (A6) are final or followed by re-checks
    df_sorted = df.sort_values(["case:concept:name", "time:timestamp"])
    df_sorted["next_same_act_resource"] = df_sorted.groupby(
        ["case:concept:name", "concept:name"]
    )["org:resource"].shift(-1)

    authority_check = (
        df_sorted[
            df_sorted["concept:name"].isin(
                [
                    "Supervisor Approval",
                    "Security Review",
                    "Pricing Exception Review",
                ]
            )
        ]
        .groupby(["org:resource", "concept:name"])
        .agg(
            total_executions=("case:concept:name", "count"),
            overridden_or_rechecked=(
                "next_same_act_resource",
                lambda x: x.notna().sum(),
            ),
            median_service_sec=("@@service_time", "median"),
        )
        .reset_index()
    )
    authority_check["recheck_rate"] = (
        authority_check["overridden_or_rechecked"]
        / authority_check["total_executions"]
    )

    # Merge role assignment with complexity and authority metrics
    res_complexity_summary["discovered_routine_role"] = res_complexity_summary[
        "org:resource"
    ].map(resource_to_role)

    return {
        "roles_full_log": roles_full,
        "roles_routine_only": roles_routine,
        "similarity_matrix": act_similarity,
        "handover_network": how_network,
        "working_together_network": wt_network,
        "subcontracting_network": sub_network,
        "resource_enriched_profile": res_complexity_summary,
        "authority_and_substitution_audit": authority_check,
    }
```

# 6. Operational interpretation

*   **Staffing and Resilience:**
    *   *De-risk Single Points of Failure:* R05_Erik (Security) and R06_Fatima (Commercial) are acute bottlenecks. Rather than relying on R13_Mateo to filter Erik's queue (which delayed two critical cases >2 days) or R07_Giorgio to cover Fatima at half speed (74m vs 38m), management should formally cross-train and certify R03_Chen as a backup Security Reviewer (capitalizing on Chen's 24 A6 touches) and provide Giorgio with dedicated commercial tooling/refresher training so his substitution speed approaches Fatima’s.
    *   *Protect Brokerage Capacity:* R13_Mateo should be staffed and measured as an Incident Coordinator/Broker, not evaluated on standard single-activity throughput.
*   **Training and Automation Boundary Control:**
    *   *Fix Temp Agent Onboarding:* The 17 reopened cases in Weeks 8–10 trace directly to R11_Kai and R12_Lina failing to record complete entitlement notes during A2. Management must either restrict Temps to cases already fully validated by R14_Nora (Fragment 1) or institute mandatory checklist validation in the UI before Temps can execute A4/A12.
    *   *Tiered Technical Progression:* Use R03_Chen’s workflow on high/critical cases as the structured training curriculum to upskill R04_Dina from medium-complexity work (0.48) to senior technical diagnostic work (0.71).
*   **Risk Control and Governance:**
    *   *Enforce Decision Authority Boundaries:* Separate *preliminary coordination* from *formal sign-off* in the workflow engine. When R13_Mateo covers for R08_Hana (Week 5), the system should log a distinct activity (`A9_Preliminary_Triage`) rather than `A9_Supervisor_Approval`, automatically queuing high-risk cases for Hana’s audit while allowing low-risk cases to proceed.

**Two Cautions Against Changing Staffing Based Only on Mined Roles:**
1.  **Do not equate high cosine similarity with resource interchangeability:** R03_Chen and R04_Dina have a 0.94 activity similarity, and R01_Amira/R02_Ben have ~0.95+ similarity with Temps R11/R12 on core activities. Replacing Chen with Dina on critical enterprise security configurations, or replacing permanent Front Office staff with uncertified Temps, will cause severe SLA breaches, security vulnerabilities, and reopen cascades because activity frequency ignores case complexity and tacit domain skill.
2.  **Do not mistake crisis-driven shadow work for sustainable capacity or formal authority:** Mined logs show R13_Mateo performing 44 Supervisor Approvals and 16 Security Reviews, and R07_Giorgio performing 72 Pricing Reviews. Treating these emergency substitutions as proof that Mateo can replace management/security specialists, or that Giorgio equals a full-time Commercial Specialist, institutionalizes hidden quality risks (e.g., 62% approval recheck rates, >2-day critical security delays, and doubled cycle times).
