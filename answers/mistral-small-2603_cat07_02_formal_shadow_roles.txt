## # 5. PM4Py role-discovery and similarity code

```python
import pandas as pd
import pm4py
from pm4py.objects.log.util import dataframe_utils
from pm4py.objects.conversion.log import converter as log_converter

# Load and format the event log
def load_and_format_event_log(file_path):
    df = pd.read_csv(file_path)
    df = dataframe_utils.convert_timestamp_columns(df)
    df = pm4py.format_dataframe(df,
                               case_id='case:concept:name',
                               activity_key='concept:name',
                               timestamp_key='time:timestamp',
                               resource_key='org:resource')
    return df

# Discover organizational roles
def discover_organizational_roles(event_log):
    # Discover roles based on resource behavior
    roles = pm4py.discover_organizational_roles(event_log)

    # Add case complexity information
    roles_with_complexity = roles.copy()
    roles_with_complexity['median_complexity'] = roles['org:resource'].map(
        lambda r: get_median_complexity(r)
    )

    return roles_with_complexity

def get_median_complexity(resource):
    # This would query the complexity data (simplified here)
    complexity_map = {
        'R01_Amira': 0.34, 'R02_Ben': 0.29, 'R03_Chen': 0.71,
        'R04_Dina': 0.48, 'R05_Erik': 0.86, 'R06_Fatima': 0.67,
        'R07_Giorgio': 0.58, 'R08_Hana': 0.79, 'R09_Ivan': 0.52,
        'R10_Julia': 0.48, 'R11_Kai': 0.18, 'R12_Lina': 0.19,
        'R13_Mateo': 0.77, 'R14_Nora': 0.11
    }
    return complexity_map.get(resource, 0.5)

# Discover activity-based resource similarity
def discover_resource_similarity(event_log):
    similarity_matrix = pm4py.discover_activity_based_resource_similarity(event_log)

    # Filter for significant similarities (cosine > 0.7)
    significant_pairs = []
    for r1 in similarity_matrix.index:
        for r2 in similarity_matrix.columns:
            if r1 < r2 and similarity_matrix.loc[r1, r2] > 0.7:
                significant_pairs.append({
                    'resource1': r1,
                    'resource2': r2,
                    'similarity': similarity_matrix.loc[r1, r2]
                })
    return pd.DataFrame(significant_pairs)

# Analyze handover and working-together networks
def analyze_networks(event_log):
    handover_net = pm4py.discover_handover_of_work_network(event_log)
    together_net = pm4py.discover_working_together_network(event_log)

    # Convert to dataframe for analysis
    handover_df = pm4py.convert_to_dataframe(handover_net)
    together_df = pm4py.convert_to_dataframe(together_net)

    return handover_df, together_df

# Main analysis pipeline
def analyze_organizational_structure(file_path):
    # Load and format data
    df = load_and_format_event_log(file_path)
    event_log = log_converter.apply(df)

    # Discover roles
    roles = discover_organizational_roles(event_log)
    print("Discovered Roles:")
    print(roles.head())

    # Discover similarities
    similarities = discover_resource_similarity(event_log)
    print("\nSignificant Resource Similarities:")
    print(similarities)

    # Analyze networks
    handover_df, together_df = analyze_networks(event_log)
    print("\nHandover Network:")
    print(handover_df.head())
    print("\nWorking-Together Network:")
    print(together_df.head())

    # Join with substitution data
    substitution_data = df.groupby('org:resource')['temporary_substitution'].sum()
    roles_with_subs = roles.merge(
        substitution_data.rename('substitution_count'),
        left_on='org:resource',
        right_index=True,
        how='left'
    )

    return roles_with_subs, similarities, handover_df, together_df
```
