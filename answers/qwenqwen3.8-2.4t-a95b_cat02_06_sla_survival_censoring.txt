def cif_at(times, codes, horizon, event_code=1):

def km_survival_at(times, events, horizon=72.0):
    times = np.asarray(times, dtype=float)
    events = np.asarray(events, dtype=int)
    s = 1.0
    event_times = np.unique(times[(events == 1) & (times <= horizon)])
    for t in event_times:
        n = np.sum(times >= t)
        d = np.sum((times == t) & (events == 1))
        if n > 0:
            s *= (1.0 - d / n)
    return s

def km_survival_at(times, events, horizon=H):
    times = np.asarray(times, dtype=float)
    events = np.asarray(events, dtype=int)
    s = 1.0
    event_times = np.unique(times[(events == 1) & (times <= horizon)])
    for t in event_times:
        n = np.sum(times >= t)
        d = np.sum((times == t) & (events == 1))
        if n > 0:
            s *= (1.0 - d / n)
    return s

def cif_at(times, codes, horizon=H, event_code=1):
    times = np.asarray(times, dtype=float)
    codes = np.asarray(codes, dtype=int)
    cif = 0.0
    surv_all = 1.0
    event_times = np.unique(times[(codes > 0) & (times <= horizon)])
    for t in event_times:
        n = np.sum(times >= t)
        d_all = np.sum((times == t) & (codes > 0))
        d_event = np.sum((times == t) & (codes == event_code))
        if n > 0:
            cif += surv_all * (d_event / n)
            surv_all *= (1.0 - d_all / n)
    return cif

def encode_survival(df_in, transfer_mode='competing_at_transfer', other_outcomes='competing', informative_open=False):
    rows = []
    for _, r in df_in.iterrows():
        t = float(r['time'])
        gap = float(r['last_event_gap_h'])
        status = r['observed_status']
        res_time = t
        res_code = 0
        any_time = t
        any_code = 0
        if status == 'resolved':
            res_code = 1
            any_code = 1
        elif status in ('withdrawn', 'rejected'):
            res_code = 2 if other_outcomes == 'competing' else 0
            any_code = 1
        elif status == 'transferred':
            if transfer_mode == 'censor_at_transfer':
                res_code = 0
                any_code = 0
            elif transfer_mode == 'competing_at_transfer':
                res_code = 2
                any_code = 1
            elif transfer_mode == 'sla_horizon_failure':
                res_code = 2
                res_time = H if t < H else t
                any_code = 1
                any_time = t
            else:
                raise ValueError('unknown transfer_mode')
        elif status == 'open':
            res_code = 0
            any_code = 0
            if informative_open and (t >= H or gap >= 24.0):
                # Sensitivity assumption only: long-gap open cases become non-resolution competing events at observed cutoff age.
                res_code = 2
                res_time = t
                any_code = 1
                any_time = t
        else:
            res_code = 0
            any_code = 0
        rows.append({'case': r['case'], 'cohort': r['cohort'], 'priority': r['priority'],
                     'res_time': res_time, 'res_code': res_code,
                     'any_time': any_time, 'any_code': any_code,
                     'observed_status': status})
    return pd.DataFrame(rows)

def scenario_summary(df_in, transfer_mode, other_outcomes='competing', informative_open=False):
    enc = encode_survival(df_in, transfer_mode, other_outcomes, informative_open)
    cif_res = cif_at(enc['res_time'], enc['res_code'], H, 1)
    km_res_net = km_survival_at(enc['res_time'], (enc['res_code'] == 1).astype(int), H)
    prob_no_terminal = km_survival_at(enc['any_time'], enc['any_code'], H)
    return {
        'transfer_mode': transfer_mode,
        'other_outcomes': other_outcomes,
        'informative_open': informative_open,
        'n_cases': len(enc),
        'cif_resolve_72': cif_res,
        'prob_not_resolved_72': 1.0 - cif_res,
        'km_net_resolve_72': km_res_net,
        'prob_no_terminal_72': prob_no_terminal,
        'n_at_risk_72': int((enc['res_time'] >= H).sum())
    }

def observed_binary_not_resolved(df_in, transfer_mode='competing_at_transfer'):
    vals = []
    for _, r in df_in.iterrows():
        t = float(r['time'])
        status = r['observed_status']
        if status == 'resolved':
            vals.append(0.0 if t <= H else 1.0)
        elif status in ('withdrawn', 'rejected'):
            vals.append(1.0)
        elif status == 'open':
            vals.append(1.0 if t >= H else np.nan)
        elif status == 'transferred':
            if transfer_mode == 'censor_at_transfer':
                vals.append(np.nan if t < H else 1.0)
            else:
                vals.append(1.0)
        else:
            vals.append(np.nan)
    s = pd.Series(vals, index=df_in['case'])
    known = s.dropna()
    return {
        'transfer_mode': transfer_mode,
        'known_success_by_72': int((known == 0).sum()),
        'known_failure_by_72': int((known == 1).sum()),
        'unknown_at_72': int(s.isna().sum()),
        'known_breach_rate': float(known.mean()) if len(known) else np.nan,
        'lower_bound_breach_rate': float((known == 1).sum() / len(s)) if len(s) else np.nan,
        'upper_bound_breach_rate': float(((known == 1).sum() + s.isna().sum()) / len(s)) if len(s) else np.nan
    }

def stratified_summary(df_in, group_col, transfer_mode='competing_at_transfer', other_outcomes='competing', informative_open=False):
    out = []
    for g, sub in df_in.groupby(group_col):
        enc = encode_survival(sub, transfer_mode, other_outcomes, informative_open)
        cif_res = cif_at(enc['res_time'], enc['res_code'], H, 1)
        out.append({
            group_col: g,
            'n_cases': len(sub),
            'cif_resolve_72': cif_res,
            'prob_not_resolved_72': 1.0 - cif_res,
            'prob_no_terminal_72': km_survival_at(enc['any_time'], enc['any_code'], H),
            'n_at_risk_72': int((enc['res_time'] >= H).sum())
        })
    return pd.DataFrame(out)

# 1. SLA estimands and time definitions

**Time origin.** The natural time origin is the `Submit` event: `t = 0` at submission.

**Observed time scale.** For each case we observe one scalar time:

- For terminal rows, `duration_or_age_h` is the time from `Submit` to the observed terminal event.
- For `open` rows, `duration_or_age_h` is the age at the observation cutoff, i.e. an administrative right-censoring time.

Let:

- `T_R` = time to `Resolve`.
- `T_W` = time to `Withdraw`.
- `T_J` = time to `Reject`.
- `T_X` = time to `Transfer` out of internal observation.
- `C` = observation cutoff / censoring time for open cases.
- `H = 72` hours = SLA horizon.

The observed time is `T_obs = min(T_R, T_W, T_J, T_X, C)` with the observed status telling which quantity was observed.

**Important distinction.** A case can be known to have breached the 72-hour SLA even if its eventual terminal event happens after 72 hours. For example, a case resolved at 76 hours is an observed late resolution, but it was unresolved at 72 hours.

## Useful estimands

The phrase “probability of breaching the SLA” is ambiguous because the SLA can be framed as internal resolution, customer resolution, or still-open workload. At least four useful estimands are:

| Estimand | Formal definition | Operational question it answers |
|---|---:|---|
| **E1. Probability of not being resolved by 72h** | `P(T_R > 72)` or `1 - CIF_R(72)` when resolution competes with other terminal states | “What fraction of cases will miss the internal resolution SLA?” Best for SLA breach monitoring and recovery actions. |
| **E2. Cumulative incidence of resolution after 72h** | `CIF_R() - CIF_R(72)` for some observation horizon ``, e.g. cutoff or maximum observed time | “How much late resolution workload should we expect after the SLA has already been missed?” Useful for post-SLA recovery capacity. |
| **E3. Probability of still having no terminal disposition at 72h** | `P(no Resolve, Withdraw, Reject, or Transfer by 72)` | “How many cases will still be internally open at the SLA horizon?” Useful for queue/backlog staffing. |
| **E4. Cause-specific cumulative incidence by 72h** | `CIF_R(72)`, `CIF_W(72)`, `CIF_J(72)`, `CIF_X(72)` | “Why do cases fail to reach resolution by 72h?” Useful for process improvement and root-cause governance. |

For E2, because the extract is frozen, we can only estimate **restricted** late resolution up to the observation cutoff. Open cases after 72 hours may still resolve later, so E2 is a lower bound unless we make assumptions about post-cutoff behavior.

## Example contributions under each estimand

|---|---|---|---|---|---|

| **S09**: open at 92h, last gap 31h | `Submit>Screen>Review` | Known unresolved at 72h; for time-to-resolution it is right-censored at 92h but remains in the 72h risk set. | Unknown whether it will eventually resolve; contributes as censored after 72h. | Known to have no terminal disposition at 72h; censored at 92h. | No observed terminal cause; censored. Long last-event gap may suggest informative censoring. |

# 2. Event and censoring encoding

The encoding must separate:

1. **Right censoring**: the event of interest has not happened by the observed time, but it could still happen later.
2. **Competing events**: an alternative terminal outcome occurs that prevents the event of interest, at least internally.
3. **Already-observed breaches**: cases known to be unresolved at 72h because their observed age or terminal time exceeds 72h without prior resolution.

## Base encoding table

| Observed status | Time used | Resolution estimand code | Competing-risk code | Any-terminal estimand code | Censoring rule | Sensitivity alternatives |
|---|---:|---|---|---|---|---|
| `resolved` | duration to `Resolve` | Event of interest: `1` | None | Terminal event: `1` | Not censored | Resolutions after 72h remain resolution events but also count as unresolved at 72h for E1. |
| `withdrawn` | duration to `Withdraw` | Competing event: `2` under base | Competing cause: withdrawal | Terminal event: `1` | Not ordinary censoring if withdrawal precludes resolution | Could be treated as censoring only for a hypothetical “net resolution” estimand, but that answers a different question. |
| `rejected` | duration to `Reject` | Competing event: `2` under base | Competing cause: rejection | Terminal event: `1` | Not ordinary censoring if rejection precludes resolution | Same as withdrawal. |
| `open` | age at cutoff | Right-censored: `0` | None | Censored: `0` | Administrative censoring at cutoff | If long inactivity suggests dormancy, treat as informative censoring, impose bounds, or model as a separate “stuck open” state. |
| `transferred` | duration to `Transfer` | Scenario-dependent | Scenario-dependent | Scenario-dependent | Scenario-dependent | See transfer assumptions below. |

## Three plausible treatments of `Transfer`

| Transfer assumption | Coding for resolution estimand | Interpretation | Operational use | Main risk |
|---|---|---|---|---|
| **A. Transfer as internal competing terminal** | Competing event at transfer time, code `2` | The internal process has ended without internal resolution. Transfer prevents internal resolution. | Internal SLA ownership: if the organization is accountable only for internal resolution, transfer is a failure/exit. | May overstate customer-level breach if external organization resolves quickly. |
| **B. Transfer as administrative right censoring** | Censored at transfer time, code `0` | Case leaves internal observation; external resolution is unknown and assumed independent of internal future resolution. | Estimating internal resolution potential in a hypothetical world where transferred cases could still resolve. | Likely optimistic and potentially biased if transfers are selected for difficulty. |
| **C. Transfer as SLA-horizon unresolved / informative external unknown** | For SLA by 72h, count as unresolved at 72h if transfer occurred before 72h; for time-to-event, can code as competing failure at 72h or keep as informative censoring with bounds. | The case has not demonstrated resolution by the SLA horizon. External outcome is unknown, so governance treats it as unresolved unless evidence exists. | Conservative SLA governance and customer-level accountability. | May overstate breach if external resolution is fast; requires external outcome data for calibration. |

For `open` cases, the same distinction matters:

- An open case at age 47h is **not yet known** to have breached the 72h SLA.
- An open case at age 92h is **known unresolved at 72h**, even though its final outcome is still unknown.
- Long `last_event_gap_h` values, such as S11 with 55h since last event or S26 with 72h, suggest possible dormant cases and make simple administrative censoring less credible.

# 4. Statistical analysis plan

## Method comparison

| Method | What it estimates | Strengths for this problem | Limitations for this problem |
|---|---|---|---|
| **Kaplan-Meier** | Survival or event probability treating other outcomes as censored | Simple; useful for hypothetical net resolution where competing events are removed | Biased for actual cumulative incidence when withdrawal, rejection, or transfer compete with resolution |
| **Aalen-Johansen / cumulative incidence** | Actual cause-specific cumulative incidence in competing risks | Answers operational SLA question; handles competing terminal outcomes; nonparametric | Requires clear coding of competing events; still limited by small sample and missing intermediate timestamps |
| **Cause-specific hazards** | Instantaneous rate of each event among cases still at risk | Useful for understanding stage-specific dynamics; can include time-varying covariates | Hazards are not direct probabilities; requires event timestamps for transitions, which are only partially available |
| **Subdistribution hazards, e.g. Fine-Gray** | Covariate effects directly on cumulative incidence | Useful if the target is policy association with actual SLA incidence | Less intuitive for operational interventions; difficult with time-varying process covariates; small data makes stable estimation hard |
| **Restricted mean time** | Expected time unresolved or expected time to event within a horizon, e.g. 72h or 96h | Robust summary; avoids reliance on a single threshold; useful for governance | Needs careful definition of event/censoring; still affected by competing-risk assumptions |
| **Landmarking** | Conditional future risk given being open and in a given state at a landmark time | Avoids immortal time; supports case-level alerts at current age/stage | Requires enough cases at each landmark/state; intermediate event times are partly missing |
| **Multi-state model** | Transition probabilities among process states and terminal outcomes | Most process-aware; can represent review loops, escalation, rework, transfer | Data are sparse; exact transition times are not fully observed; external transfer outcome unknown |

## Recommended primary approach

**Primary: nonparametric Aalen-Johansen competing-risks analysis**, with separate cause-specific cumulative incidence functions for:

- resolution,
- withdrawal,
- rejection,
- transfer under chosen assumptions,
- and administrative censoring for open cases.

This is preferred because the operational SLA question is about actual probabilities by 72h, and withdrawal, rejection, and transfer are real alternatives to resolution. The sample is small, so a nonparametric estimator with explicit scenario analysis is more defensible than a heavily modeled regression.

The primary estimands should be:

- `CIF_R(72)`: probability of resolution by 72h.
- `1 - CIF_R(72)`: probability of not being resolved by 72h.
- `S_any_terminal(72)`: probability of still having no terminal disposition by 72h.
- Cause shares at 72h for withdrawal, rejection, and transfer.

## Recommended secondary approach

**Secondary: landmark stage-specific survival analysis** for operational alerting.

At a landmark time, such as 24h, 48h, or 72h, condition on cases that are still open and know their current prefix/state. Then estimate conditional probability of resolution by 72h or within the next `k` hours given:

- current stage,
- elapsed age,
- time since last event,
- escalation status up to landmark,
- rework/review-loop status up to landmark.

If enough events were available, a cause-specific Cox model with time-dependent covariates could be used. For these 32 cases, regression should be secondary and interpreted cautiously.

`last_event_time_h = duration_or_age_h - last_event_gap_h`.

For terminal cases, the last event is the terminal event, so `last_event_gap_h = 0`.

| `path_variant` | Full prefix string | Process pattern | Useful for descriptive variants; using full future prefix for early prediction causes leakage. |

## Immortal-time and leakage risks

`Escalate` is a major example.

If we define a baseline covariate `ever_escalated = yes`, then a case must survive from `Submit` until `Escalate` to receive that label. Cases that escalate late are guaranteed not to have failed before escalation. Treating escalation as a baseline covariate in a model from `Submit` can therefore create **immortal-time bias**.

Safe alternatives:

1. Use `escalated(t)` as a **time-dependent covariate**.
2. Use a **landmark design**: at time `L`, compare only cases still open at `L`, using escalation status observed up to `L`.
3. Use a **multi-state model** where escalation is a transition, not a baseline attribute.

Similar leakage applies to:

- `rework_count`,
- repeated `Review`,
- full prefix variants,
- terminal status,
- total duration.

These are valid for describing completed cases or for predicting from the current state at cutoff, but not as baseline predictors at `Submit`.

## Proposed state representation

A compact process-aware multi-state representation is:

`Submit --> AutoCheck | Screen`

`AutoCheck --> Screen | Escalate | Resolve`

`Screen --> AutoCheck | Review | Escalate | Reject | Withdraw | Resolve`

`Review --> Resolve | Reject | Withdraw | Rework | Escalate`

`Rework --> Review`

`Escalate --> Resolve | Review | Transfer`

`Transfer --> ExternalUnknown`

Terminal/internal absorbing states:

`Resolve`, `Withdraw`, `Reject`, and optionally `Transfer` if treated as internal terminal.

In Mermaid-like plain syntax:

`Submit --> AutoCheck; Submit --> Screen; AutoCheck --> Screen; AutoCheck --> Escalate; AutoCheck --> Resolve; Screen --> AutoCheck; Screen --> Review; Screen --> Escalate; Screen --> Reject; Screen --> Withdraw; Screen --> Resolve; Review --> Resolve; Review --> Reject; Review --> Withdraw; Review --> Rework; Review --> Escalate; Rework --> Review; Escalate --> Resolve; Escalate --> Review; Escalate --> Transfer; Transfer --> ExternalUnknown; Resolve --> absorbing; Withdraw --> absorbing; Reject --> absorbing.`

# 6. Sensitivity analysis

At least three sensitivity dimensions are essential.

## Sensitivity 1: Transfer treatment

| Analysis | Transfer coding | Expected effect |
|---|---|---|
| Base internal SLA | Transfer = competing terminal | Higher probability of not being resolved by 72h; conservative for internal accountability. |
| External unknown censoring | Transfer = right-censoring at transfer | Higher estimated resolution probability; lower apparent SLA breach; relies on independent censoring assumption. |
| SLA-horizon failure | Transfer before 72h counted as unresolved at 72h | Conservative for customer-facing SLA; keeps focus on lack of demonstrated resolution by 72h. |

This can change cohort conclusions. For example, B has transfers S21 at 26h and S32 at 96h; A has S08 at 68h. If transfers are censored, B may look better because some difficult transferred cases leave the internal risk set. If transfers are failures, B’s transfer count may increase apparent SLA failure.

## Sensitivity 2: Withdrawal and rejection as competing events versus censoring

| Analysis | Withdraw/reject coding | Interpretation |
|---|---|---|
| Competing risks base | Withdraw/reject = competing events | Actual internal probability of resolution. |
| Net resolution Kaplan-Meier | Withdraw/reject = censored | Hypothetical probability if withdrawal/rejection could not occur. Usually overstates actual resolution. |

If withdrawal and rejection are treated as censored, resolution probabilities will rise, especially in cohorts or priorities with many early withdrawals/rejections such as S05, S12, S23, S06.

## Sensitivity 3: Informative open censoring

Open cases with long inactivity gaps may not be equivalent to recently active cases.

Possible sensitivity analyses:

1. **Administrative censoring only**: all open cases are non-informatively censored at cutoff.
2. **Long-gap open cases as likely failures**: open cases with large `last_event_gap_h` are treated as competing non-resolution events at their observed cutoff age or at a policy threshold.
3. **Bounds**:
   - Best case: all unknown open-below-72 cases resolve by 72h.
   - Worst case: all unknown open-below-72 cases fail by 72h.

Cases most relevant here:

- S09: open at 92h, gap 31h.
- S11: open at 81h, gap 55h.
- S25: open at 79h, gap 48h.
- S26: open at 104h, gap 72h.
- S10: open at 47h, gap 19h.

## Sensitivity 4: Cohort policy effect adjustment

Because cohort B is not randomized and rollout compliance is incomplete:

- Compare cohorts unadjusted.
- Compare cohorts stratified by priority.
- Restrict to cases with policy-consistent prefixes, e.g. B cases with early `AutoCheck`, while acknowledging that this induces selection.
- If assignment covariates were available, use weighting or matching.

## Qualitative conclusions that could reverse

1. **Is cohort B better than A?**  
   Unadjusted B may look better due to more P1 cases or different transfer handling. After priority stratification or treating transfers as failures, the conclusion could weaken or reverse.

2. **Is escalation a risk factor for breach?**  
   Naive baseline `ever_escalated` may be biased by immortal time. Escalated cases may look artificially better or worse depending on how survival time is aligned. A time-aware landmark analysis may give a different answer.

3. **Is SLA breach driven mainly by late resolution or by unresolved open/transfer cases?**  
   If transfers and long-gap open cases are censored, late resolutions such as S03, S13, S20, S28 dominate the breach story. If transfers and open-above-72 cases are treated as failures, unresolved/transfer tail risk dominates.

# 7. Operational interpretation

Survival outputs should be translated into operational actions without claiming causal effects.

## Case-level alerts

For a case currently open at age `a` in state `s`, estimate a conditional probability such as:

`P(resolve by 72 | still open at age a, current state s)`

or equivalently:

`P(breach by 72 | still open at age a, current state s) = 1 - conditional resolution probability`.

Operational rules:

- If age is already above 72h, the case is an SLA breach and should enter recovery handling.
- If age is below 72h but conditional breach probability exceeds a threshold, trigger proactive review.
- If `last_event_gap_h` is large, trigger dormancy follow-up even if the model probability is moderate.

Examples:

- S15 is open at 35h with recent activity, gap 2h; likely lower urgency.
- S10 is open at 47h with gap 19h and has been escalated; may warrant follow-up.
- S09, S11, S25, S26 are already above 72h and require breach recovery, not prediction.

## Workload actions

From survival or multi-state outputs:

- Predict the number of cases likely to remain open at 72h.
- Predict expected late-resolution workload after 72h.
- Estimate cause-specific outflows: resolution, withdrawal, rejection, transfer.
- Allocate escalation and review capacity based on stage occupancy and expected remaining time.

## SLA governance

For governance, report:

- `CIF_R(72)` by cohort and priority.
- Probability of no terminal disposition at 72h.
- Transfer rate before 72h.
- Median or restricted mean time to resolution.
- Share of breaches due to late resolution, open aging, withdrawal, rejection, and transfer.

## Avoiding causal overclaiming

Risk scores and cohort comparisons are **associational**. They should not be presented as the causal effect of policy, escalation, or priority unless the assignment mechanism is modeled credibly.

Recommended checks:

- **Calibration**: compare predicted 72h breach probabilities with observed 72h status among cases whose 72h status is known.
- **Calibration by subgroup**: priority, cohort, escalation status, and transfer assumption.
- **Fairness checks**: ensure alert thresholds do not systematically under-serve or over-alert particular priorities or cohorts. For example, P1 cases should not receive delayed alerts simply because they are more likely to be escalated or transferred.
- **Stability checks**: repeat analysis under alternative transfer and informative-censoring assumptions.

# 3. Manual Kaplan-Meier and Aalen-Johansen CIF helpers.
#    These create KM/CIF-ready summaries without dropping open cases.

def km_survival_at(times, events, horizon=H):
    """
    Kaplan-Meier survival probability at horizon.
    events: 1 = event, 0 = censoring.
    """
    times = np.asarray(times, dtype=float)
    events = np.asarray(events, dtype=int)
    s = 1.0
    event_times = np.unique(times[(events == 1) & (times <= horizon)])
    for t in event_times:
        n = np.sum(times >= t)
        d = np.sum((times == t) & (events == 1))
        if n > 0:
            s *= (1.0 - d / n)
    return s

def cif_at(times, codes, horizon=H, event_code=1):
    """
    Aalen-Johansen cumulative incidence for event_code at horizon.
    codes: 0 = censoring, 1 = event of interest, 2+ = competing events.
    """
    times = np.asarray(times, dtype=float)
    codes = np.asarray(codes, dtype=int)
    cif = 0.0
    surv_all = 1.0
    event_times = np.unique(times[(codes > 0) & (times <= horizon)])

    for t in event_times:
        n = np.sum(times >= t)
        d_all = np.sum((times == t) & (codes > 0))
        d_event = np.sum((times == t) & (codes == event_code))
        if n > 0:
            cif += surv_all * (d_event / n)
            surv_all *= (1.0 - d_all / n)
    return cif

# ------------------------------------------------------------
# 4. Encoding for at least two estimands:
#    E1/E4: time-to-resolution with competing risks.
#    E3: time-to-any-terminal-disposition.
#    Three transfer scenarios are encoded.
# ------------------------------------------------------------

def encode_survival(
    df_in,
    transfer_mode="competing_at_transfer",
    other_outcomes="competing",
    informative_open=False
):
    """
    Returns survival-ready rows.

    transfer_mode:
      - competing_at_transfer: transfer is competing terminal event at transfer time.
      - censor_at_transfer: transfer is right-censored at transfer time.
      - sla_horizon_failure: for resolution/SLA coding, transfers before 72h are
        treated as unresolved failures recognized at 72h. This is a policy
        sensitivity assumption, not an observed event timestamp.

    other_outcomes:
      - competing: withdrawal/rejection are competing events.
      - censor: withdrawal/rejection are censored for net-resolution KM.

    informative_open:
      - If True, open cases with age >= 72 or last_event_gap_h >= 12 are treated
        as competing non-resolution events at their observed cutoff age.
        This is a sensitivity assumption only.
    """
    rows = []
    for _, r in df_in.iterrows():
        t = float(r["time"])
        gap = float(r["last_event_gap_h"])
        status = r["observed_status"]

        res_time = t
        res_code = 0
        any_time = t
        any_code = 0

        if status == "resolved":
            res_code = 1
            any_code = 1

        elif status in ("withdrawn", "rejected"):
            res_code = 2 if other_outcomes == "competing" else 0
            any_code = 1

        elif status == "transferred":
            if transfer_mode == "censor_at_transfer":
                res_code = 0
                any_code = 0
            elif transfer_mode == "competing_at_transfer":
                res_code = 2
                any_code = 1
            elif transfer_mode == "sla_horizon_failure":
                # Sensitivity assumption: unresolved at the SLA horizon.
                res_code = 2
                res_time = H if t < H else t
                # For internal workload, transfer is still observed at transfer time.
                any_code = 1
                any_time = t
            else:
                raise ValueError("Unknown transfer_mode")

        elif status == "open":
            res_code = 0
            any_code = 0

            if informative_open and (t >= H or gap >= 12.0):
                # Sensitivity assumption only: long-gap/open-old cases are assumed
                # to become non-resolution competing events at observed cutoff age.
                res_code = 2
                res_time = t
                any_code = 1
                any_time = t

        else:
            res_code = 0
            any_code = 0

        rows.append({
            "case": r["case"],
            "cohort": r["cohort"],
            "priority": r["priority"],
            "observed_status": status,
            "res_time": res_time,
            "res_code": res_code,
            "any_time": any_time,
            "any_code": any_code
        })

    return pd.DataFrame(rows)

def scenario_summary(
    df_in,
    transfer_mode,
    other_outcomes="competing",
    informative_open=False
):
    enc = encode_survival(df_in, transfer_mode, other_outcomes, informative_open)

    cif_res = cif_at(enc["res_time"], enc["res_code"], H, event_code=1)
    km_res_net = km_survival_at(enc["res_time"], (enc["res_code"] == 1).astype(int), H)
    prob_no_terminal = km_survival_at(enc["any_time"], enc["any_code"], H)

    return {
        "transfer_mode": transfer_mode,
        "other_outcomes": other_outcomes,
        "informative_open": informative_open,
        "n_cases": len(enc),
        "cif_resolve_72": cif_res,
        "prob_not_resolved_72": 1.0 - cif_res,
        "km_net_resolve_72": km_res_net,
        "prob_no_terminal_72": prob_no_terminal,
        "n_at_risk_72": int((enc["res_time"] >= H).sum())
    }

def observed_binary_not_resolved(df_in, transfer_mode="competing_at_transfer"):
    """
    Descriptive observed/unknown status at exactly 72h.
    This is not a survival estimate; it shows known successes, known failures,
    and unknown open-below-72 or externally transferred cases.
    """
    vals = []
    for _, r in df_in.iterrows():
        t = float(r["time"])
        status = r["observed_status"]

        if status == "resolved":
            vals.append(0.0 if t <= H else 1.0)
        elif status in ("withdrawn", "rejected"):
            vals.append(1.0)
        elif status == "open":
            vals.append(1.0 if t >= H else np.nan)
        elif status == "transferred":
            if transfer_mode == "censor_at_transfer":
                vals.append(np.nan if t < H else 1.0)
            else:
                vals.append(1.0)
        else:
            vals.append(np.nan)

    s = pd.Series(vals, index=df_in["case"])
    known = s.dropna()

    return {
        "transfer_mode": transfer_mode,
        "known_success_by_72": int((known == 0).sum()),
        "known_failure_by_72": int((known == 1).sum()),
        "unknown_at_72": int(s.isna().sum()),
        "known_breach_rate": float(known.mean()) if len(known) else np.nan,
        "lower_bound_breach_rate": float((known == 1).sum() / len(s)) if len(s) else np.nan,
        "upper_bound_breach_rate": float(((known == 1).sum() + s.isna().sum()) / len(s)) if len(s) else np.nan
    }

def stratified_summary(
    df_in,
    group_col,
    transfer_mode="competing_at_transfer",
    other_outcomes="competing",
    informative_open=False
):
    out = []
    for g, sub in df_in.groupby(group_col):
        enc = encode_survival(sub, transfer_mode, other_outcomes, informative_open)
        cif_res = cif_at(enc["res_time"], enc["res_code"], H, event_code=1)
        out.append({
            group_col: g,
            "n_cases": len(sub),
            "cif_resolve_72": cif_res,
            "prob_not_resolved_72": 1.0 - cif_res,
            "prob_no_terminal_72": km_survival_at(enc["any_time"], enc["any_code"], H),
            "n_at_risk_72": int((enc["res_time"] >= H).sum())
        })
    return pd.DataFrame(out)

# 9. Limitations and additional data

The most important limitations are caused by missing event-level detail and the non-random cohort rollout.

## Highest-priority missing data

1. **Exact timestamps for every activity**
   - Needed to estimate stage durations, escalation timing, review-loop timing, and time-dependent covariates without bounds or assumptions.
   - Currently only total duration and last-event gap are observed.

2. **External outcomes after transfer**
   - Needed to know whether transferred cases such as S08, S21, and S32 were eventually resolved externally, rejected externally, withdrawn, or remain open.
   - Without this, transfer must be treated as competing failure, censoring, or bounded unknown.

3. **Cutoff date and submission dates**
   - Needed to distinguish calendar effects, workload periods, staffing changes, and policy rollout phases.
   - Also needed to align observation cutoff with SLA measurement windows.

4. **Assignment mechanism for cohort B policy**
   - Needed to estimate causal policy effects.
   - Should include eligibility rules, escalation-policy parameters, and reasons for assignment.

5. **Rollout compliance details**
   - Needed to distinguish intended policy, actual treatment, and spillover.
   - Cases such as B cases without `AutoCheck` and A cases with `AutoCheck` show incomplete compliance.

6. **Time-varying workload and resource data**
   - Queue length, reviewer availability, escalation capacity, and shift patterns may explain long gaps and tail risk.

7. **External SLA ownership definition**
   - Does the 72h SLA stop at transfer, or does the organization remain accountable until external resolution?
   - This determines whether transfer is a competing failure, censoring, or an external state with follow-up.

8. **Reason codes for withdrawal, rejection, and transfer**
   - Needed to distinguish customer-driven withdrawal, eligibility rejection, and capacity-driven transfer.
   - Different reasons imply different operational actions.

With those data, the preferred extension would be a time-aware multi-state model with landmark prediction, cause-specific hazards for stage transitions, and explicit external follow-up after transfer.
