*   **Benefits:** Prevents total commercial queue stalls during specialist PTO; resolves hybrid front-office/pricing requests in a single touch without cross-departmental handover overhead.
*   **Risks:** **Efficiency and margin leakage.** During Weeks 3–4, median A7 duration nearly doubled from 38 to 74 minutes, showing that certification and informal shadowing do not equal full specialist fluency. Furthermore, Giorgio’s 20 A9 Approval events warrant scrutiny to ensure informal commercial concessions stay within authorized discount bounds.

He steps into authority/specialist gaps during crunch periods—issuing 21 provisional A9 approvals in Week 5 and filtering A6 Enterprise Security referrals in Weeks 7–8.
*   **Benefits:** Keeps multi-team workflows moving when executives are unavailable; shields scarce specialists (R05_Erik) from alert fatigue and false-positive escalations; centralizes accountability for messy reopened tickets (42 A13 events; 76 critical cases).
*   **Risks:** **False sense of governance and triage false-negatives.** In Week 5, 13 of Mateo's 21 provisional approvals had to be rechecked by Hana, creating compliance risk if provisional implementations occurred before recheck. In Weeks 7–8, Mateo's security pre-filtering cut false complaints, **but two genuinely critical cases waited >2 days**, illustrating the danger of a non-certified coordinator triaging high-risk technical security threats.

import pandas as pd
import numpy as np
import pm4py

# -------------------------------------------------------------------------
# 1. Load & Format Event Log with PM4Py
# -------------------------------------------------------------------------
def prepare_event_log(raw_df: pd.DataFrame) -> pd.DataFrame:
    """
    Formats standard PM4Py columns while preserving domain-specific case/event
    attributes (case_complexity, temporary_substitution, escalation_level, etc.).
    """
    df = pm4py.format_dataframe(
        raw_df,
        case_id="case:concept:name",
        activity_key="concept:name",
        timestamp_key="time:timestamp",
        timest_format="ISO8601"
    )
    # Ensure resource column is string-typed and clean
    df["org:resource"] = df["org:resource"].astype(str)
    # Insert case/event service and waiting times to detect substitution slowdowns
    df = pm4py.insert_case_service_waiting_time(df)
    return df


# -------------------------------------------------------------------------
# 2. Separate Normal Operations from Temporary Substitution Regimes
# -------------------------------------------------------------------------
def split_substitution_regimes(df: pd.DataFrame):
    """
    Prevents temporary coverage (e.g., Giorgio covering Fatima, Mateo covering Hana)
    from distorting baseline structural role discovery.
    """
    normal_df = pm4py.filter_event_attribute_values(
        df, attribute_key="temporary_substitution", values=[False, 0, "No"], level="event"
    )
    sub_df = pm4py.filter_event_attribute_values(
        df, attribute_key="temporary_substitution", values=[True, 1, "Yes"], level="event"
    )
    return normal_df, sub_df


# -------------------------------------------------------------------------
# 3. Discover Baseline Roles, Resource Similarity & SNA Networks
# -------------------------------------------------------------------------
def discover_organizational_signals(df_full: pd.DataFrame, df_normal: pd.DataFrame):
    # A. Baseline activity-bundle role discovery (run on non-substitution log)
    # Returns list of Role objects / tuples: (activities_list, resource_originator_counts)
    baseline_roles = pm4py.discover_organizational_roles(
        df_normal,
        resource_key="org:resource",
        activity_key="concept:name",
        timestamp_key="time:timestamp",
        case_id_key="case:concept:name"
    )

    # B. Pairwise activity-profile similarity (captures 0.94 Chen-Dina, 0.71 Fatima-Giorgio)
    act_similarity_sna = pm4py.discover_activity_based_resource_similarity(
        df_normal,
        resource_key="org:resource",
        activity_key="concept:name"
    )

    # C. Handover-of-Work Network (captures Nora -> Temps/Agents -> Tech -> QA pipelines)
    how_sna = pm4py.discover_handover_of_work_network(
        df_full,
        resource_key="org:resource",
        timestamp_key="time:timestamp",
        case_id_key="case:concept:name"
    )

    # D. Working-Together Network (identifies cross-functional brokers like R13_Mateo
    # and specialist collaborators like R03_Chen <-> R05_Erik)
    wt_sna = pm4py.discover_working_together_network(
        df_full,
        resource_key="org:resource",
        case_id_key="case:concept:name"
    )

    # E. Subcontracting / In-out Rework Network (detects R08_Hana -> R13_Mateo -> R08_Hana
    # re-checks and Temp -> Reopen -> Senior Agent rework loops)
    subcontract_sna = pm4py.discover_subcontracting_network(
        df_full,
        resource_key="org:resource",
        timestamp_key="time:timestamp",
        case_id_key="case:concept:name"
    )

    return {
        "roles": baseline_roles,
        "similarity": act_similarity_sna,
        "handover": how_sna,
        "working_together": wt_sna,
        "subcontracting": subcontract_sna,
    }


# -------------------------------------------------------------------------
# 4. Enrich Discovered Roles with Complexity, Substitution & Authority Signals
# -------------------------------------------------------------------------
def build_context_aware_resource_profile(
    df_full: pd.DataFrame, roles_output, metadata_df: pd.DataFrame
) -> pd.DataFrame:
    """
    Joins naïve PM4Py role memberships with case_complexity, substitution ratios,
    service times, and recheck rates so analysts do not confuse similar activity
    counts with equal authority or competence.
    """
    # Map each resource to all baseline PM4Py activity clusters they participate in
    role_map = {}
    for role_idx, role in enumerate(roles_output):
        # Handle PM4Py Role namedtuple/object structure (.activities, .originator_importance)
        activities = getattr(role, "activities", role[0])
        originators = getattr(role, "originator_importance", role[1])
        for res, weight in originators.items():
            role_map.setdefault(res, []).append(f"Cluster_{role_idx}({len(activities)} acts)")

    # Case-level complexity per resource (deduplicated per case-resource pair)
    case_res = df_full[["case:concept:name", "org:resource", "case_complexity", "escalation_level"]].drop_duplicates()
    complexity_stats = case_res.groupby("org:resource").agg(
        cases_touched=("case:concept:name", "nunique"),
        median_complexity=("case_complexity", "median"),
        p90_complexity=("case_complexity", lambda s: np.percentile(s, 90)),
        high_escalation_share=("escalation_level", lambda s: (s >= 2).mean())
    )

    # Event-level substitution dependency & activity entropy (identifies brokers like Mateo)
    def shannon_entropy(series):
        probs = series.value_counts(normalize=True)
        return -np.sum(probs * np.log2(probs + 1e-12))

    event_stats = df_full.groupby("org:resource").agg(
        total_events=("concept:name", "count"),
        substitution_event_share=("temporary_substitution", "mean"),
        activity_entropy=("concept:name", shannon_entropy)
    )

    # Guardrail against mistaking temporary coverage for permanent authority/fluency:
    # Compare service duration & subsequent re-execution on critical governance tasks (A6, A7, A9)
    df_sorted = df_full.sort_values(["case:concept:name", "time:timestamp"])
    df_sorted["next_same_act_resource"] = (
        df_sorted.groupby(["case:concept:name", "concept:name"])["org:resource"].shift(-1)
    )
    df_sorted["was_rechecked_by_other"] = (
        df_sorted["next_same_act_resource"].notna()
        & (df_sorted["next_same_act_resource"] != df_sorted["org:resource"])
    )

    gov_acts = df_sorted[df_sorted["concept:name"].isin([
        "Security Review", "Pricing Exception Review", "Supervisor Approval"
    ])]
    authority_audit = gov_acts.groupby(["org:resource", "concept:name"]).agg(
        gov_count=("case:concept:name", "count"),
        sub_rate=("temporary_substitution", "mean"),
        recheck_rate=("was_rechecked_by_other", "mean"),  # Flags Mateo's 13/21 A9 rechecks by Hana!
        median_service_sec=("@@service_time", "median")   # Flags Giorgio's 38m -> 74m A7 slowdown!
    ).unstack(fill_value=0)
    authority_audit.columns = [f"{col}_{act}" for col, act in authority_audit.columns]

    # Combine all views with HR formal metadata (certifications, contract, tenure)
    summary = (
        complexity_stats
        .join(event_stats, how="outer")
        .join(authority_audit, how="left")
        .merge(metadata_df, on="org:resource", how="left")
    )
    summary["pm4py_clusters"] = summary["org:resource"].map(lambda r: "; ".join(role_map.get(r, [])))

    # Flag mismatches where high activity similarity hides complexity/authority divergence
    summary["role_archetype_hint"] = np.select(
        [
            summary["contract"] == "system",
            summary["activity_entropy"] > 3.3,
            (summary["total_events"] < 250) & (summary["median_complexity"] > 0.80),
        ],
        [
            "Automated Boundary",
            "Cross-Functional Broker (Do NOT force into single cluster)",
            "Rare Critical Specialist (Protect from outlier pruning)",
        ],
        default="Check Tiered Complexity & Certifications"
    )
    return summary
