# 5. PM4Py role-discovery and similarity code

```python
import pandas as pd
import pm4py

# 1. Load and format the dataframe
log_df = pd.read_csv("order_to_resolution_log.csv")
log = pm4py.format_dataframe(
    log_df,
    case_id='case:concept:name',
    activity_key='concept:name',
    timestamp_key='time:timestamp',
    object_id=None,
    variant='pm4py'
)

# 2. Discover organizational roles (frequency-based baseline)
roles_baseline = pm4py.discover_organizational_roles(
    log,
    resources='org:resource',
    min_support=0.05,
    max_roles=12
)

# 3. Compute activity-based resource similarity
similarity_df = pm4py.discover_activity_based_resource_similarity(
    log,
    resources='org:resource',
    method='cosine'
)

# 4. Inspect handover-of-work and working-together networks
handover_net = pm4py.discover_handover_of_work_network(
    log,
    case_id='case:concept:name',
    resources='org:resource'
)
working_together_net = pm4py.discover_working_together_network(
    log,
    case_id='case:concept:name',
    resources='org:resource'
)

# 5. Join role discovery with complexity and substitution indicators
# Assume log_df contains case_complexity and temporary_substitution columns
resource_stats = log_df.groupby('org:resource').agg(
    median_complexity=('case_complexity', 'median'),
    temp_sub_count=('temporary_substitution', 'sum'),
    critical_cases=('case_complexity', lambda x: (x >= 0.8).sum())
).reset_index()

# Merge with baseline roles and similarity pairs
roles_enriched = roles_baseline.merge(resource_stats, on='org:resource', how='left')

# 6. Avoid over-interpreting frequency clusters as formal authority
# Filter roles by complexity threshold and authority markers
valid_roles = roles_enriched[
    (roles_enriched['median_complexity'] > 0.4) | 
    (roles_enriched['critical_cases'] > 10) |
    (roles_enriched['org:resource'].isin(['R08_Hana', 'R05_Erik', 'R13_Mateo']))
]

# Flag shadow/bridge roles by high similarity but divergent complexity/authority
shadow_candidates = similarity_df[
    (similarity_df['similarity'] > 0.7) & 
    (abs(similarity_df['complexity_diff']) > 0.15)
]

print("Validated roles:", valid_roles)
print("Shadow/bridge candidates:", shadow_candidates)
```
