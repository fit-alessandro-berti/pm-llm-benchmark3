```
import pandas as pd
import numpy as np
import pm4py

CANON_DIAG = "Initial Diagnosis Completed"
LABEL_MAP = {"Diagnostic Assessment Completed": CANON_DIAG}
UNKNOWN = "UNKNOWN_ACTIVITY"
EDGES = {
    "order_part_to_part_received": ("Order Part", "Part Received"),
    "diagnosis_to_remote_fix": (CANON_DIAG, "Remote Fix"),
}

def load_normalize_dedup(path):
    df = pd.read_csv(path, parse_dates=["timestamp"])
    # 1. Normalize renamed diagnosis before any feature extraction.
    df["activity"] = df["activity"].replace(LABEL_MAP)
    df["unknown_activity"] = df["activity"].isna() | df["activity"].astype(str).str.contains(UNKNOWN, na=False)

    # 2. Duplicate strategy: drop exact duplicates (case, activity, timestamp), keep first.
    dup_cols = [c for c in ["case_id", "activity", "timestamp"] if c in df.columns]
    df["is_exact_duplicate"] = df.duplicated(subset=dup_cols, keep="first")
    dup_removed_by_month = (
        df[df["is_exact_duplicate"]]
        .groupby(df["timestamp"].dt.to_period("M"))
        .size()
    )
    clean = df[~df["is_exact_duplicate"]].copy()

    # 3. PM4Py format.
    log = pm4py.format_dataframe(
        clean[["case_id", "activity", "timestamp"]],
        case_id="case_id",
        activity_id="activity",
        timestamp="timestamp",
    )
    return df, clean, log, dup_removed_by_month

def case_start_cohort(log, start, end):
    # Cases whose first event occurs in the month. This is the predictive cohort.
    if log.empty:
        return log.copy()
    first_ts = log.groupby("case:case_id")["time:timestamp"].min()
    cases = first_ts[(first_ts >= start) & (first_ts < end)].index
    return log[log["case:case_id"].isin(cases)].copy()

def event_calendar_cohort(log, start, end):
    # Events occurring in the month. Useful for data-quality and activity volume, not case cohorts.
    if log.empty:
        return log.copy()
    return pm4py.filter_time_range(
        log,
        "time:timestamp",
        start.isoformat(),
        end.isoformat(),
        mode="events",
        timeframe="month",
    )

def edge_median(perf_result, edge):
    perf = perf_result[0] if isinstance(perf_result, tuple) else perf_result
    if perf is None:
        return np.nan
    if isinstance(perf, dict):
        val = perf.get(edge)
        if val is None:
            return np.nan
        return float(val[-1]) if isinstance(val, tuple) else float(val)
    try:
        mask = (perf["from_activity"] == edge[0]) & (perf["to_activity"] == edge[1])
        if mask.any():
            return float(perf.loc[mask, "median"].iloc[0])
    except Exception:
        pass
    return np.nan

def classify_variant(variant):
    seq = variant if isinstance(variant, tuple) else tuple(variant)
    if "Remote Fix" in seq:
        return "V2"
    if "Order Part" in seq or "Part Received" in seq:
        return "V3"
    if "Repair Completed" in seq:
        return "V1"
    return "unmapped/other"

def monthly_process_evidence(path="field_service_events.csv", months=None):
    raw, clean, log, dup_removed_by_month = load_normalize_dedup(path)
    if months is None:
        months = clean["timestamp"].dt.to_period("M").unique().sort_values()
        months = [str(m.to_timestamp()) for m in months]

    rows = []
    for month in months:
        start = pd.Timestamp(month)
        end = start + pd.DateOffset(months=1)

        audit_month = clean[(clean["timestamp"] >= start) & (clean["timestamp"] < end)]
        event_log = event_calendar_cohort(log, start, end)
        start_log = case_start_cohort(log, start, end)

        variants = pm4py.get_variants(start_log) if not start_log.empty else {}
        total = sum(variants.values())
        shares = {"V1": 0.0, "V2": 0.0, "V3": 0.0, "unmapped/other": 0.0}
        for v, c in variants.items():
            shares[classify_variant(v)] += c
        if total:
            shares = {k: v / total for k, v in shares.items()}

        if not start_log.empty:
            dfg_result = pm4py.discover_dfg(start_log)
            perf_result = pm4py.discover_performance_dfg(start_log)
            temporal = pm4py.extract_temporal_features_dataframe(
                start_log,
                time_unit="days",
                case_id="case:case_id",
                timestamp="time:timestamp",
                activity="concept:name",
            )
            dfg_edges = len(dfg_result[0]) if isinstance(dfg_result, tuple) else len(dfg_result)
        else:
            perf_result = temporal = None
            dfg_edges = 0

        missing_queue = (
            audit_month["queue_feature"].isna().mean()
            if "queue_feature" in audit_month.columns
            else np.nan
        )

        rows.append({
            "month": month,
            "events_in_month": len(event_log),
            "cases_started": start_log["case:case_id"].nunique() if not start_log.empty else 0,
            "unknown_activity_share": audit_month["unknown_activity"].mean() if len(audit_month) else np.nan,
            "exact_duplicates_removed": int(dup_removed_by_month.get(pd.Period(month), 0)),
            "missing_queue_share": missing_queue,
            "V1_share": shares["V1"],
            "V2_share": shares["V2"],
            "V3_share": shares["V3"],
            "unmapped_share": shares["unmapped/other"],
            "dfg_edges": dfg_edges,
            "order_part_to_part_received_median": edge_median(perf_result, EDGES["order_part_to_part_received"]),
            "diagnosis_to_remote_fix_median": edge_median(perf_result, EDGES["diagnosis_to_remote_fix"]),
            "temporal_feature_rows": len(temporal) if temporal is not None else 0,
        })
    return pd.DataFrame(rows)
```

| **Process-variant monitor** | V1/V2/V3 shares, unmapped share, source-specific variant distributions. | Alert on large mix shifts, e.g., >10pp change in V2/V3. | Distinguishes real routing/parts mix from pipeline-induced unmapped variants. |
| **Performance-DFG monitor** | Median duration on key edges, especially `Order Part -> Part Received` and diagnosis-to-remote-fix. | Alert if part-receipt edge exceeds baseline multiple, e.g., >2× Jan–Mar. | Triggers operational escalation and feature update, not necessarily retraining. |
| **Population monitor** | Partner Z share, rural/device-age covariate distributions, PSI/KS, reweighted expected breach. | Alert when new source exceeds threshold or covariate distribution shifts. | Activates segment-specific calibration and transport checks. |
| **Intervention/deployment monitor** | Record patches, routing policy changes, supplier incidents, Partner Z onboarding, outcome API backlog. | Alert when a known intervention overlaps a metric change. | Provides attribution and prevents false drift alarms. |

### Alert dependency logic

1. **If label maturity is low**, predictive and calibration alerts are marked provisional and do **not** automatically trigger retraining.
2. **If data-quality alerts fire**, the system should quarantine affected cohorts and launch reprocessing before any model action.
3. **If performance-DFG alerts fire but data quality and maturity are good**, escalate operationally and consider feature updates; retraining follows only if conditional-path tests show changed risk.
4. **If population alerts fire**, activate source/variant stratification and reweighting before global recalibration.
5. **If an intervention monitor matches the month**, attribute the change first to the intervention; require post-intervention fixed-vintage validation before retraining.

### Monthly process-evidence extraction sketch

```python
import pandas as pd
import numpy as np
import pm4py

CANON_DIAG = "Initial Diagnosis Completed"
LABEL_MAP = {"Diagnostic Assessment Completed": CANON_DIAG}
UNKNOWN = "UNKNOWN_ACTIVITY"

EDGES = {
    "order_part_to_part_received": ("Order Part", "Part Received"),
    "diagnosis_to_remote_fix": (CANON_DIAG, "Remote Fix"),
}

def load_normalize_dedup(path):
    df = pd.read_csv(path, parse_dates=["timestamp"])

    # Normalize renamed diagnosis label before any feature extraction.
    df["activity"] = df["activity"].replace(LABEL_MAP)
    df["unknown_activity"] = (
        df["activity"].isna()
        | df["activity"].astype(str).str.contains(UNKNOWN, na=False)
    )

    # Duplicate-handling strategy: drop exact duplicates (case, activity, timestamp).
    # Keep first occurrence; report removed duplicates by month for auditability.
    dup_cols = ["case_id", "activity", "timestamp"]
    df["is_exact_duplicate"] = df.duplicated(subset=dup_cols, keep="first")
    dup_removed_by_month = (
        df[df["is_exact_duplicate"]]
        .groupby(df["timestamp"].dt.to_period("M"))
        .size()
    )
    clean = df[~df["is_exact_duplicate"]].copy()

    log = pm4py.format_dataframe(
        clean[["case_id", "activity", "timestamp"]],
        case_id="case_id",
        activity_id="activity",
        timestamp="timestamp",
    )
    return clean, log, dup_removed_by_month

def case_start_cohort(log, start, end):
    # Case-start cohort: cases whose first event occurs in the month.
    # This is the appropriate cohort for predictive monitoring.
    if log.empty:
        return log.copy()
    first_ts = log.groupby("case:case_id")["time:timestamp"].min()
    cases = first_ts[(first_ts >= start) & (first_ts < end)].index
    return log[log["case:case_id"].isin(cases)].copy()

def event_calendar_cohort(log, start, end):
    # Event-calendar cohort: events occurring in the month.
    # Useful for data-quality and activity-volume monitoring, not case cohorts.
    if log.empty:
        return log.copy()
    return pm4py.filter_time_range(
        log,
        "time:timestamp",
        start.isoformat(),
        end.isoformat(),
        mode="events",
        timeframe="month",
    )

def edge_median_hours(log, edge):
    # Explicit median duration for a target edge, independent of PM4Py version differences.
    if log.empty:
        return np.nan
    d = log.copy().sort_values(["case:case_id", "time:timestamp"])
    d["next_activity"] = d.groupby("case:case_id")["concept:name"].shift(-1)
    d["next_ts"] = d.groupby("case:case_id")["time:timestamp"].shift(-1)
    mask = (d["concept:name"] == edge[0]) & (d["next_activity"] == edge[1])
    if not mask.any():
        return np.nan
    return float(
        (d.loc[mask, "next_ts"] - d.loc[mask, "time:timestamp"])
        .dt.total_seconds()
        .median()
        / 3600.0
    )

def classify_variant(variant):
    seq = variant if isinstance(variant, tuple) else tuple(variant)
    if "Remote Fix" in seq:
        return "V2"
    if "Order Part" in seq or "Part Received" in seq:
        return "V3"
    if "Repair Completed" in seq:
        return "V1"
    return "unmapped/other"

def monthly_process_evidence(path="field_service_events.csv", months=None):
    clean, log, dup_removed_by_month = load_normalize_dedup(path)

    if months is None:
        months = clean["timestamp"].dt.to_period("M").unique().sort_values()
        months = [str(m.to_timestamp()) for m in months]

    rows = []
    for month in months:
        start = pd.Timestamp(month)
        end = start + pd.DateOffset(months=1)

        audit_month = clean[(clean["timestamp"] >= start) & (clean["timestamp"] < end)]
        event_log = event_calendar_cohort(log, start, end)
        start_log = case_start_cohort(log, start, end)

        variants = pm4py.get_variants(start_log) if not start_log.empty else {}
        total = sum(variants.values())
        shares = {"V1": 0.0, "V2": 0.0, "V3": 0.0, "unmapped/other": 0.0}
        for v, c in variants.items():
            shares[classify_variant(v)] += c
        if total:
            shares = {k: v / total for k, v in shares.items()}

        if not start_log.empty:
            dfg_result = pm4py.discover_dfg(start_log)
            perf_result = pm4py.discover_performance_dfg(start_log)
            temporal = pm4py.extract_temporal_features_dataframe(
                start_log,
                time_unit="days",
                case_id="case:case_id",
                timestamp="time:timestamp",
                activity="concept:name",
            )
            dfg_edges = len(dfg_result[0]) if isinstance(dfg_result, tuple) else len(dfg_result)
            perf_edges = len(perf_result[0]) if isinstance(perf_result, tuple) else len(perf_result)
        else:
            temporal = None
            dfg_edges = 0
            perf_edges = 0

        missing_queue = (
            audit_month["queue_feature"].isna().mean()
            if "queue_feature" in audit_month.columns
            else np.nan
        )

        rows.append({
            "month": month,
            "events_in_month": len(event_log),
            "cases_started": start_log["case:case_id"].nunique() if not start_log.empty else 0,
            "unknown_activity_share": audit_month["unknown_activity"].mean() if len(audit_month) else np.nan,
            "exact_duplicates_removed": int(dup_removed_by_month.get(start.to_period("M"), 0)),
            "missing_queue_share": missing_queue,
            "V1_share": shares["V1"],
            "V2_share": shares["V2"],
            "V3_share": shares["V3"],
            "unmapped_share": shares["unmapped/other"],
            "dfg_edges": dfg_edges,
            "performance_dfg_edges": perf_edges,
            "order_part_to_part_received_median_hours": edge_median_hours(
                start_log, EDGES["order_part_to_part_received"]
            ),
            "diagnosis_to_remote_fix_median_hours": edge_median_hours(
                start_log, EDGES["diagnosis_to_remote_fix"]
            ),
            "temporal_feature_rows": len(temporal) if temporal is not None else 0,
        })

    return pd.DataFrame(rows)
```

| **Retraining on immature October labels** | October has only 55% maturity, median label delay 52d, and an outcome API backlog that disproportionately affects longer paths. Observed breach rate is 0.16, much lower than August/September. | Model learns that recent high-risk V3/part-delay cases are low risk, underpredicting future breaches. | Use fixed-vintage matured cohorts only.
