# 2. Assignment policy

The policy uses a weighted scoring algorithm to rank eligible resources for each case, respecting hard constraints and optimizing soft constraints.

```text
FUNCTION AssignResources(queue, resource_pool):
    sort queue by (Priority DESC, SLA_time_remaining ASC, Complexity DESC)
    
    FOR EACH case IN queue:
        eligible_resources = []
        
        // 1. Eligibility Filtering (Hard Constraints)
        FOR EACH resource IN resource_pool:
            IF resource.new_assignments >= 4: CONTINUE
            IF case.skill == "Security" AND resource.skill < 1: CONTINUE
            IF case.skill == "Security" AND resource.skill < 3 AND resource NOT IN [R7, R8]: 
                IF supervision_by_R7_or_R8_possible: 
                    add resource to eligible_resources WITH supervisor
                ELSE: CONTINUE
            IF case.skill == "Approval" AND case.next_activity == "Final Approval" AND resource.skill < 2: CONTINUE
            IF case.priority IN ["urgent", "high"] AND resource.skill < 2 AND NOT supervised: CONTINUE
            IF resource.overtime_risk == "high" AND case.priority != "urgent": CONTINUE
            IF resource.hours_remaining_today <= 0: CONTINUE
            
            ADD resource TO eligible_resources
            
        // 2. Scoring (Soft Constraints)
        FOR EACH resource IN eligible_resources:
            S_SLA = 1 / (case.SLA_time_remaining + 1)  // Urgency match
            S_Skill = resource.skill_level / 3
            
            IF case.current_owner == resource:
                S_Continuity = 1 - (case.handover_cost_band / 3)  // High handover cost = high penalty for change
            ELSE:
                S_Continuity = 0
                
            S_Fairness = (resource.hours_remaining_today / 8) - (resource.open_cases / 15)
            
            S_Training = 0
            IF case.priority IN ["normal", "low"] AND resource.training_priority == case.skill:
                S_Training = 1
                
            // Resilience / Bottleneck avoidance
            S_Resilience = 1 - (queue_length_for_skill / 10)
            
            // Total Score
            IF case.priority == "urgent":
                Total_Score = (S_SLA * 3) + (S_Skill * 2) + S_Continuity
            ELSE IF case.priority == "high":
                Total_Score = (S_SLA * 2) + S_Skill + S_Continuity + S_Fairness
            ELSE:
                Total_Score = S_Skill + S_Fairness + S_Training + S_Continuity
                
            resource.score = Total_Score
            
        // 3. Tie-breaking & Exception Handling
        IF case is eligible for batching (Quality Check, normal/low, >24h):
            attempt to pair with resource already assigned a Quality Check in this batch
            
        ASSIGN case to highest scoring eligible resource
        INCREMENT resource.new_assignments
        DECREMENT resource.hours_remaining_today by estimated_service_time
    END FOR
```
