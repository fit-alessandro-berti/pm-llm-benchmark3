```
def fire_transition(net, marking, trans):
    enabled = pm4py.get_enabled_transitions(net, marking)
    if trans not in enabled:
        raise ValueError("transition not enabled")
    new_mark = copy.copy(marking)
    for arc in trans.in_arcs:
        pl = arc.source
        if new_mark.get(pl, 0) < 1:
            raise ValueError(...)
        new_mark[pl] = new_mark[pl] - 1
        if new_mark[pl] == 0:
            del new_mark[pl]
    for arc in trans.out_arcs:
        pl = arc.target
        new_mark[pl] = new_mark.get(pl, 0) + 1
    return new_mark
```

```
def _fire(net, marking, trans):
    enabled = pm4py.get_enabled_transitions(net, marking)
    if trans not in enabled:
        raise ValueError("transition not enabled")
    new_marking = copy.copy(marking)
    for arc in trans.in_arcs:
        place = arc.source
        if new_marking.get(place, 0) < 1:
            raise ValueError("insufficient tokens")
        new_marking[place] -= 1
        if new_marking[place] == 0:
            del new_marking[place]
    for arc in trans.out_arcs:
        place = arc.target
        new_marking[place] = new_marking.get(place, 0) + 1
    return new_marking
```

```
def replay_witness(net, im, fm, witness, max_silent_steps=MAX_SILENT_STEPS):
    if net is None or im is None or fm is None:
        return False, "null net/marking"
    final_key = _marking_key(fm)
    visited = set()
    stack = [(copy.copy(im), 0, 0)]
    while stack:
        marking, idx, silent_run = stack.pop()
        state = (_marking_key(marking), idx, silent_run)
        if state in visited:
            continue
        visited.add(state)
        if idx == len(witness):
            if _marking_key(marking) == final_key:
                return True, "exact final marking reached"
            if silent_run < max_silent_steps:
                for t in pm4py.get_enabled_transitions(net, marking):
                    if t.label is None:
                        try:
                            stack.append((_fire(net, marking, t), idx, silent_run + 1))

                new_mark = _fire(net, marking, t)
            except Exception:
                continue

                stack.append((new_mark, idx + 1, 0))
            elif t.label is None:
                stack.append((new_mark, idx, silent_run + 1))
```

```python
import copy
import logging
import sys
from pathlib import Path

import pm4py
from pm4py.objects.powl.obj import (
    StrictPartialOrder,
    OperatorPOWL,
    Transition,
    SilentTransition,
)
from pm4py.objects.process_tree.obj import Operator

logging.basicConfig(level=logging.INFO, stream=sys.stdout)
LOG = logging.getLogger("powl_conversion_audit")

# ---------------------------------------------------------------------------
# Source POWL
# ---------------------------------------------------------------------------
register = Transition(label="Register Change")
assess_risk = Transition(label="Assess Risk")
prepare_evidence = Transition(label="Prepare Evidence")

legal_review = Transition(label="Legal Review")
skip_legal = SilentTransition()
optional_legal = OperatorPOWL(
    operator=Operator.XOR,
    children=[legal_review, skip_legal],
)

configure_trial = Transition(label="Configure Trial")
inspect_trial = Transition(label="Inspect Trial")
trial_body = StrictPartialOrder(nodes=[configure_trial, inspect_trial])
trial_body.order.add_edge(configure_trial, inspect_trial)

correct_defect = Transition(label="Correct Defect")
verification_loop = OperatorPOWL(
    operator=Operator.LOOP,
    children=[trial_body, correct_defect],
)

release_change = Transition(label="Release Change")
archive_change = Transition(label="Archive Change")

source_powl = StrictPartialOrder(
    nodes=[
        register,
        assess_risk,
        prepare_evidence,
        optional_legal,
        verification_loop,
        release_change,
        archive_change,
    ]
)

source_powl.order.add_edge(register, assess_risk)
source_powl.order.add_edge(register, prepare_evidence)
source_powl.order.add_edge(assess_risk, optional_legal)
source_powl.order.add_edge(assess_risk, verification_loop)
source_powl.order.add_edge(prepare_evidence, verification_loop)
source_powl.order.add_edge(optional_legal, release_change)
source_powl.order.add_edge(verification_loop, release_change)
source_powl.order.add_edge(release_change, archive_change)

# ---------------------------------------------------------------------------
# Output helpers
# ---------------------------------------------------------------------------
OUT = Path("conversion_output")
OUT.mkdir(exist_ok=True)

def _require_accepting_net(name, net, im, fm):
    if net is None:
        raise ValueError(f"{name}: converted net is None")
    if im is None:
        raise ValueError(f"{name}: initial marking is None")
    if fm is None:
        raise ValueError(f"{name}: final marking is None")
    return net, im, fm

def _marking_key(marking):
    """
    Stable key for a marking. Uses object id first so distinct places with the
    same name cannot be accidentally merged.
    """
    return tuple(sorted((id(place), place.name, count) for place, count in marking.items()))

def _diagnose_accepting_net(name, net, im, fm):
    _require_accepting_net(name, net, im, fm)
    print(f"\n=== diagnostics: {name} ===")

    workflow_net = "not tested"
    sound = "not tested"
    detail = ""

    try:
        workflow_net = bool(pm4py.check_is_workflow_net(net))
    except Exception as exc:
        detail += f" workflow-net-check-error={type(exc).__name__}:{exc}"

    try:
        sound_result = pm4py.check_soundness(
            net, im, fm, print_diagnostics=True
        )
        if isinstance(sound_result, tuple):
            sound = bool(sound_result[0])
            detail += f" tuple={sound_result}"
        else:
            sound = bool(sound_result)
            detail += f" result={sound_result}"
    except Exception as exc:
        sound = False
        detail += f" soundness-error={type(exc).__name__}:{exc}"

    print(f"  workflow_net={workflow_net}")
    print(f"  sound={sound}")
    if detail:
        print(f"  detail={detail.strip()}")

    return {
        "name": name,
        "workflow_net": workflow_net,
        "sound": sound,
        "detail": detail,
    }

def _save_petri_net(name, net, im, fm):
    _require_accepting_net(name, net, im, fm)
    path = str(OUT / f"{name}.png")
    pm4py.save_vis_petri_net(net, im, fm, path)
    print(f"[saved] {path}")

# ---------------------------------------------------------------------------
# 1. POWL -> accepting Petri net
# ---------------------------------------------------------------------------
net_from_powl, im_from_powl, fm_from_powl = pm4py.convert_to_petri_net(source_powl)
_require_accepting_net("net_from_powl", net_from_powl, im_from_powl, fm_from_powl)

diagnostics = {}
diagnostics["net_from_powl"] = _diagnose_accepting_net(
    "net_from_powl", net_from_powl, im_from_powl, fm_from_powl
)

pm4py.save_vis_powl(source_powl, str(OUT / "source_powl.png"))
_save_petri_net("net_from_powl", net_from_powl, im_from_powl, fm_from_powl)

# ---------------------------------------------------------------------------
# 2. Petri net -> BPMN
# ---------------------------------------------------------------------------
bpmn_from_net = pm4py.convert_to_bpmn(net_from_powl, im_from_powl, fm_from_powl)
```

# 4. Preservation and Loss Audit

| Source invariant | POWLPetri | PetriBPMN | PetriTree | BPMN round trip | Evidence or uncertainty |
|---|---|---|---|---|---|
| S1 Register first | expected preserved | expected preserved | conversion failed | expected preserved | Direct source edge registerassess and registerprepare |

| S7 Loop repeats via Correct Defect | expected preserved | ambiguous | conversion failed | expected preserved | `Operator.LOOP` children are `[trial_body, correct_defect]` |
| S8 At least one verification iteration | expected preserved | ambiguous | conversion failed | expected preserved | LOOP do-first semantics; body is first child |
| S9 Release waits for optional-legal and verification | expected preserved | ambiguous | conversion failed | expected preserved | Edges optional_legalrelease and verificationrelease |
| S10 Archive after Release | expected preserved | expected preserved | conversion failed | expected preserved | Direct edge releasearchive |
| S11 Finality | expected preserved | expected preserved | conversion failed | expected preserved | Source final marking after archive; exact final-marking equality required |
| S12 Silent skip legal hidden | expected preserved | ambiguous | conversion failed | expected preserved | `SilentTransition.label` is `None`; visible projection removes it |

For this source, **PetriTree** is marked `conversion failed` because the process-tree intermediate is not a trustworthy derivative unless the converter both succeeds and passes semantic checks. The source has a partial-order structure with overlapping joins and a loop, which is frequently outside block-structured process-tree languages. If `convert_to_process_tree` unexpectedly returns a tree, it must be treated as `ambiguous` until bounded language and witness tests are run.

Additional points:

- **True concurrency versus arbitrary interleaving:** POWL `StrictPartialOrder` states absence of order. Petri net semantics interleaves concurrent transitions. Trace-based validation can prove that both orders are present, but it cannot prove true concurrency.
- **Lack of edge between Optional Legal Review and verification:** This means neither waits for the other. A Petri net preserves this as independent enabling; a process tree may introduce sequence/choice if block structuring is forced.
- **Silent optional behavior:** `skip_legal` is invisible. In a visible trace, skipping legal review and never executing legal review look identical. State-level checks are needed to know the optional branch completed before Release.
- **Binary-loop semantics:** The loop is a redo loop, not a while loop with an explicit exit test. Correct Defect is the redo transition. A BPMN loop symbol may obscure the position of Correct Defect.
- **Join before Release Change:** Release has two incoming source edges: one from optional-legal and one from verification. This is a synchronizing join. If BPMN export renders this as an exclusive gateway or merges the two incoming branches ambiguously, the join semantics may be misread.
- **Activity-label uniqueness:** In this source all visible labels are unique. That makes witness replay easier. If duplicate labels existed, visible-trace comparison would need transition identity or label disambiguation.
- **Initial and final markings:** The derived accepting Petri net must have an initial marking corresponding to source start and final marking after Archive Change. We should never assume converters return non-null markings.
- **Gateway or routing-node ambiguity in BPMN visualization:** A place with multiple outgoing arcs can become an exclusive gateway; a transition with multiple outgoing places can become a parallel split; a transition with multiple incoming places can become a parallel join. PM4Py’s BPMN export may choose routing nodes heuristically, so gateway type is a visualization artifact, not a proof of semantics.

```python
import copy

# Reuse _marking_key and _require_accepting_net from Section 3 where possible.
# Redefine them here for standalone use.

MAX_VISIBLE_LEN = 15
MAX_SILENT_STEPS = 25
MAX_PATH_STEPS = 250

def _marking_key(marking):
    return tuple(sorted((id(place), place.name, count) for place, count in marking.items()))

def _fire_transition(net, marking, transition):
    enabled = set(pm4py.get_enabled_transitions(net, marking))
    if transition not in enabled:
        raise ValueError("transition is not enabled")

    new_marking = copy.copy(marking)
    for arc in transition.in_arcs:
        place = arc.source
        if new_marking.get(place, 0) < 1:
            raise ValueError("insufficient tokens for transition")
        new_marking[place] -= 1
        if new_marking[place] == 0:
            del new_marking[place]

    for arc in transition.out_arcs:
        place = arc.target
        new_marking[place] = new_marking.get(place, 0) + 1

    return new_marking

def bounded_visible_traces(net, im, fm,
                           max_visible_len=MAX_VISIBLE_LEN,
                           max_silent_steps=MAX_SILENT_STEPS,
                           max_path_steps=MAX_PATH_STEPS):
    """
    Exhaustive under the visited-state abstraction for visible traces up to
    max_visible_len. Hides transitions whose label is None. Requires exact
    final-marking equality for a completed trace.
    """
    if net is None or im is None or fm is None:
        return set()

    final_key = _marking_key(fm)
    results = set()
    visited = set()

    # stack items: (marking, visible_tuple, current_contiguous_silent_run, path_steps)
    stack = [(copy.copy(im), (), 0, 0)]

    while stack:
        marking, visible_tuple, silent_run, path_steps = stack.pop()

        if path_steps > max_path_steps:
            continue

        mkey = _marking_key(marking)
        state = (mkey, visible_tuple, silent_run)
        if state in visited:
            continue
        visited.add(state)

        if mkey == final_key:
            results.add(visible_tuple)
            continue

        if len(visible_tuple) >= max_visible_len:
            continue

        enabled = list(pm4py.get_enabled_transitions(net, marking))
        for transition in enabled:
            try:
                next_marking = _fire_transition(net, marking, transition)
            except Exception:
                continue

            if transition.label is None:
                if silent_run >= max_silent_steps:
                    continue
                stack.append(
                    (next_marking, visible_tuple, silent_run + 1, path_steps + 1)
                )
            else:
                stack.append(
                    (
                        next_marking,
                        visible_tuple + (str(transition.label),),
                        0,
                        path_steps + 1,
                    )
                )

    return results

def replay_witness(net, im, fm, witness,
                   max_silent_steps=MAX_SILENT_STEPS):
    """
    DFS replay of a visible trace. Silent transitions may be fired between
    visible steps, up to a bounded per-path silent run. Checks exact final
    marking after the visible trace and optional trailing silent steps.
    """
    if net is None or im is None or fm is None:
        return False, "null net/marking"

    final_key = _marking_key(fm)
    visited = set()
    stack = [(copy.copy(im), 0, 0)]  # marking, witness_index, silent_run

    while stack:
        marking, idx, silent_run = stack.pop()
        state = (_marking_key(marking), idx, silent_run)
        if state in visited:
            continue
        visited.add(state)

        if idx == len(witness):
            if _marking_key(marking) == final_key:
                return True, "exact final marking reached"

            # Allow trailing silent transitions after the last visible step.
            if silent_run < max_silent_steps:
                for transition in pm4py.get_enabled_transitions(net, marking):
                    if transition.label is None:
                        try:
                            stack.append(
                                (_fire_transition(net, marking, transition), idx, silent_run + 1)
                            )
                        except Exception:
                            pass
            continue

        if silent_run >= max_silent_steps:
            continue

        for transition in pm4py.get_enabled_transitions(net, marking):
            try:
                next_marking = _fire_transition(net, marking, transition)
            except Exception:
                continue

            if transition.label == witness[idx]:
                stack.append((next_marking, idx + 1, 0))
            elif transition.label is None:
                stack.append((next_marking, idx, silent_run + 1))

    return False, "no replay path found"
```

# 6. Visual-Semantics Review

Even if the converted BPMN model is behaviorally acceptable, the rendering can still be misread.

- **Parallel gateways:** PM4Py may insert a parallel gateway for a transition with multiple outgoing places. That can correctly suggest concurrency, but BPMN execution still interleaves. Conversely, a place with multiple outgoing transitions may be an exclusive choice. Viewers should not assume every diamond represents the same semantics.
- **Exclusive gateways:** The optional legal review is an XOR with a silent branch. In BPMN this may appear as an exclusive gateway with an unlabeled or invisible outgoing edge. A reader may miss the optionality or may think the legal branch is mandatory.
- **Loop visualization:** The verification loop may be shown as an unstructured cycle, a loop marker on a task, or a loop-back edge. The exact semantics that the body is `Configure Trial  Inspect Trial` and the redo transition is `Correct Defect` may not be obvious.
- **Silent routing nodes:** Silent transitions can disappear completely in BPMN visualization. This is acceptable for execution but risky for review. The skip-legal branch and loop-exit tau are routing artifacts that should be documented explicitly.
- **Apparent ordering of graph layout:** PM4Py’s visualizer uses a layered layout. Two nodes may appear near each other or vertically aligned without an arrow. Stakeholders may infer an order that does not exist. Only directed edges define order.

Recommended annotations:

1. Add a note on the optional legal gateway: *“Unlabeled outgoing edge = skip legal review (tau).”*
2. Add a note on the release join: *“Release Change waits for both optional-legal completion and verification loop completion.”*
3. Add a note on the verification loop: *“Do-loop: at least one iteration. Repeat only through Correct Defect.”*
4. Add a note that background `tau` nodes are silent routing nodes, not business activities.
5. Include the invariant table as documentation next to the BPMN diagram.
6. Keep the Petri net and POWL source as the authoritative semantics; treat BPMN as a stakeholder projection.

# 7. Rejection Criteria

A conversion result should be rejected or quarantined if any of the following occur:

1. **Null or missing artifact:** `net`, `im`, or `fm` is `None` after conversion.
2. **Unsoundness:** `pm4py.check_soundness(...)` fails or reports unsound behavior.
3. **Not a workflow net:** `pm4py.check_is_workflow_net(...)` returns false.
4. **Finality violation:** A positive witness ending at Archive Change does not reach exact final-marking equality.
5. **Release-before-verification admitted:** The negative trace with Release before Configure/Inspect is replayable.
6. **Zero-verification admitted:** The negative trace without Configure/Inspect and without a verification body is replayable.
7. **Required interleaving lost:** One of the valid Assess/Prepare or Legal/verification interleavings is missing from the bounded visible language.
8. **Optional behavior collapsed:** Legal Review becomes mandatory or impossible, or the silent skip is no longer available.
9. **Loop separator lost:** A repeated Configure/Inspect sequence is admitted without an intervening Correct Defect.
10. **Archive ordering violation:** Archive-before-Release or missing Archive is admitted.
11. **Process-tree silent mismatch:** The process-tree intermediate succeeds but its bounded visible language or witness results disagree with the source net.
12. **Runaway silent behavior:** Bounded exploration exceeds silent-step bounds without visible progress, suggesting a silent cycle or divergence.

# 8. Canonical Representation Recommendation

- **Authoritative storage:** Store the **source POWL model** and its semantic invariant table as the authoritative representation. The POWL source is concise, editable, and closest to the intended design. No generated BPMN, process tree, or Petri net should be edited as the source of truth.
- **Canonical operational derivative:** Also store the validated `net_from_powl` accepting Petri net as the canonical operational representation for simulation, soundness checking, and reachability analysis.
- **Stakeholder visualization:** Use the BPMN generated from `net_from_powl` as a derivative visualization, annotated with the tau skip, loop scope, and release join semantics. It is for review, not for authoritative storage.
- **Other generated artifacts:** Process tree, round-tripped Petri nets, reachability graphs, footprints, and saved images are derivatives. They must be regenerated after any source change.
- **Validation evidence required for each derivative:**
  - workflow-net check;
  - soundness check;
  - bounded visible-language comparison against `net_from_powl`;
  - replay of positive witnesses, including skip-legal if tau-aware replay is used;
  - replay of negative witnesses, especially release-before-verification and zero-verification;
  - exact final-marking test after Archive Change.
- **Process-tree intermediate:** Allowed as an experiment only. It is not authoritative. If conversion fails, that is acceptable and often expected for this source. If it succeeds, it must pass the same validation as the BPMN derivative before use.
- **Future edits:** Always edit the source POWL or its serialized source file. Regenerate `net_from_powl`, BPMN, and any process-tree artifacts through the audited conversion pipeline. Never hand-edit BPMN or Petri-net output as a way to change semantics. Store source, validation scripts, and conversion logs in version control to avoid round-trip drift.

# 9. Residual Uncertainty

Even if all tests pass, the following claims cannot be fully established by this validation:

1. **True concurrency is not proven.** Trace-based bounded exploration can show that both interleavings are possible, but it cannot prove that two activities truly occur concurrently in the underlying execution semantics.
2. **Full language equivalence is not proven.** Bounded visible-trace exploration is finite and loop iterations are capped. It cannot rule out a mismatch that appears only after more loop iterations or after a deeper silent sequence.
3. **PM4Py converter-specific semantics are not fully proven.** The black-box tests validate outputs, but they do not prove that POWL operator semantics, Petri-net conversion rules, and BPMN gateway mapping are interpreted identically across PM4Py versions.
4. **Branching and silent-step equivalence is incomplete.** Even if visible traces match, the internal branching structure, silent-step distribution, and choice points may differ without being detected by the selected tests.
5. **Stakeholder interpretation cannot be verified.** Validating the converted net does not guarantee that humans will read the BPMN gateway layout without inferring false ordering or incorrect loop semantics.
