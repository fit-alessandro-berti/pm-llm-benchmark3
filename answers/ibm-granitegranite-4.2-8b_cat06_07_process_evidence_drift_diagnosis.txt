            case_part = part_edges[part_edges['case_id'] == case_id]
            if len(case_part) >= 2:
                order_idx = case_part.index[case_part['event_name'] == 'Order Part'][0]
                recv_idx = case_part.index[case_part['event_name'] == 'Part Received'][0]
                if order_idx < recv_idx:
                    order_time = case_part.loc[order_idx, 'event_date']
                    recv_time = case_part.loc[recv_idx, 'event_date']
                    part_times.append((recv_time - order_time).total_seconds() / 3600)  # hours
        median_part_edge = pd.to_timedelta(max(part_times) if part_times else pd.NaT).total_seconds()/3600 if part_times else None  # simplified; use np.median
        # Proper: np.median(part_times) if part_times else np.nan
        
        # Label maturity: for case-start cohort, % with breach label matured by cutoff
        # Assume breach label in 'breach' column, 'breach_date' when known
        matured = df_case_start[df_case_start['breach_date'] <= month_end].nunique() / df_case_start['case_id'].nunique() if df_case_start['case_id'].nunique() > 0 else 0
        
        summary = {
            'month': month,
            'cases_started': df_case_start['case_id'].nunique(),
            'maturity_pct': matured * 100,
            'median_part_edge_h': median_part_edge,
            'variant_props': {'V1': v1_prop, 'V2': v2_prop, 'V3': v3_prop},
            'dfg_edges': list(dfg.edges()),
            'perf_dfg_edges': list(perf_dfg.edges()) if perf_dfg else None
        }
        summaries.append(summary)
    
    return pd.DataFrame(summaries)
```

Note: Simplified variant counting; in practice use `pm4py.variants.get_variants` to get frequency dict. Code avoids inventing drift detector; uses existing PM4Py functions.

  calibration_monitor     --> triggers variant-specific recalibrator re-fit if ECE >0.07 for >2 months; if ECE <0.03, no action

  performance_dfg_monitor --> alerts if median part-edge >90h for V3 for >2 months; checks DFG edge breach %; no retrain
  population_monitor     --> runs CDF shift test weekly; alerts if >10% quantile shift for any used feature; no retrain
  outcome_api_backlog_monitor --> tracks pending breach labels; alerts if >500 pending; no retrain
  intervention_monitor   --> logs policy changes (e.g., routing policy), supplier delays; feeds into process_variant_monitor
  ```
  **Dependency rules**:  
  - `pipeline_quality_monitor`  immediate recompute action (no alert chain).  
  - `label_maturity_monitor`  *only* sends Slack/email to data team ("Label maturity <70% in Oct; backlog may affect observed breach"); **does not** trigger retraining or calibrator change.  
  - `calibration_monitor`  triggers recalibrator re-fit (fast, <1hr), but **only if** `performance_dfg_monitor` shows edge-breach signal (e.g., >70h part-edge V3 breach % >0.40) *and* maturity 90% for test set. Prevents retraining on label delay alone.

- **Python code sketch** (monthly summary extraction; no drift detector; uses PM4Py functions only):

  from pm4py.filtering import filter_time_range
  from pm4py.discover import discover_dfg, discover_performance_dfg
  from pm4py.variants import get_variants
  from pm4py.extractors import extract_temporal_features_dataframe
  import numpy as np

  def monthly_audit(csv_path, base_year=2026):
      # Read with dates
      df = pd.read_csv(csv_path, parse_dates=['event_date', 'breach_date'] if 'breach_date' in df.columns else None)
      
      # Normalize diagnosis labels: both old/new to 'diagnostic_assessment_completed'
      def norm_activity(row):
          if row['activity'] in ['Initial Diagnosis Completed', 'Diagnostic Assessment Completed']:
              return 'diagnostic_assessment_completed'
          return row['activity']  # keep others as-is
      
      df['normalized_activity'] = df['activity'].apply(norm_activity)
      
      # Remove duplicates per case after diagnosis: keep first occurrence of each activity post-diagnosis
      df = df.sort_values(['case_id', 'event_date'])
      # For each case, mark events after diagnosis; drop duplicates of normalized_activity after first diagnosis
      df['is_post_diag'] = df.groupby('case_id')['normalized_activity'].transform(lambda x: (x == 'diagnostic_assessment_completed').cumsum() > 0)
      df = df[df['is_post_diag'] == False] | df.drop_duplicates(subset=['case_id', 'normalized_activity'], keep='first')
      
      # Monthly loop
      summaries = []
      for month in range(1, 13):  # 1=Jan, ..., 10=Oct
          month_start = pd.Timestamp(f'{base_year}-01-01') + pd.Timedelta(days=30*(month-1))
          month_end = month_start + pd.Timedelta(days=31)
        
          # Events in month
          df_month = filter_time_range(df, month_start, month_end)
          
          # Case-start cohort: cases created in this month
          cases_created = df_month[df_month['event_name'] == 'Case Created']['case_id'].unique()
          df_case_start = df[df['case_id'].isin(cases_created)]
          
          # Matured label %: cases with breach_date  month_end
          matured_cases = df_case_start[df_case_start['breach_date'] <= month_end]['case_id'].nunique()
          total_case_start = df_case_start['case_id'].nunique()
          maturity_pct = matured_cases / total_case_start * 100 if total_case_start > 0 else 0
          
          # Variant proportions (simple: count cases with V1/V2/V3 path; V1=has Dispatch, no Part; V2=has Remote Fix, no Part; V3=has Order Part)
          # More robust: use get_variants frequency
          variants_df = df_month[df_month['normalized_activity'] == 'diagnostic_assessment_completed']  # filter to cases with diagnosis
          if variants_df.empty:
              v1_v2_v3 = {'V1': 0, 'V2': 0, 'V3': 0}
          else:
              var_freq = get_variants(variants_df)  # returns dict with 'frequency'
              freq = var_freq.get('frequency', {}) if isinstance(var_freq, dict) else {}
              # Estimate variant per case: V1=no Part/Remote, V2=Remote, V3=Part
              v1_cases = variants_df[~variants_df['event_name'].isin(['Order Part', 'Part Received', 'Remote Fix'])]['case_id'].nunique()
              v2_cases = variants_df[variants_df['event_name'] == 'Remote Fix']['case_id'].nunique()
              v3_cases = variants_df[variants_df['event_name'] == 'Order Part']['case_id'].nunique()
              total_cases = variants_df['case_id'].nunique()
              v1_v2_v3 = {'V1': v1_cases / total_cases if total_cases > 0 else 0,
                          'V2': v2_cases / total_cases if total_cases > 0 else 0,
                          'V3': v3_cases / total_cases if total_cases > 0 else 0}
          
          # Process evidence: median part-edge time (Order Part  Part Received) for V3 cases with matured label
          part_edges = df_month[df_month['event_name'].isin(['Order Part', 'Part Received'])]
          part_times = []
          for case_id in part_edges['case_id'].unique():
              case_part = part_edges[part_edges['case_id'] == case_id]
              if len(case_part) >= 2:
                  # Find first Order Part and first Part Received in order
                  order_idx = case_part.index[case_part['event_name'] == 'Order Part'].min() if (case_part['event_name'] == 'Order Part').any() else None
                  recv_idx = case_part.index[case_part['event_name'] == 'Part Received'].min() if (case_part['event_name'] == 'Part Received').any() else None
                  if pd.notna(order_idx) and pd.notna(recv_idx) and order_idx < recv_idx:
                      order_time = case_part.loc[order_idx, 'event_date']
                      recv_time = case_part.loc[recv_idx, 'event_date']
                      part_hours = (recv_time - order_time).total_seconds() / 3600
                      part_times.append(part_hours)
          median_part_edge = np.median(part_times) if part_times else np.nan
          
          # Discover DFG and performance DFG
          try:
              dfg = discover_dfg(df_month)
              perf_dfg = discover_performance_dfg(df_month)
          except Exception:
              dfg = None
              perf_dfg = None
          
          # Build summary (no model fitting)
          summary = {
              'month': month,
              'cases_started': int(total_case_start),
              'maturity_pct': round(maturity_pct, 1),
              'median_part_edge_h': round(median_part_edge, 1) if not np.isnan(median_part_edge) else None,
              'variant_props': {k: round(v*100, 1) for k, v in v1_v2_v3.items()},  # %
              'dfg_edges': list(dfg.edges()) if dfg else [],
              'perf_dfg_edges': list(perf_dfg.edges()) if perf_dfg else [],
              'unknown_activity_share': round(df_month[df_month['normalized_activity'] == 'UNKNOWN_ACTIVITY'].shape[0] / len(df_month) * 100, 1) if 'UNKNOWN_ACTIVITY' in df_month['activity'].values else 0,
              'missing_queue_pct': round(df_month[df_month['queue_feature_missing']].shape[0] / len(df_month) * 100, 1) if 'queue_feature_missing' in df_month.columns else 0
          }
          summaries.append(summary)
      
      return pd.DataFrame(summaries)
