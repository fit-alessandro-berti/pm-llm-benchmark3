import pandas as pd
import numpy as np
import pm4py

# Config
EVENT_FILE = "applications_events.csv"
CASE_ID = "case:concept:name"
ACTIVITY = "concept:name"
TIMESTAMP = "time:timestamp"
AS_OF = pd.Timestamp("2026-02-15", tz="UTC")
HORIZON_DAYS = 90
MAX_FEED_DELAY_DAYS = 35
PREFIX_EVENT_COUNT = 3
ELAPSED_PREFIX_HOURS = 72

# Canonical activity mapping for migration rename
ACTIVITY_MAP = {
    "Identity Check": "Identity/KYC Check",
    "KYC Check": "Identity/KYC Check",
    "Credit Review (old)": "Credit Review",
    "Request Docs": "Request Documents",
    "Doc Request": "Request Documents",
    "Accepted": "Accepted",
    "Rejected": "Rejected",
    "Withdrawn": "Withdrawn",
    "Transferred": "Transferred",
    "Contract Signed": "Contract Signed",
    "Account Activated": "Account Activated",
    "Collections Contact": "Collections Contact",
    "Default Recorded": "Default Recorded",
}
TERMINAL_COMPETING = {"Rejected", "Withdrawn", "Transferred"}
FINAL_DECISION = {"Accepted", "Rejected", "Withdrawn", "Transferred"}
POST_PREFIX_PROHIBITED = [
    "final_limit", "decision_reason", "collection_status", "default_90d",
    "label_available_at", "documents_total", "data_extract_time"
]

def load_events():
    df = pd.read_csv(
        EVENT_FILE,
        parse_dates=[TIMESTAMP, "label_available_at", "data_extract_time"],
    )
    # As-of discipline: no event or label known after extraction date.
    df = df[df[TIMESTAMP] <= AS_OF]
    df[ACTIVITY] = df[ACTIVITY].replace(ACTIVITY_MAP)
    # Migration duplicates: conservative dedupe. Assumes exact duplicates are spurious.
    df = df.drop_duplicates(subset=[CASE_ID, ACTIVITY, TIMESTAMP, "org:resource"], keep="first")
    df = df.sort_values([CASE_ID, TIMESTAMP])
    df = pm4py.format_dataframe(df, case_id=CASE_ID, activity_key=ACTIVITY, timestamp_key=TIMESTAMP)
    return df

def build_case_table(df):
    cases = df.groupby(CASE_ID)
    ct = pd.DataFrame(index=cases.size().index)
    ct.index.name = CASE_ID
    ct["start_time"] = cases[TIMESTAMP].min()
    ct["last_event_time"] = cases[TIMESTAMP].max()

    act = df[df[ACTIVITY] == "Account Activated"].groupby(CASE_ID)[TIMESTAMP].min()
    ct["activation_time"] = act

    review = df[df[ACTIVITY] == "Credit Review"].groupby(CASE_ID)[TIMESTAMP].min()
    ct["first_credit_review_time"] = review

    terminal = df[df[ACTIVITY].isin(TERMINAL_COMPETING)].sort_values(TIMESTAMP)
    ct["terminal_time"] = terminal.groupby(CASE_ID)[TIMESTAMP].min()
    ct["terminal_activity"] = terminal.groupby(CASE_ID)[ACTIVITY].first()

    default_event = df[df[ACTIVITY] == "Default Recorded"].groupby(CASE_ID)[TIMESTAMP].min()
    ct["default_event_time"] = default_event

    # Outcome feed: use only labels whose availability time is on/before AS_OF.
    label = df[df["default_90d"].notna() & df["label_available_at"].notna() & (df["label_available_at"] <= AS_OF)]
    label = label.sort_values("label_available_at")
    ct["default_90d_label"] = label.groupby(CASE_ID)["default_90d"].last()
    ct["label_available_at"] = label.groupby(CASE_ID)["label_available_at"].max()

    ct["activated"] = ct["activation_time"].notna()
    ct["horizon_end"] = ct["activation_time"] + pd.Timedelta(days=HORIZON_DAYS)
    ct["feed_due"] = ct["horizon_end"] + pd.Timedelta(days=MAX_FEED_DELAY_DAYS)

    ct["default_within_90d_event"] = (
        ct["default_event_time"].notna()
        & ct["activation_time"].notna()
        & (ct["default_event_time"] <= ct["horizon_end"])
        & (ct["default_event_time"] <= AS_OF)
    )

    # Primary multi-state cohort state as of AS_OF.
    # Assumption: transferred cases are unlinked in primary analysis; linkage table would be joined later.
    def state(r):
        if r.activated and (r.default_within_90d_event or r.default_90d_label == 1):
            return "activated_default"
        if r.activated and (r.default_90d_label == 0) and (r.feed_due <= AS_OF):
            return "activated_no_default_matured"
        if r.activated and (r.feed_due > AS_OF or pd.isna(r.feed_due)):
            return "activated_unmatured_or_delayed_censored"
        if r.activated and pd.isna(r.default_90d_label) and (r.feed_due <= AS_OF):
            return "activated_feed_missing_censored"
        if r.terminal_activity == "Rejected":
            return "rejected_competing"
        if r.terminal_activity == "Withdrawn":
            return "withdrawn_competing"
        if r.terminal_activity == "Transferred":
            return "transferred_competing_or_missing"
        return "open_censored"

    ct["outcome_state"] = ct.apply(state, axis=1)
    ct["policy_period"] = np.where(ct["start_time"] >= pd.Timestamp("2025-07-01", tz="UTC"), "post_policy", "pre_policy")
    ct["migration_period"] = np.where(ct["start_time"] >= pd.Timestamp("2025-10-15", tz="UTC"), "post_migration", "pre_migration")
    return ct

def fixed_prefix_log(df, prefix_len=PREFIX_EVENT_COUNT):
    df = df.copy()
    df["event_rank"] = df.groupby(CASE_ID).cumcount() + 1
    # Stop before final decision for pre-decision early warning. If decision occurs before prefix_len, prefix ends before it.
    decision_mask = df[ACTIVITY].isin(FINAL_DECISION)
    first_decision = df[decision_mask].groupby(CASE_ID)[TIMESTAMP].min()
    prefix = df[df["event_rank"] <= prefix_len].copy()
    if not first_decision.empty:
        prefix = prefix.merge(first_decision.rename("first_decision_time"), on=CASE_ID, how="left")
        prefix = prefix[prefix[TIMESTAMP] < prefix["first_decision_time"] | prefix["first_decision_time"].isna()]
        prefix = prefix.drop(columns=["first_decision_time"])
    prefix_end = prefix.groupby(CASE_ID)[TIMESTAMP].max().rename("prefix_end_time")
    prefix = prefix.merge(prefix_end, on=CASE_ID, how="left")
    return prefix

def elapsed_prefix_log(df, hours=ELAPSED_PREFIX_HOURS):
    start = df.groupby(CASE_ID)[TIMESTAMP].min().rename("start_time")
    df = df.merge(start, on=CASE_ID, how="left")
    cutoff = df["start_time"] + pd.Timedelta(hours=hours)
    decision_mask = df[ACTIVITY].isin(FINAL_DECISION)
    first_decision = df[decision_mask].groupby(CASE_ID)[TIMESTAMP].min().rename("first_decision_time")
    df = df.merge(first_decision, on=CASE_ID, how="left")
    effective_cutoff = pd.concat([cutoff, df["first_decision_time"]], axis=1).min(axis=1)
    prefix = df[df[TIMESTAMP] <= effective_cutoff].copy()
    prefix["prefix_end_time"] = prefix.groupby(CASE_ID)[TIMESTAMP].max()
    return prefix.drop(columns=["start_time", "first_decision_time"])

def make_prefix_features(prefix_log):
    # Only prefix events are present. Prohibited post-prefix columns are dropped.
    feats = prefix_log.groupby(CASE_ID).agg(
        prefix_events=(ACTIVITY, "count"),
        prefix_start=(TIMESTAMP, "min"),
        prefix_end=(TIMESTAMP, "max"),
        n_credit_review=(ACTIVITY, lambda x: (x == "Credit Review").sum()),
        n_document_requests=(ACTIVITY, lambda x: (x == "Request Documents").sum()),
        n_resources=("org:resource", "nunique"),
        requested_amount=("case:requested_amount", "first"),
        channel=("case:channel", "first"),
        region=("case:region", "first"),
    )
    feats["prefix_duration_hours"] = (feats["prefix_end"] - feats["prefix_start"]).dt.total_seconds() / 3600.0
    # Last risk_score before prefix end; assumes risk_score is carried on events.
    last_risk = prefix_log.sort_values(TIMESTAMP).groupby(CASE_ID)["risk_score"].last()
    feats["risk_score_last_prefix"] = last_risk
    feats = feats.drop(columns=[c for c in POST_PREFIX_PROHIBITED if c in feats.columns], errors="ignore")
    return feats

def leakage_audit(prefix_log, feature_table):
    report = {}
    report["prohibited_columns_present"] = [c for c in POST_PREFIX_PROHIBITED if c in feature_table.columns]
    report["post_prefix_events"] = int((prefix_log[TIMESTAMP] > prefix_log["prefix_end_time"]).sum())
    report["final_limit_nonnull_in_prefix"] = int(prefix_log["final_limit"].notna().sum()) if "final_limit" in prefix_log else 0
    report["decision_reason_nonnull_in_prefix"] = int(prefix_log["decision_reason"].notna().sum()) if "decision_reason" in prefix_log else 0
    report["collection_status_nonnull_in_prefix"] = int(prefix_log["collection_status"].notna().sum()) if "collection_status" in prefix_log else 0
    report["default_label_in_features"] = "default_90d" in feature_table.columns
    return report

def temporal_split(case_table):
    # Primary split by case start time. Random pm4py.split_train_test is only a convenience baseline.
    train = case_table[case_table["start_time"] < pd.Timestamp("2025-04-01", tz="UTC")]
    valid = case_table[
        (case_table["start_time"] >= pd.Timestamp("2025-04-01", tz="UTC"))
        & (case_table["start_time"] < pd.Timestamp("2025-07-01", tz="UTC"))
    ]
    test_policy = case_table[
        (case_table["start_time"] >= pd.Timestamp("2025-07-01", tz="UTC"))
        & (case_table["start_time"] < pd.Timestamp("2025-10-15", tz="UTC"))
    ]
    test_migration = case_table[case_table["start_time"] >= pd.Timestamp("2025-10-15", tz="UTC")]
    return {"train": train, "valid": valid, "test_policy": test_policy, "test_migration": test_migration}

def main():

    case_table = build_case_table(df)

        _ = pm4py.get_prefixes_from_log(df, prefix_length=PREFIX_EVENT_COUNT)

        # API differences should not change the deterministic manual prefix logic above.
        pass

    audit = leakage_audit(prefix_log, feature_table)

import pandas as pd
import numpy as np
import pm4py

CASE_ID = "case:concept:name"
ACTIVITY = "concept:name"

def load_events(path="applications_events.csv"):
    df = pd.read_csv(path, parse_dates=[TS, "label_available_at", "data_extract_time"])
    df = df[df[TS] <= AS_OF]  # as-of discipline

def case_table(df):

    ct = pd.DataFrame(index=g.size().index)
    ct.index.name = CASE_ID
    ct["start_time"] = g[TS].min()

    ct["terminal_activity"] = term.groupby(CASE_ID)[ACTIVITY].first()

    ct["default_event_time"] = default
    lab = df[df["default_90d"].notna() & df["label_available_at"].notna() & (df["label_available_at"] <= AS_OF)]
    lab = lab.sort_values("label_available_at")
    ct["default_90d_label"] = lab.groupby(CASE_ID)["default_90d"].last()
    ct["label_available_at"] = lab.groupby(CASE_ID)["label_available_at"].max()
    ct["activated"] = ct["activation_time"].notna()

                                      (ct["default_event_time"] <= AS_OF))
    def state(r):
        if r.activated and (r.default_within_90d_event or r.default_90d_label == 1):
            return "activated_default"
        if r.activated and (r.default_90d_label == 0) and (r.feed_due <= AS_OF):
            return "activated_no_default_matured"
        if r.activated and ((r.feed_due > AS_OF) or pd.isna(r.feed_due)):
            return "activated_unmatured_or_delayed_censored"
        if r.activated and pd.isna(r.default_90d_label) and (r.feed_due <= AS_OF):
            return "activated_feed_missing_censored"

def fixed_prefix_log(df, k=PREFIX_K):
    d = df.copy()
    d["event_rank"] = d.groupby(CASE_ID).cumcount() + 1

    d["prefix_end_time"] = d.groupby(CASE_ID)[TS].transform("max")
    return d.drop(columns=["first_decision"])

def elapsed_prefix_log(df, hours=ELAPSED_HOURS):
    d = df.copy()

    d = d[d[TS] <= cutoff]
    d["prefix_end_time"] = d.groupby(CASE_ID)[TS].transform("max")
    return d.drop(columns=["start_time", "first_decision"])

        prefix_events=(ACTIVITY, "count"),
        prefix_start=(TS, "min"),
        prefix_end=(TS, "max"),
        n_credit_review=(ACTIVITY, lambda x: (x == "Credit Review").sum()),
        n_doc_requests=(ACTIVITY, lambda x: (x == "Request Documents").sum()),
        n_resources=("org:resource", "nunique"),
        requested_amount=("case:requested_amount", "first"),
        channel=("case:channel", "first"),
        region=("case:region", "first"),
    )
    f["prefix_duration_hours"] = (f["prefix_end"] - f["prefix_start"]).dt.total_seconds() / 3600.0
    f["risk_score_last_prefix"] = prefix.sort_values(TS).groupby(CASE_ID)["risk_score"].last()
    f = f.drop(columns=[c for c in PROHIBITED if c in f.columns], errors="ignore")
    return f

def leakage_audit(prefix, feats):
    return {
        "prohibited_columns_present": [c for c in PROHIBITED if c in feats.columns],
        "post_prefix_events": int((prefix[TS] > prefix["prefix_end_time"]).sum()),
        "final_limit_nonnull_in_prefix": int(prefix["final_limit"].notna().sum()) if "final_limit" in prefix else 0,
        "decision_reason_nonnull_in_prefix": int(prefix["decision_reason"].notna().sum()) if "decision_reason" in prefix else 0,
        "collection_status_nonnull_in_prefix": int(prefix["collection_status"].notna().sum()) if "collection_status" in prefix else 0,
        "default_90d_in_features": "default_90d" in feats.columns,
    }

def temporal_split(ct):
    # Primary temporal split. pm4py.split_train_test may be used only as a random convenience baseline.
    train = ct[ct["start_time"] < pd.Timestamp("2025-04-01", tz="UTC")]
    valid = ct[(ct["start_time"] >= pd.Timestamp("2025-04-01", tz="UTC")) &
               (ct["start_time"] < pd.Timestamp("2025-07-01", tz="UTC"))]
    test_policy = ct[(ct["start_time"] >= pd.Timestamp("2025-07-01", tz="UTC")) &
                     (ct["start_time"] < pd.Timestamp("2025-10-15", tz="UTC"))]
    test_migration = ct[ct["start_time"] >= pd.Timestamp("2025-10-15", tz="UTC")]
    return {"train": train, "valid": valid, "test_policy": test_policy, "test_migration": test_migration}

def main():
    df = load_events()
    ct = case_table(df)
    prefix = fixed_prefix_log(df, PREFIX_K)
    # PM4Py prefix utility demonstration; manual rank/decision-stop is the audited path.
    try:
        _ = pm4py.get_prefixes_from_log(df, prefix_length=PREFIX_K)
    except Exception:
        pass

    splits = temporal_split(ct)
    # Process audit on prefix-only log; full-log DFG would leak future structure.
    dfg, starts, ends = pm4py.discover_dfg(prefix)
    variants = pm4py.get_variants(prefix)
    print("Cohort counts:")

    print("Split counts:", {k: len(v) for k, v in splits.items()})
    print("Leakage audit:", audit)
    print("Prefix variants:", len(variants))
    # Later: join ct + feats and fit landmark competing-risk/survival model with censoring weights.
    return ct, feats, splits, audit

if __name__ == "__main__":
    main()

## 1. Target definition

**Primary target: current-policy “activated-default” cumulative incidence for all applications.**  
For each application, predict the probability that the case will **(i)** eventually reach `Account Activated` under the observed historical process and **(ii)** default within **90 days after activation**. This is formulated as a **multi-state / competing-outcome target**, not as a binary “default / non-default” label over all applications.

- **Unit of analysis:** one credit application (`case:concept:name`).
- **Prediction time:** early in the process, before the final credit decision. Operationally, use the end of the first `Credit Review` event, or equivalently the first `k = 3` canonical events (`Application Started`, `Identity/KYC Check`, first `Credit Review`), stopping before any final-decision activity (`Accepted`, `Rejected`, `Withdrawn`, `Transferred`). If a case terminates before reaching that point, the prefix is the last observed pre-terminal event and the case is treated as a competing outcome, not as a non-default.
- **Horizon:** default within **90 days after `Account Activated`**. Because activation happens after the prefix, the overall observation horizon is bounded by the extraction date and label-maturity rules.
- **Population:** all applications started between **1 Jan 2024** and **31 Dec 2025**, including rejected, withdrawn, transferred, open, activated-matured, and activated-unmatured cases.
- **Estimand:** for prefix information \(X_i(t_0)\),

\[
P\big(T^{\text{act}}_i < \infty,\;
      T^{\text{def}}_i \le T^{\text{act}}_i + 90\text{ days}
      \;\big|\; X_i(t_0), \text{observed policy/calendar}\big)
\]

where rejection, withdrawal, and unlinked transfer before activation are **competing events**, and open or immature cases are **censored**. This is an observational, current-policy estimand. It does **not** estimate what would happen if all rejected or withdrawn applicants were approved.

**Why this primary target is defensible.**  
It respects the fact that `default_90d` is only defined for activated accounts. It avoids the false convenience of labeling rejected, withdrawn, transferred, or open applications as “non-default.” It also matches the operational question “which applications, under today’s process, are likely to become activated accounts that default soon after activation?” rather than silently changing the question to “what would happen if we approved everyone?”

**Credible alternative target 1: conditional default among eventually activated accounts.**  
This target is

\[
P(T^{\text{def}} \le T^{\text{act}} + 90 \mid T^{\text{act}} < \infty, X(t_0)).
\]

It answers a different question: among accounts that eventually become activated, how risky are they? It is useful for portfolio monitoring after approval, but it conditions on a post-prefix selection event (`Account Activated`) and therefore cannot be interpreted as early-warning risk for every application without an additional activation model or strong selection corrections.

**Credible alternative target 2: policy-specific counterfactual default if accepted.**  
One could define “default within 90 days if the application were accepted and activated.” This is a causal/counterfactual estimand. It requires strong assumptions: no unmeasured confounding of acceptance and default, positivity, consistent activation policy, and often external data to model rejected applicants’ latent credit performance. It should be labeled as a decision-support counterfactual, not as a directly observable historical target.

## 2. Cohort construction

### Inclusion rules

Include every application that satisfies all of the following:

1. Has a valid `case:concept:name`.
2. Has an `Application Started` event between **2024-01-01** and **2025-12-31**.
3. Has at least one event with a valid `time:timestamp`.
4. Is observable under the as-of extraction date **2026-02-15**.

### Exclusion and repair rules

Do **not** exclude rejected, withdrawn, transferred, or open cases merely because they lack `default_90d`. Instead, assign them explicit censoring or competing-outcome states.

Exclude or repair only technical data problems:

- Drop events with timestamps after the as-of date.
- Remove duplicate events created around the 15 Oct 2025 migration using a conservative deduplication rule, for example same case, activity, timestamp, and resource.
- Canonicalize renamed activities using a migration mapping table.
- Exclude case-level records with impossible ordering, such as activation before application start, after audit.

### As-of and label-maturity rules

Use **2026-02-15** as the analysis cutoff.

A matured activated account is one where the 90-day post-activation outcome is known at the cutoff. Because the external outcome feed can arrive 7–35 days after the 90-day horizon, a conservative maturity rule is:

\[
\text{activation time} + 90\text{ days} + 35\text{ days} \le \text{2026-02-15}.
\]

If `label_available_at` exists, prefer it:

- Label is usable only if `label_available_at <= 2026-02-15`.
- If the horizon has matured but no reliable feed exists, treat the case as delayed/censored or feed-missing, not as non-default.

| Rejected applications | 9,100 | Competing terminal event; no credit exposure; not non-default |
| Withdrawn applications | 5,700 | Competing terminal event; not non-default |
| Transferred applications | 2,900 | Competing event or missing follow-up. Approximately 60% linkable, i.e. ~1,740 potentially linkable and ~1,160 unlinked, depending on linkage table quality |
| Still-open applications at extraction | 5,100 | Administratively right-censored before terminal decision/activation |
| **Total** | **48,000** | Matches started applications |

The current labeled ML export of **21,600** cases is therefore a selected subset: only activated accounts whose outcomes matured. It is not the full population relevant to the business question.

## 3. Censoring and missing-label strategy

The study must distinguish at least five missingness/censoring mechanisms.

### 1. Ordinary right censoring

Cases that are still open at extraction, or activated accounts whose 90-day window has not finished, are right-censored. They are not failures and not successes. They contribute follow-up until the censoring time.

Examples:

- Application still in `Credit Review` on 2026-02-15.
- Account activated on 2025-12-20; 90-day horizon has not matured by extraction.

### 2. Delayed labels

The 90-day outcome may be operationally complete but the external feed may arrive 7–35 days later. If the feed has not arrived by the as-of date, the label is delayed.

Primary rule:

- Treat delayed-label cases as censored at the as-of date.
- Do not impute them as non-default.

Sensitivity rule:

- If a trusted internal `Default Recorded` event exists before the as-of date, use it in a sensitivity analysis as an event-time marker, while acknowledging that the warehouse label may not yet be finalized.

### 3. Selection through activation

Default is only defined after activation. Therefore, non-activation is not “no default”; it is either a competing terminal state or censoring, depending on the case path.

Primary rule:

- Model activation as an intermediate state in a multi-state process.
- Rejection, withdrawal, and unlinked transfer before activation are competing events.

This avoids conditioning only on activated accounts unless that is explicitly stated as a secondary estimand.

### 4. Transfer as competing event or missing follow-up

Transferred cases are ambiguous:

- Some reappear under a new identifier on another platform.
- Only about 60% have linkage.
- Partner-originated activated accounts have elevated permanently missing outcome risk.

Primary rule:

- Treat unlinked transfers as a competing terminal state: `transferred_competing_or_missing`.

Sensitivity rules:

- Include linked transfers if the linkage is as-of and outcome maturity is known.
- Treat transfer as censoring in an alternative analysis, acknowledging informative censoring.
- Report bounds or worst-case imputation for unlinked transfers.

### 5. Permanently missing outcome feeds

Some activated accounts, especially partner-originated ones, may never receive an outcome feed. These are not automatically non-default.

Primary rule:

- If feed is permanently missing or cannot be verified, censor the case and track feed-missingness as a covariate/subgroup.

Sensitivity rules:

- Inverse-probability-of-censoring weighting using predictors of feed missingness.
- Best-case/worst-case bounds.
- Separate calibration for partner and non-partner channels.

### Primary analysis strategy

Use a **landmark competing-risk framework** at the early prefix time.

States:

1. `pre_decision`
2. `activated_no_default`
3. `activated_default`
4. `rejected`
5. `withdrawn`
6. `transferred_competing_or_missing`
7. `open_censored`
8. `activated_unmatured_or_delayed_censored`

The primary model estimates the cumulative incidence of `activated_default` from the prefix, with non-activation terminal states as competing events and immature/open cases as censored.

This can be implemented later with cause-specific hazard models, Fine-Gray subdistribution models, discrete-time survival models, or gradient-boosted multi-state survival models. PM4Py is used for process extraction and auditing, not for survival modeling itself.

### Sensitivity analyses

1. **Complete-case matured-label analysis:** use only activated accounts with matured labels plus terminal competing cases.
2. **Delayed-label sensitivity:** use internal `Default Recorded` events as event markers when warehouse labels are delayed.
3. **Transfer sensitivity:** compare treating transfers as competing events, censored, and linked-where-available.
4. **Activated-only conditional analysis:** estimate default risk conditional on activation, with explicit warning that it is selected.
5. **Feed-missingness sensitivity:** compare complete-feed, partner, and unlinked-transfer subgroups; report bounds.

### Assumption testability

| Assumption | Testability |
|---|---|
| Event timestamps are correct and ordered | Partially testable through process audits and impossible-order checks |
| Duplicate events are identifiable | Partially testable around migration period |
| Censoring independent of future default conditional on prefix/calendar | Partially testable by comparing matured vs censored prefixes, but fundamentally limited |
| Delayed feed independent of default given observed covariates | Partially testable for cases that eventually receive feeds |
| Transfer missingness independent of outcome | Mostly untestable for unlinked transfers |
| Counterfactual default if rejected applicants were approved | Fundamentally untestable from these data alone |

## 4. Prefix extraction

The prefix must contain only information available at the early prediction time. No event, field, or snapshot after the prediction time may enter the feature table.

### Primary fixed event-count prefix

Use the first `k = 3` canonical events, stopping before any final-decision activity:

1. `Application Started`
2. `Identity/KYC Check`
3. First `Credit Review`

Rules:

- Sort events by `case:concept:name`, then `time:timestamp`.
- Apply activity canonicalization before ranking.
- Remove duplicate migration events before ranking.
- Include only events strictly before the first final-decision activity: `Accepted`, `Rejected`, `Withdrawn`, `Transferred`.
- If fewer than three pre-decision events exist, use all available pre-decision events and flag the case as `short_prefix`.
- If a case is rejected or withdrawn before reaching the nominal prefix, the prefix ends at the last pre-terminal event. The terminal state is used as a competing outcome label, not as a feature.

### Elapsed-time prefix

As an alternative or sensitivity, use an elapsed-time prefix, for example all events within **72 hours** of `Application Started`.

Rules:

- Compute `prefix_cutoff = start_time + 72 hours`.
- Include events with `time:timestamp <= prefix_cutoff`.
- Stop before final decision if a final decision occurs earlier.
- If the case terminates before 72 hours, use the pre-terminal events and flag the case.
- If no event occurs after start within the window, the prefix contains only the start event.

### Handling repeated document loops

Document loops are legitimate early-process signal.

- Count each `Request Documents` event before the prediction time.
- Count repeated `Credit Review` events if they occur before the prediction time.
- Do not use later loops that occur after the prefix cutoff.
- For fixed-length prefixes, loops may be truncated if they occur after event `k`; for elapsed-time prefixes, loops within the time window are preserved.

### Handling activity renaming

The 15 Oct 2025 migration renamed activities.

- Maintain an explicit mapping from old activity names to canonical names.
- Apply the mapping before prefix construction.
- Deduplicate exact duplicate events.
- Add a `migration_period` flag for validation and subgroup monitoring.

### Handling short cases

Short cases are not errors by themselves.

- If a case has only `Application Started`, keep it with a one-event prefix.
- If it terminates quickly, assign the appropriate competing state.
- Do not impute missing early events as if they occurred normally.
- Include `prefix_length` and `short_prefix` indicators as features.

## 5. Process-aware features

Features must be available at the prefix prediction time. They fall into three groups.

### Static case attributes

These are known at or before application start:

- `case:channel`
- `case:region`
- `case:requested_amount`
- `source_system`
- Application start date
- Day of week, month, and hour
- Policy-period flag: before/after 1 July 2025
- Migration-period flag: before/after 15 October 2025

### Prefix-derived process features

These are computed only from events up to the prefix cutoff:

Control-flow:

- Activities present in prefix
- Prefix variant or sequence hash
- Whether `Identity/KYC Check` completed
- Whether first `Credit Review` occurred
- Number of events in prefix
- Number of `Credit Review` events
- Number of `Request Documents` events
- Rework indicator: repeated return to `Credit Review`
- Short-prefix indicator

Timing:

- Time since application start
- Time since last event
- Time from start to KYC
- Time from KYC to first credit review
- Prefix duration
- Inter-event time statistics: mean, max, standard deviation

Resource/queue:

- Number of distinct resources in prefix
- Number of resource handoffs
- Resource identifier, if stable and low-cardinality or encoded safely
- Queue/waiting time before each observed activity, computed only from prefix events

Risk information:

- Last `risk_score` observed before prefix cutoff
- Change in `risk_score` if multiple updates occurred before prefix cutoff
- Requested amount divided by last observed risk score, if meaningful and not leakage-prone

### Features requiring historical reference data

These may be useful but must be constructed with strict as-of discipline:

- Resource workload in the 7 or 30 days before each prefix event
- Resource experience as of the event date, not extraction date
- Historical queue time by activity and calendar period
- Historical default or activation rates by channel, region, month, or variant, computed only from training cases before the prediction time
- Variant-level historical risk, using only matured outcomes from earlier periods

### Features not allowed as primary pre-decision features

The following are not allowed as pre-decision features because they are post-prefix, post-decision, or outcome-revealing:

- `final_limit`
- `decision_reason`
- `collection_status`
- `default_90d`
- `label_available_at`
- Future events such as `Contract Signed`, `Account Activated`, `Collections Contact`, and `Default Recorded`
- Same-day nightly `documents_total` snapshots that include documents uploaded after the nominal prediction time

---

## 6. Leakage audit

| Feature or data element | Leakage or bias risk | Valid availability rule | Keep/repair/drop | Rationale |
|---|---|---|---|---|
| `final_limit` | Assigned only after final credit decision; reveals decision and credit exposure | Available only after final decision; not available at early pre-decision prefix | Drop for primary pre-decision model | It directly leaks the acceptance/limit decision and is not part of the early prefix |
| `decision_reason` | Populated at or after Accept/Reject; reveals final decision logic | Available only at or after final decision | Drop for primary pre-decision model | It is decision-time information, not early-application information |
| `collection_status` | Updated after activation and may directly reveal default | Available only after activation; default-revealing statuses must not be features | Drop | It is post-activation and can encode the outcome |
| Future events: `Contract Signed`, `Account Activated`, `Collections Contact`, `Default Recorded` | Direct future process knowledge; outcome leakage | Include only events with timestamp <= prefix prediction time | Drop from prefix features; repair by filtering | Early-warning features must not contain events that occur after the prediction point |
| `documents_total` | Nightly snapshot may include documents uploaded later on the same day than the nominal prediction time | Use only snapshots with snapshot timestamp <= prediction time, or use date strictly before prediction date | Repair if snapshot timestamp exists; otherwise drop | Same-day aggregation can silently include future information |
| Updated `risk_score` | Multiple updates may occur after prediction time; using latest value leaks future assessment | Use last `risk_score` with event timestamp <= prefix cutoff | Keep after repair | The score is valid if its observation time is before the prefix cutoff |
| Resource experience | Stored as experience at extraction time, not event time; can leak future accumulated experience | Use historical resource-experience table as of event date | Repair if historical table exists; otherwise drop | Extraction-time experience is not available at the historical event time |
| `default_90d` | Outcome label; cannot be a feature | Use only for target construction and label maturity | Drop from feature table | Direct outcome leakage |
| `label_available_at` | Reveals whether outcome feed arrived; can encode maturity or missingness | Use only for cohort and censoring logic | Drop from feature table | It is meta-information about outcome observation, not an early predictor |
| Transfer linkage / partner outcome feed | Future linkage or partner outcome may reveal post-transfer performance not available at prediction time | Use only as-of linkage; primary analysis should treat unlinked transfers as competing/missing | Drop from primary; use only in sensitivity | Transfer outcomes are partially unobserved and can introduce selective follow-up bias |

## 7. Validation design

### Why random case splits are misleading

A random case split is inappropriate because:

1. **Policy change:** applications before and after 1 July 2025 have different approval probabilities and outcome relationships.
2. **System migration:** activity names and duplicate-event behavior change after 15 October 2025.
3. **Label maturity:** recent cases are more likely to be censored or delayed; random splits can mix matured and immature cases in misleading ways.
4. **Temporal drift:** channel mix, risk policy, applicant behavior, and partner routing evolve over time.
5. **Process leakage:** random splits can place near-duplicate time periods in train and test, overstating stability.

`pm4py.split_train_test` may be used only as a convenience baseline to show how poor random splitting can be. The primary validation must be temporal.

### Temporal split by case start time

| Split | Case start range | Purpose | Label maturity handling |
|---|---|---|---|
| Train | 2024-01-01 to 2025-03-31 | Learn pre-policy baseline | Mostly matured; censor immature cases |
| Validation | 2025-04-01 to 2025-06-30 | Tune model before policy shift | Mostly matured; censor immature cases |
| Test A: policy shift | 2025-07-01 to 2025-10-14 | Evaluate robustness to credit-policy change | Use matured labels where available; censored metrics for immature cases |
| Test B: migration/late | 2025-10-15 to 2025-12-31 | Evaluate migration robustness and recent drift | Many cases censored; use survival/competing-risk metrics and matured subset only where justified |

### Rolling backtest

For deployment realism, run monthly or quarterly backtests:

- At each backtest date, train only on cases whose labels were matured by that date.
- Evaluate on subsequent cases whose labels become matured later.
- Respect the 90-day horizon plus up to 35-day feed delay.
- Re-estimate calibration periodically because policy and migration changes can shift base rates.

### Metrics for the primary competing-risk target

Use metrics appropriate for censored, competing-outcome data:

- Cumulative incidence of `activated_default`.
- Time-dependent AUC for activated-default at relevant horizons.
- Fine-Gray subdistribution AUC, if a subdistribution model is used.
- Inverse-probability-of-censoring-weighted Brier score.
- Calibration curves for cumulative incidence at 30, 60, 90 days post-activation, and at relevant application-level horizons.
 Calibration slope and intercept by policy period.
- State-proportion calibration for competing states: rejected, withdrawn, transferred, activated-no-default, activated-default.
- Precision-recall at operationally relevant thresholds, interpreted cautiously because of censoring.
- Decision-curve analysis if the model will trigger manual review or policy exceptions.

### Metrics for secondary activated-only analysis

If the conditional activated-account target is also evaluated:

- Harrell’s C-index for time to default after activation.
- Time-dependent AUC at 90 days after activation.
- Calibration at 90 days post-activation.
- IPCW Brier score.
- Subgroup calibration by channel, partner status, and policy period.

### Subgroup and shift monitoring

Evaluate separately for:

- Pre-policy vs post-policy cases.
- Pre-migration vs post-migration cases.
- Direct vs partner/channel-originated applications.
- Region and channel.
- Short-prefix vs full-prefix cases.
- Transferred vs non-transferred cases.
- Risk-score bands.

Monitor distribution shift using:

- Activation rate over time.
- Rejection/withdrawal/transfer rates.
- Prefix variant distribution.
- Feature PSI or KL divergence between train and test periods.
- Feed-missing rate by partner/channel.

## 8. Limitations and decision-use boundary

### What the study can claim

The study can claim:

- Predictive association between early process behavior and later activated-default under the observed historical process.
- Relative ranking of applications by estimated current-policy activated-default risk.
- Calibrated probabilities for the observed policy regime, subject to censoring and competing-risk assumptions.
- Differences in risk across process variants, channels, regions, and policy periods.
- The impact of activation selection on observed default rates.

### What the study cannot claim

The study cannot claim:

- The default probability for rejected, withdrawn, or never-activated applications as if they had been exposed to credit.
- The causal effect of approving a rejected application.
- A complete outcome distribution for transferred cases when linkage is missing.
- That partner-originated missing feeds are missing at random.
- That pre-policy relationships remain valid after the July 2025 policy change or October 2025 migration.

### Predictive association vs selection effects vs counterfactual risk

- **Predictive association:** early events correlate with later activated-default among observed cases.
- **Selection effect:** activated accounts are selected through credit review, applicant behavior, and transfer routing.
- **Counterfactual default risk:** what would happen if a rejected applicant were approved is not observed and requires stronger causal assumptions.

### Deployment could change the data-generating process

At least three deployment feedback effects are likely:

1. **Approval-policy feedback:** if the model causes lenders to reject or manually review high-risk applications, the activated population changes, and the historical default relationship may no longer hold.
2. **Process-intervention feedback:** if the model triggers extra document requests, faster credit review, or collections contact, the process itself changes, altering prefix distributions and timing.
3. **Routing or partner feedback:** if high-risk cases are increasingly transferred to partners, transfer volume, linkage quality, and missing-feed rates may change, making the transfer mechanism more informative and less ignorable.

Additional possible changes include applicant behavior adaptation, policy threshold recalibration, and system changes that modify event names or timestamps.

## 9. PM4Py-oriented Python implementation skeleton

```python
import pandas as pd
import numpy as np
import pm4py

CASE_ID = "case:concept:name"
ACTIVITY = "concept:name"
TS = "time:timestamp"
AS_OF = pd.Timestamp("2026-02-15", tz="UTC")
HORIZON_DAYS = 90
MAX_FEED_DELAY_DAYS = 35
PREFIX_K = 3

# Assumption: migration rename mapping is maintained outside this skeleton.
ACTIVITY_MAP = {
    "Identity Check": "Identity/KYC Check",
    "KYC Check": "Identity/KYC Check",
    "Credit Review (old)": "Credit Review",
    "Request Docs": "Request Documents",
}

TERMINAL_COMPETING = {"Rejected", "Withdrawn", "Transferred"}
FINAL_DECISION = {"Accepted", "Rejected", "Withdrawn", "Transferred"}
PROHIBITED = [
    "final_limit", "decision_reason", "collection_status", "default_90d",
    "label_available_at", "documents_total", "data_extract_time"
]

def load_events(path="applications_events.csv"):
    df = pd.read_csv(
        path,
        parse_dates=[TS, "label_available_at", "data_extract_time"]
    )
    # As-of discipline: no event or label known after extraction date.
    df = df[df[TS] <= AS_OF]
    df[ACTIVITY] = df[ACTIVITY].replace(ACTIVITY_MAP)

    # Conservative migration duplicate removal.
    # Assumption: exact same case/activity/timestamp/resource duplicates are spurious.
    df = df.drop_duplicates(
        subset=[CASE_ID, ACTIVITY, TS, "org:resource"],
        keep="first"
    )
    df = df.sort_values([CASE_ID, TS])
    return pm4py.format_dataframe(
        df,
        case_id=CASE_ID,
        activity_key=ACTIVITY,
        timestamp_key=TS
    )

def build_case_table(df):
    g = df.groupby(CASE_ID)
    ct = pd.DataFrame(index=g.size().index)
    ct.index.name = CASE_ID

    ct["start_time"] = g[TS].min()
    ct["activation_time"] = (
        df.loc[df[ACTIVITY] == "Account Activated"]
        .groupby(CASE_ID)[TS].min()
    )
    ct["first_credit_review_time"] = (
        df.loc[df[ACTIVITY] == "Credit Review"]
        .groupby(CASE_ID)[TS].min()
    )

    term = df.loc[df[ACTIVITY].isin(TERMINAL_COMPETING)].sort_values(TS)
    ct["terminal_time"] = term.groupby(CASE_ID)[TS].min()
    ct["terminal_activity"] = term.groupby(CASE_ID)[ACTIVITY].first()

    # Retained for sensitivity analysis where internal default events are trusted.
    default_event = (
        df.loc[df[ACTIVITY] == "Default Recorded"]
        .groupby(CASE_ID)[TS].min()
    )
    ct["default_event_time"] = default_event

    # Primary label uses warehouse default_90d only when label availability is as-of.
    lab = df[
        df["default_90d"].notna()
        & df["label_available_at"].notna()
        & (df["label_available_at"] <= AS_OF)
    ]
    lab = lab.sort_values("label_available_at")
    ct["default_90d_label"] = lab.groupby(CASE_ID)["default_90d"].last()
    ct["label_available_at"] = lab.groupby(CASE_ID)["label_available_at"].max()

    ct["activated"] = ct["activation_time"].notna()
    ct["horizon_end"] = ct["activation_time"] + pd.Timedelta(days=HORIZON_DAYS)
    ct["feed_due"] = ct["horizon_end"] + pd.Timedelta(days=MAX_FEED_DELAY_DAYS)

    # Sensitivity marker only; primary state below uses default_90d_label.
    ct["default_within_90d_event"] = (
        ct["default_event_time"].notna()
        & ct["activated"]
        & (ct["default_event_time"] <= ct["horizon_end"])
        & (ct["default_event_time"] <= AS_OF)
    )

    def outcome_state(r):
        if r.activated and r.default_90d_label == 1:
            return "activated_default"
        if r.activated and r.default_90d_label == 0 and r.feed_due <= AS_OF:
            return "activated_no_default_matured"
        if r.activated and (r.feed_due > AS_OF or pd.isna(r.feed_due)):
            return "activated_unmatured_or_delayed_censored"
        if r.activated and pd.isna(r.default_90d_label) and r.feed_due <= AS_OF:
            return "activated_feed_missing_censored"
        if r.terminal_activity == "Rejected":
            return "rejected_competing"
        if r.terminal_activity == "Withdrawn":
            return "withdrawn_competing"
        if r.terminal_activity == "Transferred":
            return "transferred_competing_or_missing"
        return "open_censored"

    ct["outcome_state"] = ct.apply(outcome_state, axis=1)

    ct["policy_period"] = np.where(
        ct["start_time"] >= pd.Timestamp("2025-07-01", tz="UTC"),
        "post_policy",
        "pre_policy"
    )
    ct["migration_period"] = np.where(
        ct["start_time"] >= pd.Timestamp("2025-10-15", tz="UTC"),
        "post_migration",
        "pre_migration"
    )
    return ct

def fixed_prefix_log(df, k=PREFIX_K):
    # Fixed event-count prefix before final decision.
    # Elapsed-time version would filter by start_time + timedelta instead of event_rank.
    d = df.copy()
    d["event_rank"] = d.groupby(CASE_ID).cumcount() + 1

    first_decision = (
        d.loc[d[ACTIVITY].isin(FINAL_DECISION)]
        .groupby(CASE_ID)[TS].min()
        .rename("first_decision")
    )
    d = d.merge(first_decision, on=CASE_ID, how="left")

    # Keep only pre-decision events within the first k ranks.
    d = d[
        (d["event_rank"] <= k)
        & ((d[TS] < d["first_decision"]) | d["first_decision"].isna())
    ]

    d["prefix_end_time"] = d.groupby(CASE_ID)[TS].transform("max")
    return d.drop(columns=["first_decision"])

def build_prefix_features(prefix):
    f = prefix.groupby(CASE_ID).agg(
        prefix_events=(ACTIVITY, "count"),
        prefix_start=(TS, "min"),
        prefix_end=(TS, "max"),
        n_credit_review=(ACTIVITY, lambda x: (x == "Credit Review").sum()),
        n_document_requests=(ACTIVITY, lambda x: (x == "Request Documents").sum()),
        n_resources=("org:resource", "nunique"),
        requested_amount=("case:requested_amount", "first"),
        channel=("case:channel", "first"),
        region=("case:region", "first"),
    )

    f["prefix_duration_hours"] = (
        f["prefix_end"] - f["prefix_start"]
    ).dt.total_seconds() / 3600.0

    # Last risk_score observed before prefix end.
    f["risk_score_last_prefix"] = (
        prefix.sort_values(TS).groupby(CASE_ID)["risk_score"].last()
    )

    # Prevent prohibited post-prefix columns from entering the feature table.
    f = f.drop(columns=[c for c in PROHIBITED if c in f.columns], errors="ignore")
    return f

def leakage_audit(prefix, features):
    return {
        "prohibited_columns_present": [
            c for c in PROHIBITED if c in features.columns
        ],
        "post_prefix_events": int(
            (prefix[TS] > prefix["prefix_end_time"]).sum()
        ),
        "final_limit_nonnull_in_prefix": int(
            prefix["final_limit"].notna().sum()
        ) if "final_limit" in prefix else 0,
        "decision_reason_nonnull_in_prefix": int(
            prefix["decision_reason"].notna().sum()
        ) if "decision_reason" in prefix else 0,
        "collection_status_nonnull_in_prefix": int(
            prefix["collection_status"].notna().sum()
        ) if "collection_status" in prefix else 0,
        "default_90d_in_features": "default_90d" in features.columns,
    }

def temporal_split(case_table):
    # Primary temporal split by case start time.
    # pm4py.split_train_test may be used only as a random convenience baseline.
    train = case_table[
        case_table["start_time"] < pd.Timestamp("2025-04-01", tz="UTC")
    ]
    valid = case_table[
        (case_table["start_time"] >= pd.Timestamp("2025-04-01", tz="UTC"))
        & (case_table["start_time"] < pd.Timestamp("2025-07-01", tz="UTC"))
    ]
    test_policy = case_table[
        (case_table["start_time"] >= pd.Timestamp("2025-07-01", tz="UTC"))
        & (case_table["start_time"] < pd.Timestamp("2025-10-15", tz="UTC"))
    ]
    test_migration = case_table[
        case_table["start_time"] >= pd.Timestamp("2025-10-15", tz="UTC")
    ]
    return {
        "train": train,
        "valid": valid,
        "test_policy": test_policy,
        "test_migration": test_migration,
    }

def main():
    df = load_events()
    case_table = build_case_table(df)
    prefix_log = fixed_prefix_log(df, PREFIX_K)

    # Demonstrate PM4Py fixed-length prefix extraction.
    # The manual decision-stop above is the audited production path.
    try:
        _ = pm4py.get_prefixes_from_log(df, prefix_length=PREFIX_K)
    except Exception:
        pass

    features = build_prefix_features(prefix_log)
    audit = leakage_audit(prefix_log, features)
    splits = temporal_split(case_table)

    # Process audit on prefix-only log; using the full log would leak future structure.
    dfg, start_activities, end_activities = pm4py.discover_dfg(prefix_log)
    variants = pm4py.get_variants(prefix_log)

    print("Cohort counts:")
    print(case_table["outcome_state"].value_counts(dropna=False))
    print("Split counts:", {k: len(v) for k, v in splits.items()})
    print("Leakage audit:", audit)
    print("Prefix variants:", len(variants))

    # Later modeling connection point:
    # Join case_table and features, then fit a landmark competing-risk or
    # discrete-time survival model with censoring indicators and IPCW weights.
    return case_table, features, splits, audit

if __name__ == "__main__":
    main()
```
