```python
    acts = [a.strip() for a in trace_str.split(",")]

        # source-system sequence number on the timestamped log. The supplied
        # aggregate strings carry no timestamps, so we emulate the audited
        # outcome (11/70 sampled reversals were ingestion-time artifacts)
        # by swapping the adjacent 'Notify, Decision' pair.
        swapped, i = [], 0
        while i < len(acts):

                i += 2
            else:
                swapped.append(acts[i])
                i += 1
        acts = swapped
    if drop_technical:
```

```python
def conformance_snapshot(log, net, im, fm, title):
    diag = pm4py.conformance_diagnostics_alignments(log, net, im, fm)
    fitness = pm4py.fitness_alignments(log, net, im, fm)
    log_moves, model_moves = Counter(), Counter()
    n_fit, total_cost = 0, 0
    for tr in diag:
        cost = tr["cost"]
        total_cost += cost
        if cost == 0:
            n_fit += 1
        for step in tr["alignment"]:
            lm, mm = step[0], step[1]
            if lm != ">>" and mm == ">>":
                log_moves[lm] += 1
            elif lm == ">>" and mm not in (">>", None):
                model_moves[mm] += 1
    n = len(diag)
    print(f"\nALIGNMENT [{title}]")
    print(f"  traces: {n}   fitting (cost==0): {n_fit} ({100.0*n_fit/max(n,1):.1f}%)")
    print(f"  mean raw alignment cost: {total_cost/max(n,1):.2f}")
    print(f"  pm4py.fitness_alignments: { {k: round(v, 4) if isinstance(v, float) else v for k, v in fitness.items()} }")
    print("  top log moves  : " + ", ".join(f"{a} x{c}" for a, c in log_moves.most_common(6)))
    print("  top model moves: " + ", ".join(f"{a} x{c}" for a, c in model_moves.most_common(6)))
    return {"n": n, "fitting": n_fit, "mean_cost": total_cost / max(n, 1), "fitness": fitness}
```

```python
# -*- coding: utf-8 -*-
"""
Conformance-collapse reproduction for the regulated claims process.

pip install pm4py

The supplied data are variant-frequency AGGREGATES, not the original
timestamped log, so absolute numbers will not match the aggregate
alignment report exactly (the report's own percentages are also not
reproducible as '% perfectly fitting traces'; see the printed caveats).
The code reproduces the LOGIC of the investigation:

    raw conformance
      -> label normalisation (Risk Screen -> Classify)
      -> technical-event removal (Queue Routed)
      -> source-sequence ordering (emulated: aggregates carry no timestamps)
      -> model comparison for August-September (normative vs optional sign-off)

April-May acts as a negative control: every pipeline stage must return
identical results because none of the June changes applies to it.
"""

from collections import Counter

import pm4py
from pm4py.objects.process_tree.obj import Operator, ProcessTree

# ---------------------------------------------------------------------------
# 1. Normative model: process tree -> accepting Petri net
# ---------------------------------------------------------------------------
NORMATIVE_TREE_STR = (
    "-> ( 'Receive', 'Classify', "
    "X ( * ( 'Standard Review', 'Request Info' ), "
    "-> ( 'Enhanced Review', 'Compliance Check' ) ), "
    "'Decision', 'Notify' )"
)

normative_tree = pm4py.parse_process_tree(NORMATIVE_TREE_STR)
normative_net, normative_im, normative_fm = pm4py.convert_to_petri_net(normative_tree)

# ---------------------------------------------------------------------------
# 2. Candidate model: identical, but with an OPTIONAL Supervisor Sign-off
#    between the review block and the Decision.
#    'X ( 'Supervisor Sign-off', tau )' = XOR(sign-off, silent) = optional step.
# ---------------------------------------------------------------------------
CANDIDATE_TREE_STR = (
    "-> ( 'Receive', 'Classify', "
    "X ( * ( 'Standard Review', 'Request Info' ), "
    "-> ( 'Enhanced Review', 'Compliance Check' ) ), "
    "X ( 'Supervisor Sign-off', tau ), "
    "'Decision', 'Notify' )"
)


def _pt(operator, *children):
    node = ProcessTree(operator=operator)
    for child in children:
        child.parent = node
        node.children.append(child)
    return node


def _leaf(label):
    return ProcessTree(label=label)


def _has_visible_tau(tree):
    """Detect a 'tau' that was parsed as an ACTIVITY rather than a silent step."""
    if tree.label == "tau":
        return True
    return any(_has_visible_tau(c) for c in tree.children)


try:
```

```python
VARIANTS = {...}

# 4. Transparent normalisation pipeline
# ---------------------------------------------------------------------------
LABEL_MAP = {"Risk Screen": "Classify"}   # UI release 4.2 (2026-05-31)
TECHNICAL_EVENTS = {"Queue Routed"}       # event gateway (2026-06-03)

PIPELINES = [
    ("raw",                     dict(map_labels=False, drop_technical=False, fix_order=False)),
    ("label map",               dict(map_labels=True,  drop_technical=False, fix_order=False)),
    ("label map + no tech evt", dict(map_labels=True,  drop_technical=True,  fix_order=False)),
    ("+ src-seq order (emul.)", dict(map_labels=True,  drop_technical=True,  fix_order=True)),
]


def normalize_trace(trace_str, map_labels, drop_technical, fix_order):
    acts = [a.strip() for a in trace_str.split(",")]
    if map_labels:
        acts = [LABEL_MAP.get(a, a) for a in acts]
    if fix_order:
        # EMULATION ONLY. The genuine repair re-sorts each case by the
        # source-system sequence number on the timestamped log. The variant
        # aggregates carry no timestamps, so we emulate the audited outcome
        # (11/70 sampled reversals were ingestion-time artifacts) by swapping
        # the adjacent 'Notify, Decision' pair.
        fixed, i = [], 0
        while i < len(acts):
            if i + 1 < len(acts) and acts[i] == "Notify" and acts[i + 1] == "Decision":
                fixed.extend(["Decision", "Notify"])
                i += 2
            else:
                fixed.append(acts[i])
                i += 1
        acts = fixed
    if drop_technical:
        acts = [a for a in acts if a not in TECHNICAL_EVENTS]
    return ",".join(acts)


def build_log(period, map_labels=False, drop_technical=False, fix_order=False):
    """Expand the variant-frequency table into an event log (one trace per case)."""
    strings = []
    for trace_str, count in VARIANTS[period]:
        strings.extend([normalize_trace(trace_str, map_labels, drop_technical, fix_order)] * count)
    return pm4py.parse_event_log_string(strings)
```

```python
def _iter_alignments(diag):
    for item in diag:
        if isinstance(item, dict):
            yield item
        elif isinstance(item, (list, tuple)):
            for sub in item:
                if isinstance(sub, dict):
                    yield sub


def conformance_snapshot(log, net, im, fm, title):
    diag = pm4py.conformance_diagnostics_alignments(log, net, im, fm)
    fitness = pm4py.fitness_alignments(log, net, im, fm)

    log_moves, model_moves = Counter(), Counter()
    n_traces, n_fitting, total_cost = 0, 0, 0
    for tr in _iter_alignments(diag):
        n_traces += 1
        cost = tr.get("cost", tr.get("total_cost", 0))
        total_cost += cost
        if cost == 0:
            n_fitting += 1
        for step in tr.get("alignment", []):
            lm, mm = step[0], step[1]
            if lm != ">>" and mm == ">>":
                log_moves[lm] += 1
            elif lm == ">>" and mm is not None and mm != ">>":
                model_moves[mm] += 1

    print(f"\nALIGNMENT [{title}]")
    print(f"   traces: {n_traces} | perfectly fitting: {n_fitting} "
          f"({100.0 * n_fitting / max(n_traces, 1):.1f}%) | mean raw cost: {total_cost / max(n_traces, 1):.2f}")
    try:
        print(f"   fitness_alignments: "
              f"averageFitness={fitness['averageFitness']:.3f}, "
              f"log_fitness={fitness['log_fitness']:.3f}, "
              f"percFitTraces={fitness['percFitTraces']:.3f}")
    except Exception:
        print(f"   fitness_alignments: {fitness}")
    if log_moves:
        print("   top log moves  : " + ", ".join(f"{a} x{c}" for a, c in log_moves.most_common(6)))
    if model_moves:
        print("   top model moves: " + ", ".join(f"{a} x{c}" for a, c in model_moves.most_common(6)))
    return {"n": n_traces, "fitting": n_fitting, "mean_cost": total_cost / max(n_traces, 1)}


def precision_safe(log, net, im, fm):
    try:
        return pm4py.precision_alignments(log, net, im, fm)
    except Exception:
        return pm4py.precision_token_based_replay(log, net, im, fm)
```

```text
                                              # policy_change -> normative_model is ABSENT (governance failure)
reviewer_behavior         -> true_process, business_outcomes

# hidden nodes (*)
gateway_release*          -> recorded_labels, routing_quality*
routing_quality*          -> reviewer_behavior          # misrouting triggers Manual Override corrections
extraction_completeness*  -> recorded_labels            # event loss/gain; candidate cause of "missing" checks
amount_attribute*         -> true_process               # moderator: sign-off performed iff claim >= EUR 50,000
true_process              -> source_log*, business_outcomes
source_log*               -> recorded_labels, audit_evidence*
audit_evidence*           -> hypothesis_posteriors*     # informs interpretation; does not alter the log

# measurement chain
recorded_labels           -> alignment_moves, business_outcomes   # outcome metrics are themselves log-derived
recorded_order            -> alignment_moves
normative_model           -> alignment_moves            # stale: the policy was never propagated into it
alignment_moves           -> measured_fitness
```

Three measurement defects struck the log almost simultaneously. The UI release renamed `Classify` to `Risk Screen`, turning a compliant activity into an unmatched log move on every case. The event gateway began emitting `Queue Routed`, a technical event with no model counterpart. The ETL change then ordered `Decision` and `Notify` by warehouse-ingestion time instead of the source sequence, manufacturing reversed traces. The staged recoveries — roughly +26 points from label mapping, +23 from technical-event removal, +8 from source-sequence ordering — decompose the June collapse into exactly these artifacts: `recorded_labels` and `recorded_order` moved, not `true_process`. From August, a fourth mechanism dominates the residual: the policy change legitimately extended `true_process` with `Supervisor Sign-off`, but `normative_model` was never updated, so a compliant control was scored as deviation — model obsolescence masquerading as misbehavior. What survives every repair is small but genuinely ambiguous: 55 enhanced-review cases lack a recorded compliance check, `Manual Override` use doubled, the enhanced-review share jumped from 30% to 44%, and extraction degraded (missing-resource 1.2%6.4%). Hidden variables — extraction completeness and routing quality — could make these data loss, real bypasses, or workarounds for gateway misrouting. Unchanged business outcomes prove neither innocence nor error: appeals and cycle time are coarse, lagging indicators, weakly sensitive to the rare internal-control failures that compliance itself admits its detection misses. They show that no material customer harm has surfaced; they cannot show that no violation occurred, and they equally cannot validate the log.

```python
"""
Conformance-collapse reproduction for the regulated claims process.

    pip install "pm4py>=2.3"

The supplied data are variant-frequency AGGREGATES, not the original
timestamped log, so absolute numbers will not match the aggregate alignment
report exactly (the report's own percentages cannot be reproduced as
'% perfectly fitting traces'; see printed caveats). The code reproduces the
LOGIC of the investigation:

    raw conformance
      -> label normalisation   (Risk Screen -> Classify)
      -> technical-event removal (Queue Routed)
      -> source-sequence ordering (EMULATED: aggregates carry no timestamps)
      -> model comparison for Aug-Sep (normative vs optional sign-off),

April-May acts as a negative control: every pipeline stage must return
identical results because none of the June changes applies to it.

from collections import Counter

import pm4py
from pm4py.objects.process_tree.obj import Operator, ProcessTree

print(f"pm4py version: {pm4py.__version__}")

# ---------------------------------------------------------------------------
# 1. Normative model: process tree -> accepting Petri net
# ---------------------------------------------------------------------------
NORMATIVE_TREE_STR = (
    "-> ( 'Receive', 'Classify', "
    "X ( * ( 'Standard Review', 'Request Info' ), "
    "-> ( 'Enhanced Review', 'Compliance Check' ) ), "
    "'Decision', 'Notify' )"
)
normative_tree = pm4py.parse_process_tree(NORMATIVE_TREE_STR)
normative_net, normative_im, normative_fm = pm4py.convert_to_petri_net(normative_tree)

# ---------------------------------------------------------------------------
# 2. Candidate model: same tree with an OPTIONAL Supervisor Sign-off between
#    the review block and the Decision.  X('Supervisor Sign-off', tau) means
#    XOR(sign-off, silent) = optional step.
# ---------------------------------------------------------------------------
CANDIDATE_TREE_STR = (
    "-> ( 'Receive', 'Classify', "
    "X ( * ( 'Standard Review', 'Request Info' ), "
    "-> ( 'Enhanced Review', 'Compliance Check' ) ), "
    "X ( 'Supervisor Sign-off', tau ), "
    "'Decision', 'Notify' )"
)


def _pt(operator, *children):
    node = ProcessTree(operator=operator)
    for child in children:
        child.parent = node
        node.children.append(child)
    return node


def _leaf(label):
    return ProcessTree(label=label)


def _has_visible_tau(tree):
    """Detect a 'tau' that was parsed as a visible ACTIVITY, not a silent step."""
    if tree.label == "tau":
        return True
    return any(_has_visible_tau(c) for c in tree.children)


try:
    candidate_tree = pm4py.parse_process_tree(CANDIDATE_TREE_STR)
    if _has_visible_tau(candidate_tree):
        raise ValueError("'tau' parsed as a visible activity; using fallback")
except Exception:
    # Robust fallback: XOR(activity, silent ProcessTree()) == optional activity.
    candidate_tree = _pt(
        Operator.SEQUENCE,
        _leaf("Receive"),
        _leaf("Classify"),
        _pt(Operator.XOR,
            _pt(Operator.LOOP, _leaf("Standard Review"), _leaf("Request Info")),
            _pt(Operator.SEQUENCE, _leaf("Enhanced Review"), _leaf("Compliance Check"))),
        _pt(Operator.XOR, _leaf("Supervisor Sign-off"), ProcessTree()),
        _leaf("Decision"),
        _leaf("Notify"),
    )

candidate_net, candidate_im, candidate_fm = pm4py.convert_to_petri_net(candidate_tree)

        ("Receive,Risk Screen,Standard Review,Decision,Notify", 90),
        ("Receive,Classify,Queue Routed,Standard Review,Decision,Notify", 30),

        ("Receive,Risk Screen,Queue Routed,Enhanced Review,Compliance Check,"
         "Supervisor Sign-off,Decision,Notify", 150),
        ("Receive,Risk Screen,Queue Routed,Enhanced Review,Compliance Check,Decision,Notify", 90),
        ("Receive,Risk Screen,Queue Routed,Standard Review,Supervisor Sign-off,Decision,Notify", 60),
        ("Receive,Risk Screen,Queue Routed,Standard Review,Notify,Decision", 45),

# 4. Transparent normalisation pipeline
# ---------------------------------------------------------------------------
LABEL_MAP = {"Risk Screen": "Classify"}   # UI release 4.2 (2026-05-31)
TECHNICAL_EVENTS = {"Queue Routed"}       # event gateway (2026-06-03)

PIPELINES = [
    ("raw",                      dict(map_labels=False, drop_technical=False, fix_order=False)),
    ("label map",                dict(map_labels=True,  drop_technical=False, fix_order=False)),
    ("label map + no tech evt",  dict(map_labels=True,  drop_technical=True,  fix_order=False)),
    ("+ src-seq order (emul.)",  dict(map_labels=True,  drop_technical=True,  fix_order=True)),
]


def normalize_trace(trace_str, map_labels, drop_technical, fix_order):
    acts = [a.strip() for a in trace_str.split(",")]
    if map_labels:
        acts = [LABEL_MAP.get(a, a) for a in acts]
    if fix_order:
        # EMULATION ONLY. The genuine repair re-sorts each case by the
        # source-system sequence number on the timestamped log. The variant
        # aggregates carry no timestamps, so we emulate the audited outcome
        # (11/70 sampled reversals were ingestion-time artifacts) by swapping
        # the adjacent 'Notify, Decision' pair.
        fixed, i = [], 0
        while i < len(acts):
            if i + 1 < len(acts) and acts[i] == "Notify" and acts[i + 1] == "Decision":
                fixed.extend(["Decision", "Notify"])
                i += 2
            else:
                fixed.append(acts[i])
                i += 1
        acts = fixed
    if drop_technical:
        acts = [a for a in acts if a not in TECHNICAL_EVENTS]
    return ",".join(acts)


def build_log(period, map_labels=False, drop_technical=False, fix_order=False):
    """Expand the variant-frequency table into an event log (one trace per case)."""
    strings = []
    for trace_str, count in VARIANTS[period]:
        strings.extend(
            [normalize_trace(trace_str, map_labels, drop_technical, fix_order)] * count
        )
    return pm4py.parse_event_log_string(strings)

RED_FLAG_EDGES = {
    ("Standard Review", "Notify"),       # reversed-order indicator
    ("Notify", "Decision"),              # reversed-order indicator
    ("Enhanced Review", "Decision"),     # missing compliance check indicator
    ("Enhanced Review", "Compliance Check"),
}


def dfg_snapshot(log, title):
    res = pm4py.discover_dfg(log)
    dfg = res if isinstance(res, dict) else res[0]   # version-proof unpacking
    print(f"\nDFG [{title}]")
    for (src, tgt), cnt in sorted(dfg.items(), key=lambda kv: (-kv[1], kv[0])):
        flag = ""
        if ((src, tgt) in RED_FLAG_EDGES
                or "Sign-off" in (src, tgt) or "Manual Override" in (src, tgt)):
            flag = "   <-- investigate"
        print(f"   {src} -> {tgt}: {cnt}{flag}")


def _iter_alignments(diag):
    for item in diag:
        if isinstance(item, dict):
            yield item
        elif isinstance(item, (list, tuple)):
            for sub in item:
                if isinstance(sub, dict):
                    yield sub


def conformance_snapshot(log, net, im, fm, title):
    diag = pm4py.conformance_diagnostics_alignments(log, net, im, fm)
    fitness = pm4py.fitness_alignments(log, net, im, fm)

    log_moves, model_moves = Counter(), Counter()
    n_traces, n_fitting, total_cost = 0, 0, 0
    for tr in _iter_alignments(diag):
        n_traces += 1
        cost = tr.get("cost", tr.get("total_cost", 0))
        total_cost += cost
        if cost == 0:
            n_fitting += 1
        for step in tr.get("alignment", []):
            lm, mm = step[0], step[1]
            if lm != ">>" and mm == ">>":
                log_moves[lm] += 1                      # activity in log, not in model
            elif lm == ">>" and mm is not None and mm != ">>":
                model_moves[mm] += 1                    # activity in model, not in log

    print(f"\nALIGNMENT [{title}]")
    print(f"   traces: {n_traces} | perfectly fitting: {n_fitting} "
          f"({100.0 * n_fitting / max(n_traces, 1):.1f}%) | "
          f"mean raw cost: {total_cost / max(n_traces, 1):.2f}")
    try:
        print(f"   fitness_alignments: averageFitness={fitness['averageFitness']:.3f}, "
              f"log_fitness={fitness['log_fitness']:.3f}, "
              f"percFitTraces={fitness['percFitTraces']:.3f}")
    except Exception:
        print(f"   fitness_alignments: {fitness}")
    if log_moves:
        print("   top log moves  : "
              + ", ".join(f"{a} x{c}" for a, c in log_moves.most_common(6)))
    if model_moves:
        print("   top model moves: "
              + ", ".join(f"{a} x{c}" for a, c in model_moves.most_common(6)))
    return {"n": n_traces, "fitting": n_fitting,
            "mean_cost": total_cost / max(n_traces, 1)}


def precision_safe(log, net, im, fm):
    try:
        return pm4py.precision_alignments(log, net, im, fm)
    except Exception:
        return pm4py.precision_token_based_replay(log, net, im, fm)
```

## 7. Provisional conclusion

**Primary explanation (confidence: high, ~90%).** The conformance collapse is dominated by defects in the measurement chain, not by the claims process. Three documented defects landed between May 31 and June 15: the UI release renamed `Classify` to `Risk Screen`, giving every case an unmatched log move; the event gateway injected the technical `Queue Routed` event on 86–100% of cases; and the ETL switch to ingestion-time ordering reversed `Decision` and `Notify`. The staged recoveries — 32%58% after label mapping, 81% after technical-event removal, 89% after source-sequence ordering — attribute roughly 26, 23, and 8 points of the 61-point June drop to these mechanisms, leaving June behavior near the 93% baseline. August's residual under the old model (66% after full preprocessing) is mostly model obsolescence: the August 1 policy legitimately added `Supervisor Sign-off` for high-value claims, but the normative tree was never updated; a candidate model recovers 86%. What broke first was the measurement, not the process.

**Remaining compliance concern (confidence that a genuine residual signal exists: moderate, ~55%; confidence it is material: low, unquantified).** Fifty-five enhanced-review cases lack a recorded compliance check; `Manual Override` use doubled; the enhanced-review share jumped from 30% to 44%; and extraction quality degraded (missing-resource 1.2%6.4%). These could be lost events, genuine bypasses, or workarounds for gateway misrouting. Stable appeals and incident counts are reassuring but insensitive — compliance itself concedes its detection is incomplete.

**Immediate actions.** IT: restore source-sequence ordering, tag technical events with an origin attribute, and reconcile per-case event counts against the source system. Process owners: version the normative model, encode sign-off as a data-aware  EUR 50,000 constraint, and investigate gateway routing if overrides indicate misrouting. Compliance: audit the 55 unchecked cases and the 55 override events at source, with reviewer-level analysis under due-process safeguards.

**What must not be claimed yet:** that employees are noncompliant; that the process is compliant (recovered fitness is not evidence of compliance); that the optional-sign-off model validates the process; that stable outcomes rule out violations; or any precise split between bypass and data loss for the missing checks.

## 8. Model-governance recommendation

1. **Model versioning and change-control gate.** Every policy, UI, gateway, or ETL change that can touch the process or its logs opens a model-review ticket. Conformance outputs must be stamped with model version, log-schema version, and preprocessing-pipeline version, and re-run against a frozen reference window before and after each change.
2. **Event-semantic contract.** Activity labels are a governed vocabulary owned by the process owner; a UI relabeling must never change event labels. If a label must change, ship an explicit mapping, dual-log through a transition window, and require functional-equivalence sign-off before the old label is retired.
3. **Technical-event policy.** All system-generated events carry origin/lifecycle attributes and are excluded from conformance by a whitelist of business activities. Introducing a new event type requires registration in the log schema with an explicit business/technical classification.
4. **Timestamp provenance.** Every event persists the source timestamp, the source sequence number, and the ingestion time. Conformance ordering uses the source sequence; ingestion time is metadata, never an ordering key, and any job ordering by ingestion time raises an alert.
5. **Data-aware constraints.** Amount thresholds, authorizations, and role rules (e.g., sign-off iff claim  EUR 50,000) must be encoded as data-aware or declarative constraints, or enforced through mandatory stratified conformance on the governing attribute. Control-flow-only fitness is never reported alone for a regulated process.
6. **Post-release validation.** After each release, automatically diff variant distributions and frequency DFGs against the pre-release window and run fitness *and* precision regression against a frozen reference log. Any step-change triggers an artifact-first differential diagnosis — label map  technical-event filter  ordering repair  model delta — before any behavioral conclusion is drawn or communicated.
