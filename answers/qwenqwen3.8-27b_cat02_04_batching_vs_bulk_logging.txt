```
def summarize_pm4py_batches(batches, setting, case_to_cluster):
    if batches is None:
        return pd.DataFrame([{"setting": setting, "cluster": None, "n_batches": 0, "max_batch_size": 0, "cases_in_batches": 0, "note": "no batches"}])
    if isinstance(batches, pd.DataFrame):
        out = batches.copy()
    elif isinstance(batches, list):
        # If list of dicts row-per-case
        if len(batches) == 0:
            return pd.DataFrame([{"setting": setting, "cluster": None, "n_batches": 0, "max_batch_size": 0, "cases_in_batches": 0, "note": "no batches"}])
        # If list of batch objects with case_ids
        if isinstance(batches[0], dict) and ("case_ids" in batches[0] or "cases" in batches[0]):
            rows=[]
            for b in batches:
                case_ids = b.get("case_ids", b.get("cases", []))
                rows.extend([{"case": c, "batch_id": b.get("batch_id", b.get("id", str(i)))} for i,c in enumerate(case_ids)])
            out = pd.DataFrame(rows)
        else:
            out = pd.DataFrame(batches)
    else:
        out = pd.DataFrame([batches])
    # normalize columns
    rename = {}
    for c in out.columns:
        lc = str(c).lower()
        if "case" in lc and "id" not in lc: rename[c]="case"
        elif "case_id" in lc: rename[c]="case"
        elif "batch" in lc and "id" in lc: rename[c]="batch_id"
    out = out.rename(columns=rename)
    if "case" not in out.columns:
        # maybe column named case:concept:name
        for c in out.columns:
            if "case" in str(c).lower():
                out = out.rename(columns={c:"case"}); break
    if "case" not in out.columns:
        return pd.DataFrame([{"setting": setting, "cluster": None, "n_batches": len(out), "max_batch_size": 0, "cases_in_batches": 0, "note": "could not map cases"}])
    out["cluster"] = out["case"].map(case_to_cluster)
    out = out.dropna(subset=["cluster"])
    if "batch_id" not in out.columns:
        out["batch_id"] = out.index
    grp = out.groupby(["cluster","batch_id"]).agg(n_cases=("case","nunique")).reset_index()
    summary = grp.groupby(["cluster"]).agg(n_batches=("batch_id","nunique"), max_batch_size=("n_cases","max"), cases_in_batches=("n_cases","sum")).reset_index()
    summary.insert(0, "setting", setting)
    return summary
```

```
def summarize_pm4py_batches(batches, setting, case_to_cluster):
    if batches is None:
        return pd.DataFrame([{"setting": setting, "cluster": None, "n_batches": 0, "max_batch_size": 0, "cases_in_batches": 0, "note": "no batches"}])
    if isinstance(batches, pd.DataFrame):
        out = batches.copy()
    elif isinstance(batches, list):
        if len(batches) == 0:
            return pd.DataFrame([{"setting": setting, "cluster": None, "n_batches": 0, "max_batch_size": 0, "cases_in_batches": 0, "note": "no batches"}])
        first = batches[0]
        if isinstance(first, dict) and ("case_ids" in first or "cases" in first):
            rows = []
            for i, b in enumerate(batches):
                case_ids = b.get("case_ids", b.get("cases", []))
                batch_id = b.get("batch_id", b.get("id", f"batch_{i}"))
                for c in case_ids:
                    rows.append({"case": c, "batch_id": batch_id})
            out = pd.DataFrame(rows)
        else:
            out = pd.DataFrame(batches)
    else:
        out = pd.DataFrame([batches])

    # normalize case column
    case_col = None
    for c in out.columns:
        lc = str(c).lower()
        if lc in ("case", "case_id", "case:concept:name", "case id"):
            case_col = c; break
        if "case" in lc and "id" in lc:
            case_col = c; break
        if "case" in lc:
            case_col = c; break
    if case_col is None:
        return pd.DataFrame([{"setting": setting, "cluster": None, "n_batches": len(out), "max_batch_size": 0, "cases_in_batches": 0, "note": "no case column"}])
    out = out.rename(columns={case_col: "case"})

    batch_col = None
    for c in out.columns:
        if "batch" in str(c).lower() and "id" in str(c).lower():
            batch_col = c; break
    if batch_col is None:
        out["batch_id"] = out.index
    else:
        out = out.rename(columns={batch_col: "batch_id"})

    out["cluster"] = out["case"].map(case_to_cluster)
    out = out.dropna(subset=["cluster"])
    if out.empty:
        return pd.DataFrame([{"setting": setting, "cluster": None, "n_batches": 0, "max_batch_size": 0, "cases_in_batches": 0, "note": "no mapped cases"}])
    grp = out.groupby(["cluster", "batch_id"]).agg(n_cases=("case", "nunique")).reset_index()
    summary = grp.groupby("cluster").agg(
        n_batches=("batch_id", "nunique"),
        max_batch_size=("n_cases", "max"),
        cases_in_batches=("n_cases", "sum")
    ).reset_index()
    summary.insert(0, "setting", setting)
    return summary
```

# 2. Expected temporal signatures

# 3. Consequences for performance analysis

**G1 — Apply label / RBT-7**

- Waiting time: Queue arrival to 09:00 is a plausible waiting measure if 09:00 is true batch release. If fixture staging or batch assembly time is not captured, waiting may be partially misclassified as setup.
- Service time: The defensible service time is about 3 minutes for the 4-item batch. If itemized as four independent 3-minute services, machine-minutes are overstated by roughly 4x unless the fixture truly provides four parallel slots.
- Utilization: With a 4-item fixture, the fixture can be 100% utilized for 3 minutes. If modeled as one single-slot resource doing four overlapping 3-minute jobs, utilization becomes impossible or artificially high.
- Bottleneck detection: The cluster may look fast at item throughput and hide batch-release delay, fixture capacity constraints, or upstream queueing.

**G2 — Surface inspection / T-B**

- Waiting time: Observed waiting from queue arrival to the common 10:00 start understates waiting for later items if the actual physical starts occurred after earlier completions. For example, S04 appears to wait only 6 minutes, but if it actually started after S03 completed at 10:21, true waiting is much longer.
- Service time: Using common start to individual completion inflates service for later items: S01 appears 8 minutes, S04 appears 27 minutes. Completion-to-completion intervals of roughly 6-7 minutes are more plausible for serial inspection.
- Utilization: Literal overlapping intervals imply multiple simultaneous inspections by T-B. Summing the four apparent service durations over the 10:00-10:27 window exceeds 100% utilization for one worker.
- Bottleneck detection: T-B may appear overloaded because of overlap, but the real issue may be batch start delay plus serial inspection time.

**G3 — Release review / T-C**

- Waiting time: If the staggered starts are actual, waiting from queue arrival to start is plausible. If starts are also estimated or batched, waiting is uncertain.
- Service time: Common completion at 11:30 inflates service for early items. E01 appears to take 28 minutes, but if it was actually completed before the folder was submitted, true service is shorter.
- Utilization: Overlapping intervals again imply impossible concurrency for one reviewer. The data are consistent with reviews being performed over the half-hour but submitted together.
- Bottleneck detection: T-C may be misidentified as a slow reviewer. The apparent bottleneck may instead be form submission, folder closure, or end-of-period batching.

**G4 — Pack item / T-D**

- Waiting time: Queue arrival to start is likely a true waiting measure. Waiting increases as the queue builds: later items wait while earlier items are packed.
- Service time: The 4-minute service intervals are consistent and precise. This is the cleanest service-time cluster.
- Utilization: T-D is busy from 12:05 to 12:25, about 20 minutes for five 4-minute services, implying near-100% utilization during that window.
- Bottleneck detection: T-D may be a genuine bottleneck if arrivals continue. The main risk is that a batch detector merges adjacent events and labels a serial process as a batch.

**G5 — Root-cause analysis / analyst_pool**

- Waiting time: Queue arrival to start is plausible if starts are real, but attribution to a specific analyst is unavailable.
- Service time: Individual durations are 25, 15, 20, and 10 minutes. These may be real, but they cannot be assigned to named workers.
- Utilization: As one resource, total busy time is 70 minutes within a 30-minute window, which is impossible. As a 3-analyst pool, total busy time is about 78% of pool capacity, but maximum observed concurrency is four, exceeding the documented pool size. This indicates a data-quality or clock issue.
- Bottleneck detection: The pool may be a real bottleneck, but the data cannot distinguish true overload from shared-account artifacts, clock skew, or inaccurate timestamps.

**G6 — Close ticket / T-F**

This is the clearest example of how bulk submission can create both inflated waiting and near-zero service.

- Waiting time: Observed waiting from queue arrival to `start = 17:00` is inflated because 17:00 is likely the bulk-entry time, not the actual start of work. The observed wait includes true waiting, true service, idle time, and administrative delay before the form was submitted.
- Service time: Because `start = complete = 17:00`, observed service is zero. This understates true service, which may have occurred earlier in the day.
- Utilization: T-F appears idle until 17:00 and then shows zero busy time. If the bulk entry itself is counted, it creates an artificial instantaneous spike.
- Bottleneck detection: The paradox is that T-F can look both very slow, because queue waiting is long, and very fast, because service time is zero. Queue analysis should exclude this cluster from service-time and utilization calculations unless actual work timestamps are recovered.

**G7 — Manual measure / T-G**

- Waiting time: Displayed starts at 14:00 produce negative or near-zero waits for several items that arrived after 14:00. This indicates that the start timestamps are rounded or otherwise censored, not that items were measured before arriving.
- Service time: All service times appear as 15 minutes. True service could be shorter or longer; the quarter-hour rounding rule makes the exact value unrecoverable from this data.
- Utilization: Five identical 15-minute intervals imply 75 busy minutes in a 15-minute window if treated as one worker, which is impossible. If a 5-item batch existed, utilization could be plausible, but the data do not prove that.
- Bottleneck detection: No reliable bottleneck conclusion can be drawn. Rounding can create an artificial batch and artificial overlap.

**G8 — Repair triage / T-H**

- Waiting time: Waiting is long, roughly 38-58 minutes, and is plausibly caused by the safety-check window. Because the safety system and repair queue have different clocks, the exact magnitude is uncertain.
- Service time: The 7-minute serial service intervals are consistent with a precise scanner and are likely usable.
- Utilization: T-H appears unavailable from about 15:00 to 15:35, then busy from 15:38 to 16:13. The bottleneck is capacity loss, not necessarily slow service.
- Bottleneck detection: If downtime is not modeled, T-H may appear slow because of long waits. If downtime is modeled, the service rate may be normal and the safety check or clock-skew window becomes the relevant bottleneck.

```python
def summarize_pm4py_batches(batches, setting, case_to_cluster):
    if batches is None:
        return pd.DataFrame([{
            "setting": setting,
            "cluster": None,
            "n_batches": 0,
            "max_batch_size": 0,
            "cases_in_batches": 0,
            "note": "no batches"
        }])

    if isinstance(batches, pd.DataFrame):
        out = batches.copy()
    elif isinstance(batches, list):
        if len(batches) == 0:
            return pd.DataFrame([{
                "setting": setting,
                "cluster": None,
                "n_batches": 0,
                "max_batch_size": 0,
                "cases_in_batches": 0,
                "note": "no batches"
            }])

        first = batches[0]
        if isinstance(first, dict) and ("case_ids" in first or "cases" in first):
            rows = []
            for i, b in enumerate(batches):
                case_ids = b.get("case_ids", b.get("cases", []))
                batch_id = b.get("batch_id", b.get("id", f"batch_{i}"))
                for c in case_ids:
                    rows.append({"case": c, "batch_id": batch_id})
            out = pd.DataFrame(rows)
        else:
            out = pd.DataFrame(batches)
    else:
        out = pd.DataFrame([batches])

    # If a row contains a list of cases, explode it.
    if "case" in out.columns:
        has_list = out["case"].apply(lambda x: isinstance(x, (list, tuple, set))).any()
        if has_list:
            rows = []
            for idx, r in out.iterrows():
                batch_id = r.get("batch_id", idx)
                for c in r["case"]:
                    rows.append({"case": c, "batch_id": batch_id})
            out = pd.DataFrame(rows)

    case_col = None
    for c in out.columns:
        lc = str(c).lower()
        if lc in ("case", "case_id", "case:concept:name", "case id"):
            case_col = c
            break
        if "case" in lc and "id" in lc:
            case_col = c
            break
        if "case" in lc:
            case_col = c
            break

    if case_col is None:
        return pd.DataFrame([{
            "setting": setting,
            "cluster": None,
            "n_batches": len(out),
            "max_batch_size": 0,
            "cases_in_batches": 0,
            "note": "no case column"
        }])

    out = out.rename(columns={case_col: "case"})

    batch_col = None
    for c in out.columns:
        if "batch" in str(c).lower() and "id" in str(c).lower():
            batch_col = c
            break

    if batch_col is None:
        out["batch_id"] = out.index
    else:
        out = out.rename(columns={batch_col: "batch_id"})

    out["cluster"] = out["case"].map(case_to_cluster)
    out = out.dropna(subset=["cluster"])

    if out.empty:
        return pd.DataFrame([{
            "setting": setting,
            "cluster": None,
            "n_batches": 0,
            "max_batch_size": 0,
            "cases_in_batches": 0,
            "note": "no mapped cases"
        }])

    grp = out.groupby(["cluster", "batch_id"]).agg(n_cases=("case", "nunique")).reset_index()
    summary = grp.groupby("cluster").agg(
        n_batches=("batch_id", "nunique"),
        max_batch_size=("n_cases", "max"),
        cases_in_batches=("n_cases", "sum")
    ).reset_index()
    summary.insert(0, "setting", setting)
    return summary

def overlap_pair_rate(g):
    n = len(g)
    if n < 2:
        return 0.0

    pairs = 0
    for i in range(n):
        s1 = g.iloc[i]["start"]
        e1 = g.iloc[i]["complete"]
        for j in range(i + 1, n):
            s2 = g.iloc[j]["start"]
            e2 = g.iloc[j]["complete"]
            # Half-open interval overlap: [start, complete)
            if min(e1, e2) > max(s1, s2):
                pairs += 1

    return pairs / (n * (n - 1) / 2)

def max_concurrency(g):
    events = []
    for _, r in g.iterrows():
        if r["complete"] > r["start"]:
            events.append((r["start"], 1))
            events.append((r["complete"], -1))
        # Zero-duration intervals do not create positive overlap.

    if not events:
        return 0

    # Sort so that ends are processed before starts at the same timestamp.
    events.sort(key=lambda x: (x[0], x[1]))
    cur = 0
    mx = 0
    for _, delta in events:
        cur += delta
        mx = max(mx, cur)
    return mx

def max_queue_before_start(g):
    arrivals = g["queue_arrival"].values
    starts = g["start"].values
    maxq = 0
    for s in np.sort(starts):
        q = np.sum(arrivals <= s) - np.sum(starts < s)
        maxq = max(maxq, int(q))
    return maxq
```

# 5. Robustness and alternative explanations

**Resource IDs trusted versus shared accounts**

- If resource IDs are trusted as unique workers:
  - G2 and G3 become strong evidence of logging artifacts because one worker cannot physically complete four overlapping inspections or reviews.
  - G4 and G8 remain plausible serial single-worker processes.
  - G6 becomes a single-worker bulk-entry artifact.
  - G5 is the exception: the prompt explicitly says `analyst_pool` is shared, so it should not be treated as a trusted single worker.

- If resource IDs are treated as shared accounts or pools:
  - G2 could shift from “start batching by one worker” to “multiple inspectors using a shared T-B account.”
  - G3 could shift from “one reviewer submitting a folder” to “multiple reviewers using a shared T-C account.”
  - G5 becomes a pool-capacity problem rather than an individual-worker problem.
  - G7 could become multiple measurers using a shared T-G account, although the quarter-hour rounding still dominates the uncertainty.
  - G6 could become multiple closers using a shared T-F account, but the manual bulk form still makes zero-duration service suspect.

**Timestamps exact versus interval-censored by stated precision**

- If timestamps are treated as exact:
  - G1, G4, and G8 support fairly precise service-time estimates because they come from 1-second sources and show consistent serial or batch patterns.
  - G2 and G3 equality at the minute level may still be exact, but 1-minute precision means that apparent equality could also be rounding.
  - G6 zero-duration service is exact as recorded, but operationally suspect.
  - G7 cannot be treated as exact because the documented precision is 15 minutes and a rounding rule exists.

- If timestamps are interval-censored by stated precision:
  - A displayed 1-minute timestamp should be treated as an interval of roughly one minute, with exact bounds depending on whether the source rounds, truncates, or floors.
  - A displayed 15-minute timestamp should be treated as an interval on a quarter-hour grid. G7 therefore cannot support exact waiting or service estimates.
  - G1 and G4 may have hidden sub-minute variation if the 1-second source data were displayed only to the minute in the prompt.
  - G6 remains a censored bulk event: 17:00 is a submission boundary, not a reliable work-start or work-end.

**At least three cases where the same temporal pattern supports multiple physical explanations**

1. **G1: identical start and complete for four cases**
   - Physical explanation: a genuine 4-item fixture batch.
   - Logging explanation: PLC batch release at 09:00 and batch completion at 09:03.
   - Data-quality explanation: minute-level display or event aggregation hides small true differences.

2. **G2: common start, staggered completion**
   - Start batching: one “begin group” button creates a common start, then completions are recorded individually.
   - Concurrent work: multiple inspections are genuinely performed in parallel, possibly by multiple workers if T-B is shared.
   - Sequential work with batch logging: items are inspected one after another, but only the group start is captured.

3. **G3: staggered start, common completion**
   - End batching: reviews are performed over time but submitted as one folder at 11:30.
   - Concurrent work: multiple reviews are genuinely in progress and finish around the same time.
   - Single-worker final completion: the last review truly finishes at 11:30, while earlier completions were not logged.

4. **G7: identical quarter-hour start and completion**
   - Timestamp coarsening: legacy MES rounds all values to 14:00 and 14:15.
   - Genuine batch: five items are measured together in a 15-minute batch.
   - Start/end batching: a common setup or common completion is recorded, then rounded to the quarter-hour grid.

# 6. Data-collection redesign

**Priority 0 — Fix the core event model**

- Record separate execution timestamps for each item whenever possible: actual item start, actual item pause, actual item resume, and actual item completion.
- If a process is truly batched, record both:
  - batch-level timestamps: batch start, batch completion, fixture load, fixture unload;
  - item-level timestamps where physically available.
- Record `recorded_at` or ingestion time separately from lifecycle timestamps. Never allow database insertion time to replace source event time.
- Distinguish operational batches from logging batches with explicit identifiers: `batch_id`, `group_id`, `fixture_id`, `form_submission_id`, `folder_id`.
- Capture resource identity at the action level:
  - badge or session ID for humans;
  - device ID for machines;
  - pool/session mapping for shared accounts.
- Synchronize clocks across PLC, scanner, tablet, desktop form, VDI, legacy MES, and safety systems using NTP or an equivalent mechanism. Log clock source and any known offset.

**Priority 1 — Make capacity and downtime explicit**

- Store workstation or resource capacity metadata:
  - maximum concurrent items per fixture;
  - maximum concurrent workers per shared account;
  - pool size for shared logins.
- Record pause/resume/downtime events:
  - safety checks;
  - breaks;
  - batch setup;
  - form submission pauses;
  - system outages.
- For manual or bulk forms, require:
  - actual work start and end if known;
  - submission time;
  - preparer identity;
  - a flag indicating bulk or retrospective entry.
- Store timestamp precision and rounding metadata per source and per field:
  - documented precision;
  - rounding rule: floor, ceil, nearest, undocumented;
  - raw sub-precision value if available.

**Priority 2 — Add validation and auditability**

- Add automated data-quality flags for:
  - zero-duration service;
  - negative waiting time;
  - overlap exceeding documented capacity;
  - common start or completion groups;
  - large recording lag;
  - timestamps on coarse grid boundaries.
- Preserve source insertion order and source event ID so that retrospective correction is possible.
- Prefer one row per lifecycle event rather than one row per case with multiple timestamps, because event-level logs make batching, downtime, and submission artifacts easier to separate.

# 7. Final adjudication

**High-confidence interpretations**

- **G4** can be interpreted with high confidence as true sequential processing. It is the safest cluster for waiting-time and service-time analysis, with only a minor risk that a batch detector mislabels adjacency as a batch.
- **G6** can be interpreted with high confidence as bulk submission / end batching. It should not be used for service-time or utilization conclusions without actual work timestamps.
- **G1** can be interpreted with high confidence as a physical or semi-physical 4-item batch, conditional on accepting the documented fixture capacity. It should be analyzed at batch level, not as four independent single-item services.
- **G2** can be interpreted with high confidence as start batching caused by the begin-group button. The common start should not be treated as four item-level physical starts.
- **G3** can be interpreted with high confidence as end batching or folder submission. The common completion should be treated as a submission boundary unless actual per-case completion times are recovered.

**Exclude or interval-censor in queue analysis**

- **G6** should be excluded from service-time and resource-utilization analysis. It may be retained only as censored waiting from queue arrival to the 17:00 bulk-entry boundary.
- **G7** should be interval-censored rather than analyzed as exact timestamps. Waiting and service estimates are unreliable because of quarter-hour rounding and negative displayed waits.
- **G5** should be excluded from individual-worker utilization analysis. It may be retained only as a shared pool with capacity up to three analysts, and even then it should be flagged because observed concurrency exceeds that capacity.
- **G2** and **G3** should be treated as batch-censored:
  - G2: start is batch-level; service should be estimated from completion order or per-item start data if available.
  - G3: completion is batch/submission-level; service should be censored unless actual completion times are recovered.

**Require operational validation before performance conclusions**

- **G1**: validate fixture batch behavior and confirm whether per-item physical start/completion is available.
- **G2**: validate the begin-group button semantics and whether completions correspond to individual physical inspections.
- **G3**: validate whether the 11:30 timestamp is actual completion or folder/form submission.
- **G5**: obtain badge-level or session-level analyst identity and confirm whether up to three analysts were actually active.
- **G7**: obtain the legacy MES rounding rule and raw sub-quarter-hour timestamps before estimating service or waiting.
- **G8**: align the repair queue clock with the safety-check clock and confirm whether the safety check fully explains the 15:00-15:38 wait.
- **G6**: determine whether “Close ticket” has true processing time and, if so, recover actual work start/end times from audit logs, notes, or operator records.
