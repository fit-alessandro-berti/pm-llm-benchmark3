```python
def _load(path: str) -> pd.DataFrame:
    df = pd.read_csv(path)
    n_dup = df.duplicated(subset=["case_id", "activity", "timestamp"]).sum()
    df = df.drop_duplicates(subset=["case_id", "activity", "timestamp"])  # exact-duplicate strategy
    # NOTE: near-duplicates (< 60s apart, same activity/case) need a windowed pass in production.
    df["activity"] = df["activity"].replace(RENAME_MAP)   # normalize BEFORE any pm4py/feature step
    df["timestamp"] = pd.to_datetime(df["timestamp"])
    out = pm4py.format_dataframe(df, case_id="case_id", activity_key="activity", timestamp_key="timestamp")
    out.attrs["n_exact_duplicates_removed"] = int(n_dup)
    return out

def _variant_of(activities) -> str:
    s = set(activities)
    if "Remote Fix" in s:            return "V2"
    if "Order Part" in s:            return "V3"
    if "Repair Completed" in s:      return "V1"
    return "UNMAPPED"                # audit bucket, never silently dropped

def _edge_median_hours(log, edge):
    try:
        perf, _sa, _ea = pm4py.discover_performance_dfg(log)  # (edge)->stats dict, seconds
        stats = perf.get(edge, {})
        for k in ("median", "mean"):
            if k in stats:
                return round(stats[k] / 3600.0, 1)
    except Exception:
        pass
    return None  # auditable 'not computable this month' rather than a fake number
```

```python
import pandas as pd
import pm4py

DIAG = "Initial Diagnosis Completed"
# 1-Apr-2026 migration alias: normalize BEFORE any feature extraction / DFG discovery
ALIASES = {"Diagnostic Assessment Completed": DIAG}
EDGE_PARTS = ("Order Part", "Part Received")   # supplier-sensitive
EDGE_REMOTE = (DIAG, "Remote Fix")             # V2 health check

def _load_events(path: str) -> pd.DataFrame:
    df = pd.read_csv(path)
    # Duplicate strategy: drop exact (case, activity, timestamp) duplicates globally,
    # including the 1-20 Apr overlap window the old dedup job missed.
    n_dup = int(df.duplicated(subset=["case_id", "activity", "timestamp"]).sum())
    df = df.drop_duplicates(subset=["case_id", "activity", "timestamp"])
    # Production note: near-duplicates (same case+activity within ~5 min, differing ms)
    # need a windowed pass; exact-match is the auditable first cut.
    df["activity"] = df["activity"].replace(ALIASES)
    df["timestamp"] = pd.to_datetime(df["timestamp"])
    ev = pm4py.format_dataframe(df, case_id="case_id", activity_key="activity",
                                timestamp_key="timestamp")
    ev.attrs["duplicates_removed"] = n_dup
    return ev

def _edge_median_hours_pm4py(log, edge):
    """Prefer pm4py's performance DFG; return None if the edge is absent/uncomputable."""
    try:
        perf = pm4py.discover_performance_dfg(log)
        stats = (perf[0] if isinstance(perf, tuple) else perf).get(edge, {})
        for k in ("median", "mean"):        # key names differ across pm4py versions
            if k in stats:
                return round(float(stats[k]) / 3600.0, 1)
    except Exception:
        pass
    return None

def _edge_median_hours_manual(ev: pd.DataFrame, a: str, b: str):
    """Fallback: median (a -> b) directly-paired consecutive events, per case."""
    ev = ev.sort_values(["case:concept:name", "time:timestamp"])
    nxt_act = ev.groupby("case:concept:name")["concept:name"].shift(-1)
    nxt_ts = ev.groupby("case:concept:name")["time:timestamp"].shift(-1)
    m = (ev["concept:name"] == a) & (nxt_act == b)
    if not m.any():
        return None
    return round(float((nxt_ts[m] - ev.loc[m, "time:timestamp"]).dt.total_seconds().median()) / 3600.0, 1)

def _variant_bucket(variant) -> str:
    s = set(variant)
    if "Remote Fix" in s:       return "V2"
    if "Order Part" in s:       return "V3"
    if "Repair Completed" in s: return "V1"
    return "UNMAPPED"           # kept visible for auditing, never silently dropped

def monthly_process_evidence(path: str = "field_service_events.csv") -> pd.DataFrame:
    ev = _load_events(path)
    log_all = pm4py.convert_to_event_log(ev)
    rows = []
    for m0 in pd.date_range("2026-01-01", "2026-10-01", freq="MS"):

        #     Interpretation: the model's scoring cohort; label maturity and breach
        #     rates must be reported on THIS cohort, not the calendar one.

        except Exception:

        dfg = pm4py.discover_dfg(cal)
        dfg = dfg[0] if isinstance(dfg, tuple) else dfg
        diag_handoffs = sum(v for (a, b), v in dfg.items() if a == DIAG)

        # Temporal features (case-start cohort): durations etc.
        temporal = pm4py.extract_temporal_features_dataframe(coh_log)

            "cohort": "case_start",
            "n_cases": n,
            "share_V1": round(buckets.get("V1", 0) / n, 3) if n else None,
            "share_V2": round(buckets.get("V2", 0) / n, 3) if n else None,
            "share_V3": round(buckets.get("V3", 0) / n, 3) if n else None,
            "share_unmapped": round(buckets.get("UNMAPPED", 0) / n, 3) if n else None,
```

## 5. Monitoring architecture

7. **Population:** source/partner share; covariate PSI/KL per feature; new-cluster detection (e.g., the rural/older-device segment); fairness panel computed only on matured labels within matched strata.
8. **Intervention/deployment ledger:** every routing-policy change, supplier incident, partner onboarding, and model/feature/calibration version, plus shadow scoring and rollback triggers.

```python
import pm4py

DIAG = "Initial Diagnosis Completed"
ALIASES = {"Diagnostic Assessment Completed": DIAG}   # 1-Apr-2026 migration rename
EDGE_PARTS = ("Order Part", "Part Received")          # supplier-sensitive edge
EDGE_REMOTE = (DIAG, "Remote Fix")                    # V2 health check

def _load_events(path: str) -> pd.DataFrame:
    df = pd.read_csv(path)
    # Duplicate strategy: drop exact (case, activity, timestamp) duplicates globally,
    # including the 1-20 Apr overlap the old dedup job missed. Near-duplicates
    # (same case+activity within ~5 min) need a windowed pass in production.
    n_dup = int(df.duplicated(subset=["case_id", "activity", "timestamp"]).sum())
    df = df.drop_duplicates(subset=["case_id", "activity", "timestamp"])
    df["activity"] = df["activity"].replace(ALIASES)      # normalize BEFORE feature extraction
    df["timestamp"] = pd.to_datetime(df["timestamp"])
    ev = pm4py.format_dataframe(df, case_id="case_id", activity_key="activity",
                                timestamp_key="timestamp")
    ev.attrs["duplicates_removed"] = n_dup
    return ev

def _edge_median_h_pm4py(log, edge):
    """Performance DFG first; return None if edge absent/uncomputable (version-tolerant)."""
    try:
        perf = pm4py.discover_performance_dfg(log)
        stats = (perf[0] if isinstance(perf, tuple) else perf).get(edge, {})
        for k in ("median", "mean"):                      # key names vary across pm4py versions
            if k in stats:
                return round(float(stats[k]) / 3600.0, 1)
    except Exception:
        pass
    return None

def _edge_median_h_manual(ev: pd.DataFrame, a: str, b: str):
    """Auditable fallback: median over consecutive (a -> b) event pairs, per case."""
    ev = ev.sort_values(["case:concept:name", "time:timestamp"])
    nxt_act = ev.groupby("case:concept:name")["concept:name"].shift(-1)
    nxt_ts = ev.groupby("case:concept:name")["time:timestamp"].shift(-1)
    m = (ev["concept:name"] == a) & (nxt_act == b)
    if not m.any():
        return None
    return round(float((nxt_ts[m] - ev.loc[m, "time:timestamp"]).dt.total_seconds().median()) / 3600.0, 1)

def _bucket(variant) -> str:
    s = set(variant)
    if "Remote Fix" in s:       return "V2"
    if "Order Part" in s:       return "V3"
    if "Repair Completed" in s: return "V1"
    return "UNMAPPED"           # kept visible for auditing, never silently dropped

    log_all = pm4py.convert_to_event_log(ev)
    rows = []
    for m0 in pd.date_range("2026-01-01", "2026-10-01", freq="MS"):
        m1 = m0 + pd.offsets.MonthEnd(1) + pd.Timedelta(days=1)      # exclusive end
        tag = m0.strftime("%Y-%m")

        # (a) Event-calendar cohort: events inside the calendar month.
        #     Interpretation: process STATE during the month (in-flight cases);
        #     the right lens for performance-DFG edges.
        cal = pm4py.filter_time_range(log_all, m0.strftime("%Y-%m-%d 00:00:00"),
                                      m1.strftime("%Y-%m-%d 00:00:00"), mode="events")

        # (b) Case-start cohort: cases whose FIRST event falls in the month.
        #     Interpretation: the model's scoring cohort -- label maturity and
        #     breach rates must be reported on THIS cohort, not the calendar one.
        starts = ev.groupby("case:concept:name")["time:timestamp"].min()
        cases = starts[(starts >= m0) & (starts < m1)].index
        coh = ev[ev["case:concept:name"].isin(cases)]
        coh_log = pm4py.convert_to_event_log(coh)

        buckets = {}
        try:                                   # {variant_tuple: case_count}
            for v, c in pm4py.get_variants(coh_log).items():
                buckets[_bucket(v)] = buckets.get(_bucket(v), 0) + c
        except Exception:
            pass
        n = sum(buckets.values()) or None

        dfg = pm4py.discover_dfg(cal)          # frequency DFG on the calendar cohort
        dfg = dfg[0] if isinstance(dfg, tuple) else dfg
        diag_handoffs = sum(v for (a, b), v in dfg.items() if a == DIAG)

        temporal = pm4py.extract_temporal_features_dataframe(coh_log)
        dur_cols = [c for c in temporal.columns if "duration" in c.lower()]
        u = (coh["concept:name"] == "UNKNOWN_ACTIVITY")

        rows.append({
            "month": tag, "cohort": "case_start", "n_cases": n,
            "share_V1": round(buckets.get("V1", 0) / n, 3) if n else None,
            "share_V2": round(buckets.get("V2", 0) / n, 3) if n else None,
            "share_V3": round(buckets.get("V3", 0) / n, 3) if n else None,
            "share_unmapped": round(buckets.get("UNMAPPED", 0) / n, 3) if n else None,
            "cases_with_diagnosis": int(coh.loc[coh["concept:name"] == DIAG,
                                                "case:concept:name"].nunique()),
            "unknown_activity_share": round(float(u.mean()), 4) if len(u) else 0.0,
            "parts_edge_median_h": _edge_median_h_pm4py(coh_log, EDGE_PARTS)
                                   or _edge_median_h_manual(coh, *EDGE_PARTS),
            "remote_edge_median_h": _edge_median_h_pm4py(coh_log, EDGE_REMOTE)
                                    or _edge_median_h_manual(coh, *EDGE_REMOTE),
            "median_case_duration_d": round(float(temporal[dur_cols[0]].median()) / 86400, 1)
                                      if dur_cols and len(temporal) else None,
        })
        cal_df = pm4py.convert_to_dataframe(cal)
        uc = (cal_df["concept:name"] == "UNKNOWN_ACTIVITY")
        rows.append({
            "month": tag, "cohort": "event_calendar",
            "diag_handoff_edges": diag_handoffs,
            "unknown_activity_share": round(float(uc.mean()), 4) if len(uc) else 0.0,
            "parts_edge_median_h": _edge_median_h_pm4py(cal, EDGE_PARTS)
                                   or _edge_median_h_manual(cal_df, *EDGE_PARTS),
            "remote_edge_median_h": _edge_median_h_pm4py(cal, EDGE_REMOTE)
                                    or _edge_median_h_manual(cal_df, *EDGE_REMOTE),
        })
    return pd.DataFrame(rows)   # auditable evidence only -- no model is fitted here
```

## 6. Risks of acting on the wrong diagnosis

1. **Retraining on immature labels (Sep/Oct).** The model would learn a deflated breach base rate (~0.16–0.31 instead of ~0.33+) and systematically underpredict exactly when V3 and Partner-Z risk is highest, while poisoning every future evaluation that reuses these cohorts. *Mitigation:* vintage-gated training sets; maturity thresholds per cohort; frozen-score/matured-label backtesting; embargo immature months and retrain later on a maturity calendar.

2. **Calibrating around the April pipeline bug.** Fitting isotonic/Platt on corrupted April scores bakes the corruption into the calibration layer; once features are repaired, the model is miscalibrated in the *opposite* direction, and the underlying bug is masked and persists. *Mitigation:* a hard DQ gate before any recalibration; recompute-then-reevaluate rule; feature-store vs. raw recompute sampling audits; schema-change regression tests with pinned aliases.

3. **Ignoring genuine supplier-driven concept drift.** If August–September is explained away as "just censoring and the new partner," underprediction for V3 persists, dispatch prioritization stays wrong, SLA breaches continue, and monitoring slowly normalizes the degraded state. *Mitigation:* per-variant matured-label monitors plus performance-DFG SPC alarms wired to an operational escalation path independent of ML; add the supplier lead-time feature; apply an interim V3 risk offset now.

4. **Attributing Partner Z's gap to model unfairness (or dismissing it) without stratified evidence.** Declaring unfairness on the immature 0.45-vs-0.27 gap risks damaging partner relations and triggering unnecessary interventions; dismissing it risks ignoring genuine under-service of rural customers. Both are plausible until composition, omitted features, and label maturity are separated. *Mitigation:* fairness panels only on matured labels, within variant and matched covariate strata; transport decomposition separating composition from within-stratum effects; add rural-distance and device-age features before drawing conclusions.

5. **Treating June's benign mix shift as model failure and retraining unnecessarily.** A full retrain in response to June would overfit a policy artifact, consume the team's capacity, and — worse — likely train on data still containing corrupted April rows. *Mitigation:* within-variant stability tests before any escalation; the intervention ledger attributing the shift to the routing policy; recalibrate-or-reweight first, retrain only on evidence.

6. **Reading October's 0.16 as genuine improvement.** Relaxing SLAs, celebrating KPIs, or standing down the supplier escalation on the basis of a censoring artifact would be followed by a painful restatement when the vintage matures near ~0.33. *Mitigation:* never publish cohort metrics below the maturity threshold without a "provisional" flag; automatic restatement when vintages mature; keep the supplier escalation keyed to process evidence (155 h edge), not to the immature outcome rate.
