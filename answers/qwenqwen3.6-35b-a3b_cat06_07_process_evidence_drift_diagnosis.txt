import pandas as pd
import pm4py
from datetime import datetime

def audit_monthly_process(csv_path, month_start, month_end):
    """
    Extracts auditable monthly process summaries.
    Normalizes labels, handles duplicates, and computes DFG/performance metrics.
    """
    # 1. Read and normalize
    df = pd.read_csv(csv_path)
    
    # Normalize renamed diagnosis labels
    df['activity'] = df['activity'].replace(
        {'Initial Diagnosis Completed': 'Diagnostic Assessment Completed'}
    )
    
    # Duplicate handling strategy: Keep first occurrence per case-activity pair
    # assuming duplicates are artifacts of migration
    df = df.sort_values(['case_id', 'time_stamp']).drop_duplicates(
        subset=['case_id', 'activity'], keep='first'
    )
    
    # Format for PM4Py
    log = pm4py.format_dataframe(df, case_id='case_id', activity_key='activity', timestamp_key='time_stamp')
    
    # 2. Filter to month cohort
    month_log = pm4py.filter_time_range(log, month_start, month_end)
    
    # 3. Extract variants
    variants = pm4py.get_variants(month_log)
    variant_counts = variants.value_counts()
    
    # 4. Discover DFG and Performance DFG
    # Focus on V3 path: Order Part -> Part Received
    # Filter to V3 cases for performance analysis
    v3_log = month_log[month_log['variant'].str.contains('Order Part.*Part Received', na=False)]
    
    dfg, dfg_edges = pm4py.discover_dfg(v3_log, activity_key='activity')
    perf_dfg = pm4py.discover_performance_dfg(v3_log, activity_key='activity')
    
    # Extract median duration for critical edge
    edge = ('Order Part', 'Part Received')
    if edge in perf_dfg:
        median_duration = perf_dfg[edge]
    else:
        median_duration = None
        
    # 5. Temporal features extraction (auditable summary)
    # Use extract_temporal_features_dataframe to get case-level stats
    # Note: This returns features, we aggregate for summary
    features_df = pm4py.extract_temporal_features_dataframe(v3_log, activity_key='activity')
    
    # Summaries
    summary = {
        'month': f"{month_start.strftime('%Y-%m')}",
        'case_count': len(month_log),
        'variant_distribution': variant_counts.to_dict(),
        'v3_median_part_delay_hours': median_duration / 3600 if median_duration else None,
        'v3_case_count': len(v3_log),
        'dfg_edges': len(dfg_edges)
    }
    
    return summary
