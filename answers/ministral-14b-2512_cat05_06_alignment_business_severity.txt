## 7. **PM4Py Reproduction and Aggregation Code**

```python
import pm4py
from pm4py.objects.log.importer.xes import importer as xes_importer
from pm4py.objects.conversion.log import converter as log_converter
from pm4py.objects.petri_net.importer import importer as pnml_importer
from pm4py.objects.petri_net.utils import petri_net_to_df
from pm4py.algo.conformance_diagnostics.alignments.algorithm import algorithm as alignments
from pm4py.objects.log.util import dataframe_utils
import pandas as pd

# --- Step 1: Parse Process Tree ---
process_tree_str = """
-> ( 'Register Claim', 'Validate Coverage', 'Assess Loss', X ( -> ( 'Fraud Screen', 'Manual Approve' ), 'Auto Approve' ), 'Pay Claim', 'Notify Claimant' )
"""
tree = pm4py.parse_process_tree(process_tree_str)
pn = pm4py.convert_to_petri_net(tree)

# --- Step 2: Create Event Log (Example: CL-602) ---
log_data = {
    "case:concept:name": ["CL-602"],
    "concept:name": [
        "Register Claim", "Validate Coverage", "Assess Loss",
        "Manual Approve", "Pay Claim", "Notify Claimant"
    ],
    "time:timestamp": pd.date_range("2023-01-01", periods=6)
}
log = log_converter.apply(dataframe_utils.convert_to_dataframe(log_data))

# --- Step 3: Align Log to Petri Net ---
alignments_result = alignments.apply(log, pn, variant=alignments.Variants.ALPHA)
alignments_df = pd.DataFrame(alignments_result)

# --- Step 4: Aggregate Move Patterns ---
move_summaries = []
for case_id, trace in log:
    for alignment in alignments_result[case_id]:
        moves = alignment[1]
        for move in moves:
            move_summaries.append({
                "case": case_id,
                "move_type": move[0].__class__.__name__,
                "activity": str(move[0]),
                "cost": move[1],
                "amount": {"CL-602": 480000, "CL-604": 90000}[case_id],
                "risk": {"CL-602": "HIGH", "CL-604": "MEDIUM"}[case_id],
                "policy_violation": "P2" if case_id == "CL-602" else "P3" if case_id == "CL-604" else None
            })

move_df = pd.DataFrame(move_summaries)
risk_priority = move_df.groupby(["case", "policy_violation", "move_type"]).agg({
    "cost": "sum",
    "amount": "first",
    "risk": "first"
}).reset_index()

# --- Step 5: Risk-Prioritization Evidence Table ---
risk_priority["severity"] = risk_priority.apply(
    lambda x: 5 if x["policy_violation"] in ["P2", "P3"] and x["risk"] == "HIGH"
    else 3 if x["policy_violation"] in ["P2", "P3"] and x["risk"] == "MEDIUM"
    else 1, axis=1
)
risk_priority["confidence"] = risk_priority.apply(
    lambda x: "High" if x["move_type"] == "S" else "Medium" if x["move_type"] == "M" else "Low", axis=1
)

print(risk_priority[["case", "policy_violation", "move_type", "cost", "amount", "risk", "severity", "confidence"]])
```
