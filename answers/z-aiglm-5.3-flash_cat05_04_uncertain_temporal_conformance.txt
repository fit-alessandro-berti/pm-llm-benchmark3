def _secs(x): return timedelta(seconds=float(x))

def build_intervals():
    """Return {(case,activity): [(utc_start, utc_end), ...]} preserving unions."""
    out = {}
    for case, act, sysname, parts, sem in EVENTS:
        (rlo, rhi), (a, b), tzname = CFG[sysname]
        disp = TS(*parts)
        # candidate localizations (fold enumeration only where relevant)
        locals_ = []
        if sysname in FOLD_SENSITIVE and (parts[0], parts[1], parts[2]) == FOLD_DATE \
           and 2 <= disp.hour <= 2:
            for fold in (0, 1):
                lo = disp.replace(tzinfo=ZONES[tzname], fold=fold)
                locals_.append(lo)
        else:
            locals_.append(disp.replace(tzinfo=ZONES[tzname]))
        ivals = []
        for loc in locals_:
            # resolution-widened clock window (naive-equivalent in clock frame)
            wlo = loc + _secs(rlo)
            whi = loc + _secs(rhi)   # half-open top; second-granularity-safe here
            # clock offset: true = recorded - o, o in [a,b] -> shift [-b,-a]
            t_lo = (wlo - _secs(b)).astimezone(UTC)
            t_hi = (whi - _secs(a)).astimezone(UTC)
            ivals.append((t_lo, t_hi))
        if sysname == "MOBILE-LON":
            # Scenario A (PRIMARY): device stamps submission -> occurrence
            # estimate shifts EARLIER by [0, ENTRY_DELAY_S]. Stated openly.

        # normalize to seconds-resolution display strings
        ivals = sorted(set((l.isoformat(), h.isoformat()) for l, h in ivals))
        out[(case, act)] = [(pd.Timestamp(l), pd.Timestamp(h)) for l, h in ivals]
    return out

# ---- interval durations ------------------------------------------------------
def dur_union(A, B):
    """Cartesian union of feasible durations B-A across interval options."""
    ds = []
    for alo, ahi in A:
        for blo, bhi in B:
            ds.append((blo - ahi, bhi - alo))
    ds.sort()
    merged = []
    for lo, hi in ds:
        if merged and lo <= merged[-1][1]:
            merged[-1] = (merged[-1][0], max(merged[-1][1], hi))
        else:
            merged.append((lo, hi))
    return merged

PAIRS = [("Alert_Received","Investigation_Start","K1"),
         ("Investigation_Start","Containment_Complete","K2"),
         ("Alert_Received","Containment_Complete","K3"),
         ("Containment_Complete","Close_Incident","K4")]
PROFILE = {"K1": (600,180), "K2": (2100,600),
           "K3": (2700,720), "K4": (86400,21600)}
SLA = {"S1": ("Alert_Received","Investigation_Start",900),
       "S2": ("Investigation_Start","Containment_Complete",2700),
       "S3": ("Containment_Complete","Close_Incident",129600)}

def cls_band(ds, kind, zeta):
    mu, sd = PROFILE[kind]; lo, hi = mu - zeta*sd, mu + zeta*sd
    if all(h <= hi for _, h in ds) and all(l >= lo for l, _ in ds): return "CERTAIN_PASS"
    if all(h < lo or l > hi for l, h in ds):
        return "CERTAIN_VIOLATION"   # every piece wholly outside; for 1-piece sets band-adjacent
    return "POSSIBLE_VIOLATION"

def cls_sla(ds, limit, causal_clamp=False):
    ds = [(max(l, 0.0), max(h, 0.0)) for l, h in ds] if causal_clamp else ds
    if all(h <= limit for _, h in ds): return "CERTAIN_PASS"
    if all(l > limit for l, _ in ds): return "CERTAIN_VIOLATION"
    return "POSSIBLE_VIOLATION"

## 7. Governance Recommendation

1. **UTC storage invariant.** Canonical store in UTC (or epoch nanos); local renderings are views. This alone collapses the OPS/NY/London/Berlin conversion surface and prevents the 2025-10-26-style rewrites.

2. **Raw timestamp retention.** Keep the original literal (`"02:50"` with no fold flag) alongside the canonical value; never overwrite source truth with corrected values. Every normalization in §2 must be reproducible years later — today's *derived* intervals (T-404's start, T-407's receipt) must be persisted with their inputs.

3. **Source offset telemetry.** Continuous, alertable clock-offset reporting per source (the thing missing for T-409 and weak for VENDOR-NY). Offsets exceeding declared bounds should quarantine the source's timestamps from automated SLA scoring — auto-degrading classifications to POSSIBLE/INDETERMINATE rather than fabricating precision.

4. **Resolution metadata.** Every field carries precision semantics (`floor-minute`, `round-10s`, `second-exact±0.5`). Schemas should reject bare datetimes lacking precision markers; consumers must apply §1-style expansion mechanically, not by analyst folklore.

5. **Start/complete semantics.** Declare contractually which lifecycle verb a device emits (SENSOR emits completions). Heterogeneous semantic mapping belongs in the pipeline, with lineage recorded — this is exactly the trap behind T-407 (Alert_Sent  Alert_Received) and would bite equally on any In_Progress/work-order streams.

6. **DST fold control.** During fall-back windows: capture the UTC offset (or fold index) at write time; if unavoidable, flag records in ambiguous local ranges as two-hypothesis objects. Never let downstream analytics silently pick a fold.

7. **Causal identifiers.** Link related evidence explicitly: message-id connecting Alert_SentAlert_Received, acknowledgment receipts, ingest sequence numbers tied to occurrence-position guarantees. T-407's unlogged causal premise and T-405's order-based fold inference are both stand-ins for a proper provenance graph; explicit references convert INDETERMINATE into decidable.

8. **Mobile/manual-entry policy.** Mark user-entered timestamps as such and require a form-level declaration of "this is the event time" vs. "stamp now" — the two interpretations yielded opposite S2 verdicts for T-406. Recall-error caps belong in the metadata, not in analysts' heads.
