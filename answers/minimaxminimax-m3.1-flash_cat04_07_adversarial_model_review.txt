```python
import pm4py
from pm4py.objects.powl import Transition, SilentTransition, OperatorPOWL, StrictPartialOrder

def _mandatory(name):
    return Transition(name=name)

def _optional(name, skip_name):
    """Exactly one of: the activity happens, or an explicit (silent) skip."""
    return OperatorPOWL(OperatorPOWL.XOR, [Transition(name=name), SilentTransition(name=skip_name)])

def build_third(is_external=None):
    """is_external: True/False -> specialize the Supplier Review requirement;
       None -> generic backbone (guard decides legality)."""
    register = Transition(name='Register Dossier')
    credit_1 = Transition(name='Credit Check')          # initial execution
    tech_1  = Transition(name='Technical Review')       # initial execution
    correct = Transition(name='Correct Dossier')
    credit_2 = Transition(name='Credit Check')          # re-execution after correction
    tech_2  = Transition(name='Technical Review')       # re-execution after correction
    expedite = Transition(name='Expedite Review')
    manager = Transition(name='Manager Approval')
    auto    = Transition(name='Auto Approval')
    release = Transition(name='Release Order')

    checks = StrictPartialOrder(nodes=[credit_1, tech_1])          # P2: both, either order

    recheck = OperatorPOWL(OperatorPOWL.XOR, [                     # P6: which checks repeat
        StrictPartialOrder(nodes=[credit_2]),
        StrictPartialOrder(nodes=[tech_2]),
        StrictPartialOrder(nodes=[credit_2, tech_2]),
    ])
    correction_stage = OperatorPOWL(OperatorPOWL.LOOP, [            # 0..n corrections
        StrictPartialOrder(nodes=[correct, recheck]),
    ])

    if is_external is True:
        supplier = Transition(name='Supplier Review')                # P3: mandatory for external
    else:
        supplier = _optional('Supplier Review', 'supplier_skipped')

    supplier_step = supplier if is_external is True else supplier
    exp_step = _optional('Expedite Review', 'expedite_not_required')
    approval = OperatorPOWL(OperatorPOWL.XOR, [manager, auto])       # P5: exactly one

    model = StrictPartialOrder(nodes=[register, checks, correction_stage,
                                      supplier_step, exp_step, approval, release])
    model.order.add_edge(register, checks)
    model.order.add_edge(checks, correction_stage)
    model.order.add_edge(correction_stage, supplier_step)
    model.order.add_edge(correction_stage, exp_step)
    model.order.add_edge(supplier_step, approval)
    model.order.add_edge(exp_step, approval)
    model.order.add_edge(approval, release)
    return model

model_third = build_third(is_external=None)
```

```python
def policy_check(trace):
    """trace: list of dicts with 'concept:name' and optional attributes
       returns dict(verdict=..., violations=[...], undetermined=[...], info=[...])
    """
    acts = [e['concept:name'] for e in trace]
    attrs = trace[0] if trace else {}
    is_external = attrs.get('is_external', None)
    V, U, I = [], [], []

    # P1
    if not acts or acts[0] != 'Register Dossier':
        V.append('P1:first_event')
    if not acts or acts[-1] != 'Release Order':
        V.append('P1:last_event')
    if acts.count('Register Dossier') != 1 or acts.count('Release Order') != 1:
        V.append('P1:cardinality')

    # P5 exactly one approval, and it must be before Release
    n_app = sum(a in APPROVALS for a in acts)
    if n_app != 1:
        V.append('P5:exactly_one_approval')
    app_idx = [i for i,a in enumerate(acts) if a in APPROVALS]
    rel_idx = acts.index('Release Order') if 'Release Order' in acts else len(acts)
    if app_idx and app_idx[0] > rel_idx:
        V.append('P5:approval_after_release')

    # P2 mandatory checks before approval, each at least once
    for m in MANDATORY:
        if m not in acts:
            V.append('P2:missing_'+m.replace(' ','_'))
    if app_idx:
        for m in MANDATORY:
            for i,a in enumerate(acts):
                if a == m and i > app_idx[0]:
                    V.append('P2:check_after_approval:'+m)
    # P7 + P6
    for i,a in enumerate(acts):
        if a in MANDATORY:
            first = acts.index(a)
            if i > first:
                # a repetition: require an intervening correction
                if 'Correct Dossier' not in acts[:i]:
                    V.append('P7:repeat_without_correction:'+a)
                else:
                    # minimality only checkable with the attribute
                    aff = _affected_of_last_correction(trace, i)
                    if aff is None:
                        U.append('P6:affected_checks_unknown')
                    elif set(aff) != _rechecks_since_last_correction(acts, i):
                        V.append('P6:recheck_set_mismatch:'+a)
    # P3
    if 'Supplier Review' not in acts:
        if is_external is True:
            V.append('P3:supplier_review_missing_external')
        elif is_external is None:
            U.append('P3:supplier_review_absent_attribute_missing')
    else:
        if is_external is False:
            I.append('P3:supplier_review_on_internal_order')
    # P4 expedite
    if 'Expedite Review' in acts:
        e = acts.index('Expedite Review')
        for m in MANDATORY:
            if m in acts and acts.index(m) > e:
                V.append('P4:expedite_before_'+m.replace(' ','_'))
        if app_idx and e > app_idx[0]:
            V.append('P4:expedite_after_approval')

        if a == 'Correct Dossier':

            if not any(s in MANDATORY for s in seg):
                V.append('P6:correction_without_recheck')
    verdict = 'VIOLATION' if V else ('UNDETERMINED' if U else 'COMPLIANT')
    return {'verdict': verdict, 'violations': V, 'undetermined': U, 'info': I}
```

# 1. Review Scope and Assumptions

## 1.1 Intended uses, distinguished

The board is being asked to compare two artifacts that were never built for the same purpose. The evaluation criterion is use-case dependent, and the reported numbers are only interpretable once the use case is fixed.

| Use case | What must be true | Dominant error | Who is hurt |
|---|---|---|---|
| **E1 Exploration / process discovery** | High recall of observed behavior; cheap to run; tolerant of log noise | Missing a real variant | Analyst, low harm |
| **E2 Conformance enforcement / gating** | **Soundness** w.r.t. policy (no illegal trace accepted) first; then low false-rejection | False acceptance | Customer, regulator, bank |
| **E3 Predictive monitoring / alerting** | High recall on illegal traces, tolerable false-alarm rate, and a usable reason code | Missing an illegal trace | Ops |
| **E4 Stakeholder communication / training** | Human-readable, low comprehension burden, no misleading messages | Misleading simplicity | All business users |
| **E5 Simulation / what-if** | Unbiased transition probabilities, no structural artifacts (e.g., arbitrary branch weights) | Biased estimates | Planning |
| **E6 Variant analytics / root cause** | Occurrence-level identity (which occurrence, in which case) | Merging distinct nodes | Analyst, IT |

**Asymmetry of error cost.** In a credit-checked dossier process, a false acceptance (releasing an order with no Credit Check) is a control failure with expected loss orders of magnitude above a false rejection (an order is held for re-review and released a day later). This asymmetry means: the *enforcement* artifact must be optimized for soundness first, and false-rejection tolerance is bought deliberately. The *exploratory* artifact should be permissive because it is not used to make release decisions. Conflating the two artifacts into one "best model" is the central methodological error in the submission.

1. The log contains the nine activity names in the policy vocabulary; the log was not shown to either team in a form that permits independent recomputation.
2. The claimed "80/20 case split" is case-level, not trace-level; this is asserted, not evidenced (no seeds, no manifest, no held-out case list).
3. The metrics are PM4Py defaults of *unknown version*; `fitness_alignments` is trace- vs event-level and `precision_alignments` is path-dependent in ways that differ across PM4Py releases.
4. **No `external/internal` attribute exists** (P3) and **no "affected check" attribute exists** (P6). Neither input model can express these facts; this is a *data* limitation, not a modeling defect.

| **Duplicate-label count** (S 0 / C 33, definition undocumented) | C duplicates activity labels across branches; S does not | How many *distinct nodes* are affected; whether duplication is semantically typed or accidental; impact on alignment attribution, impact analysis, ownership, and simulation branch weights | "Definition undocumented" — 33 could mean extra nodes beyond first occurrence, or total nodes sharing a label, or pairs; these differ by ~an order of magnitude | **High** (maintainability and impact analysis, F3) and **medium** (it may also contaminate C's precision figure) |
| **Stakeholder comprehension, informal 1–5** (S 4.4 / C 1.9) | S reads as simpler to non-experts; C reads as heavy | Task-level correctness, time-to-answer, or error rate; whether respondents understood the *semantics* rather than the notation; whether the scale was anchored | n unknown; role composition unknown; self-selection; no anchoring; order/learning effects; "simplicity" of notation  "correctness" of model — a model that is easy to read *because* it says less is easy and wrong | Medium for E4 (communication), **zero** for E2 (enforcement) — comprehension must not be traded against soundness |
| *Derived:* **trainholdout fitness drop** (S 0.011 / C 0.229) | C's fit is variant-specific; S's is not | Whether C's drop is overfitting or a split artifact; whether S's stability is a virtue or vacuity | Requires 5 seeds and case-level grouping to attribute | **High** — the primary discriminator |
| *Derived:* **policy-probe acceptance** (B1–B8, §5) | Direct evidence of admitted/forbidden behavior | Statistical generalization from 8 probes | Probes are a policy test set, not a sample | **Highest** — this is the only evidence class tied to P1–P7 |

**Reading of the table as a whole.** The metric set is dominated by measures of *fit to a log that is known to be incomplete* (F4), and contains **no measure of fit to the policy**, which is the actual specification for the enforcement use case. A model can score 0.99/0.98/0.91 on S's row and be *catastrophically wrong* for compliance (it releases orders with no credit check). Conversely, a model can score 0.744 and still be *more correct* than the log. The board should therefore treat the reported metrics as evidence about **log-exploration utility** (where S genuinely leads) and demand a **policy-conformance measurement** before any compliance claim is made (see §9).

# 3. Adversarial Review of Model S

```text
->( 'Register Dossier',
    +( X('Credit Check', tau), X('Technical Review', tau),
       X('Supplier Review', tau), X('Expedite Review', tau) ),
    X('Manager Approval', 'Auto Approval'),
    'Release Order' )
```

Semantics: a strict sequence; in the middle, four *independent* exclusive choices (act-or-skip) executed in an unordered parallel region; then exactly one approval; then release. The admitted behavior set is: any subset of the four activities, in **any interleaving**, with any one of the two approvals.

## 3.1 Strongest case for Model S

1. **It matches the log as it actually is.** F4 reports that some Supplier Review events may be missing from the log. A model fitted to that log must either penalize its own log (C's 0.744 holdout) or absorb the loss. S absorbs it at zero cost. For exploration (E1, E6) that is the *right* engineering choice: a discovery artifact that rejects a third of reality is not usable for variant analysis.
2. **Operational stability.** The 0.011 trainholdout drop is the single most favorable number in the submission. Under a changing dossier process, a model that needs re-enumeration every time a new combination appears (F2) is a maintenance liability, not an asset.
3. **Human factors.** 4.4/5 comprehension, one screen, no branch enumeration, no duplicate labels: S is the only artifact in this dossier that a business user can validate unaided, and it can be used as a *communication overlay* on top of an enforcing model (E4).
4. **Honest intent, honestly scoped.** Its defect set is transparent: permissive, non-enforcing, non-ordering. A team can say what it is for. C claims "audit-ready", which is a claim about authority that its own metrics do not support.
5. **Cost.** 0.018 s/case and a tiny net: S can be re-run on every new log within minutes, enabling tight discovery–design loops.

## 3.2 Strongest case against Model S

S is not "robust to missing events" in any compliance-relevant sense. It is *indifferent* to missing events, and for the specific events that matter (mandatory checks), indifference is indistinguishable from compliance. A control that cannot tell "Credit Check was performed and not logged" from "Credit Check was never performed" is not a control; it is a screen.

Concretely, S **converts an evidence gap into an apparent compliance signal**. This is the worst failure mode available to a governance board: silent, systematic, and undetectable from the metrics.

## 3.3 Concrete admitted behaviors (traces, not prose)

| # | Trace admitted by S at cost 0 | Why it is undesirable / policy-ambiguous | Policy |
|---|---|---|---|
| A1 | `Register Dossier  Manager Approval  Release Order` | No credit or technical check at all; released to market | violates P2 (probe B3) |
| A2 | `Register Dossier  Auto Approval  Release Order` | Same, via the other approval route; auto-approval on a dossier with zero checks | violates P2, P5 used as an amplifier |
| A3 | `Register Dossier  Expedite Review  Credit Check  Technical Review  Manager Approval  Release Order` | Urgent review before the checks it presumably depends on | violates P4 (probe B4) |
| A4 | `Register Dossier  Supplier Review  Expedite Review  Credit Check  Technical Review  Auto Approval  Release Order` | Both optional reviews precede both mandatory checks; also auto-approval + expedite + supplier in one order S cannot distinguish from a legitimate one | violates P4; spurious (P4 vs P3 interaction) |
| A5 | `Register Dossier  Credit Check  Technical Review  Manager Approval  Release Order` **for an external order whose Supplier Review event was lost in logging** | S cannot distinguish "internal order" from "external order with a lost event"; F4 says such losses exist. S certifies both identically | violates P3 (silently) |
| A6 | `Register Dossier  Technical Review  Supplier Review  Credit Check  Expedite Review  Auto Approval  Release Order` | Interleavings of supplier review *inside* the mandatory check window are accepted as normal | violates P2/P4 intent; unmonitorable |
| A7 | `Register Dossier  Credit Check  Technical Review  Correct Dossier  Technical Review  Manager Approval  Release Order` | **Rejected** by S (no `Correct Dossier` in the model; 1 deletion cost) although P6 makes it potentially legitimate | false rejection of legal behavior (probe B6) |
| A8 | `Register Dossier  Credit Check  Technical Check  Credit Check  Manager Approval  Release Order` | Rejected *incidentally* (branch transitions fire at most once) rather than by an explicit suspicious-repeat rule | P7 unenforced in principle |

**A7/A8 are the uncomfortable point: S is not merely permissive, it is asymmetrically permissive.** It over-accepts on mandatory checks and ordering (A1–A6) while under-accepting on corrections (A7). "Permissive" therefore mis-describes S; the accurate description is *semantically under-specified*: it captures "the four labels plus one approval" and nothing else.

## 3.4 Intentional noise tolerance vs. accidental semantic flaw

| Intentional (defensible under E1) | Accidental (defect under every use case) |
|---|---|
| Absorbing missing events so the model matches a noisy log | Making **mandatory** checks skippable (P2) — this is not noise tolerance, it is a policy inversion |
| Tolerating activity ordering variance in a genuinely concurrent phase | Imposing *no* order where the policy imposes one (Expedite after both checks, P4) |
| One-token-per-activity, so every observed label is present exactly once | Dropping `Correct Dossier` from the language, making legitimate rework unexplainable (P6, F6) |
| Keeping the net small so it stays readable and fast | Silently dropping the external/internal condition (P3), converting "conditionally required" into "never required" |
| A cheap exploratory sketch | Being presented to the board as a candidate *control* |

# 4. Adversarial Review of Model C

## 4.1 What Model C actually is

`X` over eight sequences. Its admitted behavior is the union of exactly eight traces (plus their trace-level alignments). Enumerating its commitments: check order {Credit-first, Tech-first}; Supplier Review {absent, present}; Expedite Review {absent, present}; approval {Manager, Auto}; correction {one rigid variant}. Note the couplings the eight branches silently introduce:

- **Auto Approval occurs only in one branch**, and only with Credit-first, no Supplier, no Expedite. P5 says nothing about check order or reviews. This coupling is invented.
- **Expedite Review occurs only with Credit-first** (branches 5, 6). Tech-first + Expedite is *not* in the model.
- **Supplier Review before Expedite only**; the reverse ordering is not representable, although P4 explicitly permits both.
- **Correction exists in exactly one branch**, and that branch forces *both* checks to repeat, with Credit-first and Manager Approval. P6 requires that *only affected* checks repeat.

C is therefore not a policy model. It is a **catalogue of eight observed variants**, and the branch set encodes correlations that have no basis in P1–P7.

## 4.2 Strongest case for Model C

1. **It is the only artifact with a directly checkable control property.** Every branch visibly contains both mandatory checks before a single approval. The negative probes B3, B4, B8 are all correctly rejected. For a regulator-facing claim ("no order is released without both checks"), C *can* be read by a human to demonstrate it; S demonstrably cannot.
2. **Deterministic verdicts.** With no silent branches, a given log trace either aligns at zero cost or it does not — no alignment-path ambiguity. That is exactly the property you want when the alignment result is used as *evidence* in an audit file (F5's underlying worry, though F5 attacks the wrong metric).
3. **Audit-readiness is a real requirement.** In regulated processes, a model that enumerates and fixes the approved variants is a legitimate artifact — "the board approved these eight order shapes" — provided it is labelled as an *approved-variants catalogue*, not as a complete policy.
4. **It at least models rework.** C is not blind to `Correct Dossier` (branch 7); S is entirely blind.
5. **High training precision (0.941)** is genuine evidence of *structural faithfulness to observed data*: C's alignments rarely need insertions or deletions, which is useful for variant analytics (E6) once node identity is repaired.

## 4.3 Strongest case against Model C

1. **Branch enumeration is a sample, not a specification.** The legal space under P1–P7 (ignoring rework variants) is roughly: 2 check orders × 2 supplier × 2 expedite × 2 approval = 16 baseline shapes, plus 4 correction patterns  ~64, and P4's "expedite before *or* after supplier" doubles the review-ordering dimension. C enumerates 8 of these and then **asserts that the other ~56 are illegal**. That assertion is false, and the holdout drop (0.9730.744) is the measurement of that falsehood.
2. **It rejects legal behavior (F2), concretely.** Probe B2 (`Technical  Credit  Auto Approval`) is legal under P2 and P5 but is rejected by C — solely because C's single Auto branch is Credit-first. Probe B5 (Supplier + Expedite + Auto) is rejected for the same reason. In a gating deployment these are blocked legitimate orders, and the operators will route around the control.
3. **Duplicate labels (33) are a structural defect, not a cosmetic one.** With 33 nodes sharing 9 labels: (i) alignment sync attribution becomes ambiguous (which node was executed?), so C's 0.941 precision is not attributable to individual nodes; (ii) impact analysis ("if we change Supplier Review, what breaks?") requires label-level, not node-level, reasoning and produces false matches; (iii) ownership cannot be assigned to a branch; (iv) play-out assigns each of the 8 branches probability 1/8, which is a fabricated distribution for simulation (E5).
4. **Precision is probably a memorization artifact.** F5: precision was computed only on observed positive traces. If the eight branches were *enumerated from* those traces, the metric is close to tautological. The discriminating experiment is: (i) compute precision on positives *not* used for enumeration (leave-one-cell-out), and (ii) measure the fraction of training traces that match a branch exactly. If that fraction is 1, C is a lookup table with a Petri-net costume. The generalization drop is consistent with this hypothesis; it is not yet proof.
5. **It inherits S's worst blind spot while looking rigorous.** Because C contains both branches *with* and *without* Supplier Review, it silently "optionalizes" Supplier Review with no condition on external/internal. So an external order with no supplier review aligns at cost 0 — the identical control failure as S, now disguised by precision and branch structure. **Both models fail P3 in exactly the same way; C just looks more precise.**
6. **Correction semantics are wrong in substance.** Branch 7 requires *both* checks to repeat after a correction, directly contradicting P6's "only the checks affected by the correction should repeat." It therefore **accepts** an over-recheck (adversarial trace T5 below) and **rejects** every partial-recheck variant. F6 is fully unresolved: C *appears* to handle rework and does not.
7. **Maintainability (F2, F3).** Any policy change or newly observed combination triggers branch re-enumeration; the branch count multiplies (adding the Auto-approval dimension alone would take 8  16, and adding correction patterns  32+). Each addition re-introduces duplicate labels and inflates the arc-degree simplicity metric, so the artifact's headline numbers get *worse* as it becomes *more correct*. The metric penalizes the remedy.
8. **Runtime is a symptom, not a cost.** 0.440 s vs 0.018 s is 24× for 8 branches; the growth is combinatorial, and alignment time on a large log is a production concern for E3.

# 7. Third Alternative

**Chosen strategy:** *a block-structured procedural backbone (POWL) + attribute-conditioned guard layer (declarative policy checks) + an explicit `UNDETERMINED` verdict class*, with per-case specialization into attribute-conditioned nets where the attribute is known. This is a hybrid of options (b) and (d) in the brief. Rationale: the *structure* (ordering, cardinality, exactly-one) is fully expressible in a compact procedural net and should be enforced there; the *data-conditional* facts (external/internal; which check a correction affected) cannot be learned from the log and must be evaluated at decision time by an explicit, testable rule.

### 7.1 Design commitments (derived from the pathology analysis)

1. **No `tau` branch on any mandatory activity.** The only silent transitions in the model are the *documented* skips of two genuinely optional tasks, and the loop-exit of the correction stage. A missing Credit Check must cost something.
2. **One activity label may appear more than once only with a typed reason**, and the number of duplicates is capped at 2 (Credit Check, Technical Review: initial execution vs. re-execution). No whole-process branch enumeration.
3. **Approval is decoupled** from check order and reviews (fixes C's coupling; satisfies P5 minimally).
4. **Expedite Review is unordered w.r.t. Supplier Review** but strictly after the mandatory checks (satisfies P4 exactly).
5. **The guard can only *escalate*, never silently certify.** Missing data  `UNDETERMINED`, never `COMPLIANT`.
6. **P7 is a first-class output** ("suspicious repeat"), not a side effect of the net topology.

### 7.2 Compact formal description

```
Third := Register  Checks  CorrectionStage  Supplier  Expedite  Approval  Release

Checks          := {Credit Check, Technical Review}            both required, unordered      (P2)
CorrectionStage := (Correct Dossier  Recheck)*               0..n iterations                 (P6)
Recheck         := {Credit} | {Technical} | {Credit, Technical}  exactly the affected subset
Supplier        := Supplier Review |                           only when justified by attr  (P3)
Expedite        := Expedite Review |                           genuinely optional            (P4)
Approval        := Manager Approval  Auto Approval             exactly one                     (P5)
Release         := Release Order                                 last                            (P1)
```

Where `` (silent skip) is *not* a claim of optionality: it is a placeholder whose legality is decided by the guard, and whose presence/absence is always reported in the audit record.

Ordering edges: `Register < Checks < CorrectionStage < {Supplier, Expedite} < Approval < Release`, with **no** edge between `Supplier` and `Expedite` (P4 permits either order).

### 7.3 Self-contained PM4Py code

```python
# ---------------------------------------------------------------
# model_third : attribute-aware procedural backbone + policy guard
# ---------------------------------------------------------------
import pm4py
from pm4py.objects.powl import Transition, SilentTransition, OperatorPOWL, StrictPartialOrder


def _optional(activity_name, skip_name):
    """Exactly one of: the activity occurs, or an explicit (silent) skip is recorded.
    The silent branch is a PLACEHOLDER whose legality is decided by the guard layer.
    It is never evidence that the activity was 'not required'."""
    return OperatorPOWL(OperatorPOWL.XOR,
                        [Transition(name=activity_name),
                         SilentTransition(name=skip_name)])


def build_third(is_external=None):
    """
    is_external = None  -> generic backbone (guard decides Supplier Review legality)
    is_external = True  -> specialised net: Supplier Review is MANDATORY
    is_external = False -> specialised net: Supplier Review is optional (silent skip allowed)
    """
    register = Transition(name='Register Dossier')

    # --- P2: mandatory checks, either order, NOT skippable (no tau) -------------
    credit_1 = Transition(name='Credit Check')      # initial execution
    tech_1   = Transition(name='Technical Review')   # initial execution
    checks   = StrictPartialOrder(nodes=[credit_1, tech_1])

    # --- P6: correction / recheck stage -----------------------------------------
    # The re-check occurrences deliberately REUSE the log labels, so alignment sees
    # one event stream. Only 2 duplicate labels exist (Credit, Technical), and they
    # are semantically typed: occurrence #1 = initial, #2+ = re-check.
    correct  = Transition(name='Correct Dossier')
    credit_2 = Transition(name='Credit Check')       # re-execution
    tech_2   = Transition(name='Technical Review')    # re-execution
    recheck  = OperatorPOWL(OperatorPOWL.XOR, [
        StrictPartialOrder(nodes=[credit_2]),                  # credit only
        StrictPartialOrder(nodes=[tech_2]),                   # technical only
        StrictPartialOrder(nodes=[credit_2, tech_2]),          # both
    ])
    # 0..n corrections.  NOTE: PM4Py POWL loops may break at ANY point of the body;
    # the guard layer therefore independently rejects 'Correct Dossier' with no
    # following re-check (P6).  If your PM4Py version exposes OperatorPOYL
    # (exit only at end of body), prefer it and the guard becomes belt-and-braces.
    correction_stage = OperatorPOWL(OperatorPOWL.LOOP, [
        StrictPartialOrder(nodes=[correct, recheck]),
    ])

    # --- P3: Supplier Review ---------------------------------------------------
    if is_external is True:
        supplier = Transition(name='Supplier Review')     # mandatory for external
    else:
        supplier = _optional('Supplier Review', 'supplier_review_skipped')

    # --- P4: Expedite Review (optional, after both checks, before approval) ----
    expedite = _optional('Expedite Review', 'expedite_not_required')

    # --- P5: exactly one approval, decoupled from everything else -------------
    approval = OperatorPOWL(OperatorPOWL.XOR, [
        Transition(name='Manager Approval'),
        Transition(name='Auto Approval'),
    ])

    release = Transition(name='Release Order')

    model = StrictPartialOrder(nodes=[register, checks, correction_stage,
                                      supplier, expedite, approval, release])
    model.order.add_edge(register, checks)
    model.order.add_edge(checks, correction_stage)
    model.order.add_edge(correction_stage, supplier)
    model.order.add_edge(correction_stage, expedite)   # P4: after BOTH mandatory checks
    model.order.add_edge(supplier, approval)            # P4: no edge supplier<->expedite
    model.order.add_edge(expedite, approval)
    model.order.add_edge(approval, release)
    return model


model_third = build_third(is_external=None)          # generic backbone (default)
model_third_external = build_third(is_external=True) # specialised: Supplier mandatory
```

### 7.4 Declarative policy dictionary and guard layer

```python
POLICY = {
 'P1': {'first': 'Register Dossier', 'last': 'Release Order',
        'cardinality': {'Register Dossier': 1, 'Release Order': 1}},
 'P2': {'mandatory_before_approval': ['Credit Check', 'Technical Review'], 'order': 'free'},
 'P3': {'activity': 'Supplier Review', 'condition_attr': 'is_external',
        'when_true': 'required', 'when_false': 'not_required'},
 'P4': {'optional': ['Expedite Review'], 'after': ['Credit Check', 'Technical Review'],
        'before_approval': True, 'order_vs_supplier': 'free'},
 'P5': {'exactly_one_of': ['Manager Approval', 'Auto Approval']},
 'P6': {'correction': 'Correct Dossier', 'recheck': 'only_affected',
        'attr': 'affected_checks'},
 'P7': {'repeat_without_correction': 'suspicious'},
}

MANDATORY = {'Credit Check', 'Technical Review'}
APPROVALS = {'Manager Approval', 'Auto Approval'}


def policy_check(trace, case_attrs=None):
    """trace: list of event dicts with 'concept:name' (+ optional 'affected_checks').
    Returns verdict in {COMPLIANT, VIOLATION, UNDETERMINED} plus reason codes.
    The guard NEVER silently certifies: missing data => UNDETERMINED."""
    acts = [e['concept:name'] for e in trace]
    attrs = dict(case_attrs or {})
    if trace:
        attrs.update({k: v for k, v in trace[0].items()
                      if k not in ('concept:name', 'concept:timestamp')})
    is_external = attrs.get('is_external', None)          # may be None -> unknown
    V, U, I = [], [], []

    # ---- P1 ------------------------------------------------------------------
    if not acts or acts[0] != 'Register Dossier':
        V.append('P1:first_event')
    if not acts or acts[-1] != 'Release Order':
        V.append('P1:last_event')
    for a in ('Register Dossier', 'Release Order'):
        if acts.count(a) != 1:
            V.append('P1:cardinality:%s' % a)

    # ---- P5 ------------------------------------------------------------------
    app = [i for i, a in enumerate(acts) if a in APPROVALS]
    if len(app) != 1:
        V.append('P5:exactly_one_approval')
    rel = acts.index('Release Order') if 'Release Order' in acts else len(acts)
    app_i = app[0] if app else rel
    if app and app[0] > rel:
        V.append('P5:approval_after_release')

    # ---- P2 ------------------------------------------------------------------
    for m in MANDATORY:
        if m not in acts:
            V.append('P2:missing:%s' % m)
        for i, a in enumerate(acts):
            if a == m and i > app_i:
                V.append('P2:check_after_approval:%s' % m)

    # ---- P6 / P7 -------------------------------------------------------------
    for i, a in enumerate(acts):
        if a not in MANDATORY:
            continue
        if i == acts.index(a):
            continue                                  # first (initial) execution
        # a repetition: must be preceded by a correction
        corr = [j for j in range(i) if acts[j] == 'Correct Dossier']
        if not corr:
            V.append('P7:repeat_without_correction:%s' % a)
        else:
            last = corr[-1]
            affected = _affected_after(trace, last)     # None if attribute missing
            repeated = set(x for x in acts[last + 1:i + 1] if x in MANDATORY)
            if affected is None:
                U.append('P6:affected_checks_unknown')
            elif set(affected) != repeated:
                V.append('P6:recheck_set_mismatch:%s' % a)
    for i, a in enumerate(acts):
        if a == 'Correct Dossier':
            seg = acts[i + 1:app_i]
            if not any(s in MANDATORY for s in seg):
                V.append('P6:correction_without_recheck')

    # ---- P3 ------------------------------------------------------------------
    if 'Supplier Review' not in acts:
        if is_external is True:
            V.append('P3:supplier_review_missing_but_external')
        elif is_external is None:
            U.append('P3:supplier_review_absent_and_attribute_missing')
    else:
        if is_external is False:
            I.append('P3:supplier_review_on_internal_order')   # policy reading: not forbidden

    # ---- P4 ------------------------------------------------------------------
    if 'Expedite Review' in acts:
        e = acts.index('Expedite Review')
        for m in MANDATORY:
            if m in acts and acts.index(m) > e:
                V.append('P4:expedite_before:%s' % m)
        if e > app_i:
            V.append('P4:expedite_after_approval')

    verdict = 'VIOLATION' if V else ('UNDETERMINED' if U else 'COMPLIANT')
    return {'verdict': verdict, 'violations': V, 'undetermined': U, 'info': I}
```

**Declared semantics (item 4 of the brief), explicitly:**

| Requirement | Procedural layer | Guard layer | Residual risk |
|---|---|---|---|
| **Mandatory checks** | Credit Check and Technical Review are non-silent, both in an unordered `StrictPartialOrder`; no skip branch exists. A trace lacking either cannot align at cost 0. | Confirms presence and pre-approval position (P2). | None for presence; alignment cost is the only signal if the log is incomplete |
| **Optional Supplier Review** | XOR(activity, silent placeholder). Silent branch  "not required" | `is_external=True` + missing  **VIOLATION**; `is_external=None` + missing  **UNDETERMINED**; `is_external=False` + present  INFO (policy reading: P3 says "required *only* for external", so presence on an internal order is atypical, not illegal — *must be confirmed with the policy owner; one-line change if the intended reading is "forbidden for internal"*) | If `is_external` never arrives, Supplier Review stays permanently UNDETERMINED  must be fixed as a data workstream, not a model change |
| **Expedite Review ordering** | Positioned after the correction stage (hence after both mandatory checks **and** after any re-check) and before the approval XOR; **no** edge to Supplier Review | Verifies position and pre-approval position (P4) | Ordering relative to Supplier Review is unconstrained, as P4 requires |
| **Exactly one approval** | XOR over two non-silent transitions; no skip branch; no coupling to anything else | Counts approvals (P5) | None |
| **Correction / re-check** | LOOP over `Correct Dossier  {Credit | Technical | Credit+Technical}`; 0..n corrections; **re-check subset is a modeled choice, not inferred from the log** | P6 minimality: re-executed set must equal `affected_checks`; without the attribute  UNDETERMINED; a correction with no re-check  VIOLATION | POWL loop break-at-any-point permits `Correct Dossier` with no re-check in the *net*; the guard rejects it. Documented limitation |
| **Suspicious repeat (P7)** | Not structurally possible for the *initial* checks (each fires once) | Explicit `P7:repeat_without_correction` reason code | In the re-check branch, repeats *are* possible; the guard distinguishes them via the correction context |

### 7.5 Limitations caused by the missing attributes (stated, not papered over)

1. **`is_external` missing.** The procedural layer cannot make Supplier Review conditionally mandatory *as learned*; only the guard can, per case. Until the attribute is available, every case lacking Supplier Review is `UNDETERMINED`, **not** compliant. This is deliberate: it is the explicit form of "do not silently treat every activity as optional".
2. **`affected_checks` missing.** P6 minimality (only affected checks repeat) is **unverifiable**. The model can check *internal consistency* (a correction is followed by at least one re-check; no repeat without a correction — P7) and can *enumerate* the three legal re-check shapes, but it cannot certify that the chosen subset was the affected one. The correct output is `UNDETERMINED` with reason `P6:affected_checks_unknown`.
3. **Consequence for metrics.** Any fitness/precision figure for the third model is still computed against an incomplete log; the policy-conformance suite (§9.3) is therefore the primary evidence, not fitness.
4. **Consequence for the guard.** The `UNDETERMINED` rate must be an operational KPI. If it is high, the fix is attribute backfill, not model redesign.
5. **Loop semantics.** PM4Py's POWL loop permits early break; a version with `OperatorPOYL` (exit only at the end) would remove the need for the guard to cover "correction without re-check". Verify availability in the deployed version.
6. **Duplicate labels.** Exactly two labels are duplicated (Credit Check, Technical Review), each twice, semantically typed as initial vs. re-check, with the convention **occurrence #1 = initial**. This must be published, because alignment node attribution and impact analysis depend on it. (Contrast C: 33 duplicates, untyped, arising from whole-process branching.)

```python
from pm4py.objects.petrinet.utils import apply_transition

def enumerate_behaviour(net, im, fm, max_len=14, max_traces=20000, silent_label='<silent>'):
    """BFS over reachable markings; yields observable activity sequences (bounded)."""
    start = im
    seen, out, queue = {frozenset(start.items()): ()}, [], [(start, ())]
    while queue:
        marking, seq = queue.pop()
        if len(seq) >= max_len or len(out) >= max_traces:
            continue
        for t in net.transitions:
            try:
                nm, nvis = apply_transition(net, im, fm, t, enable_activity=True)
            except Exception:
                continue
            key = frozenset(nm.items())
            nseq = seq + ((t.label,) if t.label and t.label != silent_label else ())
            if key not in seen:
                seen[key] = nseq
                queue.append((nm, nseq))
                if t.label and t.label != silent_label:
                    out.append(nseq)
    return out
```

Then: `illegal_admitted = [s for s in behaviour if policy_check(list_of_events(s))['verdict'] == 'VIOLATION']` and `legal_rejected = [s for s in legal_grid_traces if not accepted]`. **Version note:** `apply_transition` and silent-transition labelling differ across PM4Py releases — pin the version and verify the enumerator against a hand-checked trace before trusting it.

### 9.9 Stakeholder comprehension protocol

* **Participants:** 18, stratified across compliance (4), operations (5), IT architecture (3), internal audit (3), business/requestor (3); no one who authored either model.
* **Instrument:** 8 task-based items derived from P1–P7 and B1–B8, e.g., *"This order was expedited, then credit-checked, then released. Under the policy, is this order correctly handled? Yes/No/Unsure."* Plus 2 items on the *model artifact* ("how many ways can this order be approved?").
* **Metrics:** accuracy per item, median time-to-answer, "Unsure" rate (a proxy for false confidence), and confidence calibration. A high confidence with a wrong answer is worse than a slow correct answer.
* **Design:** paired within-subject, randomized model order, warm-up task excluded; pre-register the analysis plan; report per-role accuracy, not just a mean.
* **Explicitly:** comprehension is a *communication* criterion (E4). It must never be traded against soundness (E2); a target of 0.8 task accuracy and 1 min median time on the target audience.

### 9.10 Decision rule — reject despite good aggregate metrics

A model is **rejected for compliance use** if **any** of the following holds, regardless of fitness, precision or simplicity:

1. **Any** trace in `NEGATIVE_SUITE` is accepted at zero cost (hard gate; a single false acceptance is disqualifying).
2. False-rejection rate on the legal suite > 5% **and** the rejections are not explained by a documented, attribute-driven rule.
3. Corruption-detection rate < 1.0 for the removal of a mandatory check.
4. `UNDETERMINED` rate on real cases > 20% at the time of deployment (for a gating use case).
5. Trainholdout fitness gap > 0.05 across 5 seeds *and* no explanation other than "unseen combinations" — because that is a completeness failure by construction.
6. Precision is reported **only** on positive observed traces (process failure: the metric is not admissible as compliance evidence).
7. No published seeds/manifest/versions/costs (process failure: results are not reproducible; the submission is not admissible).
8. Duplicate labels without a published occurrence-index convention and an impact-analysis procedure.
9. Comprehension task accuracy < 0.8 in a role that will operate the control.

# 11. Dissenting View

**Rebuttal (210 words).**

The recommendation may be an over-engineered answer to a data problem. The third alternative's decisive advantage — conditional Supplier Review and P6 minimality — depends entirely on `is_external` and `affected_checks`, and the dossier contains neither. Its guard layer will therefore emit `UNDETERMINED` for exactly the cases the business most needs decided: external orders whose Supplier Review event was lost in logging, and corrections whose scope is unknown. A control that answers "unknown" to a large fraction of real dossiers is, operationally, a control that has been switched off, and its 0.5-page of policy code plus a specialized-model factory is a maintenance surface far larger than Model C's eight branches. Model C, conversely, may be more right than the evidence allows: if this is a tightly controlled dossier process where deviations genuinely require sign-off, then "unseen behavior is non-compliant until approved" is the correct governance stance, and its 0.744 holdout fitness may be a *split artifact* rather than overfitting — one seed was reported, no manifest was published, and a single re-split could move that number substantially. Finally, Model S's permissiveness may be the honest reading of reality: if the log is incomplete, then a strict model flags the *log* rather than the business, and a model that rejects 25% of real orders for missing Supplier Review events (F4) would destroy trust faster than it would improve control.

**What would change my decision.** (i) Evidence that `is_external` and `affected_checks` can be recovered, derived, or reliably backfilled — then the conditional design is not speculative and becomes the right answer. (ii) A re-splitting study (5 seeds, published manifests) showing C's holdout fitness is stable near 0.74 rather than collapsing further — that would confirm completeness failure. (iii) Demonstration that the third model's `UNDETERMINED` rate cannot be brought below a threshold (e.g., 20%) within an acceptable backfill effort, which would make a staged rollout with C-as-fixture and S-as-exploration the pragmatic choice. (iv) A policy ruling that P3's condition is derivable from existing fields, or that Supplier Review is optional for *all* orders — which would collapse the third model's advantage. (v) Evidence that no curated negative suite can be maintained, in which case policy conformance cannot be measured at all and neither model may be used for enforcement; the board's only honest options would then be exploratory use plus manual audit sampling.
