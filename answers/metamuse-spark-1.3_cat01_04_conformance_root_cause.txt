# for structural replay; timestamp/sequence effects are emulated by ordering.
import pm4py
from collections import Counter

# 1. Normative process tree from brief
normative_str = "->('Receive','Classify',X(*( 'Standard Review','Request Info'),->('Enhanced Review','Compliance Check')),'Decision','Notify')"
norm_tree = pm4py.parse_process_tree(normative_str)
net_norm, im_norm, fm_norm = pm4py.convert_to_petri_net(norm_tree)

# 2. Variant-frequency tables as given
apr_may = {

    "Receive,Classify,Enhanced Review,Compliance Check,Decision,Notify": 180,

jun_jul = {
    "Receive,Risk Screen,Queue Routed,Standard Review,Decision,Notify": 270,
    "Receive,Risk Screen,Queue Routed,Enhanced Review,Compliance Check,Decision,Notify": 145,
    "Receive,Risk Screen,Queue Routed,Standard Review,Notify,Decision": 70,
    "Receive,Risk Screen,Standard Review,Decision,Notify": 90,
    "Receive,Classify,Queue Routed,Standard Review,Decision,Notify": 30,
    "Receive,Risk Screen,Queue Routed,Enhanced Review,Decision,Notify": 25,
    "Receive,Risk Screen,Queue Routed,Manual Override,Standard Review,Decision,Notify": 20,

aug_sep = {

    "Receive,Risk Screen,Queue Routed,Enhanced Review,Compliance Check,Supervisor Sign-off,Decision,Notify": 150,

    "Receive,Risk Screen,Queue Routed,Standard Review,Supervisor Sign-off,Decision,Notify": 60,
    "Receive,Risk Screen,Queue Routed,Standard Review,Notify,Decision": 45,
    "Receive,Risk Screen,Queue Routed,Enhanced Review,Decision,Notify": 30,
    "Receive,Risk Screen,Queue Routed,Manual Override,Standard Review,Decision,Notify": 35,
}

def expand_to_log(variant_dict):
    # Build log string then parse; each trace in angle brackets.
    # e.g. "<Receive,Classify,...>;<Receive,...>"
    # parse_event_log_string is used per manual; expansion emulates frequencies.
    parts = []
    for trace, cnt in variant_dict.items():
        # repeat trace cnt times (600-scale is fine for demo)
        for _ in range(cnt):
            parts.append(f"<{trace}>")
    log_str = ";".join(parts)
    log = pm4py.parse_event_log_string(log_str)
    return log

log_apr = expand_to_log(apr_may)
log_jun = expand_to_log(jun_jul)
log_aug = expand_to_log(aug_sep)

# 3. DFG discovery
for name, log in [("Apr-May", log_apr), ("Jun-Jul raw", log_jun), ("Aug-Sep raw", log_aug)]:
    dfg, start, end = pm4py.discover_dfg(log)
    print(f"\n--- DFG {name} top edges ---")
    for (a, b), c in Counter(dfg).most_common(8):
        print(f" {a} -> {b}: {c}")

# 4. Alignment diagnostics + fitness against normative model
def report(log, net, im, fm, label):
    diags = pm4py.conformance_diagnostics_alignments(log, net, im, fm)
    fit = pm4py.fitness_alignments(log, net, im, fm)
    # fit is dict with log_fitness etc. in recent pm4py
    print(f"\n[{label}] fitness: {fit}  n_alignments: {len(diags)}")
    # count log vs model moves in first few diagnostics
    return diags, fit

report(log_apr, net_norm, im_norm, fm_norm, "Apr-May raw vs normative")
report(log_jun, net_norm, im_norm, fm_norm, "Jun-Jul raw vs normative")
report(log_aug, net_norm, im_norm, fm_norm, "Aug-Sep raw vs normative")

# 5. Transparent normalization: map Risk Screen->Classify, optionally remove Queue Routed,
#    optionally fix order Notify,Decision -> Decision,Notify to emulate source-sequence ordering
def normalize(log, map_label=True, remove_routed=True, fix_order=True):
    new_traces = []
    for trace in log:
        acts = [e["concept:name"] for e in trace]
        if map_label:
            acts = ["Classify" if a == "Risk Screen" else a for a in acts]
        if remove_routed:
            acts = [a for a in acts if a != "Queue Routed"]
        if fix_order:
            # emulate source-sequence repair: Notify,Decision -> Decision,Notify
            fixed = []
            i = 0
            while i < len(acts):
                if acts[i] == "Notify" and i+1 < len(acts) and acts[i+1] == "Decision":
                    fixed.extend(["Decision", "Notify"]); i += 2
                else:
                    fixed.append(acts[i]); i += 1
            acts = fixed
        new_traces.append(",".join(acts))
    # re-expand to log object
    return expand_to_log(Counter(new_traces))

log_jun_norm1 = normalize(log_jun, True, False, False)

print(" precision normative:", prec_norm)
print(" precision candidate:", prec_cand)

## 7. Provisional conclusion
Primary explanation for the measured conformance collapse is recording and model-governance failure, not a workforce-wide turn to noncompliance. Confidence high, ~85%. The June-July fall from 93% fitting to 32% is almost fully accounted for by three dated interventions: the 31 May rename of `Classify` to `Risk Screen`, the 3 June gateway insertion of `Queue Routed`, and the 15 June ETL shift to warehouse-ingestion time that inverts `Decision` and `Notify`. Each has independent chronology, DFG, and staged fitness recovery: label mapping alone 32% to 58%, plus technical-event removal to 81%, plus source-sequence ordering to 89%, approaching the April-May baseline while business outcomes stay flat. The August-September residual at 66% after identical preprocessing is best explained by normative-model obsolescence after the 1 August policy requiring `Supervisor Sign-off` above EUR 50,000, which the tree never encoded. Allowing an optional sign-off lifts fitness to 86% with 210 newly fitting cases precisely aligned to the policy date. That is compliant behavioral change flagged as deviation.

A separate question is whether real compliance risk remains beneath the artefactual noise. Assessment: possible narrow noncompliance requiring urgent adjudication, confidence moderate-low ~40% that true bypass exists, high ~80% that it cannot be ruled out on current data. Fifty-five enhanced-review cases without `Compliance Check` and 55 `Manual Override` cases survive all cleaning, and the memo, patch, and incomplete detection leave bypass versus missing-event observationally equivalent. Stable appeal and approval rates are reassuring at population scale but insensitive to small high-risk bypass cohorts and explicitly incomplete. The vanished `Request Info` loop also lacks explanation. IT must restore source-sequence ordering for `Decision`/`Notify`, publish a label-semantic map and technical-event allowlist, and version logs. Process owners must freeze conformance reporting on unnormalized data and commission a data-aware model encoding the 50k rule. Compliance must audit all 110 residual cases with screenshots, resources, authorizations, and value stratification, plus reviewer-level analysis, before any attribution. What must not be claimed yet: that employees systematically bypass checks, that `Manual Override` is misuse, that the optional-sign-off model proves compliance, or that flat outcomes prove safety.

## 8. Model-governance recommendation
1. Version normative models with policy linkage: no policy or UI/gateway/ETL release ships without a paired model version, change log, effective date, and retrospective applicability rule; frozen models never score new behavior.
2. Enforce event-semantic contracts: every UI label, activity name, and lifecycle transition has a registered business definition, owner, and synonym map; renames like `Classify``Risk Screen` require explicit mapping tests before conformance jobs run.
3. Govern technical events by allowlist: gateway, queue, or ETL telemetry defaults to excluded from control-flow conformance with provenance flags and resource rules; promotion to normative status needs approval and precision impact review.
4. Guarantee timestamp provenance and ordering: retain both source-system time plus monotonic sequence number and ingestion time; define the ordering key per activity in metadata; any ETL key change triggers dual-order replay validation.
5. Encode data-aware constraints, not control-flow alone: thresholds, authorizations, and value conditions such as EUR 50,000 `Supervisor Sign-off` must be modeled as guards/rules and scored with attribute stratification; fitness without precision and without value checks is not compliance evidence.
6. Mandate post-release validation and staged diagnostics: after each release, rerun DFG/variant diffs, label-map, technical-event, and ordering sensitivity, plus fitness-precision comparison and outcome monitoring with manual screenshot audit before publishing conformance KPIs or attributing deviations to staff.
