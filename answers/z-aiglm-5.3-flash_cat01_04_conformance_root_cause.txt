"""

from collections import Counter
import pandas as pd
import pm4py
from pm4py.objects.process_tree.obj import ProcessTree, Operator

# ---------------------------------------------------------------- 1. Models
NORMATIVE_TREE_STR = (
    "-> ( 'Receive', 'Classify', "
    "X ( * ( 'Standard Review', 'Request Info' ), "
    "-> ( 'Enhanced Review', 'Compliance Check' ) ), "
    "'Decision', 'Notify' )"
)

normative_tree = pm4py.parse_process_tree(NORMATIVE_TREE_STR)
net, im, fm = pm4py.convert_to_petri_net(normative_tree)


def _leaf(label):
    return ProcessTree(label=label)

def _tau():
    return ProcessTree()

def _xor(children):
    n = ProcessTree(operator=Operator.XOR)
    for c in children:
        c.parent = n
        n.children.append(c)
    return n

def _seq(children):
    n = ProcessTree(operator=Operator.SEQUENCE)
    for c in children:
        c.parent = n
        n.children.append(c)
    return n


def build_candidate_tree_with_optional_signoff():
    """
    Normative flow with an optional 'Supervisor Sign-off' immediately before
    'Decision'. Purely control-flow: it CANNOT express the EUR-50k guard,
    so it accepts traces with unneeded sign-offs and traces without it;
    fitness will rise partly through overgeneralization.
    """
    review_choice = _xor([
        pm4py.parse_process_tree("* ( 'Standard Review', 'Request Info' )"),
        pm4py.parse_process_tree("-> ( 'Enhanced Review', 'Compliance Check' )"),
    ])
    return _seq([
        _leaf("Receive"), _leaf("Classify"), review_choice,
        _xor([_tau(), _leaf("Supervisor Sign-off")]),
        _leaf("Decision"), _leaf("Notify"),
    ])

candidate_tree = build_candidate_tree_with_optional_signoff()

def evaluate(name, df_or_log, net_, im_, fm_, do_precision=True):
    res = {"scenario": name}
    try:
        algs = pm4py.conformance_diagnostics_alignments(df_or_log, net_, im_, fm_)

        res["alignments"] = len(algs)
        perfect = sum(1 for a in algs if float(a.get("fitness", 0.0)) >= 1.0 - 1e-9)
        res["perfect_traces"] = f"{perfect}/{len(algs)}"
        fit = pm4py.fitness_alignments(df_or_log, net_, im_, fm_)

        res["mean_cost"] = round(sum(c for c in costs if c)/max(1,len(costs)),4)
    except Exception as e:
        res["error"] = repr(e)
    return res

def dominant_deviations(algs, topn=6):
    lm, mm, asy = Counter(), Counter(), 0
    for a in algs:
        if not isinstance(a, dict): continue

        diags = pm4py.conformance_diagnostics_alignments(logobj, model_net, model_im, model_fm)  # req. 5
        perfect = sum(1 for a in diags if isinstance(a, dict) and float(a.get("fitness", 0.0)) >= 1 - 1e-9)
        out["traces"] = len(diags)
        out["optimal"] = perfect
        lm, mm = Counter(), Counter()
        for a in diags:
            for lp, rp in (a.get("alignment") or []):
                l = lp if isinstance(lp, str) else None
                r = rp if (isinstance(rp, str) and rp != ">>") else None
                if l and r and l != r: continue
                if l: lm[l] += 1
                elif r: mm[r] += 1
        out["dominant_log_moves"] = lm.most_common(5)
        out["dominant_model_moves"] = mm.most_common(5)
        fit = pm4py.fitness_alignments(logobj, model_net, model_im, model_fm)
        out["fitness"] = {k: (round(v, 4) if isinstance(v, (int, float)) else v) for k, v in fit.items()}
        out["precision_alignments"] = round(pm4py.precision_alignments(logobj, model_net, model_im, model_fm), 4)
    except Exception as exc:
        out["error"] = repr(exc)
    return out

def dfg_summary(tag, logobj, topn=10):                              # requirement 4
    try:

## 7. Provisional conclusion

**Primary explanation for the measured conformance collapse** (confidence: high, 0.85 that the preponderance of the Jun-Jul collapse is sensor-level): a compound measurement artifact deployed in mid-2026. The UI release of May 31 substituted the `Classify` label with `Risk Screen`; the June 3 gateway injection threaded `Queue Routed` into most traces; the June 15 ETL switch inverted `Decision`/`Notify` in exactly 70 traces. The tiered repairs reproduce almost mechanically the diagnostic ladder — mapping recovers +26 points, technical-event removal +23, sequence ordering +8, taking Jun-Jul from 32% to 89%, essentially the April baseline of 93%. Independent evidence (40-case audit, screenshot verification in 11 reversed cases, DFG counts matching variant counts precisely) triangulates each mechanism. Blaming reviewers for the measured drop would confuse the instrument with the subject.

From August, a different, partially real phenomenon enters (confidence: moderate-high, 0.8): the EUR 50k policy created a genuine `Supervisor Sign-off` step that the frozen normative model cannot express, so even fully compliant conduct generates deviations; an optional-branch candidate lifts fitness to 86% but only by overgeneralizing, and its precision cost must be shown before anyone celebrates.

**Separate compliance assessment** (confidence: low, and symmetrically undecided): a residue of 25 (Jun-Jul) and 30 (Aug-Sep) enhanced-review traces without a recorded `Compliance Check` remains. Outcomes are flat and the relative unchecked share actually declined (14.7%11.1%), arguing against an escalating bypass wave; yet extraction visibly degraded in the same window (resource-attribution failures quintupled), so missing events are an equally live explanation, and incident detection is admittedly incomplete.

**Immediate actions** — IT: reconstruct event provenance (origin-type attribute, source timestamps, sequence ordering) and republish conformance only on the normalized log; Process owners: version the model (v1.1) with a data-guarded sign-off and decide formally whether `Manual Override` is a legitimate modeled activity; Compliance: run a dual-source, full-census audit of every no-check enhanced trace before any personnel implication, and commit to filing findings either way.

**What must not be claimed yet**: that employees bypassed compliance checks (no audit); that the process truly deteriorated in June; that restored fitness certifies health; that the optional-sign-off model proves policy adherence; that stable appeals exonerate behavior — detection lags.

## 8. Model-governance recommendation

1. **Model versioning.** Every normative model carries a semantic version, an owner, and a changelog binding it to policy/UI releases that affect its scope. Any process-affecting organizational change triggers a mandatory impact review of the conformance baseline; every published conformance report states the exact model version; superseded models are archived with validity windows so historical fitness figures remain interpretable.
2. **Event-semantic contracts.** Applications emit canonical activity identifiers from a jointly governed registry; UI display labels are presentation-only and may never alter stream semantics. Renames require a registered, versioned alias table and a compatibility smoke test before rollout; unregistered relabelings are treated as data incidents.
3. **Technical-event policy.** Every event carries an immutable `origin_type  {human, system, external}` attribute set at emission. Conformance pipelines exclude system events by default through a centrally maintained filter; enabling a new emitter (like the `Queue Routed` gateway event) requires registration, and orphan system events discovered in logs raise alerts.
4. **Timestamp provenance.** The business-meaningful event time is captured at the source and is immutable; ingestion time is stored in a separate, clearly named column and is never substituted. Conformance jobs assert an ordering-consistency SLA using source sequence numbers, and any clock-domain switch or extractor change must pass an ordering-equality test before touching conformance outputs.
5. **Data-aware constraints.** Policies conditioned on case data (e.g., EUR thresholds) are expressed as executable guards — data-aware extensions or stratified mandatory conformance per attribute band. A control-flow-only model that cannot encode a policy condition is explicitly flagged "not policy-complete," and its fitness figures may not be cited as compliance evidence.
6. **Post-release validation.** An automated regression harness computes a variant fingerprint, DFG distance, and tiered fitness/precision on a shadow window after every UI, ETL, or gateway deployment. Metric ruptures freeze conformance publication and open a triage ticket; release sign-off requires an explicit "intended/unintended" disposition of any observed change before numbers reach compliance consumers.
