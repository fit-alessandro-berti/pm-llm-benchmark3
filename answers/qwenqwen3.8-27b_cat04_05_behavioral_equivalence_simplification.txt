```
def bounded_visible_traces(net, im, fm, max_visible=12, max_analyze=3, analyze_label='Analyze Revision'):
    fm = _normalize(fm)
    traces = set()
    queue = deque([(im, (), 0)])
    visited = set()
    while queue:
        m, trace, ac = queue.popleft()
        m_key = tuple(sorted(m.items()))
        key = (m_key, trace, ac)
        if key in visited:
            continue
        visited.add(key)
        if m == fm:
            if len(trace) <= max_visible and ac <= max_analyze:
                traces.add(trace)
            continue
        for t in pm4py.get_enabled_transitions(net, m):
            label = getattr(t, 'label', None)
            silent = _is_silent(label)
            if not silent:
                if len(trace) >= max_visible:
                    continue
                if label == analyze_label:
                    if ac >= max_analyze:
                        continue
                    new_ac = ac + 1
                else:
                    new_ac = ac
                new_trace = trace + (label,)
            else:
                new_trace = trace
                new_ac = ac
            new_m = _fire(net, m, t)
            if new_m is None:
                continue
            nkey = (tuple(sorted(new_m.items())), new_trace, new_ac)
            if nkey not in visited:
```

```
def normalize_marking(m):
    return {p: c for p, c in m.items() if c > 0}

def is_silent(label):
    if label is None:
        return True
    try:
        return label == pm4py.TAU or str(label) == 'tau'
    except Exception:
        return str(label) == 'tau'

def get_initial_marking(net):
    try:
        return normalize_marking(pm4py.get_initial_marking(net))
    except Exception:
        pass
    for p in net.places:
        if str(p.name) in {'source', 'initial', 'im', 'start'}:
            return {p: 1}
    no_in = [p for p in net.places if not p.inputs]
    if len(no_in) == 1:
        return {no_in[0]: 1}
    raise RuntimeError('Could not infer initial marking')

def get_final_marking(net):
    try:
        return normalize_marking(pm4py.get_final_marking(net))
    except Exception:
        pass
    for p in net.places:
        if str(p.name) in {'sink', 'final', 'fm', 'end'}:
            return {p: 1}
    no_out = [p for p in net.places if not p.outputs]
    if len(no_out) == 1:
        return {no_out[0]: 1}
    raise RuntimeError('Could not infer final marking')

def to_accepting_petri_net(tree):
    res = pm4py.convert_to_petri_net(tree)
    if isinstance(res, tuple):
        if len(res) == 3:
            net, im, fm = res
        elif len(res) == 2:
            net, im = res
            fm = get_final_marking(net)
        else:
            net = res[0]
            im = get_initial_marking(net)
            fm = get_final_marking(net)
    else:
        net = res
        im = get_initial_marking(net)
        fm = get_final_marking(net)
    return net, normalize_marking(im), normalize_marking(fm)

def fire(net, marking, transition):
    m = dict(marking)
    for p in transition.inputs:
        m[p] = m.get(p, 0) - 1
        if m[p] < 0:
            return None
    for p in transition.outputs:
        m[p] = m.get(p, 0) + 1
    return normalize_marking(m)

def bounded_visible_traces(net, im, fm, max_visible=12, max_analyze=3, analyze_label='Analyze Revision'):
    fm = normalize_marking(fm)
    traces = set()
    queue = deque([(im, (), 0)])
    visited = set()
    while queue:
        m, trace, analyze_count = queue.popleft()
        m_key = tuple(sorted(m.items()))
        key = (m_key, trace, analyze_count)
        if key in visited:
            continue
        visited.add(key)
        if m == fm:
            if len(trace) <= max_visible and analyze_count <= max_analyze:
                traces.add(trace)
            continue
        for t in pm4py.get_enabled_transitions(net, m):
            label = getattr(t, 'label', None)
            silent = is_silent(label)
            if not silent:
                if len(trace) >= max_visible:
                    continue
                if label == analyze_label:
                    if analyze_count >= max_analyze:
                        continue
                    new_analyze_count = analyze_count + 1
                else:
                    new_analyze_count = analyze_count
                new_trace = trace + (label,)
            else:
                new_trace = trace
                new_analyze_count = analyze_count
            new_m = fire(net, m, t)
            if new_m is None:
                continue
            nkey = (tuple(sorted(new_m.items())), new_trace, new_analyze_count)
            if nkey not in visited:
                queue.append((new_m, new_trace, new_analyze_count))
    return traces
```

# 1. Equivalence Notions

Let the visible alphabet be:

```text
 = {
  'Open Case',
  'Analyze Revision',
  'Check Revision',
  'Approve Revision',
  'Rework Case',
  'Close Case'
}
```

Let `` denote the silent label. For a complete execution path, its visible trace is the word over `` obtained by deleting all `` events.

## Visible trace equivalence after hiding `tau`

For a model `M`, define:

```text
L_vis(M) = { (path) | path is a complete execution of M,  removes  }
```

Two models are visible-trace equivalent iff:

```text
L_vis(M_A) = L_vis(M_B)
```

This is language equality over complete visible traces. It is not bisimulation, not branching equivalence, and not a business abstraction.

**Duplicate visible labels.** In all formal notions below, duplicate visible labels are **not** distinguished by occurrence identity. A trace is a word over labels. The first `Analyze Revision` and the second `Analyze Revision` are distinguished only by their **position** in the word, not by unique IDs. Under P2, repeated `Check Revision` events are further collapsed to a Boolean fact.

## Bounded visible-language equivalence under P3

Under the bounded testing view P3:

```text
L_vis^P3(M) =
  {
      L_vis(M)
    | ||  12
      and count_('Analyze Revision')  3
  }
```

Two models are P3-equivalent iff:

```text
L_vis^P3(M_A) = L_vis^P3(M_B)
```

This is a finite, decidable comparison if the bounded language is explored exhaustively. It is **not** a proof of unbounded visible-trace equivalence. If the bounded languages differ, the models are not equivalent in general. If they are equal, that is only evidence within the bound.

## Business-relevant equivalence under P1

P1 is an operational compliance view. Define a compliance predicate `C_P1()` for a complete visible trace ``.

`C_P1()` holds iff:

1. `` starts with `Open Case` and ends with `Close Case`.
2. The interior between `Open Case` and `Close Case` can be segmented by `Rework Case` into `n  1` blocks:
   ```text
   B1 Rework B2 Rework ... Rework Bn
   ```
   where there is no `Rework Case` inside any `Bi`.
3. For every block `Bi`:
   ```text
   count_Bi('Analyze Revision') = 1
   count_Bi('Check Revision') = 1
   ```

This captures the stated P1 requirement that every analysis attempt has its own check before the next `Rework Case` or before `Close Case`. It does **not** by itself force `Check Revision` to occur after `Analyze Revision`; Model A also allows `Check Revision` before `Analyze Revision` inside the same iteration.

`Approve Revision` is not hidden under P1. Its exact position in the visible trace is part of the observed behavior. If a stricter business rule is required, for example “approval must occur after check,” that rule can be added to `C_P1`. The verdict below does not depend on adding that stricter rule, because Models A and B already differ on check counts and check placement.

Two models are P1-equivalent iff they induce the same set of observed pairs:

```text
{ (, C_P1()) |   L_vis(M) }
```

Thus, a model that accepts a non-compliant trace, or accepts a compliant trace that the other model rejects, is not P1-equivalent to the other.

## Abstracted dashboard equivalence under P2

Under P2, the dashboard projection `D()` of a complete visible trace `` is:

```text
D() =
  (
    first_is_Open = ([0] = 'Open Case'),
    last_is_Close = ([-1] = 'Close Case'),
    n_analyze = count_('Analyze Revision'),
    check_occurred = count_('Check Revision') > 0,
    rework_occurred = count_('Rework Case') > 0
  )
```

`Approve Revision` and `` are ignored. Repeated `Check Revision` events are collapsed to the Boolean fact `check_occurred`. The exact interleaving of `Analyze Revision` and `Check Revision` is ignored.

Two models are P2-equivalent iff:

```text
{ D() |   L_vis(M_A) } = { D() |   L_vis(M_B) }
```

## Effect of optional `Approve Revision` and hidden ``

The construct:

```text
X('Approve Revision', )
```

means that either `Approve Revision` occurs visibly or `` occurs silently. After hiding ``, this behaves like an optional visible `Approve Revision`.

This optional approval affects the exact visible language:

- It creates traces with and without `Approve Revision`.
- In Model A, the optional approval can occur only after both `Analyze Revision` and `Check Revision` in the same iteration.
- In Model B, the optional approval can occur after `Analyze Revision` but before the single global `Check Revision`.

Optional approval and silent `` therefore change the set of visible traces, but they do not make Models A and B equivalent under P1 or P3.

# 5. Bounded Comparison Code

The following Python code is self-contained. It uses an exhaustive breadth-first exploration over markings, not random playout. It therefore distinguishes itself from `pm4py.play_out(...)`: it explores all enabled transitions up to the P3 bound, including silent transitions, and records only complete traces that reach the final marking.

```python
import pm4py
from collections import deque, Counter

TREE_A = """
->(
  'Open Case',
  *(
    ->(
      +('Analyze Revision', 'Check Revision'),
      X('Approve Revision', tau)
    ),
    'Rework Case'
  ),
  'Close Case'
)
"""

TREE_B = """
->(
  'Open Case',
  +(
    'Check Revision',
    *(
      ->(
        'Analyze Revision',
        X('Approve Revision', tau)
      ),
      'Rework Case'
    )
  ),
  'Close Case'
)
"""


def normalize_marking(m):
    """Normalize a marking to a dict with only positive token counts."""
    if isinstance(m, dict):
        return {p: c for p, c in m.items() if c > 0}
    # Fallback: if a marking is represented as a set of places, assume one token each.
    return {p: 1 for p in m}


def full_marking(net, m):
    """Return a marking dict containing all places, with zeros where absent."""
    return {p: m.get(p, 0) for p in net.places}


def is_silent(label):
    if label is None:
        return True
    try:
        return label == pm4py.TAU or str(label) == "tau"
    except Exception:
        return str(label) == "tau"


def get_initial_marking(net):
    try:
        return normalize_marking(pm4py.get_initial_marking(net))
    except Exception:
        pass

    for p in net.places:
        if str(p.name) in {"source", "initial", "im", "start"}:
            return {p: 1}

    no_in = [p for p in net.places if not p.inputs]
    if len(no_in) == 1:
        return {no_in[0]: 1}

    raise RuntimeError("Could not infer initial marking")


def get_final_marking(net):
    try:
        return normalize_marking(pm4py.get_final_marking(net))
    except Exception:
        pass

    for p in net.places:
        if str(p.name) in {"sink", "final", "fm", "end"}:
            return {p: 1}

    no_out = [p for p in net.places if not p.outputs]
    if len(no_out) == 1:
        return {no_out[0]: 1}

    raise RuntimeError("Could not infer final marking")


def to_accepting_petri_net(tree):
    """
    Convert a process tree to an accepting Petri net and initial/final markings.
    Handles PM4Py versions that return either a net or a tuple.
    """
    res = pm4py.convert_to_petri_net(tree)

    if isinstance(res, tuple):
        if len(res) == 3:
            net, im, fm = res
        elif len(res) == 2:
            net, im = res
            fm = get_final_marking(net)
        else:
            net = res[0]
            im = get_initial_marking(net)
            fm = get_final_marking(net)
    else:
        net = res
        im = get_initial_marking(net)
        fm = get_final_marking(net)

    return net, normalize_marking(im), normalize_marking(fm)


def fire(net, marking, transition):
    """Fire a transition manually and return the normalized successor marking."""
    m = dict(marking)

    for p in transition.inputs:
        m[p] = m.get(p, 0) - 1
        if m[p] < 0:
            return None

    for p in transition.outputs:
        m[p] = m.get(p, 0) + 1

    return normalize_marking(m)


def bounded_visible_traces(
    net,
    im,
    fm,
    max_visible=12,
    max_analyze=3,
    analyze_label="Analyze Revision",
):
    """
    Exhaustively enumerate complete visible traces up to:
      - max_visible visible events
      - max_analyze occurrences of Analyze Revision

    Silent transitions are still explored after the visible bound is reached,
    because a trace may need silent steps to reach the final marking.
    """
    fm = normalize_marking(fm)
    traces = set()
    queue = deque([(im, (), 0)])
    visited = set()

    while queue:
        m, trace, analyze_count = queue.popleft()

        m_key = tuple(sorted(m.items()))
        state_key = (m_key, trace, analyze_count)

        if state_key in visited:
            continue

        visited.add(state_key)

        # Complete trace: final marking reached.
        if m == fm:
            if len(trace) <= max_visible and analyze_count <= max_analyze:
                traces.add(trace)
            continue

        for t in pm4py.get_enabled_transitions(net, full_marking(net, m)):
            label = getattr(t, "label", None)
            silent = is_silent(label)

            if not silent:
                if len(trace) >= max_visible:
                    continue

                if label == analyze_label:
                    if analyze_count >= max_analyze:
                        continue
                    new_analyze_count = analyze_count + 1
                else:
                    new_analyze_count = analyze_count

                new_trace = trace + (label,)
            else:
                new_trace = trace
                new_analyze_count = analyze_count

            new_m = fire(net, m, t)
            if new_m is None:
                continue

            next_key = (
                tuple(sorted(new_m.items())),
                new_trace,
                new_analyze_count,
            )

            if next_key not in visited:
                queue.append((new_m, new_trace, new_analyze_count))

    return traces


def fmt(trace):
    return " -> ".join(trace)


# Parse trees.
tree_a = pm4py.parse_process_tree(TREE_A)
tree_b = pm4py.parse_process_tree(TREE_B)

models = {"A": tree_a, "B": tree_b}
nets = {}

# Convert and check soundness.
for name, tree in models.items():
    net, im, fm = to_accepting_petri_net(tree)
    nets[name] = (net, im, fm)
    sound = pm4py.check_soundness(net, im, fm, print_diagnostics=True)
    print(f"Model {name} soundness: {sound}")

net_a, im_a, fm_a = nets["A"]
net_b, im_b, fm_b = nets["B"]

# Exhaustive bounded visible trace enumeration.
traces_a = bounded_visible_traces(net_a, im_a, fm_a)
traces_b = bounded_visible_traces(net_b, im_b, fm_b)

only_a = sorted(traces_a - traces_b, key=lambda x: (len(x), x))
only_b = sorted(traces_b - traces_a, key=lambda x: (len(x), x))

print("\nExhaustive bounded visible traces:")
print("Bound: at most 12 visible events, at most 3 Analyze Revision events")
print(f"Model A complete bounded visible traces: {len(traces_a)}")
print(f"Model B complete bounded visible traces: {len(traces_b)}")
print(f"Traces unique to A: {len(only_a)}")
print(f"Traces unique to B: {len(only_b)}")

print("\nFirst 20 traces unique to Model A:")
for tr in only_a[:20]:
    print("A-only:", fmt(tr))

print("\nFirst 20 traces unique to Model B:")
for tr in only_b[:20]:
    print("B-only:", fmt(tr))
```

This BFS is exhaustive for the stated P3 bound because it explores every enabled transition from every reachable state of the form:

```text
(marking, visible prefix, analyze count)
```

up to the visible-length and analyze-count bounds. It is not a random or heuristic playout.

# 6. Stochastic and Footprint Comparison

## Footprints

`pm4py.discover_footprints(...)` can provide evidence about which activities can co-occur in some trace. However, footprints ignore order, multiplicity, and exact control-flow dependencies.

For example, both models may have a footprint containing:

```text
{'Analyze Revision', 'Check Revision', 'Rework Case'}
```

because both can produce some trace containing those labels. But footprints cannot distinguish:

- Model A: one check per analysis attempt;
- Model B: one global check for the entire loop.

Therefore, equal footprints do not establish visible-trace equivalence.

Plausible code:

```python
def normalize_footprints(footprints):
    """
    Normalize footprints to a set of frozensets.
    Handles common return shapes: list of sets, set of sets, or dict.
    """
    if isinstance(footprints, dict):
        iterable = footprints.keys()
    else:
        iterable = footprints

    normalized = set()
    for fp in iterable:
        try:
            normalized.add(frozenset(fp))
        except TypeError:
            normalized.add(frozenset([fp]))
    return normalized


foot_a = pm4py.discover_footprints(tree_a)
foot_b = pm4py.discover_footprints(tree_b)

norm_foot_a = normalize_footprints(foot_a)
norm_foot_b = normalize_footprints(foot_b)

print("Footprints equal:", norm_foot_a == norm_foot_b)
print("Only in A:", norm_foot_a - norm_foot_b)
print("Only in B:", norm_foot_b - norm_foot_a)
```

## Stochastic language and EMD

A stochastic comparison can add evidence only if transition or trace probabilities are explicitly declared. Without declared probabilities, stochastic language comparison is not meaningful.

If one declares, for illustration, a uniform distribution over enabled transitions at each marking, then a stochastic comparison can be attempted. This uniform choice is **not** inferred from data and should not be interpreted as a business probability.

Plausible code, depending on the exact PM4Py version:

```python
def make_uniform_transition_probability_oracle(net):
    """
    Declares a uniform distribution over enabled transitions.
    This is an arbitrary modeling assumption, not an empirical probability.
    """
    def probability(marking):
        enabled = pm4py.get_enabled_transitions(net, full_marking(net, marking))
        if not enabled:
            return {}
        p = 1.0 / len(enabled)
        return {t: p for t in enabled}
    return probability
```

## Why low EMD or equal footprints do not prove equivalence

- Footprints are sets of activities. They do not record how many times an activity occurs.
- Footprints do not record order. A trace with `Check` before `Rework` and a trace with `Check` after `Rework` may have the same footprint.
- EMD measures distributional distance. A low EMD can occur when most traces are similar while rare but decisive counterexamples differ.
- Stochastic EMD requires probabilities. Without declared probabilities, any stochastic comparison is arbitrary.
- None of these methods proves language equality or inclusion.

# 7. Complexity Comparison

The following code computes process-tree structural counts and calls `pm4py.simplicity_petri_net(...)`. It assumes the helper functions from Section 5 are available.

```python
from collections import Counter


def tree_node_count(node):
    return 1 + sum(tree_node_count(c) for c in node.children)


def tree_depth(node):
    if not node.children:
        return 0
    return 1 + max(tree_depth(c) for c in node.children)


def tree_leaves(node):
    if not node.children:
        return [node.label]
    out = []
    for c in node.children:
        out.extend(tree_leaves(c))
    return out


def is_silent_label(label):
    if label is None:
        return True
    try:
        return label == pm4py.TAU
    except Exception:
        pass
    return str(label) == "tau"


def report_complexity(tree, name):
    labels = tree_leaves(tree)
    visible = [l for l in labels if not is_silent_label(l)]
    silent = [l for l in labels if is_silent_label(l)]

    counter = Counter(labels)
    duplicated_labels = [l for l, c in counter.items() if c > 1]
    extra_duplicate_occurrences = sum(c - 1 for c in counter.values() if c > 1)

    net, im, fm = to_accepting_petri_net(tree)

    try:
        simplicity = pm4py.simplicity_petri_net(net, im, fm)
    except Exception as e:
        simplicity = f"unavailable: {e}"

    print(name, {
        "process_tree_nodes": tree_node_count(tree),
        "depth_root_is_0": tree_depth(tree),
        "visible_leaves": len(visible),
        "silent_leaves": len(silent),
        "duplicated_leaf_labels": duplicated_labels,
        "extra_duplicate_leaf_occurrences": extra_duplicate_occurrences,
        "petri_net_simplicity": simplicity,
    })


report_complexity(tree_a, "Model A")
report_complexity(tree_b, "Model B")
```

## Derived exact structural values

These values are derived directly from the process trees.

| Metric | Model A | Model B | Notes |
|---|---:|---:|---|
| Process-tree node count | 12 | 12 | Includes operators and leaves. |
| Operator count | 5 | 5 | Sequence, loop, inner sequence, parallel/exclusive, exclusive. |
| Depth, root = 0 | 4 | 5 | Model B is deeper because `Check Revision` is outside the loop and the approval branch is nested under the inner sequence. |
| Visible leaves | 6 | 6 | `Open Case`, `Analyze Revision`, `Check Revision`, `Approve Revision`, `Rework Case`, `Close Case`. |
| Silent leaves | 1 | 1 | `tau`. |
| Duplicate leaf labels, labels appearing more than once | 0 | 0 | No leaf label is structurally duplicated. |
| Extra duplicate leaf occurrences | 0 | 0 | Repeated occurrences arise dynamically from the loop, not from duplicate leaves. |
| Petri-net simplicity | computed by code | computed by code | The exact PM4Py simplicity score is version-dependent; the code above prints it. |

Structurally, Models A and B have the same number of leaves and operators. Model A is shallower than Model B. Any node/arc-based simplicity metric should therefore treat Model A as at least as simple as Model B, and the code above can confirm the PM4Py-specific score.

# 8. Simplified Model

I choose a **P1-oriented compromise** as the target semantics:

- preserve the operational compliance invariant that every analysis attempt has its own check before the next rework or close;
- preserve the P2 dashboard summary;
- make the check a post-analysis step, i.e. require `Analyze Revision` before `Check Revision`;
- keep `Approve Revision` optional and visible;
- accept that the exact visible language of Model A is not preserved, because traces where `Check Revision` precedes `Analyze Revision` are removed.

This is appropriate if the business meaning of `Check Revision` is to inspect the analyzed revision. It is a stricter and simpler operational model than Model A, and much closer to P1 than Model B.

## Compact process tree

```text
->(
  'Open Case',
  *(
    ->(
      'Analyze Revision',
      'Check Revision',
      X('Approve Revision', tau)
    ),
    'Rework Case'
  ),
  'Close Case'
)
```

# 9. Final Recommendation

- **Model A** is the suitable original model for **P1 operational compliance** if the business accepts that `Check Revision` may occur either before or after `Analyze Revision` within the same attempt. It enforces one check per analysis attempt and keeps approval visible.
- **Model B** is **not suitable** for P1. It allows a single global check to cover multiple analysis attempts and allows the check to occur after rework or after approval.
- For **P2 executive dashboard**, Models A and B are equivalent under the stated abstraction. The simplified model S is also P2-equivalent and is structurally simpler than both originals.
- For **P3 bounded testing**, Models A and B are not equivalent. The simplified model S is also not equivalent to either original if the goal is to preserve the exact visible language of A or B. S should be used for P3 only if its stricter language is the intended test target.
- The simplified model S should **replace Model B** if the goal is a simple dashboard-compatible model that also enforces one check per analysis attempt. It may **replace Model A** for P1 only if the business agrees that `Check Revision` must follow `Analyze Revision`. It should **not replace Model A** if check-before-analysis interleaving is a real operational possibility or if exact P3 equivalence to Model A is required.
- A single simplification cannot preserve all three views simultaneously if P3 means exact visible-language equivalence to the originals and P1 includes Model A’s local parallelism between analysis and check. S preserves the P1 compliance core and P2 dashboard abstraction, but it deliberately weakens the exact visible language by fixing the order of analysis and check.

**Warning about bounded comparison.** The P3 comparison is bounded to at most 12 visible events and at most 3 `Analyze Revision` occurrences. Finding a difference within that bound proves that the models are not equivalent in general. However, failing to find a difference within that bound would **not** prove unbounded visible-trace equivalence. Footprint equality, stochastic similarity, or low Earth Mover’s Distance also do not establish trace equivalence.
