   pairs=over=0; overlap_sec=0.0
   for i,j in combinations(idx,2):
      s = max(g.loc[i,'start_dt'], g.loc[j,'start_dt'])
      e = min(g.loc[i,'complete_dt'], g.loc[j,'complete_dt'])
      pairs +=1
      if e > s:
         over +=1; overlap_sec += (e-s).total_seconds()
   return over/pairs if pairs else np.nan, overlap_sec

def on_boundary(series, precision):
   if precision == '1 sec': return pd.Series(True, index=series.index)
   if precision == '1 min': return series.dt.second.eq(0)
   if precision == '15 min': return series.dt.second.eq(0) & series.dt.minute.mod(15).eq(0)
   return pd.Series(np.nan, index=series.index)

def diag_cluster(g):
   g = g.sort_values('sequence_no')
   n = len(g)
   overlap_rate, overlap_sec = overlap_metrics(g)
   start_unique = g['start_dt'].nunique()
   complete_unique = g['complete_dt'].nunique()
   max_start_group = g.groupby('start_dt').size().max() if n else 0
   max_complete_group = g.groupby('complete_dt').size().max() if n else 0
   if n > 1:
      adjacency_exact = int((g['start_dt'].values[1:] == g['complete_dt'].values[:-1]).sum())
      adjacency_rate = adjacency_exact/(n-1)
   else: adjacency_rate=np.nan
   recording_lag_median = (g['recorded_at_dt'] - g['complete_dt']).dt.total_seconds().median()
   precision = g['precision'].iloc[0]
   start_boundary = float(on_boundary(g['start_dt'], precision).mean())
   complete_boundary = float(on_boundary(g['complete_dt'], precision).mean())
   queue_wait_start_median_min = ((g['start_dt'] - g['queue_arrival_dt']).dt.total_seconds()/60).median()
   first_start = g['start_dt'].min()
   arrivals_before_first_start = int((g['queue_arrival_dt'] < first_start).sum())
   queue_after_start = int((g['queue_arrival_dt'] > g['start_dt']).sum())
   zero_duration = int((g['start_dt'] == g['complete_dt']).sum())
   shared_resource = int((g['resource'].isin(['analyst_pool'])).sum())
   return pd.Series({
      'n_cases': n,
      'resource': g['resource'].iloc[0],
      'precision': precision,
      'overlap_pair_rate': round(overlap_rate,3) if pd.notna(overlap_rate) else np.nan,
      'overlap_seconds_total': int(overlap_sec),
      'distinct_starts': start_unique,
      'distinct_completes': complete_unique,
      'max_start_group': max_start_group,
      'max_complete_group': max_complete_group,
      'adjacency_rate': round(adjacency_rate,3) if pd.notna(adjacency_rate) else np.nan,
      'median_recording_lag_sec': recording_lag_median,
      'start_on_precision_boundary': start_boundary,
      'complete_on_precision_boundary': complete_boundary,
      'median_queue_to_start_min': queue_wait_start_median_min,
      'arrivals_before_first_start': arrivals_before_first_start,
      'queue_after_start_flags': queue_after_start,
      'zero_duration_flags': zero_duration,
      'shared_resource_flags': shared_resource
   })

diag = df.groupby('cluster').apply(diag_cluster).reset_index()

G6-like data can create both inflated waiting and near-zero service because the administrative stamp decouples recording from execution. If waiting is computed as `start - queue_arrival`, all accumulated queue time is pushed into waiting because the artificial start is 17:00. If service time is computed as `complete - start`, it becomes zero because the same artificial stamp is used for both lifecycle boundaries. Thus the same bulk-submission behavior simultaneously produces long apparent waiting and zero apparent processing, which is a classic sign of end-of-shift or bulk data entry rather than true instantaneous service.

def format_log(frame):
    """
    Build a PM4Py-compatible interval-oriented dataframe.
    Completion is the main time:timestamp; start_timestamp is retained for interval semantics.
    """
    f = frame.copy()
    f["case:concept:name"] = f["case"].astype(str)
    f["concept:name"] = f["activity"]
    f["org:resource"] = f["resource"]
    f["time:timestamp"] = f["complete_dt"]
    f["start_timestamp"] = f["start_dt"]

    try:
        formatted = pm4py.format_dataframe(
            f,
            case_id="case:concept:name",
            activity_key="concept:name",
            timestamp_key="time:timestamp",
            start_timestamp_key="start_timestamp",
            resource_key="org:resource",
        )
    except TypeError:
        formatted = pm4py.format_dataframe(
            f,
            case_id="case:concept:name",
            activity_key="concept:name",
            timestamp_key="time:timestamp",
            resource_key="org:resource",
        )

    if "start_timestamp" not in formatted.columns:
        formatted["start_timestamp"] = f["start_timestamp"].values

    return formatted

def call_discover(log, merge_distance, min_batch_size):
    """
    Call pm4py.discover_batches defensively because API keyword arguments can differ by version.
    merge_distance is assumed to be in seconds.
    """
    attempts = [
        dict(
            case_id="case:concept:name",
            activity_key="concept:name",
            timestamp_key="time:timestamp",
            start_timestamp_key="start_timestamp",
            resource_key="org:resource",
            merge_distance=merge_distance,
            min_batch_size=min_batch_size,
        ),
        dict(
            activity_key="concept:name",
            timestamp_key="time:timestamp",
            start_timestamp_key="start_timestamp",
            resource_key="org:resource",
            merge_distance=merge_distance,
            min_batch_size=min_batch_size,
        ),
        dict(
            activity_key="concept:name",
            timestamp_key="time:timestamp",
            resource_key="org:resource",
            merge_distance=merge_distance,
            min_batch_size=min_batch_size,
        ),
        dict(
            merge_distance=merge_distance,
            min_batch_size=min_batch_size,
        ),
        dict(),
    ]

    last_error = None
    for kwargs in attempts:
        try:
            return pm4py.discover_batches(log, **kwargs)
        except TypeError as exc:
            last_error = exc
        except Exception as exc:
            last_error = exc

    return f"error: {last_error}"

def batch_summary(result):
    """
    Compactly summarize PM4Py batch output without assuming a fixed return schema.
    """
    if isinstance(result, str) and result.startswith("error"):
        return result
    if result is None:
        return "none"
    if isinstance(result, tuple):
        return "tuple(" + "; ".join(batch_summary(x) for x in result) + ")"
    if isinstance(result, pd.DataFrame):
        if result.empty:
            return "none"

        label_col = next(
            (c for c in ["batch_type", "category", "type", "label", "batch_category"] if c in result.columns),
            None,
        )
        act_col = next(
            (c for c in ["activity", "concept:name", "activity_key"] if c in result.columns),
            None,
        )

        try:
            if label_col is not None and act_col is not None:
                grp = result.groupby([label_col, act_col]).size().reset_index(name="n")
                return "; ".join(
                    f"{r[label_col]}:{r[act_col]}={r['n']}" for _, r in grp.iterrows()
                )[:250]
            if label_col is not None:
                counts = result[label_col].value_counts()
                return "; ".join(f"{idx}={cnt}" for idx, cnt in counts.items())[:250]
            if act_col is not None:
                counts = result[act_col].value_counts()
                return "; ".join(f"{idx}={cnt}" for idx, cnt in counts.items())[:250]
        except Exception:
            pass

        return f"df_rows={len(result)}"

    if isinstance(result, dict):
        parts = []
        for key, value in result.items():
            if isinstance(value, pd.DataFrame):
                parts.append(f"{key}=df[{len(value)}]")
            elif isinstance(value, (list, tuple, set)):
                parts.append(f"{key}=len{len(value)}")
            else:
                parts.append(f"{key}={value}")
        return "; ".join(parts)[:250]

    if isinstance(result, (list, tuple, set)):
        return f"list_len={len(result)}"

    return str(result)[:250]

def overlap_metrics(g):
    pairs = 0
    overlapping_pairs = 0
    overlap_seconds = 0.0

    for i, j in combinations(g.index, 2):
        s = max(g.loc[i, "start_dt"], g.loc[j, "start_dt"])
        e = min(g.loc[i, "complete_dt"], g.loc[j, "complete_dt"])
        pairs += 1
        if e > s:
            overlapping_pairs += 1
            overlap_seconds += (e - s).total_seconds()

    rate = overlapping_pairs / pairs if pairs > 0 else np.nan
    return rate, overlap_seconds

def on_precision_boundary(ts, precision):
    if precision == "1 sec":
        return pd.Series(True, index=ts.index)
    if precision == "1 min":
        return ts.dt.second.eq(0)
    if precision == "15 min":
        return ts.dt.second.eq(0) & ts.dt.minute.mod(15).eq(0)
    return pd.Series(np.nan, index=ts.index)

def diag_cluster(g):
    g = g.sort_values("sequence_no")
    n = len(g)

    overlap_rate, overlap_seconds = overlap_metrics(g)

    distinct_starts = int(g["start_dt"].nunique())
    distinct_completes = int(g["complete_dt"].nunique())
    max_start_group = int(g.groupby("start_dt").size().max()) if n > 0 else 0
    max_complete_group = int(g.groupby("complete_dt").size().max()) if n > 0 else 0

    if n > 1:
        prev_complete = g["complete_dt"].shift(1)
        prev_duration = (g["complete_dt"] - g["start_dt"]).shift(1)
        adjacent = (g["start_dt"] == prev_complete) & (prev_duration > pd.Timedelta(0))
        adjacency_rate = float(adjacent.sum()) / (n - 1)
    else:
        adjacency_rate = np.nan

    recording_lag_sec = (g["recorded_at_dt"] - g["complete_dt"]).dt.total_seconds()
    queue_to_start_min = (g["start_dt"] - g["queue_arrival_dt"]).dt.total_seconds() / 60.0

    first_start = g["start_dt"].min()
    arrivals_before_first_start = int((g["queue_arrival_dt"] < first_start).sum())
    max_queue_wait_to_first_start_min = float(
        ((first_start - g["queue_arrival_dt"]).dt.total_seconds() / 60.0).max()
    )

    precision = g["precision"].iloc[0]

    return pd.Series({
        "n_cases": n,
        "resource": g["resource"].iloc[0],
        "precision": precision,
        "overlap_pair_rate": round(overlap_rate, 3) if pd.notna(overlap_rate) else np.nan,
        "overlap_seconds_total": int(overlap_seconds),
        "distinct_starts": distinct_starts,
        "distinct_completes": distinct_completes,
        "max_start_group": max_start_group,
        "max_complete_group": max_complete_group,
        "adjacency_rate": round(adjacency_rate, 3) if pd.notna(adjacency_rate) else np.nan,
        "median_recording_lag_sec": recording_lag_sec.median(),
        "start_on_precision_boundary": float(on_precision_boundary(g["start_dt"], precision).mean()),
        "complete_on_precision_boundary": float(on_precision_boundary(g["complete_dt"], precision).mean()),
        "median_queue_to_start_min": queue_to_start_min.median(),
        "max_queue_wait_to_first_start_min": max_queue_wait_to_first_start_min,
        "arrivals_before_first_start": arrivals_before_first_start,
        "queue_after_start_flags": int(g["queue_after_start"].sum()),
        "zero_duration_flags": int(g["zero_duration"].sum()),
        "shared_resource_flags": int((g["resource"] == "analyst_pool").sum()),
    })

# Independent pandas diagnostics by cluster.
diag_rows = []
for cluster, g in df.groupby("cluster"):

# 5. Robustness and alternative explanations

**Assumption A1 — resource trust.** If resource IDs are trusted as individual workers or machines, then G1’s identical intervals are strong evidence of a machine batch, G4 and G8 are strong evidence of serial work, and G5 becomes anomalous because one logical resource appears to perform overlapping work. If resource IDs are treated as potentially shared, G5 cannot support person-level utilization or conformance conclusions, and any resource-level bottleneck inference becomes weaker. Shared-account treatment also makes G5’s four-way overlap less evidence of one person multitasking and more evidence of missing identity or session semantics.

**Assumption A2 — exact timestamps.** If timestamps are treated as exact, G1, G4, and G8 are highly interpretable because their 1-second scanner/PLC patterns are internally coherent. G2 and G3 remain interpretable as logging patterns, but exactness does not prove that the common start or common completion corresponds to physical work. G6 remains clearly suspicious because exact zero-duration service for seven cases spread over hours is operationally implausible. G7 is not credible under exactness because queue arrivals occur after the recorded start.

**Assumption A3 — interval-censoring by stated precision.** If timestamps are interval-censored by their stated precision, confidence shifts. For 1-minute sources, G2, G3, G5, and G6 tolerate up to about a minute of hidden variation, which does not materially change the diagnosis because the patterns span many minutes or hours. For G7, 15-minute precision makes the observed 14:00 and 14:15 stamps weak evidence about actual execution. A conservative bound would treat the observed start as a quarter-hour-rounded value and the observed completion as another quarter-hour-rounded value, with true times constrained by queue arrival and nonnegative duration. For G6, interval-censoring by 1 minute is insufficient; the real issue is semantic, not granularity.

**Assumption A4 — PM4Py batch labels are temporal detections.** PM4Py may label G1, G6, and G7 as batches because of identical or tightly clustered timestamps, and G4/G8 as sequential batches because of adjacency. Those labels should not be treated as proof of physical batching. They must be reconciled with resource capacity, source semantics, and lifecycle quality flags.

At least three cases where the same temporal pattern supports multiple physical explanations:

1. **G1: identical start and end times.** This can mean a genuine simultaneous machine batch, an automated cycle whose PLC records all items together, or a common setup plus common completion. The fixture-capacity context supports genuine batching, but per-item physical evidence is absent.

2. **G2: identical starts with staggered ends.** This can mean a common setup/group start followed by serial work, concurrent multitasking with several inspections open, or an administrative “begin group” stamp that does not reflect true individual starts. The separate completion buttons support serial completion, but the common start is ambiguous.

3. **G3: staggered starts with identical ends.** This can mean end batching/bulk submission, concurrent work where all reviews are finished just before submission, or a form semantics where completion records the folder submission rather than each review’s true end. The worker statement strongly supports bulk submission, but true work timing remains hidden.

4. **G5: extensive overlaps under one resource ID.** This can mean shared-account artifact, genuine parallel work by multiple analysts, or intervals that include idle/open-session time rather than active analysis. The stated pool capacity of up to three analysts conflicts with the observed four-way overlap, increasing uncertainty.

5. **G7: identical quarter-hour starts and completions.** This can mean timestamp coarsening, an actual grouped measurement session, or serial work collapsed by rounding. The documented rounding rule makes coarsening the most defensible explanation, but the true underlying pattern cannot be recovered from the supplied data alone.

| G5 | Not interpretable as single-resource work; shared-account artifact is likely. | Do not use for single-resource utilization or person-level waiting; model as a capacity pool only if identity cannot be recovered. | Obtain badge-level or session-level analyst identity and confirm maximum concurrent analysts. |
| G6 | High confidence that the timestamps are bulk administrative stamps, not physical service intervals. | Exclude from point-estimate service-time and utilization analysis; if retained, interval-censor true work somewhere before 17:00 and treat waiting to 17:00 as administrative backlog only. | Confirm when tickets were actually worked and whether actual start/complete can be reconstructed. |

Overall, the clusters that can be used most safely for conventional queue analysis are **G4** and, after downtime adjustment, **G8**. **G1** can be used if analyzed as a batch cycle. **G2** and **G3** require interval or batch-aware treatment. **G5** should not be used for resource-level performance without identity decomposition. **G6** and **G7** should be excluded from point-estimate queueing metrics or converted to interval-censored bounds before any performance conclusion.
