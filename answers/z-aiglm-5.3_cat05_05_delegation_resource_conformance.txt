def business_day_deadline(trigger_ts, n_days=2):
    """Deadline = end of the n-th business day (Mon-Fri, no site holidays in
    this window) after the trigger date.  np.busday convention."""
    d = np.datetime64(trigger_ts.date(), "D")
    due = np.busday_offset(d, n_days, roll="forward")
    return pd.Timestamp(str(due)).tz_localize("UTC") + pd.Timedelta(days=1) - pd.Timedelta(seconds=1)

## 2. Rule Interpretation

Let `H(e)` = canonical human of the acting resource of event *e* after alias normalization and service-account attribution ( if unresolvable). Roles are functions of time: `Role(h, t)` from the role-interval table.

- **RR1 (drafter/approver separation):** violated for case *c* iff `H(draft) = H(approve)  `. If either side is , the case is *unverifiable*, not a violation. Runs on normalized humans, so alias-split self-approval is caught.
- **RR2 (implementer/verifier separation):** same logic for Implement_Change vs Verify_Implementation. A service-account verification is checkable only when attribution is reliable; otherwise flag "RR2 unverifiable (RR5)".
- **RR3 (high-risk approval authority):** for HIGH-risk cases, Approve_Change at time *t* requires `Role(H, t)  QM`, **or** a delegation *D* such that: (a) `start(D)  t  end(D)`; (b) scope(D) covers the change (equipment scope **and** activity type as written); (c) if D requires principal absence, the principal is absent/formally unavailable at *t*; (d) no prohibited subdelegation occurred (delegations are personal; an instrument that conveys another delegate's authority onward without permission is void); and (e) the instrument is authenticated by evidence **created before first reliance** (see below). LOW-risk cases fall outside RR3.
- **RR4 (emergency substitution):** for HIGH-risk emergency changes: approval by a standing IC at event time or a valid IC delegation (same validity conditions); implementation after approval; and Retrospective_QM_Review performed by a QM or valid QM delegate (scope- and activity-matched) within two business days. The exception is justified only where the emergency flag and safety rationale are credible — a manual, evidence-based judgment.
  - **Business days:** Mon–Fri excluding site holidays (none in the period; 2026-08-15/16 are weekend days). Deadline = 23:59 UTC of the **second business day after the implementation date** (trigger Tue 08-11  deadline Thu 08-13 23:59). Sensitivity: under any plausible alternative convention (inclusive counting), CHG-510's review (08-12) remains timely and CHG-505's review (08-14) remains late — the conclusions are convention-robust.
- **RR5 (identity evidence):** service accounts are not humans; conformance attaches only via reliable session attribution (strong/weak grading above). Aliases must be normalized before SoD checks; unattributed actions are neither violations nor conformance.
- **RR6 (suspicion vs proof):** repeated pairs, reciprocity, shared workstations, and concentrated handovers are indicators requiring corroboration (communications, benefit, forensic attribution). No collusion finding is made from collaboration frequency alone.

**Action by a principal while on leave.** Role intervals are not suspended by leave: a QM acting during recorded leave still holds QM authority, so the act is conformant **if attributed to her**. Leave status matters in three other ways: (i) attribution risk — the act must be tied to the principal by session evidence (Asha's 08-13 VPN window 09:15–09:40 spans her 09:22 approval, corroborating but not proving her claim, since credential sharing cannot be excluded); (ii) delegation conditions — demonstrated principal activity inside an absence window can undermine a delegation's absence condition (here, no delegate approval falls inside Asha's 09:15–09:40 session, so no delegate act is directly invalidated; the formal leave record is treated as controlling, with the conflict logged); (iii) leave-policy compliance — working while on leave may breach HR rules, which is reportable but outside RR1–RR6.

**Can a later database entry prove an earlier delegation?** No — not alone. The D1 QMS record created 2026-08-12 09:15 cannot, by itself, establish that D1 existed on 2026-08-11 09:00 (post-hoc creation/backdating during migration cannot be excluded). The evidentiary standard is contemporaneity: an authentic instrument timestamped **before first reliance**. The signed PDF with trusted timestamp 2026-08-09 18:00 satisfies this for D1; the later DB record is corroboration and a data-quality item, not proof. Conversely, D2's controlled record was created before its interval — good practice.

def four_eyes_violations(frame, res_col, act_a, act_b):
    work = frame.copy()
    work["org:resource"] = work[res_col]           # identity under test
    have = set(frame.loc[frame["concept:name"] == act_a, "case:concept:name"]) \
         & set(frame.loc[frame["concept:name"] == act_b, "case:concept:name"])
    try:
        ok = set(pm4py.filter_four_eyes_principle(work, act_a, act_b)["case:concept:name"])
    except Exception:
        ok = set()
        for case in have:                          # manual fallback
            g = work[work["case:concept:name"] == case]
            ra = g.loc[g["concept:name"] == act_a, "org:resource"]
            rb = g.loc[g["concept:name"] == act_b, "org:resource"]
            if len(ra) and len(rb) and ra.iloc[0] != rb.iloc[0]:
                ok.add(case)
    return sorted(have - ok)

def business_day_deadline(trigger_ts, n_days=2):

    return (pd.Timestamp(str(due)).tz_localize("UTC")
            + pd.Timedelta(days=1) - pd.Timedelta(seconds=1))

def handover_fallback():
    pairs = {}
    for _, g in df_human.sort_values("time:timestamp").groupby("case:concept:name"):
        seq = list(g["org:resource"])
        for a, b in zip(seq, seq[1:]):
            if a != b:
                pairs[(a, b)] = pairs.get((a, b), 0) + 1
    for (a, b), w in sorted(pairs.items(), key=lambda kv: -kv[1]):
        print(f"  {a:>24s} -> {b:<24s} weight={w}")

def working_fallback():
    pairs = {}
    for _, g in df_human.groupby("case:concept:name"):
        who = sorted(set(g["org:resource"]))
        for i in range(len(who)):
            for j in range(i + 1, len(who)):
                pairs[(who[i], who[j])] = pairs.get((who[i], who[j]), 0) + 1
    for (a, b), w in sorted(pairs.items(), key=lambda kv: -kv[1]):
        print(f"  {a:>24s} ~ {b:<24s} weight={w}")

def show_network(net, title, fallback):
    try:
        edges = sorted(((u, v, d.get("weight", 0)) for u, v, d in net.edges(data=True)),
                       key=lambda x: -x[2])
        print(f"\n=== {title} (top edges by weight) ===")
        for u, v, w in edges[:15]:
            print(f"  {u:>24s} -> {v:<24s} weight={w}")
        if not edges:
            print("  (no edges)")
    except Exception as exc:
        print(f"\n=== {title}: discovery failed ({exc}); pandas fallback ===")
        fallback()
