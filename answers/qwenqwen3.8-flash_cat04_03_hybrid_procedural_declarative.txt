import pm4py
from datetime import timedelta

def _event_resources(df, activity):
    if activity not in set(df["activity"]):
        return set(), False
    if "resource" not in df.columns:
        return set(), True
    vals = df.loc[df["activity"] == activity, "resource"]
    missing = vals.isna().any()
    return set(vals.dropna()), missing

def _first_ts(df, activity):
    ts = df.loc[df["activity"] == activity, "timestamp"]
    return None if ts.empty else ts.min()

def validate_r9_r2_r10_r3_r11(log_df, emergency_df=None):
    log_df = log_df.copy()
    log_df["timestamp"] = pd.to_datetime(log_df["timestamp"], errors="coerce")
    findings = []
    for case_id, g in log_df.groupby("case_id"):
        g = g.sort_values("timestamp")
        acts = set(g["activity"])
        # R9
        verify_res, verify_missing = _event_resources(g, "Verify Bank Account")
        manager_res, manager_missing = _event_resources(g, "Manager Approval")
        compliance_res, compliance_missing = _event_resources(g, "Compliance Approval")
        capture_res, capture_missing = _event_resources(g, "Capture Master Data")
        activate_res, activate_missing = _event_resources(g, "Activate Vendor")
        if ("Verify Bank Account" in acts or "Manager Approval" in acts or "Compliance Approval" in acts or "Capture Master Data" in acts or "Activate Vendor" in acts) and ("resource" not in g.columns):
            findings.append((case_id, "R9", "fail", "resource attribute missing"))
        else:
            if "Activate Vendor" in acts:
                if manager_res and compliance_res:
                    findings.append((case_id, "R2/R8", "fail", "both approvals"))
                approval_res = manager_res if manager_res else compliance_res
                if not approval_res:
                    findings.append((case_id, "R2", "fail", "activation without approval"))
                if verify_res & approval_res:
                    findings.append((case_id, "R9", "fail", "verify/approval same resource"))
            if "Reject Vendor" in acts and ("Manager Approval" in acts or "Compliance Approval" in acts):
                findings.append((case_id, "R2", "fail", "approval in rejected case"))
            if capture_res & activate_res:
                findings.append((case_id, "R9", "fail", "capture/activate same resource"))
        # R10 counts
        counts = g["activity"].value_counts()
        for a in ["Activate Vendor", "Reject Vendor", "Release First Payment"]:
            if counts.get(a, 0) > 1:
                findings.append((case_id, "R10", "fail", f"{a} >1"))
        # R3 matching
        open_req = 0
        for row in g.itertuples():
            if row.activity == "Request Clarification":
                if open_req > 0:
                    findings.append((case_id, "R3", "fail", "second request before receive"))
                open_req += 1
            elif row.activity == "Receive Clarification":
                if open_req == 0:
                    findings.append((case_id, "R3", "fail", "orphan receive"))
                open_req -= 1
            elif row.activity in {"Classify Risk", "Reject Vendor", "Activate Vendor", "Close Onboarding"}:
                if open_req > 0:
                    findings.append((case_id, "R3", "fail", "unresolved clarification before outcome"))
                    open_req = 0
        # R11
        activate_ts = _first_ts(g, "Activate Vendor")
        if activate_ts is not None:
            change_ts = sorted(g.loc[g["activity"] == "Change Bank Details", "timestamp"])
            for ct in [t for t in change_ts if t > activate_ts]:
                suspend_ts = sorted(g.loc[g["activity"] == "Suspend Vendor", "timestamp"])
                if not any(t > ct for t in suspend_ts):
                    findings.append((case_id, "R11", "fail", "post-activation change without suspend"))
                release_ts = sorted(g.loc[g["activity"] == "Release First Payment", "timestamp"])
                verify_ts = sorted(g.loc[g["activity"] == "Verify Bank Account", "timestamp"])
                for rt in [t for t in release_ts if t > ct]:
                    if not any(vt > ct and vt < rt for vt in verify_ts):

        # emergency

        act_ts = _first_ts(g, "Activate Vendor")
        legal_ts = _first_ts(g, "Obtain Legal Opinion")
        if esc_ts is not None and act_ts is not None and (legal_ts is None or legal_ts > act_ts):
            if emergency_df is None:
                findings.append((case_id, "R5/R6", "fail", "activation before legal opinion; no emergency attestation"))

                    findings.append((case_id, "R5/R6", "fail", "legal opinion not within 24h"))

# 1. Policy Interpretation and Conflict Resolution

## Material assumptions

1. **Event-log schema.**  
   The primary log contains at least:
   - `case_id`
   - `activity`
   - `timestamp`
   - `resource`

   The supplied test traces use `Activity@hour[resource]`, so `hour` is interpreted as a relative timestamp from case start. For temporal-profile checks, hours are converted to seconds.

2. **Case scope.**  
   Each trace is one vendor-onboarding case. Cross-case behavior is not modeled.

3. **Final outcome interpretation.**  
   “Final outcome” in R3 is interpreted as the first terminal business decision:
   - `Reject Vendor`, or
   - `Activate Vendor`.

   `Close Onboarding` is treated as the administrative terminal event after the business outcome.

4. **Applicable approval.**  
   “Applicable approval” means:
   - `Manager Approval` for the low-risk accepted route;
   - `Compliance Approval` for the high-risk accepted route.

   If both approvals occur, the case violates R2 and R8.

5. **Concurrency.**  
   `Capture Master Data`, `Run Sanctions Screening`, and `Verify Bank Account` are modeled as concurrent after receipt. Any interleaving is allowed, but all three must precede `Classify Risk`.

6. **Clarification cycles.**  
   R3 is modeled conservatively as a block-structured clarification loop before classification:
   - `Request Clarification` must be followed by `Receive Clarification` before the next request and before the next major gate.
   - Multiple cycles are allowed.
   - This is an approximation: the requirement says “before any final outcome,” not explicitly “before classification,” but the test traces place clarification before classification.

7. **Sanctions escalation.**  
   `Sanctions Escalation` is optional. If it occurs:
   - it must be after `Run Sanctions Screening` (R7);
   - `Obtain Legal Opinion` must eventually occur (R4);
   - under the implemented conservative interpretation, `Obtain Legal Opinion` must precede `Activate Vendor` (R5).

8. **Post-activation behavior.**  
   `Change Bank Details`, `Suspend Vendor`, and any post-activation re-verification or later payment behavior are not part of the stable procedural backbone. They are handled by declarative and data-aware layers.

9. **Temporal expectations.**  
   R12 defines normative temporal profiles, but they are warnings, not hard control-flow violations.

10. **Missing attributes.**  
    If a required attribute is missing, the relevant check is treated as non-evaluable in exploratory mode but fail-closed in production enforcement. For emergency activation, missing authorization data means the exception cannot be validated.

## R5/R6 conflict

R5 and R6 are incompatible if both are treated as purely activity-sequence rules:

- **R5:** after `Sanctions Escalation`, `Obtain Legal Opinion` must precede `Activate Vendor` under all circumstances.
- **R6:** emergency activation may occur before `Obtain Legal Opinion`, but only with joint authorization outside the event log, and the opinion must arrive within 24 hours.

The event log contains no `emergency` attribute and no joint-authorization evidence. Therefore, an activity-only model cannot distinguish:

1. a legitimate emergency activation; from
2. an unauthorized activation that violates R5.

### Resolution A — Conservative legal precedence, implemented

**Choice:** R5 is authoritative for procedural and hard control-flow validation.

- If `Sanctions Escalation` occurs and `Activate Vendor` occurs, `Obtain Legal Opinion` must occur before activation.
- Emergency activation is not accepted from activity data alone.
- A separate external emergency-authorization check can downgrade the finding only if trusted off-log evidence exists.
- Because the supplied log lacks that evidence, T6 is treated as a hard violation in the implemented model.

**Compliance risk:**

- False positives for legitimate emergency activations.
- Operational pressure to create off-log workarounds.
- Possible business delay for vendors who could be safely activated under a properly authorized emergency process.

### Resolution B — Attested emergency exception

**Choice:** R5 is relaxed to a warning or policy-dependent condition when an emergency activation is externally attested.

- Activity-only layers encode only R4: escalation eventually leads to legal opinion.
- A data-aware layer accepts activation before legal opinion only if external attributes prove:
  - `emergency_activation = true`;
  - a valid joint authorization exists;
  - authorized roles are sufficient;
  - `Obtain Legal Opinion` arrives within 24 hours after activation.

**Compliance risk:**

- If the external attestation system is weak, this under-detects unauthorized activations.
- The event log alone cannot prove compliance.
- The model becomes policy-dependent rather than log-conformant.

### Chosen implemented resolution

The implemented model uses **Resolution A**: conservative legal precedence.

The reason is that R5 is stated as an absolute legal requirement, while R6 depends on evidence that is explicitly absent from the log. Silently allowing activation before legal opinion would make the model unable to distinguish authorized emergencies from violations.

For T6, the verdict is therefore:

- **Implemented conservative model:** `fail`.
- **Alternative attested-exception model:** `policy-dependent` if no attestation is present; `pass` or `warning` if a valid emergency attestation is present and legal opinion arrives within 24 hours.

DECLARE is the more flexible authoritative layer for precedence. |
| R8 | DECLARE + procedural | DECLARE: `noncoexistence: ("Enhanced Due Diligence", "Manager Approval")`, `noncoexistence: ("Compliance Approval", "Manager Approval")`. Procedural exclusive low/high branches. | These are hard co-occurrence prohibitions. DECLARE is well suited. | Procedural already prevents the combinations in the backbone. DECLARE protects against model drift or post-backbone additions. |
| R9 | Resource/data-aware validation | Custom: `Verify Bank Account` resource must differ from applicable approval resource; `Capture Master Data` resource must differ from `Activate Vendor` resource. Optional cross-check with `pm4py.filter_four_eyes_principle`. | Resource segregation cannot be expressed in activity-only DECLARE, ProcessTree, POWL, or log skeleton. | Requires `resource` attribute. Missing resource means fail-closed in enforcement. |
| R10 | Log Skeleton + procedural | Log Skeleton: `activ_freq: {"Activate Vendor": {0,1}, "Reject Vendor": {0,1}, "Release First Payment": {0,1}}`. Procedural each event occurs at most once. | At-most-once is naturally expressed by allowed frequency sets. | DECLARE has no clean at-most-once template without forcing exactly-once or absence. |
| R11 | Data-aware validation + conservative DECLARE | Custom: if `Change Bank Details` occurs after `Activate Vendor`, require later `Suspend Vendor`; if any later `Release First Payment` occurs after change, require another `Verify Bank Account` between change and release. DECLARE conservative: `response: ("Change Bank Details", "Suspend Vendor")`. | The rule is conditional on post-activation timing and later payment behavior. Activity-only DECLARE cannot condition on “after activation” or “later release.” | DECLARE response overconstrains changes that occur before activation. Custom check is authoritative. |
| R12 | Temporal profile | `temporal_profile` with three activity pairs and zeta = 2.0. | These are statistical temporal expectations, not hard control-flow constraints. | Temporal deviations are warnings. Missing pairs are not evaluated. |

# 3. Procedural Backbone

## 3.1 Compact formal description

The procedural backbone is represented as a process tree:

```text
Receive Onboarding Request
 parallel(
    Capture Master Data,
    Run Sanctions Screening,
    Verify Bank Account
  )
 loop(
    body: Request Clarification  Receive Clarification,
    exit: Classify Risk
  )
 exclusive_choice(
    Reject Vendor  Close Onboarding,
    Manager Approval  Activate Vendor  Release First Payment  Close Onboarding,
    exclusive_choice(
      Enhanced Due Diligence  Compliance Approval  Activate Vendor  Release First Payment  Close Onboarding,
      Sanctions Escalation  Enhanced Due Diligence  Obtain Legal Opinion  Compliance Approval  Activate Vendor  Release First Payment  Close Onboarding
    )
  )
```

The loop exit is `Classify Risk`, allowing zero or more clarification cycles before classification.

The high-risk escalation branch encodes the conservative legal interpretation:

```text
Sanctions Escalation  Enhanced Due Diligence  Obtain Legal Opinion  Compliance Approval  Activate Vendor
```

This enforces R5 for escalation cases but is stricter than R4 alone.

## 3.2 PM4Py process-tree code

```python
from pm4py.objects.process_tree.obj import (
    ProcessTree,
    ProcessTreeModelActivity,
    ProcessTreeModelOperator,
)


def event(name: str) -> ProcessTree:
    node = ProcessTree()
    node.operator = ProcessTreeModelOperator.OPERATOR_EVENT
    node.activity = ProcessTreeModelActivity(name=name)
    return node


def sequence(*children: ProcessTree) -> ProcessTree:
    node = ProcessTree()
    node.operator = ProcessTreeModelOperator.OPERATOR_SEQUENCE
    node.children = list(children)
    return node


def exclusive(*children: ProcessTree) -> ProcessTree:
    node = ProcessTree()
    node.operator = ProcessTreeModelOperator.OPERATOR_XOR
    node.children = list(children)
    return node


def parallel(*children: ProcessTree) -> ProcessTree:
    node = ProcessTree()
    node.operator = ProcessTreeModelOperator.OPERATOR_PARALLEL
    node.children = list(children)
    return node


def loop(body: ProcessTree, exit_node: ProcessTree) -> ProcessTree:
    node = ProcessTree()
    node.operator = ProcessTreeModelOperator.OPERATOR_LOOP
    node.children = [body, exit_node]
    return node


procedural_model = sequence(
    event("Receive Onboarding Request"),

    parallel(
        event("Capture Master Data"),
        event("Run Sanctions Screening"),
        event("Verify Bank Account"),
    ),

    loop(
        sequence(
            event("Request Clarification"),
            event("Receive Clarification"),
        ),
        event("Classify Risk"),
    ),

    exclusive(
        sequence(
            event("Reject Vendor"),
            event("Close Onboarding"),
        ),

        sequence(
            event("Manager Approval"),
            event("Activate Vendor"),
            event("Release First Payment"),
            event("Close Onboarding"),
        ),

        exclusive(
            sequence(
                event("Enhanced Due Diligence"),
                event("Compliance Approval"),
                event("Activate Vendor"),
                event("Release First Payment"),
                event("Close Onboarding"),
            ),

            sequence(
                event("Sanctions Escalation"),
                event("Enhanced Due Diligence"),
                event("Obtain Legal Opinion"),
                event("Compliance Approval"),
                event("Activate Vendor"),
                event("Release First Payment"),
                event("Close Onboarding"),
            ),
        ),
    ),
)
```

If a PM4Py version uses different operator enum names, replace the constants with the equivalent sequence, exclusive-choice, parallel, loop, and event operators.

## 3.3 Explanation of procedural behavior

### Concurrency

`Capture Master Data`, `Run Sanctions Screening`, and `Verify Bank Account` are in a parallel branch after `Receive Onboarding Request`. This allows any interleaving, including the order in T2:

```text
Run Sanctions Screening  Verify Bank Account  Capture Master Data
```

All three must complete before the loop exit `Classify Risk`.

### Choices

After classification, the model has three mutually exclusive primary outcomes:

1. **Rejected:**
   ```text
   Reject Vendor  Close Onboarding
   ```

2. **Low-risk accepted:**
   ```text
   Manager Approval  Activate Vendor  Release First Payment  Close Onboarding
   ```

3. **High-risk accepted:**
   - no escalation:
     ```text
     Enhanced Due Diligence  Compliance Approval  Activate Vendor  Release First Payment  Close Onboarding
     ```
   - escalation under conservative legal precedence:
     ```text
     Sanctions Escalation  Enhanced Due Diligence  Obtain Legal Opinion  Compliance Approval  Activate Vendor  Release First Payment  Close Onboarding
     ```

This enforces R1, R2, R5 conservatively, and R8.

### Clarification loops

The loop:

```text
loop(Request Clarification  Receive Clarification, Classify Risk)
```

allows zero or more complete request/response cycles before classification. It enforces:

- no `Classify Risk` after an unanswered request;
- no second request before a receive;
- no orphan receive before a request.

This is a conservative block-structured approximation of R3. If clarification may occur after classification in real operations, the procedural model would need a more complex accepting Petri net or POWL precedence structure.

### Terminal outcomes

`Close Onboarding` occurs after either:

- `Reject Vendor`, or
- `Release First Payment`.

This matches the backbone.

### Post-activation behavior deliberately excluded

`Change Bank Details`, `Suspend Vendor`, and any post-activation re-verification or later payment behavior are excluded from the procedural backbone.

Reasons:

1. They are not part of the stable onboarding sequence.
2. Including them in the tree would make the model either overly permissive or overly complex.
3. R11 is conditional on post-activation timing and is better handled by DECLARE plus data-aware checks.

The procedural model therefore does not evaluate T1–T8 for R11 because none of the test traces contains post-activation changes.

## 4.1 DECLARE rule semantics

All rules are normative, so `support` and `confidence` are set to `1.0`. These values should not be interpreted as empirical discovery confidence.

| Template | Key | Direction and semantics | Requirement | Notes |
|---|---|---|---|---|
| `init` | `"Receive Onboarding Request"` | The trace must start with this activity. | Backbone | Unary template. |
| `response` | `("Request Clarification", "Receive Clarification")` | Every occurrence of `Request Clarification` must eventually be followed by `Receive Clarification`. | R3 | Does not enforce one-to-one matching across multiple requests. |
| `response` | `("Sanctions Escalation", "Obtain Legal Opinion")` | Every escalation must eventually be followed by legal opinion. | R4 | Does not enforce legal opinion before activation. |
| `response` | `("Manager Approval", "Activate Vendor")` | Every manager approval must eventually be followed by activation. | Backbone/R2 | Does not require activation to be preceded by manager approval. |
| `response` | `("Compliance Approval", "Activate Vendor")` | Every compliance approval must eventually be followed by activation. | Backbone/R2 | Does not require activation to be preceded by compliance approval. |
| `response` | `("Activate Vendor", "Release First Payment")` | Every activation must eventually be followed by first-payment release. | Backbone | Fails T8 because activation occurs without release. |
| `response` | `("Reject Vendor", "Close Onboarding")` | Every rejection must eventually be followed by close. | Backbone | Passes T3 and T8 for this rule. |
| `response` | `("Release First Payment", "Close Onboarding")` | Every release must eventually be followed by close. | Backbone | Passes T1, T2, T4, T6, T7. |
| `response` | `("Change Bank Details", "Suspend Vendor")` | Every change must eventually be followed by suspension. | R11, conservative approximation | Overconstrains changes that occur before activation. |
| `precedence` | `("Request Clarification", "Receive Clarification")` | Every receive must be preceded by a request. | R3 | Prevents orphan receives. |
| `precedence` | `("Run Sanctions Screening", "Sanctions Escalation")` | Every escalation must be preceded by screening. | R7 | Unary source/target orientation: source must occur before target. |
| `precedence` | `("Classify Risk", "Activate Vendor")` | Every activation must be preceded by classification. | Backbone | Does not distinguish low/high risk. |
| `precedence` | `("Classify Risk", "Reject Vendor")` | Every rejection must be preceded by classification. | Backbone |  |
| `precedence` | `("Enhanced Due Diligence", "Compliance Approval")` | Every compliance approval must be preceded by enhanced due diligence. | Backbone | High-risk route. |
| `noncoexistence` | `("Reject Vendor", "Activate Vendor")` | The two activities must never both occur in the same trace, in any order. | R1 | Stronger than `nonsuccession`, which would only forbid one order. |
| `noncoexistence` | `("Enhanced Due Diligence", "Manager Approval")` | The two activities must never both occur. | R8 |  |
| `noncoexistence` | `("Compliance Approval", "Manager Approval")` | The two activities must never both occur. | R2/R8 | Does not enforce exactly one approval in accepted cases. |

### Why alternate/chain templates were not used

- `chainresponse` and `chainprecedence` require immediate adjacency. Clarification responses and legal opinions may be separated by other events, so chain templates would be too strict.
- `alternate response` and `alternate precedence` are useful for alternating patterns, but R3 allows other activities between request and receive and allows multiple cycles. The chosen procedural loop plus custom matching is clearer.
- `nonsuccession` would forbid only one direction. For R1 and R8, co-occurrence in either direction is invalid, so `noncoexistence` is appropriate.

# 5. Log-Skeleton Model

```python
log_skeleton = {
    "equivalence": set(),

    "always_before": {
        ("Receive Onboarding Request", "Classify Risk"),
        ("Capture Master Data", "Classify Risk"),
        ("Run Sanctions Screening", "Classify Risk"),
        ("Verify Bank Account", "Classify Risk"),

        ("Request Clarification", "Receive Clarification"),

        ("Run Sanctions Screening", "Sanctions Escalation"),
        ("Sanctions Escalation", "Obtain Legal Opinion"),

        ("Classify Risk", "Activate Vendor"),
        ("Classify Risk", "Reject Vendor"),

        ("Manager Approval", "Activate Vendor"),
        ("Compliance Approval", "Activate Vendor"),
        ("Enhanced Due Diligence", "Compliance Approval"),

        ("Activate Vendor", "Release First Payment"),
        ("Reject Vendor", "Close Onboarding"),
        ("Release First Payment", "Close Onboarding"),
    },

    "always_after": set(),

    "never_together": {
        ("Reject Vendor", "Activate Vendor"),
        ("Enhanced Due Diligence", "Manager Approval"),
        ("Compliance Approval", "Manager Approval"),
    },

    "activ_freq": {
        "Receive Onboarding Request": {1},
        "Classify Risk": {1},

        "Activate Vendor": {0, 1},
        "Reject Vendor": {0, 1},
        "Release First Payment": {0, 1},

        "Manager Approval": {0, 1},
        "Compliance Approval": {0, 1},
        "Close Onboarding": {0, 1},
    },

    "directly_follows": set(),
}
```

## 5.1 Orientation and semantics

- `always_before` uses ordered pairs `(A, B)`.
  - Meaning: if both activities occur, `A` must occur before `B`.
  - If `A` or `B` is absent, the constraint may be vacuously satisfied depending on the PM4Py implementation.

- `always_after` is intentionally empty.
  - The same ordering constraints are already expressed in `always_before`.
  - Populating both directions can create orientation ambiguity or redundant failures.

- `never_together` uses unordered activity pairs.
  - Meaning: the two activities must not both occur in the same trace.

- `activ_freq` maps activities to allowed occurrence-count sets.
  - `{0, 1}` means at most once.
  - `{1}` means exactly once.

- `directly_follows` is empty.
  - Direct-follows constraints are too brittle here because of concurrency, optional clarification, and optional escalation.

- `equivalence` is empty.
  - No two activities are globally equivalent in the complete process. For example, `Activate Vendor` can co-occur with `Manager Approval` or `Compliance Approval`, but not both.

## 5.2 Overlap and disagreement with other layers

### Overlap

The log skeleton duplicates several DECLARE constraints:

- `Request Clarification` before `Receive Clarification`;
- `Run Sanctions Screening` before `Sanctions Escalation`;
- `Sanctions Escalation` before `Obtain Legal Opinion`;
- `Activate Vendor` before `Release First Payment`;
- `Reject Vendor` and `Activate Vendor` never together;
- enhanced due diligence and manager approval never together;
- compliance approval and manager approval never together.

This redundancy is intentional. Log skeleton is simple and useful for fast screening and at-most-once checks.

### Disagreement

The log skeleton does not encode conditional R5 precedence:

```text
If Sanctions Escalation occurs, Obtain Legal Opinion must precede Activate Vendor.
```

It only encodes:

```text
Sanctions Escalation before Obtain Legal Opinion
```

Therefore T6 passes the log skeleton even though it fails the procedural layer under the conservative legal interpretation.

This is not an error; it reflects the limitation of activity-only skeleton constraints for conditional precedence.

# 6. Temporal Profile

```python
zeta = 2.0

temporal_profile = {
    ("Request Clarification", "Receive Clarification"): (
        48 * 3600,
        24 * 3600,
    ),
    ("Sanctions Escalation", "Obtain Legal Opinion"): (
        12 * 3600,
        6 * 3600,
    ),
    ("Activate Vendor", "Release First Payment"): (
        72 * 3600,
        36 * 3600,
    ),
}
```

## 6.1 Why temporal deviations are warnings

R12 states that compliance considers these expectations normative enough to flag but not automatically reject a case. Therefore:

- A temporal deviation does not override hard control-flow violations.
- A temporal warning does not by itself make a case non-conformant.
- A case may pass DECLARE and procedural checks but still receive a temporal warning.

Temporal profiles are statistical. They can have legitimate outliers, especially for:

- vendor response times;
- legal opinion delays;
- operational payment scheduling.

The temporal layer is therefore used for:

- monitoring;
- prioritization;
- trend detection;
- escalation to human review.

It is not used as an automatic rejection mechanism.

# 7. Resource- and Data-Aware Checks

Activity-only models cannot express:

- resource segregation;
- emergency authorization;
- conditional approval counts;
- request/receive matching across cycles;
- post-activation conditional behavior;
- missing external attributes.

The following code is executable-style pseudocode for a pandas event log.

```python
import pandas as pd
import pm4py
from datetime import timedelta


REQUIRED_COLUMNS = {"case_id", "activity", "timestamp"}
RESOURCE_COLUMN = "resource"


def _first_timestamp(df: pd.DataFrame, activity: str):
    if activity not in set(df["activity"]):
        return None
    ts = df.loc[df["activity"] == activity, "timestamp"]
    if ts.empty:
        return None
    return ts.min()


def _resources_for(df: pd.DataFrame, activity: str):
    """
    Returns:
      (resource_set, missing_attribute_flag)

    If the activity does not occur, resource_set is empty and missing flag is False.
    If the activity occurs but resource column is absent or values are missing,
    missing flag is True.
    """
    if activity not in set(df["activity"]):
        return set(), False

    if RESOURCE_COLUMN not in df.columns:
        return set(), True

    vals = df.loc[df["activity"] == activity, RESOURCE_COLUMN]
    missing = vals.isna().any()
    return set(vals.dropna()), missing


def validate_resource_and_data_layer(
    log_df: pd.DataFrame,
    emergency_df: pd.DataFrame | None = None,
    enforce_conservative_legal_precedence: bool = True,
):
    """
    Returns a findings DataFrame with columns:
      case_id, requirement, result, detail

    result values:
      pass, fail, warning, not evaluated, policy-dependent
    """
    missing_cols = REQUIRED_COLUMNS - set(log_df.columns)
    if missing_cols:
        raise ValueError(f"Missing required log columns: {missing_cols}")

    log_df = log_df.copy()

    # If timestamps are numeric hours, do not convert to datetime.
    # This version assumes datetime-compatible timestamps.
    log_df["timestamp"] = pd.to_datetime(log_df["timestamp"], errors="coerce")

    findings = []

    for case_id, g in log_df.groupby("case_id"):
        g = g.sort_values("timestamp")
        acts = set(g["activity"])

        # ---------------------------------------------------------------
        # R2/R1/R8 conditional approval and terminal outcome checks
        # ---------------------------------------------------------------
        reject = "Reject Vendor" in acts
        activate = "Activate Vendor" in acts
        manager = "Manager Approval" in acts
        compliance = "Compliance Approval" in acts

        if reject and activate:
            findings.append((case_id, "R1", "fail", "Reject and Activate in same case"))

        if activate:
            if manager and compliance:
                findings.append((case_id, "R2/R8", "fail", "Both approvals in accepted case"))
            if not manager and not compliance:
                findings.append((case_id, "R2", "fail", "Activation without applicable approval"))
        elif reject:
            if manager or compliance:
                findings.append((case_id, "R2", "fail", "Approval in rejected case"))

        # ---------------------------------------------------------------
        # R9 segregation of duties
        # ---------------------------------------------------------------
        verify_res, verify_missing = _resources_for(g, "Verify Bank Account")
        manager_res, manager_missing = _resources_for(g, "Manager Approval")
        compliance_res, compliance_missing = _resources_for(g, "Compliance Approval")
        capture_res, capture_missing = _resources_for(g, "Capture Master Data")
        activate_res, activate_missing = _resources_for(g, "Activate Vendor")

        resource_missing = any(
            [
                verify_missing,
                manager_missing,
                compliance_missing,
                capture_missing,
                activate_missing,
            ]
        )

        if resource_missing:
            findings.append(
                (
                    case_id,
                    "R9",
                    "fail",
                    "Resource attribute missing for one or more segregation checks",
                )
            )
        else:
            approval_res = manager_res | compliance_res

            if verify_res and approval_res:
                if verify_res & approval_res:
                    findings.append(
                        (
                            case_id,
                            "R9",
                            "fail",
                            "Verify Bank Account and applicable approval share resource",
                        )
                    )

            if capture_res and activate_res:
                if capture_res & activate_res:
                    findings.append(
                        (
                            case_id,
                            "R9",
                            "fail",
                            "Capture Master Data and Activate Vendor share resource",
                        )
                    )

        # ---------------------------------------------------------------
        # R10 at-most-once checks
        # ---------------------------------------------------------------
        counts = g["activity"].value_counts()
        for activity in [
            "Activate Vendor",
            "Reject Vendor",
            "Release First Payment",
        ]:
            if counts.get(activity, 0) > 1:
                findings.append(
                    (case_id, "R10", "fail", f"{activity} occurs more than once")
                )

        # ---------------------------------------------------------------
        # R3 request/receive matching
        # ---------------------------------------------------------------
        open_requests = 0
        gate_activities = {
            "Classify Risk",
            "Reject Vendor",
            "Activate Vendor",
            "Close Onboarding",
        }

        for row in g.itertuples():
            if row.activity == "Request Clarification":
                if open_requests > 0:
                    findings.append(
                        (
                            case_id,
                            "R3",
                            "fail",
                            "Second request before prior receive",
                        )
                    )
                open_requests += 1

            elif row.activity == "Receive Clarification":
                if open_requests == 0:
                    findings.append(
                        (case_id, "R3", "fail", "Receive without prior request")
                    )
                open_requests -= 1

            elif row.activity in gate_activities:
                if open_requests > 0:
                    findings.append(
                        (
                            case_id,
                            "R3",
                            "fail",
                            "Unresolved clarification before gate/outcome",
                        )
                    )
                    open_requests = 0

        # ---------------------------------------------------------------
        # R11 post-activation change behavior
        # ---------------------------------------------------------------
        activate_ts = _first_timestamp(g, "Activate Vendor")

        if activate_ts is not None:
            change_ts_list = sorted(
                g.loc[g["activity"] == "Change Bank Details", "timestamp"]
            )
            post_activation_changes = [t for t in change_ts_list if t > activate_ts]

            for change_ts in post_activation_changes:
                suspend_ts_list = sorted(
                    g.loc[g["activity"] == "Suspend Vendor", "timestamp"]
                )
                if not any(t > change_ts for t in suspend_ts_list):
                    findings.append(
                        (
                            case_id,
                            "R11",
                            "fail",
                            "Post-activation change without later suspension",
                        )
                    )

                release_ts_list = sorted(
                    g.loc[g["activity"] == "Release First Payment", "timestamp"]
                )
                verify_ts_list = sorted(
                    g.loc[g["activity"] == "Verify Bank Account", "timestamp"]
                )

                later_releases = [t for t in release_ts_list if t > change_ts]
                for release_ts in later_releases:
                    intervening_verify = any(
                        change_ts < verify_ts < release_ts
                        for verify_ts in verify_ts_list
                    )
                    if not intervening_verify:
                        findings.append(
                            (
                                case_id,
                                "R11",
                                "fail",
                                "Release after change without re-verification",
                            )
                        )

        # ---------------------------------------------------------------
        # R5/R6 emergency activation condition
        # ---------------------------------------------------------------
        escalation_ts = _first_timestamp(g, "Sanctions Escalation")
        activation_ts = _first_timestamp(g, "Activate Vendor")
        legal_ts = _first_timestamp(g, "Obtain Legal Opinion")

        activation_before_legal = (
            escalation_ts is not None
            and activation_ts is not None
            and (legal_ts is None or legal_ts > activation_ts)
        )

        if activation_before_legal:
            if enforce_conservative_legal_precedence:
                findings.append(
                    (
                        case_id,
                        "R5/R6",
                        "fail",
                        "Activation before legal opinion without enforceable emergency attestation",
                    )
                )
            else:
                if emergency_df is None or emergency_df.empty:
                    findings.append(
                        (
                            case_id,
                            "R5/R6",
                            "policy-dependent",
                            "Emergency attestation data unavailable",
                        )
                    )
                else:
                    rec = emergency_df.loc[emergency_df["case_id"] == case_id]

                    if rec.empty:
                        findings.append(
                            (
                                case_id,
                                "R5/R6",
                                "policy-dependent",
                                "No emergency attestation record",
                            )
                        )
                    else:
                        row = rec.iloc[0]
                        emergency = bool(row.get("emergency_activation", False))
                        authorization = row.get("joint_authorization_id", None)
                        roles = set(row.get("authorizing_roles", []) or [])

                        if not emergency or authorization is None or len(roles) < 2:
                            findings.append(
                                (
                                    case_id,
                                    "R5/R6",
                                    "fail",
                                    "Emergency attestation missing or insufficient",
                                )
                            )
                        elif legal_ts is None:
                            findings.append(
                                (
                                    case_id,
                                    "R5/R6",
                                    "fail",
                                    "No legal opinion after emergency activation",
                                )
                            )
                        elif legal_ts > activation_ts + timedelta(hours=24):
                            findings.append(
                                (
                                    case_id,
                                    "R5/R6",
                                    "fail",
                                    "Legal opinion arrived more than 24 hours after emergency activation",
                                )
                            )
                        else:
                            findings.append(
                                (
                                    case_id,
                                    "R5/R6",
                                    "warning",
                                    "Attested emergency activation; legal opinion within 24 hours",
                                )
                            )

    return pd.DataFrame(
        findings,
        columns=["case_id", "requirement", "result", "detail"],
    )
```

# 9. Contradictions, Redundancies, and Blind Spots

## 9.1 Duplicated enforcement across layers

Several requirements are intentionally enforced in more than one layer:

- **R1:** procedural exclusive branch, DECLARE `noncoexistence`, log skeleton `never_together`.
- **R3:** procedural loop, DECLARE response/precedence, custom matching.
- **R4:** procedural escalation branch, DECLARE response.
- **R7:** procedural placement after classification, DECLARE precedence.
- **R8:** procedural exclusive routes, DECLARE noncoexistence, log skeleton never-together.
- **R10:** procedural at-most-once structure, log skeleton `activ_freq`, custom counts.

This is defense-in-depth. However, duplication can produce contradictory findings. For example, T6 passes DECLARE because DECLARE only encodes R4, but fails procedural because procedural encodes conservative R5.

The deployment design must therefore preserve layer-level verdicts rather than collapse them into a single boolean.

## 9.2 Constraints that become inconsistent under loops or post-activation changes

### R3 and loops

DECLARE `response(Request Clarification, Receive Clarification)` can be satisfied by one receive after multiple requests if the semantics are not carefully checked. The procedural loop and custom matching are needed to enforce one-to-one cycles.

### R10 versus R11

R10 says:

> `Release First Payment` occurs at most once.

R11 says:

> another `Verify Bank Account` must occur before any later `Release First Payment`.

These are inconsistent if “later `Release First Payment`” means the same activity label in the same case. If `Release First Payment` can occur at most once, there cannot be a later `Release First Payment` in that case.

Possible resolutions:

1. Treat R10 as applying only to the first onboarding payment and add a distinct activity such as `Release Subsequent Payment`.
2. Relax R10 to allow `Release First Payment` multiple times only after suspension and re-verification.
3. Interpret “later release” as occurring in a new case or re-onboarding case.

The implemented model enforces R10 as stated, so R11’s later-release condition is vacuous unless a distinct payment activity or relaxed frequency rule is introduced.

### R11 and DECLARE response

The conservative DECLARE rule:

```text
Change Bank Details -> Suspend Vendor
```

applies to every change, including changes before activation. R11 only requires suspension for changes after activation. Therefore the DECLARE rule is an overconstraint. It is acceptable as a warning approximation, but the custom post-activation check is authoritative.

## 9.3 Vacuity in declarative rules

Many declarative rules pass vacuously:

- R4 is vacuous when no `Sanctions Escalation` occurs.
- R7 is vacuous when no `Sanctions Escalation` occurs.
- R11 is vacuous when no `Change Bank Details` occurs.
- Temporal profile rules are not evaluated when one activity in the pair is absent.
- `noncoexistence` rules can pass because one activity is absent.

Vacuous passes should not be interpreted as positive compliance evidence. They should be reported as “not applicable” or “not evaluated” in user-facing reports.

## 9.4 Missing case attributes

The supplied log lacks several attributes needed for full policy validation:

1. **Emergency activation flag.**  
   Required for R6.

2. **Joint authorization identifier.**  
   Required to prove emergency activation was authorized.

3. **Authorizing roles.**  
   Required to verify joint authorization.

4. **Clarification request identifiers.**  
   Required for robust matching of request and response cycles.

5. **Risk classification outcome.**  
   The log has `Classify Risk`, but not whether the result was low or high risk. The model infers route from subsequent approvals.

6. **Payment sequence.**  
   Required to distinguish first payment from later payments if R11 is made non-vacuous.

7. **Bank account identifier.**  
   Required to know whether `Verify Bank Account` verifies the same account after `Change Bank Details`.

## 9.5 Timestamp-only limitations

Timestamps can establish order and duration, but cannot by themselves establish:

- whether activities were intended to be concurrent;
- whether a verification relates to the same bank account;
- whether a legal opinion relates to the same escalation;
- whether an activation was emergency-authorized;
- whether a later payment is a first payment or subsequent payment;
- whether a clarification response answers the immediately preceding request.

Therefore, activity-only conformance is necessary but not sufficient for full policy compliance.

## 9.6 Risk of treating support/confidence metadata as normative truth

The DECLARE dictionary uses:

```python
"support": 1.0,
"confidence": 1.0,
```

This is a modeling convention, not empirical evidence. If a DECLARE model is discovered from a log, support and confidence reflect observed frequency, not policy correctness. Treating discovered metadata as normative truth can:

- encode deviations as rules;
- miss rare but mandatory controls;
- overfit to historical behavior;
- hide policy violations that are common in the log.

In this task, the rules are normative requirements, so support/confidence should be governed by policy, not inferred from data.

# 10. Deployment Recommendation

## 10.1 Canonical procedural representation

Use the process tree as the human-readable canonical procedural specification.

For machine execution, consider converting the process tree to an accepting Petri net if your PM4Py tooling provides more stable conformance semantics for loops and optional branches. The normative behavior remains the same, but the Petri net makes token flow explicit.

The canonical procedural model should represent only the stable backbone:

- receipt;
- concurrent checks;
- clarification loop;
- classification;
- exclusive terminal outcome;
- activation/payment/close;
- conservative escalation branch with legal opinion before activation.

Post-activation changes should not be forced into the procedural tree unless the organization decides they are part of the canonical onboarding process.

## 10.2 Authoritative policy layer

There is no single authoritative layer for all requirements. The authoritative layer depends on the requirement type:

| Requirement type | Authoritative layer |
|---|---|
| Stable sequence, exclusive outcomes, at-most-once backbone | Procedural model + log skeleton |
| Eventual response, precedence, noncoexistence | DECLARE |
| Resource segregation | Resource/data-aware validation |
| Emergency authorization | External data-aware validation |
| Conditional approval counts | Data-aware validation |
| Post-activation conditional behavior | Data-aware validation |
| Temporal expectations | Temporal profile, warning-only |

For R5/R6, the implemented authoritative choice is conservative legal precedence. The emergency exception is not authoritative from the event log alone.

## 10.3 Recommended conformance-check order

Run checks in this order:

1. **Schema and attribute completeness.**
   - Verify `case_id`, `activity`, `timestamp`.
   - Verify `resource` if R9 is enforced.
   - Verify external emergency attributes if R6 is enabled.

2. **Procedural backbone conformance.**
   - Detect impossible sequences, missing classification, invalid outcomes, and conservative R5 violations.

3. **DECLARE conformance.**
   - Detect missing responses, precedence violations, and forbidden co-occurrences.

4. **Log-skeleton conformance.**
   - Detect at-most-once violations and simple ordering violations.

5. **Resource/data-aware conformance.**
   - Detect four-eyes violations, conditional approval-count violations, request/receive matching issues, post-activation issues, and emergency-policy issues.

6. **Temporal-profile evaluation.**
   - Compute warnings only.
   - Do not allow temporal warnings to override hard control-flow failures.

7. **Aggregation.**
   - Produce a layered verdict, not a single boolean.
   - Identify the earliest decisive violation.
   - Mark policy-dependent findings separately.

## 10.4 Surfacing contradictory findings

When layers disagree, expose the disagreement explicitly.

Example for T6:

```text
Procedural: fail — activation before legal opinion after escalation.
DECLARE: pass — legal opinion eventually occurred.
Log Skeleton: pass — escalation occurred before legal opinion.
Temporal: pass — legal opinion arrived within expected statistical bounds.
Resource/data: fail — no emergency attestation available.
Overall: fail under conservative policy.
Policy note: if emergency attestation is supplied and valid, downgrade to warning or policy-dependent.
```

This avoids silently combining incompatible policies.

## 10.5 Recommended reporting format

For each case, report:

1. **Overall status:**
   - `pass`
   - `fail`
   - `warning`
   - `policy-dependent`
   - `not evaluated`

2. **Layer statuses:**
   - procedural
   - DECLARE
   - log skeleton
   - temporal
   - resource/data

3. **Earliest decisive violation:**
   - activity;
   - timestamp;
   - requirement;
   - layer;
   - explanation.

4. **Required remediation:**
   - add missing attribute;
   - provide emergency authorization;
   - reconcile R10/R11;
   - resolve unanswered clarification;
   - enforce segregation of duties;
   - review temporal outlier.

5. **Policy interpretation:**
   - conservative legal precedence;
   - attested emergency exception;
   - warning-only temporal policy.

This layered approach preserves the distinction between control-flow constraints, resource/data constraints, temporal expectations, and unresolved policy conflicts.
