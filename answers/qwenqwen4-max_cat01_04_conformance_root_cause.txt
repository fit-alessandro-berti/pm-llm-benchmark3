            log.append(trace)
    base = datetime(2026, 1, 1, tzinfo=timezone.utc)
    for i, trace in enumerate(log):
        trace.attributes["concept:name"] = f"{period}-{i:05d}"
        trace.attributes["period"] = period
        for j, event in enumerate(trace):
            event["extracted_index"] = j          # the ONLY order we actually possess
            if "time:timestamp" not in event:
                event["time:timestamp"] = base + timedelta(hours=i, seconds=j)
    return log

# 3. Transparent normalization pipeline
# --------------------------------------------------------------------------- #
def _filter_out(log, labels):
    try:
        return pm4py.filter_event_attribute_values(
            log, "concept:name", set(labels), level="event", retain=False)
    except Exception:
        out = EventLog(attributes=log.attributes)
        for trace in log:
            new = Trace(attributes=trace.attributes)
            for event in trace:
                if event["concept:name"] not in labels:
                    new.append(event)
            if len(new):
                out.append(new)
        return out


def normalize(log, map_labels=True, remove_technical=True, order_repair=False,
              also_drop=()):
    out = copy.deepcopy(log)
    if map_labels:
        for trace in out:
            for event in trace:
                old = event["concept:name"]
                if old in LABEL_MAP:
                    event["concept:name_original"] = old   # provenance preserved
                    event["concept:name"] = LABEL_MAP[old]
    if remove_technical:
        out = _filter_out(out, TECHNICAL)
    if also_drop:                                   # DIAGNOSTIC ONLY / illegitimate
        out = _filter_out(out, also_drop)
    if order_repair:
        # WHAT-IF: assumes the ETL hypothesis (decision truly precedes notify).
        # It cannot validate the hypothesis; it bounds recoverable fitness.
        for trace in out:
            events = list(trace)
            names = [e["concept:name"] for e in events]
            for k in range(len(names) - 1):
                if names[k] == "Notify" and names[k + 1] == "Decision":
                    events[k], events[k + 1] = events[k + 1], events[k]
                    names[k], names[k + 1] = names[k + 1], names[k]
            trace[:] = events
    return out

# 4. Alignment diagnostics + fitness (+ optional precision)
# --------------------------------------------------------------------------- #
def _get(dic, keys, default=None):
    for key in keys:
        if isinstance(dic, dict) and key in dic:
            return dic[key]
    return default


def alignment_report(log, model, tag, period=None, condition=None, precision=False):
    net, im, fm = model
    diags = pm4py.conformance_diagnostics_alignments(log, net, im, fm)
    costs = [_get(d, ("cost",), 0) for d in diags]
    fits = [_get(d, ("fitness",), 1.0) for d in diags]
    strict = 100.0 * sum(1 for c in costs if c == 0) / max(len(costs), 1)
    log_moves, model_moves = Counter(), Counter()
    for diag in diags:
        for move in _get(diag, ("alignment",), []):
            lab_log, lab_mod = move[0], move[1]
            if lab_mod == ">>" and lab_log is not None:
                log_moves[lab_log] += 1
            elif lab_log == ">>" and lab_mod is not None:
                model_moves[lab_mod] += 1
    official = pm4py.fitness_alignments(log, net, im, fm)
    result = {
        "tag": tag, "cases": len(log),
        "strict_fit_%": round(strict, 1),
        "mean_trace_fit": round(sum(fits) / max(len(fits), 1), 3),
        "mean_cost": round(sum(costs) / max(len(costs), 1), 3),
        "pm4py_fitness": _get(official, ("log_fitness", "percentage_of_fitting_traces",
                                        "averageFitness", "percFitTraces")),
        "log_moves": log_moves.most_common(5),
        "model_moves": model_moves.most_common(5),
    }
    if precision:
        for name, fn in (("precision_alignments", pm4py.precision_alignments),
                         ("precision_tbr", pm4py.precision_token_based_replay)):
            try:
                result[name] = round(float(fn(log, net, im, fm)), 3)
            except Exception as exc:
                result[name] = f"n/a ({type(exc).__name__})"
    if period and condition:
        result["reported_fit_%"] = REPORTED_FIT.get((period, condition), "n/a")
    print(f"\n--- {tag} ---")
    for key, value in result.items():
        if key != "tag":
            print(f"  {key:18s}: {value}")
    return result


def dfg_report(log, tag, top=8):
    dfg, start, end = pm4py.discover_dfg(log)
    print(f"\n[DFG] {tag}: start={dict(start)} end={dict(end)}")
    for (a, b), n in sorted(dfg.items(), key=lambda kv: -kv[1])[:top]:
        print(f"   {a:22s} -> {b:22s} {n}")
    for probe in (("Standard Review", "Notify"), ("Notify", "Decision"),
                  ("Enhanced Review", "Decision"), ("Risk Screen", "Queue Routed"),
                  ("Compliance Check", "Supervisor Sign-off"),
                  ("Standard Review", "Supervisor Sign-off")):
        if probe in dfg:
            print(f"   probe {probe} = {dfg[probe]}")
    return dfg


def provenance_check(log, period):
    n_events = sum(len(t) for t in log)
    mean = n_events / max(len(log), 1)
    print(f"[provenance] {period}: cases={len(log)} events={n_events} "
          f"mean={mean:.2f} reported={REPORTED_MEAN_EVENTS[period]}")
    if abs(mean - REPORTED_MEAN_EVENTS[period]) > 0.05:
        print("   WARNING: events/case mismatch -> truncated variants or unlisted events.")


def stratify_by_value(log, threshold=50000):
    hi, lo = EventLog(), EventLog()
    for trace in log:
        amount = trace.attributes.get("amount", trace.attributes.get("case:amount"))
        if amount is None:
            print("[stratification] UNAVAILABLE: case amount not in this reconstruction. "
                  "Data-aware conformance cannot be evaluated; join the attribute first.")
            return None
        (hi if float(amount) > threshold else lo).append(trace)
    return hi, lo

## 7. Provisional conclusion

**Primary explanation of the measured collapse (confidence: high, ~85%).** The June–July drop is overwhelmingly a *measurement* event, not a behavioural one. Three dated, independently scoped pipeline changes each wrote a distinct signature into the same observable: a partially rolled-out client rename (`Classify``Risk Screen`), injection of a resource-less gateway event (`Queue Routed`, 530/650 cases), and replacement of source time by warehouse-ingestion time for exactly `Decision` and `Notify`. The staged ladder (32588189) matches those mechanisms one-for-one, the inversion footprint equals the ETL blast radius, the 40-case audit confirms the rename, and the reversal counts match the DFG exactly. The August–September drop has a *different* dominant cause: the normative tree was never updated for the 1 August EUR 50,000 sign-off policy (6686). Baseline adequacy in April–May, with the same tree, rules out chronic model obsolescence or endemic noncompliance as the trigger.

**Separate assessment of a real compliance problem (confidence: low-moderate, ~40–45% that some genuine bypasses exist).** 55 enhanced reviews (25 + 30) have no recorded compliance check, against 180/180 in the baseline. No other activity ever goes missing, so generic ETL loss is unlikely; this is either a compliance-emitter/channel loss or targeted bypass. The rate did not rise after June and did not fall after the memo — evidence that fits neither a misconduct wave nor a memo effect. `Manual Override` on the standard path plausibly reflects gateway-induced routing rework, i.e. an IT change causing real behaviour. The 60 sign-offs on the standard path also raise a routing/authorization question.

**Immediate actions.** IT: re-extract with source timestamps and sequence numbers, publish the technical-event allowlist and per-activity emitter/resource provenance, and reconcile the events-per-case and fitness-definition discrepancies. Process owners: version the model with an effective date and a value-guarded sign-off branch. Compliance: dual-source reconcile all 55 check-less cases plus a stratified random sample of the 59 unaudited reversals; check signer authorization.

**Must not be claimed yet:** that employees became noncompliant; that all 70/45 reversals are artefacts; that 86% fitness with an optional sign-off demonstrates compliance; that flat appeals exonerate anyone; or that any absolute reported percentage is reproducible.

## 8. Model-governance recommendation

1. **Model versioning with effective dates.** Every normative model carries a version, owner, effective-from/to date and a link to the policy text it encodes; conformance jobs must select the model version valid *per case timestamp*, and any policy change automatically opens a model-update ticket with an SLA. Never check August cases against a May model.
2. **Event-semantic contracts.** Each activity has a registered contract: emitting system, business meaning, allowed UI aliases, resource semantics, lifecycle and cardinality. Label renames require a versioned alias mapping approved by the process owner plus evidence that the business function is unchanged; unmapped labels fail the pipeline loudly instead of silently becoming log moves.
3. **Technical-event policy.** A maintained allowlist classifies every event as business, technical or lifecycle. Technical events are retained in the raw log with their emitter identity but excluded from business conformance by explicit, logged rules — never by "drop unknown labels". Real user actions (e.g. `Manual Override`) may never be filtered to improve fitness.
4. **Timestamp provenance and ordering authority.** Persist source timestamp, ingestion timestamp, extractor version and a monotone per-case source sequence number; ordering must use the authoritative source field, with ingestion time kept only for lineage. Emit automated alerts for inversions, ties, batch-window clustering and per-activity clock drift.
5. **Data-aware constraints and authorization.** Guard conditions (amount > EUR 50,000), resource/role authorizations and four-eyes rules must be expressed as data-aware constraints (e.g. guarded/DMN-augmented models) or enforced by stratified conformance. Fitness gains from optional activities must always be reported alongside precision, generalization and per-stratum results to expose overpermissive models.
6. **Post-release validation gate.** Every UI, gateway, ETL or policy release triggers a shadow conformance run on a pre/post window: variant and DFG diff, events-per-case and resource-nullity deltas, alignment move attribution, change-point detection on daily fitness, and a targeted manual audit. Publish numerator/denominator definitions, and require sign-off that observed fitness movement is decomposed into measurement versus behaviour before any compliance conclusion is drawn.
