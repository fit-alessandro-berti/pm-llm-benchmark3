```python
# Revised normative model M1 + policy-overlay governance (self-contained; no file I/O)
from datetime import datetime, timedelta
from pm4py.objects.powl.obj import StrictPartialOrder, OperatorPOWL, Transition, SilentTransition
from pm4py.objects.process_tree.obj import Operator

# ---------------------------------------------------------------------
# 1. Control-flow backbone (POWL)
# ---------------------------------------------------------------------
# Scope note: this backbone governs FULL_ONBOARDING cases only.
# PRE_SCREEN (rejected leads), REACTIVATION and WITHDRAWAL are separate
# case types (Section 6, scope rules); forcing them through one POWL
# would require silent-transition hacks that hide policy, so they are
# modeled as scoped sibling models / data-guarded variants instead.

def T(label):
    return Transition(label=label)

def make_info_request_loop():
    """Zero or more Request_More_Info cycles.

    Approximation: XOR( tau , LOOP(do=Request_More_Info, redo=tau) )
    => RMI may occur 0..n times wherever the loop node is placed.
    Known limitations (enforced in the data layer, not in POWL):
      * unbounded iterations (cap via data rule, e.g. <= 5);
      * does not guarantee the *answer* arrives before activation
        (data rule: no Activate while info_requests_open > 0);
      * placement concurrent with the checks models 'RMI while
        checking', matching observed practice (e.g. SUP-707).
    """
    rmi = T("Request_More_Info")
    loop = OperatorPOWL(operator=Operator.LOOP, children=[rmi, SilentTransition()])
    return OperatorPOWL(operator=Operator.XOR, children=[SilentTransition(), loop])

def make_standard_route():
    sanctions, quality, info = T("Sanctions_Check"), T("Quality_Review"), make_info_request_loop()
    gather = StrictPartialOrder(nodes=[sanctions, quality, info])          # concurrent, incl. RMI
    manual, activate = T("Manual_Procurement_Approval"), T("Activate_Supplier")
    route = StrictPartialOrder(nodes=[gather, manual, activate])
    route.order.add_edge(gather, manual)
    route.order.add_edge(manual, activate)
    return route

def make_low_risk_auto_route():
    # PD-3-style: sanctions + automated master-data validation first,
    # Auto_Approve replaces manual approval, Quality_Review may follow
    # (provisional) activation. The 'within 2 business days' deadline and
    # the LOW-risk/DE/approved-group guard are DATA rules, not POWL.
    sanctions, mdv, info = T("Sanctions_Check"), T("Master_Data_Validation"), make_info_request_loop()
    gather = StrictPartialOrder(nodes=[sanctions, mdv, info])
    auto, activate, quality = T("Auto_Approve"), T("Activate_Supplier"), T("Quality_Review")
    route = StrictPartialOrder(nodes=[gather, auto, activate, quality])
    route.order.add_edge(gather, auto)
    route.order.add_edge(auto, activate)
    route.order.add_edge(activate, quality)
    return route

def make_emergency_route():
    # FD-7-style: sanctions is a NON-WAIVABLE precondition of activation;
    # activation may precede quality review and (retrospective) manual
    # approval, which are concurrent with each other but must both
    # complete before notification/closure. Spend cap, daily monitoring
    # and the 2-business-day deadline are DATA rules.
    sanctions, activate = T("Sanctions_Check"), T("Activate_Supplier")
    quality, retro = T("Quality_Review"), T("Manual_Procurement_Approval")  # approval_phase=retrospective
    retro_block = StrictPartialOrder(nodes=[quality, retro])
    route = StrictPartialOrder(nodes=[sanctions, activate, retro_block])
    route.order.add_edge(sanctions, activate)
    route.order.add_edge(activate, retro_block)
    return route

register = T("Register_Supplier")
decision = OperatorPOWL(
    operator=Operator.XOR,
    children=[make_standard_route(), make_low_risk_auto_route(),
              make_emergency_route(), T("Reject_Supplier")],
)
notify, close = T("Notify_Requester"), T("Close_Case")

M1 = StrictPartialOrder(nodes=[register, decision, notify, close])
M1.order.add_edge(register, decision)
M1.order.add_edge(decision, notify)
M1.order.add_edge(notify, close)

print(M1)
```

- SUP-703 was ineligible for PD-3 (HIGH-risk, foreign, new) and transacted EUR 180,000; SUP-708 activated 40 minutes before sanctions clearance with no overlay; SUP-712's requester and approver resolve to one human via two aliases; SUP-715 recorded no COI disclosure; SUP-711's duplicate Register events share one source transaction ID.
- Notification missingness rose 2%  29% exactly at the CRM migration; ~70% mirror rate.
- 312 Auto_Approve cases in June–July: 289 eligible, 14 clearly ineligible, 9 unknown; 47 Request_More_Info cases, none activating before evidence arrived; 6 post-pilot Auto_Approve cases in early August.

**Disputed (depend on authority, scope, identity, or logging assumptions):**
- Whether PD-3 validly *suspended* CS-9 for eligible cases (steering-committee minutes vs. the Compliance observer's later denial).
- Whether a factual emergency on 2026-07-13 extends FD-7 without a written extension (SUP-705).
- Whether rejected leads are M0 cases (SUP-709); whether reactivations require full re-onboarding (SUP-710); whether QA_Clearance  Quality_Review at case time (SUP-713).
- Whether SUP-706's CRM message is attributable to the case (mutable, late-entered link).
- Whether M0's high deviation rate evidences noncompliance or obsolescence.

**Unproved on this record:** master-data validation for pilot cases; the 9 unknown-eligibility cases; deliberate concealment in SUP-712; actual notification in SUP-706.

**Burden-of-proof approach:**
- *Declaring noncompliance* requires (i) a binding norm applicable to the case **at event time** (authority + effective window + scope), (ii) a genuine deviation **after** data correction (dedup, synonym mapping, CRM corroboration), and (iii) materiality (the deviation touches a control objective: sanctions-first, approval gate, SoD, COI, caps). Doubt arising from scope ambiguity or known instrumentation failure resolves *against* a noncompliance finding — but is recorded as a model/scope/data defect requiring remediation.
- *Accepting an exception* places the burden on its proponent to prove, with **contemporaneous** records: (i) the instrument was in force at event time; (ii) every eligibility predicate, from an eligibility snapshot; (iii) every safeguard of the instrument (caps, monitoring, deadlines, retrospective steps); (iv) the behavior stayed within the instrument's scope. Failure of any element defeats the exception. Post-hoc belief (SUP-714) and factual necessity without extension (SUP-705) cannot satisfy (i).
- *Evidence hierarchy:* immutable system timestamps and stable IDs > mirrored records > mutable late-entered links > recollection.

## 2. Prosecution Brief

**Formal authority and hierarchy.** CS-9 is the only controlled, approved norm. PD-3 and FD-7 are time-boxed overlays, not amendments — and the Compliance observer expressly denied approving any weakening of CS-9. Whatever authority a steering committee has to run a *monitored exception*, it cannot silently rewrite the controlled standard, and it certainly cannot outlive its own window. Outside an overlay's predicates and dates, CS-9 controls in full.

**Expiry is a bright line.** FD-7 "expired automatically." SUP-705 (2026-07-13) repeats the emergency pattern with no written extension; "the flood emergency still factually existed" converts every deadline into a negotiation. SUP-714 (2026-08-02) and the six August Auto_Approve cases rest on a belief that permanent approval was "imminent." A norm that can be extended by expectation is not a norm.

**Deviations hidden by M0.** Conformance to M0 is not compliance. SUP-712 fits M0 perfectly while its requester and approver are the same human behind two aliases — a direct breach of CS-9's SoD rule. SUP-715 fits M0 while a spousal relationship in one reporting line went undisclosed contrary to COI policy. M0's silence is not a safe harbor; the resource rules bind independently of the model.

**Normalization of exceptions.** A pilot framed as a "monitored exception" processed 312 cases in two months, including 14 clearly ineligible and 9 undocumented — and continued after expiry. SUP-703 is the ratchet made visible: an operator "selected pilot" for a HIGH-risk new foreign supplier, bypassing the manual gate entirely, with a EUR 180,000 first purchase. **High-risk cases.** SUP-703 and SUP-708 (both flagged by Internal Audit). SUP-708 breaches the one invariant *no overlay ever waived* — even FD-7 kept sanctions-before-activation. "No transactions occurred before clearance" is luck, not control: the control is ex ante, and sanctions exposure is typically strict-liability.

**Desirable outcomes do not retroactively amend policy.** SUP-707's diligence, SUP-713's equivalent content, and SUP-714's would-have-qualified eligibility are irrelevant to what the norm required at the time. The remedy is prospective amendment, not retroactive blessing.

**Concessions.** The prosecution does not contend that check *order* is a deviation (M0 concurrency), that SUP-707's info requests are misconduct, that SUP-711 is anything but a migration duplicate, that SUP-706 proves a missed notification given the 2%29% instrumentation break, or that M0's requirement of Quality_Review before rejection is sound — CS-9 itself permits abbreviated pre-screening (SUP-709). ## 6. Revised Normative Architecture

**Layer 0 — Case typing and scope (organizational policy + data attribute `case_type`).** Four types: `FULL_ONBOARDING` (the backbone below); `PRE_SCREEN` (intake leads; may end in Reject_Supplier after Sanctions_Check alone — resolving SUP-709 consistently with CS-9's abbreviation clause); `REACTIVATION` (Sanctions_Check + approval mandatory; Quality_Review only if prior certification lapsed — resolving SUP-710); `WITHDRAWAL` (requester-initiated; Close without decision). *Not in POWL:* forcing these into one model would hide policy in silent transitions.

**Layer 1 — Control-flow backbone (POWL).** Concurrent checks, three approval routes (standard / low-risk auto / emergency-provisional), an RMI loop, reject, notify-before-close. POWL is the right formalism for ordering, concurrency, choice, and the loop approximation — nothing more.

**Layer 2 — Data-aware guards (DECLARE / log-skeleton-style constraints + pandas checks).** Route guards (`Auto` only if risk=LOW  country=DE  group approved, snapshot at decision time); the non-waivable precedence Sanctions  Activate across *all* routes; timed response (Quality_Review  2 business days after provisional activation); spend cap  EUR 20,000 until retrospective approval; no activation while info requests are open; notification existence with CRM correlation. These are attribute/time/amount conditions POWL cannot express.

**Layer 3 — Resource and COI rules (organizational + pandas identity checks).** Alias-resolved requester  approver; reviewer–approver COI disclosure; four-eyes for HIGH-risk; effective-dated delegation registry.

**Layer 4 — Policy-overlay governance (organizational).** Versioned overlays with `rule_id`/`version`, effective windows, **hard expiry enforced by the workflow engine** (route unavailable after `to` date) or a written extension *before* lapse; eligibility snapshots captured at decision; a monitored-exception register; a ratification path for pilots (PD-3  permanent amendment proposal).

**Layer 5 — Data collection (data engineering).** Immutable append-only log; stable IDs; dedup keys; ontology registry; mirror-rate SLA monitoring (alert when mirror rate drops, not when auditors ask).

```python
# =====================================================================
# M1: Revised normative backbone (POWL) + policy-overlay governance.
# Self-contained: no file I/O. Requires pm4py.
# =====================================================================
from datetime import datetime, timedelta
from pm4py.objects.powl.obj import (
    StrictPartialOrder, OperatorPOWL, Transition, SilentTransition,
)
from pm4py.objects.process_tree.obj import Operator

# SCOPE NOTE: this backbone governs FULL_ONBOARDING cases only.
# PRE_SCREEN (rejected leads), REACTIVATION and WITHDRAWAL are separate
# case types (Layer 0). Encoding them here would require silent-transition
# hacks that hide policy inside the model; they are governed by scope
# rules and small sibling models instead.

def T(label):
    return Transition(label=label)

def make_info_request_loop():
    """Zero or more Request_More_Info cycles, placed CONCURRENT with the
    checks (matching observed practice, e.g. SUP-707: RMI between
    Sanctions_Check and Quality_Review).

    Approximation: XOR( tau , LOOP(do=Request_More_Info, redo=tau) ).
    Known limitations, enforced in the DATA layer rather than POWL:
      * iterations are unbounded here (data rule: cap, e.g. <= 5);
      * the loop does not guarantee the ANSWER arrives before activation
        (data rule: no Activate_Supplier while info_requests_open > 0);
      * placement permits RMI only before the decision, per N5.
    """
    rmi = T("Request_More_Info")
    loop = OperatorPOWL(operator=Operator.LOOP,
                        children=[rmi, SilentTransition()])
    return OperatorPOWL(operator=Operator.XOR,
                        children=[SilentTransition(), loop])

def make_standard_route():
    sanctions, quality, info = (T("Sanctions_Check"), T("Quality_Review"),
                                make_info_request_loop())
    gather = StrictPartialOrder(nodes=[sanctions, quality, info])  # concurrent
    manual, activate = T("Manual_Procurement_Approval"), T("Activate_Supplier")
    route = StrictPartialOrder(nodes=[gather, manual, activate])
    route.order.add_edge(gather, manual)
    route.order.add_edge(manual, activate)
    return route

def make_low_risk_auto_route():
    # PD-3-style: Sanctions + automated Master_Data_Validation first;
    # Auto_Approve replaces manual approval; Quality_Review may follow
    # (provisional) activation. The LOW-risk/DE/approved-group guard and
    # the 2-business-day quality deadline are DATA rules, not POWL.
    sanctions, mdv, info = (T("Sanctions_Check"), T("Master_Data_Validation"),
                            make_info_request_loop())
    gather = StrictPartialOrder(nodes=[sanctions, mdv, info])
    auto, activate = T("Auto_Approve"), T("Activate_Supplier")
    quality = T("Quality_Review")  # retrospective; deadline enforced in data
    route = StrictPartialOrder(nodes=[gather, auto, activate, quality])
    route.order.add_edge(gather, auto)
    route.order.add_edge(auto, activate)
    route.order.add_edge(activate, quality)
    return route

def make_emergency_route():
    # FD-7-style: Sanctions_Check is a NON-WAIVABLE precondition of
    # activation; Quality_Review and retrospective Manual_Procurement_
    # Approval may follow activation, concurrently, but both must complete
    # before notification/closure. Spend cap, daily monitoring and the
    # 2-business-day deadline are DATA rules. The retrospective approval is
    # distinguished by event attribute approval_phase='retrospective'.
    sanctions, activate = T("Sanctions_Check"), T("Activate_Supplier")
    retro_block = StrictPartialOrder(nodes=[T("Quality_Review"),
                                            T("Manual_Procurement_Approval")])
    route = StrictPartialOrder(nodes=[sanctions, activate, retro_block])
    route.order.add_edge(sanctions, activate)
    route.order.add_edge(activate, retro_block)
    return route

register = T("Register_Supplier")
decision = OperatorPOWL(
    operator=Operator.XOR,
    children=[make_standard_route(), make_low_risk_auto_route(),
              make_emergency_route(), T("Reject_Supplier")],
)
notify, close = T("Notify_Requester"), T("Close_Case")

M1 = StrictPartialOrder(nodes=[register, decision, notify, close])
M1.order.add_edge(register, decision)
M1.order.add_edge(decision, notify)
M1.order.add_edge(notify, close)
print(M1)

def d(s):
    return datetime.fromisoformat(s)

def active_overlays(ts):
    """Overlay is active from 00:00 on 'from' through 23:59:59 on 'to'."""
    return [o for o in OVERLAYS
            if d(o["from"]) <= ts and
            (o["to"] is None or ts < d(o["to"]) + timedelta(days=1))]

def route_authority(route, snapshot, ts):
    """Return the rule_id authorizing `route` at time `ts`, else None.
    Route choice is UNCONSTRAINED in the POWL; authority is enforced here
    and should additionally be hard-blocked by the workflow engine."""
    if route == "standard":
        return "CS-9"
    for o in active_overlays(ts):
        if o["route"] == route and all(snapshot.get(k) == v
                                       for k, v in o["guard"].items()):
            return o["rule_id"]
    return None

def sanctions_before_activation(events):
    """The one invariant no overlay waives (cf. SUP-708)."""
    s = [e["ts"] for e in events if e["activity"] == "Sanctions_Check"]
    a = [e["ts"] for e in events if e["activity"] == "Activate_Supplier"]
    return bool(s and a) and min(s) < min(a)

def dedup(events):
    """Migration-duplicate removal via immutable source transaction ID
    (cf. SUP-711)."""
    seen, out = set(), []
    for e in events:
        key = (e["case"], e.get("source_txn_id", id(e)))
        if key not in seen:
            seen.add(key)
            out.append(e)
    return out

    return (ALIAS_REGISTRY.get(requester_raw, requester_raw)
```

**Event/data-collection proposal.** Every event carries: `case_id`, `case_type`, `activity` (ontology-registry label with effective-dated synonyms, e.g. QA_ClearanceQuality_Review), `effective_time` (immutable, when it happened), `record_time` (immutable, when logged), `source_system`, `source_txn_id` (dedup key), `actor_id` (canonical, alias-resolved), `actor_role`, `rule_version_id`/`overlay_id` stamped on guarded events (Auto_Approve, provisional Activate), `eligibility_snapshot` (risk, country, group status at decision time), `amount`/`currency`, `approval_phase`, `delegation_ref`, `coi_disclosure_id`, and `info_request_id/status`. Notifications carry a stable UUID `notification_id` plus an **immutable `case_id` foreign key written by the CRM at send time**; nightly mirroring with an SLA alert on mirror rate. Storage is append-only; mutable text links like SUP-706's are deprecated. Emergency cases link one monitoring record per day. This schema makes every element of the exception burden of proof — authority, eligibility, safeguards, scope — checkable from contemporaneous records rather than reconstructed belief.
