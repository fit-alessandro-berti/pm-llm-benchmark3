def format_log_for_analysis(log: pm4py.EventLog) -> pd.DataFrame:

def slice_monthly_windows(df: pd.DataFrame, start_date: str, end_date: str) -> List[pd.DataFrame]:
    """Filters dataframe into monthly windows."""
    df['time:timestamp'] = pd.to_datetime(df['time:timestamp'])
    monthly_logs = []
    current_start = pd.to_datetime(start_date)
    current_end = pd.to_datetime(end_date)
    
    while current_start <= current_end:
        next_month = current_start.replace(day=28) + pd.Timedelta(days=4)
        next_month = next_month - pd.Timedelta(days=next_month.day)
        
        mask = (df['time:timestamp'] >= current_start) & (df['time:timestamp'] < next_month)
        monthly_logs.append(df[mask])
        current_start = next_month
    return monthly_logs

def compute_monthly_dfgs(monthly_dfs: List[pd.DataFrame]) -> List:
    """Discovers DFG for each month."""
    dfgs = []
    for df in monthly_dfs:
        dfg = pm4py.discover_dfg(df)
        dfgs.append(dfg)
    return dfgs
