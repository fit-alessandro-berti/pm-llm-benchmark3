    x = np.asarray(x, float); return x.var(ddof=0)/x.mean()

    x = np.asarray(x, float); x = x - x.mean()
    return float((x[:-1]*x[1:]).sum()/(x**2).sum())

Change-point heuristic:

```python
def best_split(values, dates):
    v = np.asarray(values, float)
    best = None
    for k in range(1, len(v)):
        sse = ((v[:k]-v[:k].mean())**2).sum() + ((v[k:]-v[k:].mean())**2).sum()
        if best is None or sse < best[0]: best = (sse, k)
    sse_tot = ((v - v.mean())**2).sum()
    return dates[best[1]-1], dates[best[1]], 1-best[0]/sse_tot  # boundary: last of left, first of right, R^2-ish gain
```

def binseg(dates, vals, depth=0, max_splits=2, min_seg=3, res=None):
    if res is None: res=[]
    v=np.asarray(vals,float)
    if len(v)<2*min_seg or depth>=max_splits: return res
    best=None
    for k in range(min_seg, len(v)-min_seg+1):
        s=sse(vals[:k])+sse(vals[k:])
        if best is None or s<best[0]: best=(s,k)
    tot=sse(vals)
    gain=1-best[0]/tot
    if gain<0.10: return res   # stop if little structure left
    res.append((dates[best[1]-1].date(), dates[best[1]].date(), round(gain,2)))
    binseg(dates[:best[1]], vals[:best[1]], depth+1, max_splits, min_seg, res)
    binseg(dates[best[1]:], vals[best[1]:], depth+1, max_splits, min_seg, res)
    return res
```

def sse(v):
    v = np.asarray(v, dtype=float)
    return float(((v - v.mean())**2).sum())

def binseg(dates, vals, depth=0, max_splits=3, min_seg=3,
           min_gain=0.25, min_drop=5.0, res=None):
    """Greedy binary segmentation on SSE. Heuristic only: thresholds are
    arbitrary knobs, and mean-shift methods will also 'find' structure in
    noise/short spikes. Corroborate with documented events before believing."""
    if res is None:
        res = []
    v = np.asarray(vals, dtype=float)
    if len(v) < 2*min_seg or depth >= max_splits:
        return res
    base = sse(v)
    best = None
    for k in range(min_seg, len(v)-min_seg+1):
        s = sse(v[:k]) + sse(v[k:])
        if best is None or s < best[0]:
            best = (s, k)
    s, k = best
    gain, drop = 1 - s/base, base - s
    if gain < min_gain or drop < min_drop:
        return res
    res.append((str(dates[k-1].date()), str(dates[k].date()), round(gain, 2)))
    binseg(dates[:k], vals[:k], depth+1, max_splits, min_seg, min_gain, min_drop, res)
    binseg(dates[k:], vals[k:], depth+1, max_splits, min_seg, min_gain, min_drop, res)
    return res
```
