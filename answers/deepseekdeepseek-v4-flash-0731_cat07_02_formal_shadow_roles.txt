# 5. PM4Py role-discovery and similarity code

The following code shows how to load, format, and analyze the log with PM4Py. It emphasizes that activity-frequency clusters must be cross-referenced with case complexity and substitution flags.

```python
import pandas as pd
import pm4py

# Load raw event log (CSV or dataframe)
df = pd.read_csv('order_to_resolution_log.csv')

# PM4Py expects standard column names; we provide them via format_dataframe
df = pm4py.format_dataframe(
    df,
    case_id='case:concept:name',
    activity_key='concept:name',
    timestamp_key='time:timestamp',
    resource_key='org:resource',
    start_timestamp_key=None
)

# Optionally convert to event log object
event_log = pm4py.convert_to_event_log(df)

# 1. Discover organizational roles (activity-resource frequency based)
roles = pm4py.discover_organizational_roles(event_log)
# Output is a dictionary: (role_label) -> list of resources
# Role label is from a simple activity-frequency grouping, so use with caution.

# 2. Activity-based resource similarity
# Requires a list of resources and an event log
resources = df['org:resource'].unique().tolist()
sim_matrix = pm4py.discover_activity_based_resource_similarity(
    event_log,
    resources=resources
)
# This returns a pandas DataFrame of cosine similarities over normalized activity counts.

# 3. Resource networks
# Handover of work network (who passes work to whom)
ho_net = pm4py.discover_handover_of_work_network(event_log)
# Working together network (resources that co-occur in cases)
wt_net = pm4py.discover_working_together_network(event_log)
# Subcontracting network (resource delegates work to another in same case)
sub_net = pm4py.discover_subcontracting_network(event_log)

# 4. Join role discovery with case complexity and substitution indicators
# First aggregate case complexity per resource
case_attr = df.groupby('case:concept:name').agg(
    case_complexity=('case_complexity', 'first'),
    temporary_substitution=('temporary_substitution', 'max'),  # 0/1 indicator
    escalation_level=('escalation_level', 'max')
).reset_index()

# Merge resource-level metrics
resource_metrics = df.merge(case_attr, on='case:concept:name') \
    .groupby('org:resource').agg(
        total_events=('concept:name', 'count'),
        avg_complexity=('case_complexity', 'mean'),
        max_complexity=('case_complexity', 'max'),
        substitution_events=('temporary_substitution', 'sum'),
        high_escalation=('escalation_level', 'lambda x: (x>=2).sum())
    )

# Add role label from discovered roles (join on resource)
# roles dict: role_label -> list of resources, invert it
resource_to_role = {}
for role, members in roles.items():
    for r in members:
        resource_to_role[r] = role

resource_metrics['discovered_role'] = resource_metrics.index.map(resource_to_role)

# Display table for inspection
print(resource_metrics.sort_values('avg_complexity', ascending=False))

# 5. Avoid over-interpreting activity frequencies as authority
# Example: compare R08_Hana (approval) and R13_Mateo (coordinator)
# Both have high A9 counts, but Mateo's are preliminary while Hana's are final.
# Use the temporary_substitution and escalation_level fields:
# - Count A9 events by resource and by whether the case was later rechecked (escalation)
prelim_approval = df[(df['concept:name'] == 'A9 Supervisor Approval')]
approval_summary = prelim_approval.groupby('org:resource').agg(
    approval_count=('concept:name', 'count'),
    avg_escalation=('escalation_level', 'mean')
)
# If a resource has high approval_count but also high avg_escalation, it may be a filter/preparer,
# not final authority. Use this to challenge clustering output.

# 6. Clustering log by variants to detect rework patterns
variants = pm4py.get_variants(event_log)
# Inspect fragments involving Reopen (A13) to understand quality issues.

# 7. Service time analysis (optional)
# service_time = pm4py.get_service_time(event_log) # depending on PM4Py version
# Use to quantify impact of substitution (e.g., pricing duration during Fatima absence)

# Important: The discovered roles are purely activity-frequency clusters.
# They DO NOT encode authority, risk, or formal job titles.
# Always join with case_complexity, substitution flags, and escalation_level
# before making staffing decisions.
```
