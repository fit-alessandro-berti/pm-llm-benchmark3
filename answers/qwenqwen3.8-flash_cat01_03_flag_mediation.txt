def build_event_log(cases):
   events=[]
   for _, row in cases.iterrows():
       case=row['case']
       base={'flag': row['flag'], ...}
       def add(concept, timestamp, start=None, resource=None):
           if pd.isna(timestamp): return
           events.append({
              'case:concept:name': case,
              'concept:name': concept,
              'time:timestamp': timestamp,
              'start_timestamp': start if pd.notna(start) else timestamp,
              'org:resource': resource,
              **base

       add('Close', row['close'], row['close'], None)
   log=pd.DataFrame(events)
   log=log.sort_values(['case:concept:name','time:timestamp']).reset_index(drop=True)

## 1. Intervention definitions and estimands

I use **interventional indirect effects** and **separable policy-component effects**, not natural direct/indirect effects, because the flag is embedded in a policy bundle: priority triage, mandatory review, escalation routing, shared queues, and human judgment. “Set flag = 0” is not a well-defined atomic manipulation unless we specify what the organization does when the flag is hidden, when risk_score remains visible, and when capacity is shared.

Interference arises because `manual_review` and `escalation` consume shared capacity: a flagged case’s occurrence or duration changes `manual_queue_wait` and `escalation_queue_wait` for concurrent cases, so a case’s outcome depends on other cases’ flags. Exposure-induced confounding arises if the flag changes later variables that are also confounders for downstream mediators and outcomes, e.g., priority triage changes triage findings, which affect `manual_review` and decision quality, or protected capacity changes arrival_load. Treatment-confounder feedback is present when early handling affects congestion that affects later handling: `manual_review -> manual_queue_wait -> cycle_time` and queue wait -> service assignment; `escalation -> escalation_queue_wait -> senior_service`. Also `complexity` and `risk_score` are pre-flag confounders, but `arrival_load` can be correlated with flag timing and affects queues. These features violate simple natural mediation and require interventional or interference-aware estimands.

def build_event_log(cases: pd.DataFrame) -> pd.DataFrame:
    """
    Reshape the wide case table into an event-level log.

    Important: absent optional stages are not fabricated. A Manual Review event
    is added only if manual_done exists; Escalation only if escalation_done exists.
    """
    events = []
    attr_cols = ["risk_score", "flag", "complexity", "arrival_load", "outcome"]

    for _, row in cases.iterrows():
        base = {c: row[c] for c in attr_cols}

        def add(concept, timestamp, start=None, resource=None):
            if pd.isna(timestamp):
                return
            start_val = timestamp if pd.isna(start) else start
            events.append(
                {
                    "case:concept:name": row["case"],
                    "concept:name": concept,
                    "time:timestamp": timestamp,
                    "start_timestamp": start_val,
                    "org:resource": resource,
                    **base,
                }
            )

        # Register is represented by arrival.
        add("Register", row["arrival"], row["arrival"], None)

        # Triage completion event, with triage start available.
        add("Triage", row["triage_done"], row["triage_start"], row["triage_resource"])

        # Optional manual review completion event.
        add("Manual Review", row["manual_done"], row["manual_start"], row["review_resource"])

        # Optional escalation completion event.
        add("Escalation", row["escalation_done"], row["escalation_start"], None)

        # Close event.
        add("Close", row["close"], row["close"], None)

    log = pd.DataFrame(events)
    if not log.empty:
        log = log.sort_values(["case:concept:name", "time:timestamp"]).reset_index(drop=True)
    return log


def discover_and_print_dfgs(log: pd.DataFrame, label: str) -> None:
    """
    Discover frequency and performance DFGs.

    This is descriptive process mining, not causal identification.
    """
    if log.empty:
        print(f"No events for {label}")
        return

    try:
        dfg_result = pm4py.discover_dfg(
            log,
            case_id_key="case:concept:name",
            activity_key="concept:name",
            timestamp_key="time:timestamp",
        )
        dfg = dfg_result[0] if isinstance(dfg_result, tuple) else dfg_result

        print(f"\nFrequency DFG ({label}) first 10 edges:")
        if hasattr(dfg, "items"):
            for edge, count in list(dfg.items())[:10]:
                print(" ", edge, count)
        else:
            print(" ", dfg)

        perf_result = pm4py.discover_performance_dfg(
            log,
            case_id_key="case:concept:name",
            activity_key="concept:name",
            timestamp_key="time:timestamp",
        )
        perf = perf_result[0] if isinstance(perf_result, tuple) else perf_result

        print(f"\nPerformance DFG ({label}) first 10 edges:")
        if hasattr(perf, "items"):
            for edge, value in list(perf.items())[:10]:
                print(" ", edge, value)
        else:
            print(" ", perf)

    except Exception as exc:
        print(f"DFG discovery failed for {label}: {exc}")


def compute_stage_times(cases: pd.DataFrame) -> pd.DataFrame:
    """
    Calculate explicit waiting/service intervals from timestamps.

    Absent stages remain NaN; they are not treated as zero-length observed stages.
    """
    s = cases.copy()

    def diff_minutes(end_col: str, start_col: str) -> pd.Series:
        return (s[end_col] - s[start_col]).dt.total_seconds() / 60.0

    s["arrival_to_triage_start_min"] = diff_minutes("triage_start", "arrival")
    s["triage_service_min"] = diff_minutes("triage_done", "triage_start")

    s["triage_done_to_manual_start_min"] = np.nan
    mask = s["triage_done"].notna() & s["manual_start"].notna()
    s.loc[mask, "triage_done_to_manual_start_min"] = (
        (s.loc[mask, "manual_start"] - s.loc[mask, "triage_done"]).dt.total_seconds() / 60.0
    )

    s["manual_service_min"] = np.nan
    mask = s["manual_start"].notna() & s["manual_done"].notna()
    s.loc[mask, "manual_service_min"] = (
        (s.loc[mask, "manual_done"] - s.loc[mask, "manual_start"]).dt.total_seconds() / 60.0
    )

    s["manual_done_to_escalation_start_min"] = np.nan
    mask = s["manual_done"].notna() & s["escalation_start"].notna()
    s.loc[mask, "manual_done_to_escalation_start_min"] = (
        (s.loc[mask, "escalation_start"] - s.loc[mask, "manual_done"]).dt.total_seconds() / 60.0
    )

    s["escalation_service_min"] = np.nan
    mask = s["escalation_start"].notna() & s["escalation_done"].notna()
    s.loc[mask, "escalation_service_min"] = (
        (s.loc[mask, "escalation_done"] - s.loc[mask, "escalation_start"]).dt.total_seconds() / 60.0
    )

    # Residual is after the last observed stage completion.
    last_done = s["escalation_done"].copy()
    last_done = last_done.where(last_done.notna(), s["manual_done"])
    last_done = last_done.where(last_done.notna(), s["triage_done"])
    s["residual_to_close_min"] = (s["close"] - last_done).dt.total_seconds() / 60.0

    s["total_cycle_min"] = (s["close"] - s["arrival"]).dt.total_seconds() / 60.0
    return s


def get_subgroups(s: pd.DataFrame) -> dict:
    return {
        "Flag=0": s["flag"] == 0,
        "Flag=1": s["flag"] == 1,
        "Complexity=Low": s["complexity"] == "Low",
        "Complexity=Medium": s["complexity"] == "Medium",
        "Complexity=High": s["complexity"] == "High",
        "Flag=1, arrival_load=High": (s["flag"] == 1) & (s["arrival_load"] == "High"),
        "Flag=1, arrival_load=Normal": (s["flag"] == 1) & (s["arrival_load"] == "Normal"),
        "Flag=1, Escalation=Yes": (s["flag"] == 1) & (s["escalation_done"].notna()),
        "Flag=1, Escalation=No": (s["flag"] == 1) & (s["escalation_done"].isna()),
    }


def print_stage_summaries(s: pd.DataFrame) -> None:
    subgroups = get_subgroups(s)
    for name, mask in subgroups.items():
        sub = s[mask]
        n = len(sub)
        print(f"\nSubgroup: {name} (n={n})")
        print(
            "  support: "
            f"manual_review={int(sub['manual_done'].notna().sum())}, "
            f"escalation={int(sub['escalation_done'].notna().sum())}"
        )
        for col in INTERVAL_COLS:
            obs = sub[col].dropna()
            if len(obs) == 0:
                print(f"  {col}: NA (support=0)")
            else:
                print(
                    f"  {col}: median={obs.median():.0f}, "
                    f"mean={obs.mean():.0f}, support={len(obs)}"
                )


def warn_small_subgroups(s: pd.DataFrame) -> None:
    subgroups = get_subgroups(s)
    for name, mask in subgroups.items():
        n = s[mask].shape[0]
        if n < 5:
            warnings.warn(f"Small subgroup {name}: n={n}. Medians are unstable.")
        for col in INTERVAL_COLS:
            support = s[mask][col].notna().sum()
            if 0 < support < 5:
                warnings.warn(
                    f"Small stage support in {name}: {col} has support={support}. "
                    "Do not interpret this stage median causally."
                )

## 8. Reversal conditions

1. Hidden risk: if `risk_score` and `complexity` omit important underlying risk, and the omitted risk both causes the flag and downstream delay, the apparent flag effect may be confounding rather than policy effect.

2. Queue interference: if flagged cases mainly slow the system by consuming shared manual-review or senior-team capacity, then a case-level mediation analysis without concurrent queue state will misattribute delay to the flag itself.

3. Decision quality: if manual review or escalation prevents adverse decisions, a policy that shortens cycle time may still be harmful; the preferred mechanism would reverse if cycle-time gains come with more false approvals.

4. Timestamp semantics: if `start_timestamp` represents queue entry rather than actual work start, or if close timestamps are administrative rather than decision completion, stage decompositions may mix waiting, service, and non-process delays.

5. Positivity or lack of overlap: if there are too few comparable low-risk flagged cases or high-risk unflagged cases, local-threshold or within-risk comparisons cannot estimate the flag effect for the whole population.

6. Behavior change by reviewers: if reviewers learn to over-escalate or rubber-stamp flagged cases, the observed policy effect may reflect human adaptation rather than the original flag-handling mechanism.

## 9. Final synthesis

Directly observed: flagged cases start triage much sooner and finish triage somewhat faster than unflagged cases, but they almost always enter manual review, wait longer for review, are escalated more often, and have longer total cycle times; escalated flagged cases are the slowest. Plausible mechanism: the flag is a policy bundle—priority triage plus mandatory review and escalation routing. The triage acceleration is real but small; downstream capacity constraints and routing create the net delay. Unidentified: causal effects of the flag, manual-review occurrence, manual waiting, or escalation because risk_score is imperfect, complexity and human judgment select cases, queues interfere, and no counterfactual policy is observed. Natural direct/indirect effects are not credible here; interventional or interference-aware estimands are needed. Most useful next evidence: a stratified pilot of P2—retain priority triage but make manual review discretionary with high-risk overrides—randomized within risk and load strata, with capacity guardrails and quality auditing. This isolates whether mandatory review occurrence, rather than flag salience or triage, drives delay, while measuring whether any cycle-time gain is bought with decision-quality or unflagged-case harm. If P2 shortens cycle without quality loss, queueing/occurrence is confirmed; if not, capacity or escalation must be targeted.
