# 5. PM4Py role-discovery and similarity code

```python
import pandas as pd
import pm4py

# 1. Load and format the dataframe
df = pd.read_csv("event_log.csv")
df = pm4py.format_dataframe(df, 
                            case_id='case:concept:name', 
                            activity_key='concept:name', 
                            timestamp_key='time:timestamp',
                            resource_key='org:resource')

# 2. Discover organizational roles based on activity profiles
# Note: This uses frequency-based profiling and may group shadow roles with their primary team.
roles = pm4py.discover_organizational_roles(df)

# Inspect roles
for role in roles:
    print(f"Role: {role.name}, Resources: {role.resources}")

# 3. Activity-based resource similarity
# Computes cosine similarity over normalized activity counts.
sim_matrix = pm4py.discover_activity_based_resource_similarity(df)

# 4. Handover, working together, and subcontracting networks
# These help identify cross-functional brokers and shadow roles.
handover_net = pm4py.discover_handover_of_work_network(df)
working_together_net = pm4py.discover_working_together_network(df)
subcontracting_net = pm4py.discover_subcontracting_network(df)

# 5. Join role-discovery output with case complexity and temporary substitution
# We need to prove that similar profiles != similar authority or risk.
# Extract median case complexity per resource
complexity_df = df.groupby('org:resource')['case_complexity'].median().reset_index()
complexity_df.columns = ['org:resource', 'median_complexity']

# Extract temporary substitution counts
temp_sub_df = df[df['temporary_substitution'] == True].groupby('org:resource').size().reset_index()
temp_sub_df.columns = ['org:resource', 'temp_sub_count']

# Map resources to discovered roles
resource_role_map = []
for role in roles:
    for res in role.resources:
        resource_role_map.append({'org:resource': res, 'discovered_role': role.name})

role_df = pd.DataFrame(resource_role_map)
role_df = role_df.merge(complexity_df, on='org:resource', how='left')
role_df = role_df.merge(temp_sub_df, on='org:resource', how='left')

# 6. Avoid over-interpreting activity-frequency clusters
# We flag resources where complexity deviates significantly from the role average.
role_avg_complexity = role_df.groupby('discovered_role')['median_complexity'].mean().reset_index()
role_avg_complexity.columns = ['discovered_role', 'role_avg_complexity']
role_df = role_df.merge(role_avg_complexity, on='discovered_role', how='left')
role_df['complexity_deviation'] = role_df['median_complexity'] - role_df['role_avg_complexity']

# Flag outliers for manual review
outliers = role_df[abs(role_df['complexity_deviation']) > 0.15]
print("Resources requiring manual review (complexity outliers):")
print(outliers)

# Similarly, check subcontracting network to find shadow roles:
# E.g., if R07_Giorgio frequently hands over to R06_Fatima or vice versa,
# it indicates a shadow relationship, not just a similar profile.
```
