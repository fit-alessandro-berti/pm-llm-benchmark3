```python
from itertools import permutations

def linear_extensions(nodes, edges):
    """All topological orderings of `nodes` given precedence pairs in `edges`."""
    preds = {n: set() for n in nodes}
    for a, b in edges:
        preds[b].add(a)
    result = []
    def rec(prefix, remaining):
        if not remaining:
            result.append(list(prefix))
            return
        for n in sorted(remaining, key=str):  # deterministic
            if preds[n] <= set(prefix):
                rec(prefix + [n], remaining - {n})
    rec([], set(nodes))
    return result
```

```python
def linear_extensions(elements, precedence):
    """All topological orderings of `elements` under `precedence` pairs."""
    preds = {e: set() for e in elements}
    for u, v in precedence:
        preds[v].add(u)
    out, stack = [], []
    def rec(remaining):
        if not remaining:
            out.append(tuple(stack))
            return
        for e in sorted(remaining):
            if preds[e] <= set(stack):
                stack.append(e)
                rec(remaining - {e})
                stack.pop()
    rec(set(elements))
    return out

def middle_language(edge_pairs):
    """Exact language of one partial-order middle block."""
    base = ["C", "E"]
    lang = set()
    for with_s in (False, True):
        for with_x in (False, True):
            elems = list(base) + (["S"] if with_s else []) + (["X"] if with_x else [])
            prec = [(u, v) for u, v in edge_pairs if u in elems and v in elems]
            for ext in linear_extensions(elems, prec):
                lang.add(ext)
    return lang
```

```python
def trace_fits_net(net, im, fm, trace_str):

    try:
```

```python
def linear_extensions(elements, precedence):
    preds = {e: set() for e in elements}
    for u, v in precedence:
        preds[v].add(u)
    out, stack = [], []
    def rec(remaining):
        if not remaining:
            out.append(tuple(stack)); return
        for e in sorted(remaining):
            if preds[e] <= set(stack):
                stack.append(e); rec(remaining - {e}); stack.pop()
    rec(set(elements))
    return out
```

```python
# (a) Exact enumeration. Valid only because every candidate is loop-free,
#     hence its language is finite. For models with loops the language is
#     infinite and exhaustive comparison is impossible.
def linear_extensions(elements, precedence):
    preds = {e: set() for e in elements}
    for u, v in precedence:
        preds[v].add(u)
    out, stack = [], []

    def rec(remaining):
        if not remaining:
            out.append(tuple(stack))
            return
        for e in sorted(remaining):
            if preds[e] <= set(stack):
                stack.append(e)
                rec(remaining - {e})
                stack.pop()

    rec(set(elements))
    return out


def middle_language(trigger):
    """Exact language of one partial-order middle block (S and X optional)."""
    lang = set()
    for with_s in (False, True):
        for with_x in (False, True):
            elems = ["C", "E"] + (["S"] if with_s else []) + (["X"] if with_x else [])
            prec = [(u, v) for (u, v) in [(trigger, "S"), ("C", "X"), ("E", "X")]
                    if u in elems and v in elems]
            lang |= set(linear_extensions(elems, prec))
    return lang
```
