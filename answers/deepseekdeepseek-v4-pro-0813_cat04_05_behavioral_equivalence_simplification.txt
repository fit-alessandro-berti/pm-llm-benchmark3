```python
import pm4py
from pm4py.objects.petri_net.obj import PetriNet, Marking
import copy

def is_silent(trans):
    label = trans.label
    return label is None or str(label).lower() == "tau"

def visible_label(trans):
    if is_silent(trans):
        return None
    return str(trans.label)

def marking_key(marking):
    # sort place names and token counts for canonical representation
    items = sorted((p.name, count) for p, count in marking.items() if count > 0)
    return tuple(items)

def fire(net, marking, trans):
    new_marking = copy.copy(marking)
    for arc in trans.in_arcs:
        src = arc.source
        new_marking[src] -= 1
        if new_marking[src] == 0:
            del new_marking[src]
    for arc in trans.out_arcs:
        tgt = arc.target
        new_marking[tgt] = new_marking.get(tgt, 0) + 1
    return new_marking

def bounded_visible_traces(net, im, fm, max_visible=12, max_analyze=3):
    results = set()
    # queue elements: (marking, visible_trace_tuple, analyze_count)
    q = [(copy.copy(im), (), 0)]
    # visited for termination: (marking_key, visible_trace_tuple)
    visited = set()
    while q:
        marking, trace, analyze_count = q.pop(0)
        key = (marking_key(marking), trace)
        if key in visited:
            continue
        visited.add(key)
        if marking == fm:
            results.add(trace)
            # In a sound net final marking has no enabled transitions; do not continue.
            continue
        enabled = pm4py.get_enabled_transitions(net, marking)
        for t in enabled:
            label = visible_label(t)
            new_trace = trace
            new_analyze = analyze_count
            if label is not None:
                if len(trace) >= max_visible:
                    continue
                new_trace = trace + (label,)
                if label == 'Analyze Revision':
                    new_analyze += 1
                    if new_analyze > max_analyze:
                        continue
            # else silent: no visible change; allow only if no infinite silent loop via visited
            new_marking = fire(net, marking, t)
            q.append((new_marking, new_trace, new_analyze))
    return results
```

```python
import copy
import pm4py

MODEL_A = """
->(
  'Open Case',
  *(
    ->(
      +('Analyze Revision', 'Check Revision'),
      X('Approve Revision', tau)
    ),
    'Rework Case'
  ),
  'Close Case'
)
"""

MODEL_B = """
->(
  'Open Case',
  +(
    'Check Revision',
    *(
      ->(
        'Analyze Revision',
        X('Approve Revision', tau)
      ),
      'Rework Case'
    )
  ),
  'Close Case'
)
"""

MAX_VISIBLE = 12
MAX_ANALYZE = 3

def is_silent(trans):
    label = trans.label
    return label is None or str(label).lower() == "tau"

def visible_label(trans):
    return None if is_silent(trans) else str(trans.label)

def marking_key(marking):
    return tuple(sorted((p.name, cnt) for p, cnt in marking.items() if cnt > 0))

def fire_transition(net, marking, trans):
    new_marking = copy.copy(marking)
    for arc in trans.in_arcs:
        new_marking[arc.source] -= 1
        if new_marking[arc.source] == 0:
            del new_marking[arc.source]
    for arc in trans.out_arcs:
        new_marking[arc.target] = new_marking.get(arc.target, 0) + 1
    return new_marking

def exhaustive_bounded_visible_traces(net, im, fm, max_visible=MAX_VISIBLE, max_analyze=MAX_ANALYZE):
    found = set()
    queue = [(copy.copy(im), (), 0)]
    visited = set()
    while queue:
        marking, trace, analyze_count = queue.pop(0)
        key = (marking_key(marking), trace)
        if key in visited:
            continue
        visited.add(key)

        if marking == fm:
            found.add(trace)
            continue

        for trans in pm4py.get_enabled_transitions(net, marking):
            label = visible_label(trans)
            new_trace = trace
            new_analyze_count = analyze_count
            if label is not None:
                if len(trace) >= max_visible:
                    continue
                new_trace = trace + (label,)
                if label == "Analyze Revision":
                    new_analyze_count += 1
                    if new_analyze_count > max_analyze:
                        continue
            new_marking = fire_transition(net, marking, trans)
            queue.append((new_marking, new_trace, new_analyze_count))
    return found

def main():
    tree_a = pm4py.parse_process_tree(MODEL_A)
    tree_b = pm4py.parse_process_tree(MODEL_B)

    net_a, im_a, fm_a = pm4py.convert_to_petri_net(tree_a)
    net_b, im_b, fm_b = pm4py.convert_to_petri_net(tree_b)

    print("Soundness A:", pm4py.check_soundness(net_a, im_a, fm_a, print_diagnostics=False))
    print("Soundness B:", pm4py.check_soundness(net_b, im_b, fm_b, print_diagnostics=False))

    traces_a = exhaustive_bounded_visible_traces(net_a, im_a, fm_a)
    traces_b = exhaustive_bounded_visible_traces(net_b, im_b, fm_b)

    print(f"P3 bounded traces A: {len(traces_a)}")
    print(f"P3 bounded traces B: {len(traces_b)}")
    unique_a = sorted(traces_a - traces_b)
    unique_b = sorted(traces_b - traces_a)
    print(f"Unique to A: {len(unique_a)}")
    for t in unique_a[:20]:
        print("  A-only:", t)
    print(f"Unique to B: {len(unique_b)}")
    for t in unique_b[:20]:
        print("  B-only:", t)

    print("A and B bounded languages equal?", traces_a == traces_b)

if __name__ == "__main__":
    main()
```

# 1. Equivalence Notions

A trace is a finite sequence of visible activity labels. Repeated labels are distinguished by **occurrence position**, not by a separate event identity. For example, in  
`Open, Analyze, Rework, Analyze, Close`, the two `Analyze` events are two positions with the same label.

- **Visible trace equivalence after hiding `tau`**  
  Given a process tree, delete all `tau`-labelled silent events from complete traces. Two models are equivalent iff their sets of complete visible traces are equal.

- **Bounded visible-language equivalence under P3**  
  Hide `tau`, retain all visible labels. Compare only complete visible traces with at most 12 visible events and at most three `Analyze Revision` occurrences.  
  \[
  L_M^{\text{P3}} = \{ t \mid t \text{ is a complete visible trace of } M,\ |t| \le 12,\ \#\text{Analyze}(t) \le 3\}.
  \]
  Two models are P3-equivalent iff \(L_A^{\text{P3}} = L_B^{\text{P3}}\). This is an exact comparison of a finite bounded language, not a statistical sample.

- **Business-relevant equivalence under P1**  
  Hide `tau`, retain `Approve Revision`. Split the trace between `Open Case` and `Close Case` into revision segments separated by `Rework Case`. A trace is P1-valid iff:
  1. `Open Case` is first and `Close Case` last;
  2. every revision segment contains exactly one `Analyze Revision` and exactly one `Check Revision`;
  3. `Rework Case` separates revision attempts;
  4. `Approve Revision` is optional, visible, and its placement is retained.

  P1-equivalence means the models generate the same set of P1-valid complete visible traces. I also report whether all generated traces are P1-valid. The written P1 does not force `Analyze` before `Check` within a segment; if that stricter order is intended, it is called out as a caveat.

- **Abstracted dashboard equivalence under P2**  
  Hide `tau` and `Approve Revision`. Map each complete trace to the dashboard record:
  \[
  (\text{Open}, \text{AnalyzeCount}=n,\ \text{ReworkOccurred}=n>1,\ \text{CheckOccurred}=\text{True}, \text{Close}).
  \]
  Repeated `Check Revision` occurrences collapse to a Boolean “a check occurred”. Exact interleaving of `Analyze Revision` and `Check Revision` is not recorded. Two models are P2-equivalent iff their sets of dashboard records are equal.

# 5. Bounded Comparison Code

The following code is self-contained. It parses both trees, converts them to Petri nets, checks soundness, and performs **exhaustive bounded BFS** over markings up to the P3 bounds. It is not a random playout: it expands every enabled transition, including silent transitions, and records complete visible traces reaching the final marking.

```python
import copy
import pm4py

MODEL_A = """
->(
  'Open Case',
  *(
    ->(
      +('Analyze Revision', 'Check Revision'),
      X('Approve Revision', tau)
    ),
    'Rework Case'
  ),
  'Close Case'
)
"""

MODEL_B = """
->(
  'Open Case',
  +(
    'Check Revision',
    *(
      ->(
        'Analyze Revision',
        X('Approve Revision', tau)
      ),
      'Rework Case'
    )
  ),
  'Close Case'
)
"""

MAX_VISIBLE = 12
MAX_ANALYZE = 3


def is_silent(trans):
    label = trans.label
    return label is None or str(label).lower() == "tau"


def visible_label(trans):
    return None if is_silent(trans) else str(trans.label)


def marking_key(marking):
    return tuple(sorted((p.name, cnt) for p, cnt in marking.items() if cnt > 0))


def fire_transition(net, marking, trans):
    new_marking = copy.copy(marking)
    for arc in trans.in_arcs:
        new_marking[arc.source] -= 1
        if new_marking[arc.source] == 0:
            del new_marking[arc.source]
    for arc in trans.out_arcs:
        new_marking[arc.target] = new_marking.get(arc.target, 0) + 1
    return new_marking


def exhaustive_bounded_visible_traces(net, im, fm,
                                      max_visible=MAX_VISIBLE,
                                      max_analyze=MAX_ANALYZE):
    found = set()
    queue = [(copy.copy(im), (), 0)]  # (marking, visible_trace_tuple, analyze_count)
    visited = set()

    while queue:
        marking, trace, analyze_count = queue.pop(0)

        # Key includes visible trace to avoid cutting alternative prefixes
        key = (marking_key(marking), trace)
        if key in visited:
            continue
        visited.add(key)

        if marking == fm:
            found.add(trace)
            continue

        for trans in pm4py.get_enabled_transitions(net, marking):
            label = visible_label(trans)
            new_trace = trace
            new_analyze_count = analyze_count

            if label is not None:
                if len(trace) >= max_visible:
                    continue
                new_trace = trace + (label,)
                if label == "Analyze Revision":
                    new_analyze_count += 1
                    if new_analyze_count > max_analyze:
                        continue

            new_marking = fire_transition(net, marking, trans)
            queue.append((new_marking, new_trace, new_analyze_count))

    return found


def main():
    tree_a = pm4py.parse_process_tree(MODEL_A)
    tree_b = pm4py.parse_process_tree(MODEL_B)

    net_a, im_a, fm_a = pm4py.convert_to_petri_net(tree_a)
    net_b, im_b, fm_b = pm4py.convert_to_petri_net(tree_b)

    print("Soundness A:", pm4py.check_soundness(net_a, im_a, fm_a, print_diagnostics=False))
    print("Soundness B:", pm4py.check_soundness(net_b, im_b, fm_b, print_diagnostics=False))

    traces_a = exhaustive_bounded_visible_traces(net_a, im_a, fm_a)
    traces_b = exhaustive_bounded_visible_traces(net_b, im_b, fm_b)

    print(f"P3 bounded traces A: {len(traces_a)}")
    print(f"P3 bounded traces B: {len(traces_b)}")

    unique_a = sorted(traces_a - traces_b)
    unique_b = sorted(traces_b - traces_a)

    print(f"Unique to A: {len(unique_a)}")
    for t in unique_a[:20]:
        print("  A-only:", t)

    print(f"Unique to B: {len(unique_b)}")
    for t in unique_b[:20]:
        print("  B-only:", t)

    print("A and B bounded languages equal?", traces_a == traces_b)


if __name__ == "__main__":
    main()
```

The BFS handles silent transitions by keeping them out of the visible trace. It cuts silent loops only when the same marking and same visible trace are reached again. This is exhaustive for the finite bounded language under P3.

```python
def is_tau(label):

def tree_stats(tree):
    # Leaf
    if tree.operator is None:
        visible = not is_tau(tree.label)
        return {
            "nodes": 1,
            "depth": 0,
            "visible_leaves": 1 if visible else 0,
            "silent_leaves": 0 if visible else 1,
            "labels": [str(tree.label)] if visible else []
        }

    child_stats = [tree_stats(c) for c in tree.children]
    nodes = 1 + sum(s["nodes"] for s in child_stats)
    depth = 1 + max(s["depth"] for s in child_stats)
    visible_leaves = sum(s["visible_leaves"] for s in child_stats)
    silent_leaves = sum(s["silent_leaves"] for s in child_stats)

    labels = []
    for s in child_stats:
        labels.extend(s["labels"])

    return {
        "nodes": nodes,
        "depth": depth,
        "visible_leaves": visible_leaves,
        "silent_leaves": silent_leaves,
        "labels": labels
    }
```

# 8. Simplified Model

Target semantics: **P1 operational compliance with explicit sequential order** — a clearly defined compromise.  
Each revision attempt must be:

`Analyze Revision -> Check Revision -> optional Approve Revision`

then either exit to `Close Case` or do `Rework Case` and repeat.

```python
SIMPLIFIED_P1 = """
->(
  'Open Case',
  *(
    ->(
      'Analyze Revision',
      'Check Revision',
      X('Approve Revision', tau)
    ),
    'Rework Case'
  ),
  'Close Case'
)
"""

simplified_model = pm4py.parse_process_tree(SIMPLIFIED_P1)
```

This model is simpler than both originals:

- Original A: 12 nodes, depth 4, includes `+` parallel between Analyze and Check.
- Original B: 12 nodes, depth 5, includes top-level `+` parallel between Check and the loop.
- Simplified P1 model: 11 nodes, depth 4, no parallel operator; it enforces Analyze  Check  optional Approve.

| Property | Preservation status | Explanation |
|---|---|---|
| Open first, Close last | Exact | Root sequence enforces it. |
| Rework separates revisions | Exact | Loop redo child is `Rework Case`. |
| One Analyze per revision attempt | Exact | Loop body always contains one Analyze. |
| One Check per revision attempt after Analyze | Exact under target | Sequence `Analyze -> Check` enforces the strict P1 interpretation. |
| Approve optional | Exact | `X('Approve Revision', tau)` makes approval optional. |
| Approve visible placement | Exact for target | Approval occurs only after Check. |
| Silent `tau` behavior | Exact | Hidden `tau` branch is preserved. |
| Analyze/Check concurrency and order freedom | Lost | Model A’s `+` allowed `Check, Analyze`; this model intentionally disallows it. |
| B’s one global Check concurrent with the loop | Lost | This model rejects that design for operational compliance. |
| P2 dashboard facts | Exact | It still has Open, at least one Analyze/Check, countable Analyze, rework boolean, Close. |
| P3 raw trace language | Not preserved | The simplified model is stricter than Model A and not language-equivalent to either original. |

**Regression traces** (admitted and valid under the target semantics):

1. `Open Case, Analyze Revision, Check Revision, Close Case`
2. `Open Case, Analyze Revision, Check Revision, Approve Revision, Rework Case, Analyze Revision, Check Revision, Close Case`
3. `Open Case, Analyze Revision, Check Revision, Rework Case, Analyze Revision, Check Revision, Approve Revision, Close Case`

**Negative traces** (must not be admitted under the target semantics):

1. `Open Case, Analyze Revision, Rework Case, Analyze Revision, Check Revision, Close Case` — first analysis lacks its own Check before Rework.
2. `Open Case, Check Revision, Analyze Revision, Close Case` — Check before Analyze violates the strict P1 compromise.
3. `Open Case, Analyze Revision, Check Revision, Approve Revision, Check Revision, Close Case` — duplicate Check in the same revision segment.

# 9. Final Recommendation

For **P1 operational compliance**, Model A is closer than Model B because it puts Check inside the revision loop. However, Model A’s parallel `+` between Analyze and Check does not enforce Analyze-before-Check. If the P1 rule is read strictly as “Analyze then Check before Rework/Close,” the simplified sequential model is the appropriate replacement for Model A.

For **P2 executive dashboard**, Models A and B are equivalent only under the stated dashboard abstraction. Either original can be used if the dashboard is the only concern, because both produce the same dashboard records: Open, positive Analyze count, Rework boolean, Check occurred, Close.

For **P3 bounded testing**, the raw visible languages are not equivalent. The bounded comparison is strong evidence of a real semantic difference, but it is not a proof of unbounded equivalence. A bounded result should always be accompanied by structural reasoning or an unbounded language argument.

The simplified P1-sequential model should replace Model A when operational compliance with strict Analyze-before-Check is required. It should not be treated as a language-level simplification of both originals for all views: it deliberately narrows Model A’s interleaving and rejects Model B’s global Check design. A single simplification cannot preserve all three views simultaneously.
