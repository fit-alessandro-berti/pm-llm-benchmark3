```python
def retro_ok(case, res_key=RES, act_key=ACT, ts_key=TS):
    emg = [e for e in case if e[act_key] == "Emergency Override"]
    retro = [e for e in case if e[act_key] == "Retrospective Justification"]

    if not emg:
        return True

    for e in emg:
        ok = False
        for r in retro:
            if r[ts_key] >= e[ts_key] and r[res_key] != e[res_key]:
                busdays = pd.bdate_range(e[ts_key], r[ts_key], freq="B").size - 1
                if 0 <= busdays <= 3:
                    ok = True
        if not ok:
            return False

    return True

bad_retro = [case for case in log if not retro_ok(case)]
```

# 6. Cautious audit-report wording

The event log shows a mixture of control exceptions, emergency-handling patterns, and data-quality limitations rather than direct evidence of collusion.
