def fcfs_sim(ready, service, n_servers):
    order = np.argsort(ready, kind="stable")   # ties broken by case order
    free = np.zeros(n_servers)
    start = np.empty(len(ready)); end = np.empty(len(ready))
    for k in order:
        s = int(np.argmin(free))          # earliest-free server; ties -> lowest index
        st = max(ready[k], free[s])
        start[k] = st; end[k] = st + service[k]
        free[s] = st + service[k]
    return start, end, start - ready
```

def shapley_stage(spans: pd.DataFrame) -> pd.DataFrame:
    # v(S) = max_{i in S} span_i ;3 branches -> weights 1/3, 1/6, 1/6, 1/3
    out = {}
    for i in BRANCHES:
        others = [b for b in BRANCHES if b != i]
        s_i = spans[i]
        term_empty = spans[i] / 3.0
        term_single = sum((spans[i] - spans[j]).clip(lower=0) / 6.0 for j in others)
        max_o = spans[others].max(axis=1)
        term_full = (spans.max(axis=1) - max_o) / 3.0
        out[i] = term_empty + term_single + term_full
    return pd.DataFrame(out)
```

    order = np.argsort(ready, kind="stable")
    free = np.zeros(n_servers)
