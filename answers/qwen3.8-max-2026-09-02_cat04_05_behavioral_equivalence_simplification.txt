def bounded_visible_traces(net, im, fm, max_visible=12, max_analyze=3, max_total_steps=64):
    inputs, outputs = precompute(net)
    start_key = (marking_key(im), 0,0, tuple())
    q=deque([(im,0,0,tuple(),0)])
    seen={start_key}
    lang=set()
    truncated=False
    while q:
      m,v,a,trace,steps=q.popleft()
      if steps > max_total_steps: truncated=True; continue
      if m == fm:
         lang.add(trace)
         continue
      enabled = pm4py.get_enabled_transitions(net, m)
      for t in enabled:
         label = getattr(t, 'label', None)
         is_visible = label is not None and str(label).lower() != 'tau'
         nv = v + (1 if is_visible else 0)
         na = a + (1 if is_visible and label == 'Analyze Revision' else 0)
         if nv > max_visible or na > max_analyze: continue
         try: nm = fire_transition(inputs, outputs, m, t)
         except ValueError: continue
         ntrace = trace + ((label,) if is_visible else tuple())
         key=(marking_key(nm), nv, na, ntrace)
         if key not in seen:
            seen.add(key)
            q.append((nm,nv,na,ntrace,steps+1))
    return lang, truncated

def petri_io(net):
    inputs = {}
    outputs = {}
    for t in net.transitions:
       inputs[t]=[]; outputs[t]=[]
    for arc in net.arcs:
       if isinstance(arc.source, PetriNet.Place) and isinstance(arc.target, PetriNet.Transition):
           inputs[arc.target].append(arc.source)
       elif isinstance(arc.source, PetriNet.Transition) and isinstance(arc.target, PetriNet.Place):
           outputs[arc.source].append(arc.target)
    return inputs, outputs

    inputs, outputs = petri_io(net)
    lang=set()
    start=(im,0,0,tuple(),0)
    q=deque([start])
    seen={(mkey(im),0,0,tuple())}
    truncated=False
    while q:
       marking, vlen, acount, trace, steps = q.popleft()
       if steps > max_total_steps:
          truncated=True
          continue
       if marking == fm:
          lang.add(trace)
          continue
       enabled = pm4py.get_enabled_transitions(net, marking)
       for t in enabled:
          label = t.label
          visible = label is not None and str(label).lower() != 'tau'
          nv = vlen + (1 if visible else 0)

def tree_metrics(tree):
    node_count = 0
    leaf_count = 0
    visible_leaf_count = 0
    silent_leaf_count = 0

    max_depth = 0
    def walk(n, depth):
        nonlocal node_count, leaf_count, visible_leaf_count, silent_leaf_count, max_depth
        node_count += 1
        max_depth = max(max_depth, depth)
        children = getattr(n, 'children', [])

            leaf_count += 1
            label = getattr(n, 'label', None)

                silent_leaf_count += 1
            else:

                labels.append(str(label))
        else:
            for c in children:
                walk(c, depth + 1)
    walk(tree, 0)

def petri_io(net):
    inputs = {t: [] for t in net.transitions}
    outputs = {t: [] for t in net.transitions}
    for arc in net.arcs:
        if isinstance(arc.source, PetriNet.Place) and isinstance(arc.target, PetriNet.Transition):
            inputs[arc.target].append(arc.source)
        elif isinstance(arc.source, PetriNet.Transition) and isinstance(arc.target, PetriNet.Place):
            outputs[arc.source].append(arc.target)
    return inputs, outputs

def fire(marking, t, inputs, outputs):
    new = dict(marking)
    for p in inputs.get(t, []):
        tokens = new.get(p, 0)
        if tokens < 1:
            raise ValueError("transition not enabled")
        if tokens == 1:
            new.pop(p, None)
        else:
            new[p] = tokens - 1
    for p in outputs.get(t, []):
        new[p] = new.get(p, 0) + 1
    return Marking(new)

def marking_key(m):
    return frozenset(m.items())

def bounded_visible_language(net, im, fm, max_visible=12, max_analyze=3, max_total_steps=80):
    inputs, outputs = petri_io(net)
    language = set()
    start_trace = tuple()
    queue = deque([(im, 0, 0, start_trace, 0)])
    seen = {(marking_key(im), 0, 0, start_trace)}
    truncated = False

    while queue:
        marking, visible_len, analyze_count, trace, steps = queue.popleft()
        if steps > max_total_steps:
            truncated = True
            continue

        # Complete trace: explicit final-marking test.
        if marking == fm:
            language.add(trace)
            continue

        for t in pm4py.get_enabled_transitions(net, marking):
            label = getattr(t, "label", None)
            visible = label is not None and str(label).lower() != "tau"
            next_visible_len = visible_len + (1 if visible else 0)
            next_analyze_count = analyze_count + (1 if visible and str(label) == "Analyze Revision" else 0)

            if next_visible_len > max_visible or next_analyze_count > max_analyze:
                continue

            try:
                next_marking = fire(marking, t, inputs, outputs)
            except ValueError:
                continue

            next_trace = trace + ((str(label),) if visible else tuple())
            key = (marking_key(next_marking), next_visible_len, next_analyze_count, next_trace)
            if key not in seen:
                seen.add(key)
                queue.append((next_marking, next_visible_len, next_analyze_count, next_trace, steps + 1))

    return language, truncated

def petri_io(net):
    inputs = defaultdict(list)
    outputs = defaultdict(list)
    for arc in net.arcs:
        w = getattr(arc, "weight", 1)
        if isinstance(arc.source, PetriNet.Place) and isinstance(arc.target, PetriNet.Transition):
            inputs[arc.target].append((arc.source, w))
        elif isinstance(arc.source, PetriNet.Transition) and isinstance(arc.target, PetriNet.Place):

    for p, w in outputs.get(t, []):
        new[p] = new.get(p, 0) + w
    return Marking(new)

# 1. Equivalence Notions

I use the following observation boundary before any verdict:

- Complete runs are compared from the initial marking/state to the final marking/state.
- `tau` is hidden.
- The visible alphabet is:
  - `Open Case`
  - `Analyze Revision`
  - `Check Revision`
  - `Approve Revision`
  - `Rework Case`
  - `Close Case`

Let `proj` be the projection that removes `tau`.

Useful shorthand for the visible languages after hiding `tau`:

- Model A:

```text
L(A) =
  Open Case
  ( (Analyze Revision Check Revision | Check Revision Analyze Revision)
    (Approve Revision | )
    ( Rework Case
      (Analyze Revision Check Revision | Check Revision Analyze Revision)
      (Approve Revision | )
    )*
  )
  Close Case
```

- Model B:

```text
L(B) =
  Open Case
  shuffle(
    Check Revision,
    Analyze Revision (Approve Revision | )
    ( Rework Case Analyze Revision (Approve Revision | ) )*
  )
  Close Case
```

Here `shuffle` means arbitrary interleaving of the single `Check Revision` event with the entire loop execution.

## Visible trace equivalence after hiding `tau`

Two models are visible-trace equivalent iff:

```text
proj(traces(Model 1)) = proj(traces(Model 2))
```

Duplicate visible labels are treated as repeated labels in a sequence. They are distinguished by occurrence position in the trace, but they do not carry unique activity-instance identities unless explicitly modeled in event data. Therefore:

```text
Open, Analyze, Check, Rework, Analyze, Check, Close
```

is different from:

```text
Open, Analyze, Check, Rework, Analyze, Close
```

because the second `Check Revision` occurrence is present in the first trace and absent in the second.

## Bounded visible-language equivalence under P3

Under P3, compare only complete visible traces `w` such that:

```text
|w| <= 12
count(w, Analyze Revision) <= 3
```

The bounded language is:

```text
L_{12,3}(M) =
  { w in proj(traces(M))
    | length(w) <= 12 and count_Analyze(w) <= 3 }
```

Models are P3-bounded equivalent iff:

```text
L_{12,3}(A) = L_{12,3}(B)
```

Duplicate labels are still distinguished by sequence position and count. `Check Revision` occurrences are not collapsed under P3.

## Business-relevant equivalence under P1

P1 is an operational compliance view. A reasonable formalization is:

- A case starts with `Open Case` and ends with `Close Case`.
- Each revision attempt contains one `Analyze Revision`.
- Each revision attempt must have its own `Check Revision` before the next `Rework Case` or before `Close Case`.
- `Rework Case` separates two analysis attempts.
- `Approve Revision` is optional.
- The number and placement of `Approve Revision` occurrences remain visible and potentially relevant.

Because the labels themselves do not contain revision-instance IDs, “its own check” can only be checked by occurrence count and ordering. Under a literal reading, Model A provides one `Check Revision` per loop iteration. Under a stricter audit reading, the check should occur after the analysis it checks; Model A’s parallelism also allows `Check Revision` before `Analyze Revision`, which may be undesirable. I will refer to the stricter version as P1+ where relevant.

Model B has only one `Check Revision` in the entire case. Therefore, for two or more `Analyze Revision` occurrences, Model B cannot satisfy “every analysis revision has its own Check Revision.”

## Abstracted dashboard equivalence under P2

P2 defines an abstraction function `D` on complete visible traces.

For a visible trace `w`:

```text
D(w) = (
  open_first(w),
  close_last(w),
  check_occurred(w) = count(w, Check Revision) > 0,
  analyze_count(w)  = count(w, Analyze Revision),
  rework_occurred(w) = count(w, Rework Case) > 0
)
```

P2 hides:

- `Approve Revision`
- `tau`
- exact interleaving of `Analyze Revision` and `Check Revision`
- repeated `Check Revision` occurrences, collapsed to the Boolean `check_occurred`

Models are P2-equivalent iff:

```text
D(L(A)) = D(L(B))
```

Under this abstraction, both models produce, for every `k >= 1`:

```text
analyze_count = k
check_occurred = true
rework_occurred = (k > 1)
```

So they are equivalent only under this stated abstraction.

# 2. Structural Comparison

| Feature | Model A | Model B | Semantic consequence |
|---|---|---|---|
| Loop scope | The loop body contains `Analyze Revision`, `Check Revision`, and optional `Approve Revision`. | The loop body contains only `Analyze Revision` and optional `Approve Revision`; `Check Revision` is outside the loop. | In A, checking is repeated with each revision iteration. In B, checking is a case-level concurrent activity. |
| Minimum `Check Revision` occurrences | 1 | 1 | Both guarantee at least one check because the loop executes at least once and B’s parallel check executes once. |
| Maximum `Check Revision` occurrences | Unbounded: one per loop iteration | Exactly 1 | A can have two or more checks; B cannot. This is the main trace-language difference. |
| Possible placement of `Check Revision` | Inside each iteration, concurrent with that iteration’s `Analyze Revision`, before optional approval and before the next `Rework Case` or `Close Case`. | One single check can be interleaved anywhere relative to the whole loop: before the first analysis, between analyses, during rework, after the final analysis, or just before `Close Case`. | B permits a check to occur before any analysis or after all analyses. A does not permit one check to serve multiple analyses. |
| Optional approval occurrences | Zero or one per iteration, after both `Analyze Revision` and `Check Revision` of that iteration. | Zero or one per iteration, after the iteration’s `Analyze Revision`, but independent of the single global `Check Revision`. | B can produce `Analyze Revision, Approve Revision, Check Revision, Close Case`; A cannot. |
| Concurrency | `Analyze Revision` and `Check Revision` are concurrent inside each iteration. | `Check Revision` is concurrent with the entire revision loop. | B has more global interleavings; A has only local per-iteration interleavings. |
| Join before `Close Case` | `Close Case` occurs after the loop completes. | `Close Case` occurs after both parallel branches complete: the single `Check Revision` branch and the loop branch. | B can postpone the only check until immediately before `Close Case`. |
| Visible cardinality invariant | `count(Check Revision) = count(Analyze Revision)` for complete traces. | `count(Check Revision) = 1`, while `count(Analyze Revision) >= 1`. | This invariant alone separates the unbounded visible trace languages. |

# 3. Verdicts

Unbounded visible-trace equivalence after hiding `tau` is also not equivalent. The P3 counterexamples below are already witnesses of unbounded inequivalence.

| View | Verdict | Confidence | Decisive reason | Caveat |
|---|---|---|---|---|
| P1 | not equivalent | High | Model B has exactly one `Check Revision`, while Model A has one check per revision iteration. With two or more analyses, B cannot provide an own check for each analysis. | If P1 were weakened to “at least one case-level check,” the verdict could change. If P1+ requires check after analysis, Model A’s parallelism is also too permissive. |
| P2 | equivalent only under stated abstraction | High | The dashboard collapses all `Check Revision` occurrences to `true`, hides approvals, and ignores exact interleaving. Both models yield `analyze_count = k`, `check_occurred = true`, `rework_occurred = k > 1` for every `k >= 1`. | If the dashboard retained check count, check-per-analysis, approval placement, or exact ordering, they would not be equivalent. |
| P3 | not equivalent | High | There are bounded complete visible traces within `<= 12` visible events and `<= 3` analyses that are accepted by one model and not the other. | Bounded equivalence is evidence only. Here the bound is sufficient to find differences, but equality under a bound would not prove unbounded equivalence. |

# 4. Counterexample Traces

| Trace ID | Trace | Model A | Model B | First decisive semantic difference | Relevant view |
|---|---|---|---|---|---|
| T-A1 | `Open Case, Analyze Revision, Check Revision, Rework Case, Analyze Revision, Check Revision, Close Case` | yes | no | Model A repeats `Check Revision` once per iteration; Model B can emit only one `Check Revision`. Contains two analysis attempts. | P1, P3 |
| T-A2 | `Open Case, Check Revision, Analyze Revision, Approve Revision, Rework Case, Check Revision, Analyze Revision, Close Case` | yes | no | Model A has two checks, one in each iteration. Model B cannot produce the second check. Contains two analysis attempts. | P1, P3 |
| T-B1 | `Open Case, Analyze Revision, Rework Case, Analyze Revision, Check Revision, Close Case` | no | yes | Model B uses one global check after two analyses. Model A requires a check before the first `Rework Case`. Contains two analysis attempts. | P1, P3 |
| T-B2 | `Open Case, Analyze Revision, Approve Revision, Check Revision, Close Case` | no | yes | Model B can place the single check after approval. Model A requires `Check Revision` before `Approve Revision` in the only iteration. | P1, P3 |
| T-B3 | `Open Case, Check Revision, Analyze Revision, Rework Case, Analyze Revision, Close Case` | no | yes | Model B’s single check occurs before the first analysis and is then reused implicitly for the whole case. Model A requires a second check for the second analysis. Contains two analysis attempts. | P1, P3 |

Neither language includes the other globally.

- For exactly one `Analyze Revision` and no visible approval, Model A’s behavior is largely included in Model B’s behavior, because B’s single check can be interleaved before or after the analysis.
- For multiple analyses, Model A produces multiple checks, which Model B cannot produce.
- Model B produces multiple-analysis traces with only one check, which Model A cannot produce.
- Model B also produces one-analysis traces where the check occurs after `Approve Revision`, which Model A cannot produce.

Optional `Approve Revision` and hidden `tau` matter:

- `X('Approve Revision', tau)` means an approval may be visible or silently skipped.
- When `tau` is chosen, the trace contains no `Approve Revision`. This shortens visible traces and affects P3 length bounds.
- When `Approve Revision` is chosen, its position is visible under P1 and P3. Trace T-B2 depends on this: B allows `Analyze Revision, Approve Revision, Check Revision`, while A requires the check before approval.
- Under P2, `Approve Revision` is hidden, so T-B2’s ordering difference disappears. That is one reason P2 can regard the models as equivalent while P1 and P3 do not.

# 5. Bounded Comparison Code

The following code performs exhaustive bounded exploration over Petri-net markings under the P3 bound. It does not use `pm4py.play_out`. If `pm4py.play_out` were used instead, the result would be a sample or heuristic playout, not an exhaustive bounded language, because loop exits, exclusive choices, and parallel interleavings may not all be visited.

```python
import pm4py
from pm4py.objects.petri_net.obj import PetriNet, Marking
from collections import deque, defaultdict

TREE_A = (
    "->('Open Case', "
    "*(->(+('Analyze Revision', 'Check Revision'), X('Approve Revision', tau)), 'Rework Case'), "
    "'Close Case')"
)

TREE_B = (
    "->('Open Case', "
    "+('Check Revision', *(->('Analyze Revision', X('Approve Revision', tau)), 'Rework Case')), "
    "'Close Case')"
)

tree_a = pm4py.parse_process_tree(TREE_A)
tree_b = pm4py.parse_process_tree(TREE_B)

net_a, im_a, fm_a = pm4py.convert_to_petri_net(tree_a)
net_b, im_b, fm_b = pm4py.convert_to_petri_net(tree_b)

for name, (net, im, fm) in {
    "A": (net_a, im_a, fm_a),
    "B": (net_b, im_b, fm_b),
}.items():
    sound = pm4py.check_soundness(net, im, fm, print_diagnostics=False)
    print(f"Model {name} sound: {sound}")


def petri_io(net):
    """
    Precompute input and output arcs with weights.
    """
    inputs = defaultdict(list)
    outputs = defaultdict(list)

    for arc in net.arcs:
        weight = getattr(arc, "weight", 1)

        if isinstance(arc.source, PetriNet.Place) and isinstance(arc.target, PetriNet.Transition):
            inputs[arc.target].append((arc.source, weight))
        elif isinstance(arc.source, PetriNet.Transition) and isinstance(arc.target, PetriNet.Place):
            outputs[arc.source].append((arc.target, weight))

    return inputs, outputs


def fire(marking, transition, inputs, outputs):
    """
    Fire a transition manually from a marking.
    """
    new_marking = dict(marking)

    for place, weight in inputs.get(transition, []):
        tokens = new_marking.get(place, 0)
        if tokens < weight:
            raise ValueError("Transition is not enabled.")

        if tokens == weight:
            new_marking.pop(place, None)
        else:
            new_marking[place] = tokens - weight

    for place, weight in outputs.get(transition, []):
        new_marking[place] = new_marking.get(place, 0) + weight

    return Marking(new_marking)


def marking_key(marking):
    return frozenset(marking.items())


def bounded_visible_language(
    net,
    im,
    fm,
    max_visible=12,
    max_analyze=3,
    max_total_steps=80
):
    """
    Exhaustive bounded visible-language exploration.

    - Hides tau transitions.
    - Counts only visible events toward max_visible.
    - Counts only Analyze Revision toward max_analyze.
    - Accepts only complete traces reaching fm.
    - Uses a step cap to avoid unexpected silent cycles.
    """
    inputs, outputs = petri_io(net)

    language = set()
    start_trace = tuple()
    start_key = (marking_key(im), 0, 0, start_trace)

    queue = deque([(im, 0, 0, start_trace, 0)])
    min_steps = {start_key: 0}
    truncated = False

    while queue:
        marking, visible_len, analyze_count, trace, steps = queue.popleft()

        current_key = (marking_key(marking), visible_len, analyze_count, trace)
        if steps > min_steps.get(current_key, steps):
            continue

        if steps > max_total_steps:
            truncated = True
            continue

        # Explicit final-marking equality check.
        if marking == fm:
            language.add(trace)
            continue

        enabled = pm4py.get_enabled_transitions(net, marking)

        for transition in enabled:
            label = getattr(transition, "label", None)

            visible = label is not None and str(label).lower() != "tau"

            next_visible_len = visible_len + (1 if visible else 0)
            next_analyze_count = (
                analyze_count + 1
                if visible and str(label) == "Analyze Revision"
                else analyze_count
            )

            if next_visible_len > max_visible:
                continue

            if next_analyze_count > max_analyze:
                continue

            try:
                next_marking = fire(marking, transition, inputs, outputs)
            except ValueError:
                continue

            next_trace = trace + ((str(label),) if visible else tuple())
            next_steps = steps + 1

            next_key = (
                marking_key(next_marking),
                next_visible_len,
                next_analyze_count,
                next_trace,
            )

            if next_steps < min_steps.get(next_key, float("inf")):
                min_steps[next_key] = next_steps
                queue.append(
                    (
                        next_marking,
                        next_visible_len,
                        next_analyze_count,
                        next_trace,
                        next_steps,
                    )
                )

    return language, truncated


lang_a, trunc_a = bounded_visible_language(net_a, im_a, fm_a)
lang_b, trunc_b = bounded_visible_language(net_b, im_b, fm_b)

only_a = lang_a - lang_b
only_b = lang_b - lang_a

print("\nBounded comparison under P3")
print("Bound: <= 12 visible events and <= 3 Analyze Revision occurrences")
print(f"Exhaustive within step cap: {not (trunc_a or trunc_b)}")
print(f"Model A bounded traces: {len(lang_a)}")
print(f"Model B bounded traces: {len(lang_b)}")
print(f"Traces unique to Model A: {len(only_a)}")
print(f"Traces unique to Model B: {len(only_b)}")


def show(trace_set, prefix, limit=20):
    for trace in sorted(trace_set, key=lambda x: (len(x), x))[:limit]:
        print(prefix, " -> ".join(trace))


print("\nFirst traces unique to Model A:")
show(only_a, "A-only:")

print("\nFirst traces unique to Model B:")
show(only_b, "B-only:")

print("\nThis is exhaustive bounded state exploration, not random playout.")
print("pm4py.play_out would only provide sampled evidence unless combined with an exhaustive separate language construction.")
```

Important implementation notes:

- Silent transitions are fired but not appended to the visible trace.
- `visible_len` counts only visible labels.
- `analyze_count` counts only `Analyze Revision`.
- A trace is included only when the current marking equals the final marking.
- Repeated markings are handled through `min_steps`, which keeps the shortest path to the same marking and visible prefix. This prevents silent-cycle explosion while still preserving distinct visible prefixes.
- If `trunc_a` or `trunc_b` is `True`, the step cap was reached and the exploration should not be considered complete for the requested bound.

# 6. Stochastic and Footprint Comparison

Footprints and stochastic distances can add behavioral evidence, but they are weaker than trace-language equivalence.

## Footprints

`pm4py.discover_footprints(...)` computes pairwise activity relations such as causality, concurrency, and conflict. Footprints abstract away from:

- exact trace sequences,
- occurrence counts,
- loop cardinality,
- probabilities,
- and silent-step details.

Therefore, even if footprints were equal, that would not prove trace equivalence. In this example, footprint equality would still hide the decisive fact that Model A can repeat `Check Revision` once per iteration while Model B has exactly one `Check Revision`.

```python
# Assumes tree_a, tree_b, net_a, net_b from the previous section.

def safe_footprints(tree, net):
    try:
        return pm4py.discover_footprints(tree)
    except Exception:
        return pm4py.discover_footprints(net)

fp_a = safe_footprints(tree_a, net_a)
fp_b = safe_footprints(tree_b, net_b)

print("Footprints equal:", fp_a == fp_b)
```

## Stochastic language and Earth Mover’s Distance

A stochastic language is a probability distribution over traces. It can be useful if:

- you have an observed event log,
- you have declared transition probabilities or rates,
- or you have a stochastic Petri net semantics.

Do not invent transition probabilities for the exclusive choice or loop exit unless those probabilities are part of the business model or estimated from data.

If a stochastic language is available, `pm4py.compute_emd(...)` can compare distributions. A low Earth Mover’s Distance can indicate that the likely behavior is close, but it does not establish trace equivalence because:

- rare distinguishing traces may have low probability,
- bounded or sampled languages may omit differences,
- zero or low distance over a bounded language does not prove unbounded equality,
- stochastic semantics may assign zero probability to traces that are structurally possible,
- and EMD depends on the chosen trace distance metric.

```python
# Illustrative only.
# This requires a PM4Py version and input type where get_stochastic_language is meaningful.
# Do not assign probabilities unless they are declared or estimated.

def stochastic_emd_if_available(obj_a, obj_b):
    try:
        lang_a = pm4py.get_stochastic_language(obj_a)
        lang_b = pm4py.get_stochastic_language(obj_b)
        return pm4py.compute_emd(lang_a, lang_b)
    except Exception as exc:
        print("Stochastic EMD not computed:", exc)
        return None

emd_value = stochastic_emd_if_available(tree_a, tree_b)
print("Stochastic EMD:", emd_value)
```

If you want to compare observed behavior instead of model-implied stochastic behavior, use event logs and derive stochastic languages from those logs. That avoids inventing model probabilities.

# 7. Complexity Comparison

The following exact values are derived from the logical process-tree syntax, counting every operator node and every leaf node. Depth uses root depth `0`.

| Metric | Model A | Model B | Comment |
|---|---:|---:|---|
| Total process-tree nodes | 12 | 12 | Operators plus leaves. |
| Operator nodes | 5 | 5 | A: `->`, `*`, `->`, `+`, `X`. B: `->`, `+`, `*`, `->`, `X`. |
| Leaf nodes | 7 | 7 | Includes the silent `tau` leaf. |
| Maximum depth, root = 0 | 4 | 5 | B is deeper because the loop is nested inside the parallel operator. |
| Visible leaves | 6 | 6 | `Open Case`, `Analyze Revision`, `Check Revision`, `Approve Revision`, `Rework Case`, `Close Case`. |
| Silent leaves | 1 | 1 | The `tau` alternative. |
| Duplicate visible leaf labels in tree | 0 | 0 | Duplicate occurrences arise during execution because of the loop, not because the tree repeats leaf labels. |
| Duplicate `Check Revision` occurrences in traces | Unbounded, equal to number of iterations | Exactly 1 | This is the main behavioral difference. |

The following code recomputes structural metrics and prints Petri-net simplicity. The exact simplicity value can depend on the PM4Py version and Petri-net translation, so the code reports it rather than assuming a numeric value.

```python
from collections import Counter

def tree_metrics(tree):
    node_count = 0
    leaf_count = 0
    visible_leaf_count = 0
    silent_leaf_count = 0
    max_depth = 0
    visible_labels = []

    def walk(node, depth):
        nonlocal node_count
        nonlocal leaf_count
        nonlocal visible_leaf_count
        nonlocal silent_leaf_count
        nonlocal max_depth

        node_count += 1
        max_depth = max(max_depth, depth)

        children = getattr(node, "children", [])

        if not children:
            leaf_count += 1
            label = getattr(node, "label", None)

            if label is None or str(label).lower() == "tau":
                silent_leaf_count += 1
            else:
                visible_leaf_count += 1
                visible_labels.append(str(label))
        else:
            for child in children:
                walk(child, depth + 1)

    walk(tree, 0)

    label_counts = Counter(visible_labels)
    duplicate_visible_leaf_occurrences = sum(
        count - 1
        for count in label_counts.values()
        if count > 1
    )

    return {
        "node_count": node_count,
        "operator_count": node_count - leaf_count,
        "leaf_count": leaf_count,
        "visible_leaf_count": visible_leaf_count,
        "silent_leaf_count": silent_leaf_count,
        "max_depth_root_0": max_depth,
        "duplicate_visible_leaf_occurrences": duplicate_visible_leaf_occurrences,
        "visible_labels": sorted(label_counts.keys()),
    }

print("Model A metrics:", tree_metrics(tree_a))
print("Model B metrics:", tree_metrics(tree_b))

for name, (net, im, fm) in {
    "A": (net_a, im_a, fm_a),
    "B": (net_b, im_b, fm_b),
}.items():
    try:
        simplicity = pm4py.simplicity_petri_net(net, im, fm)
    except TypeError:
        simplicity = pm4py.simplicity_petri_net(net)
    except Exception as exc:
        simplicity = f"not computed: {exc}"

    print(f"Model {name} Petri-net simplicity: {simplicity}")
```

Structurally, Model B is slightly deeper because the loop is nested inside the parallel branch. Model A and Model B have the same number of logical nodes and leaves, but their wiring differs, and that wiring changes the trace language substantially.

# 8. Simplified Model

I choose P1, more precisely P1+, as the most appropriate target semantics for a compliance-oriented model. The simplification removes the parallel operator between `Analyze Revision` and `Check Revision` and enforces the stricter order:

```text
Analyze Revision -> Check Revision -> optional Approve Revision
```

This is simpler than Model A and much closer to the operational compliance intent than Model B.

Simplified process tree:

```text
->(
  'Open Case',
  *(
    ->(
      'Analyze Revision',
      'Check Revision',
      X('Approve Revision', tau)
    ),
    'Rework Case'
  ),
  'Close Case'
)
```

PM4Py code:

```python
import pm4py

simplified_model = pm4py.parse_process_tree(
    "->('Open Case', "
    "*(->('Analyze Revision', 'Check Revision', X('Approve Revision', tau)), 'Rework Case'), "
    "'Close Case')"
)

s_net, s_im, s_fm = pm4py.convert_to_petri_net(simplified_model)

print(
    "Simplified model sound:",
    pm4py.check_soundness(s_net, s_im, s_fm, print_diagnostics=False)
)
```

## Structural comparison with the simplified model

| Model | Total nodes | Operator nodes | Maximum depth, root = 0 | Visible leaves | Silent leaves | Check semantics |
|---|---:|---:|---:|---:|---:|---|
| Model A | 12 | 5 | 4 | 6 | 1 | One check per iteration, concurrent with analysis. |
| Model B | 12 | 5 | 5 | 6 | 1 | One global check concurrent with the whole loop. |
| Simplified | 11 | 4 | 4 | 6 | 1 | One check per iteration, after analysis. |

The simplified model is structurally simpler than Model A because it removes the parallel operator. It is also less deep than Model B. It is not behaviorally identical to Model A, but the difference is intentional: it removes the compliance-questionable possibility that a check occurs before the analysis it is supposed to check.

## Preservation table

| Property or view | Status in simplified model | Explanation |
|---|---|---|
| `Open Case` first, `Close Case` last | exact | Preserved by the outer sequence. |
| At least one `Analyze Revision` | exact | Binary loop executes the first child at least once. |
| At least one `Check Revision` | exact | Each loop iteration contains one check. |
| One check per analysis | exact | Each iteration has one analysis and one check. |
| Check after analysis | exact, strengthened | The simplified model enforces `Analyze Revision` before `Check Revision`. Model A allowed the reverse order because of parallelism. |
| `Rework Case` separates analysis attempts | exact | Rework remains the loop repeat branch. |
| Optional approval | exact | `X('Approve Revision', tau)` is preserved. |
| Approval count and placement | exact for P1/P1+ | One optional approval per iteration, after the check. |
| P2 dashboard abstraction | exact | The simplified model still yields `check_occurred = true`, `analyze_count = k`, `rework_occurred = k > 1`. |
| P3 visible-trace equivalence to Model A | lost | The simplified model forbids `Check Revision` before `Analyze Revision`. |
| P3 visible-trace equivalence to Model B | lost | The simplified model repeats checks per iteration, while Model B has only one check. |
| Model B’s single global check behavior | lost | Intentionally lost because it violates P1 for multiple analyses. |

## Regression traces for the simplified model

These traces should be accepted.

| Trace ID | Trace | Simplified model | Why it should pass |
|---|---|---|---|
| R1 | `Open Case, Analyze Revision, Check Revision, Close Case` | yes | Single iteration, approval silently skipped. |
| R2 | `Open Case, Analyze Revision, Check Revision, Approve Revision, Rework Case, Analyze Revision, Check Revision, Close Case` | yes | Two iterations; first approved, second silently skipped. |
| R3 | `Open Case, Analyze Revision, Check Revision, Rework Case, Analyze Revision, Check Revision, Approve Revision, Close Case` | yes | Two iterations; first silently skipped, second approved. |

## Negative traces for the simplified model

These traces should be rejected.

| Trace ID | Trace | Simplified model | Why it should fail |
|---|---|---|---|
| N1 | `Open Case, Check Revision, Analyze Revision, Close Case` | no | Check occurs before analysis. This may be allowed by Model A, but not by the stricter P1+ simplified model. |
| N2 | `Open Case, Analyze Revision, Rework Case, Analyze Revision, Check Revision, Close Case` | no | The first analysis attempt has no check before `Rework Case`. This trace is possible in Model B but violates P1. |
| N3 | `Open Case, Analyze Revision, Check Revision, Approve Revision, Check Revision, Close Case` | no | Extra check without a new analysis/rework iteration. Not generated by the simplified loop structure. |

A single simplification cannot preserve all three views. The simplified model preserves P1/P1+ and the P2 dashboard abstraction, but it intentionally sacrifices exact P3 trace equivalence to both original models.

# 9. Final Recommendation

| View | Suitable original model | Should the simplified model replace it? |
|---|---|---|
| P1 operational compliance | Model A is closer than Model B because it repeats the check per revision iteration. However, Model A allows `Check Revision` before `Analyze Revision`. | If the business requires check-after-analysis, replace Model A with the simplified model. Do not use Model B for P1. |
| P2 executive dashboard | Model A and Model B are equivalent under the stated dashboard abstraction. | The simplified model is also equivalent under P2 and is structurally simpler, so it can be used for dashboard purposes. |
| P3 bounded testing | Neither Model A nor Model B is equivalent. Bounded testing should report differences. | The simplified model should not be justified by P3 equivalence. It is justified only if P1/P1+ is the target semantics. |

Warning about bounded comparison:

- Bounded comparison under P3 can prove that two models differ if a distinguishing trace is found inside the bound.
- It cannot prove unbounded equivalence if no difference is found.
- The chosen bound, `<= 12` visible events and `<= 3` analyses, is enough to expose the main differences here, but a different bound could hide some differences.
- Exhaustive bounded state exploration is stronger than random playout, but it is still not a substitute for an unbounded semantic proof.
