```python
"""
Interval-aware temporal conformance prototype + lossy midpoint comparison via PM4Py.
Standard library + pandas + pm4py. No external files.
"""

import itertools
from datetime import datetime, timedelta, timezone
from zoneinfo import ZoneInfo
import pandas as pd

# ---------------------------------------------------------------
# 0. Constants / semantics of sources
# ---------------------------------------------------------------
UTC = timezone.utc

# offset bounds in seconds: (a, b) with clock_offset = recorded - true in [a, b]
SOURCES = {
    "HUB":       {"tz": "UTC",                 "rounding": "second",  "offset": (-1, 1)},
    "OPS-BER":   {"tz": "Europe/Berlin",       "rounding": "floor_minute", "offset": (-90, 120)},
    "SENSOR":    {"tz": "UTC",                 "rounding": "nearest10", "offset": (-5, 5)},
    "VENDOR-NY": {"tz": "America/New_York",    "rounding": "floor_minute", "offset": (-180, 180)},
    "LEGACY-BER":{"tz": "Europe/Berlin",       "rounding": "floor_minute", "offset": (-60, 60)},
    "MOBILE-LON":{"tz": "Europe/London",       "rounding": "nearest_minute", "offset": (-300, 300)},
}

def clock_interval(local_dt, rounding):
    """Resolution uncertainty: naive local display -> [lo, hi) local clock interval."""
    if rounding == "second":              # exact to displayed second, +/-0.5s measurement
        lo = local_dt - timedelta(seconds=0.5); hi = local_dt + timedelta(seconds=0.5)
    elif rounding == "floor_minute":
        lo = local_dt.replace(second=0, microsecond=0)
        hi = lo + timedelta(minutes=1)    # exclusive
    elif rounding == "nearest_minute":
        lo = local_dt - timedelta(seconds=30); hi = local_dt + timedelta(seconds=30)
```

```python
def clock_interval(local_naive: datetime, rounding: str):
    """Step 1 (resolution): displayed local value -> [lo, hi) local clock interval.
    Upper bounds from floor/rounding semantics are exclusive; we keep them as
    instants and rely on the note that no classification flips on open/closed ends."""
    if rounding == "second":                 # displayed second exact +/- 0.5 s measurement
        return local_naive - timedelta(seconds=0.5), local_naive + timedelta(seconds=0.5)
    if rounding == "floor_minute":           # 10:14 means clock in [10:14:00, 10:15:00)
        lo = local_naive.replace(second=0, microsecond=0)
        return lo, lo + timedelta(minutes=1)
    if rounding == "nearest_minute":
        return local_naive - timedelta(seconds=30), local_naive + timedelta(seconds=30)
    if rounding == "nearest10":
        return local_naive - timedelta(seconds=5), local_naive + timedelta(seconds=5)
    raise ValueError(rounding)


def utc_candidates(lo_local, hi_local, tzname):
    """Step 2 (time zone): local clock interval -> UTC interval(s).
    DST-ambiguous local times yield one interval per fold (fold=0 earlier offset,
    fold=1 later offset). Non-ambiguous times collapse to a single interval."""
    tz = ZoneInfo(tzname)
    out = []
    for fold in (0, 1):
        lo_u = lo_local.replace(fold=fold, tzinfo=tz).astimezone(UTC)
        hi_u = (hi_local - SEC).replace(fold=fold, tzinfo=tz).astimezone(UTC) + SEC
        if not out or (lo_u, hi_u) != out[-1]:
            out.append((lo_u, hi_u))
    return out
```

```python
def duration_intervals(feas, case, act_from, act_to):
    """Feasible durations act_to - act_from as a union of intervals (seconds)."""
    A, B = feas.get((case, act_from)), feas.get((case, act_to))
    if not A or not B:
        return None
    out = []
    for (alo, ahi), (blo, bhi) in itertools.product(A, B):
        d_lo = (blo - ahi).total_seconds()
        d_hi = (bhi - alo).total_seconds()
        out.append((d_lo, d_hi))
    # keep disjoint structure; drop duplicates
    return out
```

```python
"""
================================================================================
Interval-aware temporal conformance prototype (standard library + pandas)
plus a deliberately LOSSY midpoint-only comparison through PM4Py.

-no external files are read.
- PM4Py diagnostics are point-based: PM4Py does NOT natively model timestamp
  uncertainty, rounding, clock offsets, or DST folds. The midpoint log below is
  labelled as a lossy approximation and is used only as a secondary comparison.
================================================================================
"""

from __future__ import annotations

import itertools
from datetime import datetime, timedelta, timezone
from zoneinfo import ZoneInfo

import pandas as pd

try:
    import pm4py
    HAVE_PM4PY = True
except ImportError:                       # interval analysis still runs without pm4py
    HAVE_PM4PY = False

UTC = timezone.utc
EPS = timedelta(seconds=1)                # treats half-open bounds as instants (documented)

# ---------------------------------------------------------------------------
# 1) Source semantics
#    clock_offset = recorded_clock - true_clock in [a, b] seconds
#    => true = recorded - offset in [recorded - b, recorded - a]
# ---------------------------------------------------------------------------
SOURCES = {
    "HUB":        dict(tz="UTC",              rounding="second",         offset=(-1, 1)),
    "OPS-BER":    dict(tz="Europe/Berlin",    rounding="floor_minute",   offset=(-90, 120)),
    "SENSOR":     dict(tz="UTC",              rounding="nearest10",      offset=(-5, 5)),
    "VENDOR-NY":  dict(tz="America/New_York", rounding="floor_minute",   offset=(-180, 180)),
    "LEGACY-BER": dict(tz="Europe/Berlin",    rounding="floor_minute",   offset=(-60, 60)),
    "MOBILE-LON": dict(tz="Europe/London",    rounding="nearest_minute", offset=(-300, 300)),
}


def clock_interval(local_naive: datetime, rounding: str):
    """STEP 1 - timestamp resolution: displayed local value -> [lo, hi) clock time."""
    if rounding == "second":            # exact to the second; +/-0.5 s measurement error
        return local_naive - timedelta(seconds=0.5), local_naive + timedelta(seconds=0.5)
    if rounding == "floor_minute":      # 10:14 -> clock in [10:14:00, 10:15:00)
        lo = local_naive.replace(second=0, microsecond=0)
        return lo, lo + timedelta(minutes=1)
    if rounding == "nearest_minute":    # 13:14 -> clock in [13:13:30, 13:14:30)
        return local_naive - timedelta(seconds=30), local_naive + timedelta(seconds=30)
    if rounding == "nearest10":         # 08:51:10 -> clock in [08:51:05, 08:51:15)
        return local_naive - timedelta(seconds=5), local_naive + timedelta(seconds=5)
    raise ValueError(f"unknown rounding {rounding!r}")


def utc_candidates(lo_local: datetime, hi_local: datetime, tzname: str):
    """STEP 2 - time zone: local clock interval -> UTC interval(s).
    DST-ambiguous local times (Berlin 2025-10-26 02:00-02:59) produce TWO
    alternative intervals (fold=0 = first pass / CEST, fold=1 = second pass / CET)."""
    tz = ZoneInfo(tzname)
    out = []
    for fold in (0, 1):
        lo_u = lo_local.replace(fold=fold, tzinfo=tz).astimezone(UTC)
        hi_u = hi_local.replace(fold=fold, tzinfo=tz).astimezone(UTC)
        if (lo_u, hi_u) not in out:     # unambiguous times collapse to one interval
            out.append((lo_u, hi_u))
    return out


def apply_offset(intervals, offset_bounds):
    """STEP 3 - clock correction: true = clock - offset, offset in [a, b]."""
    a, b = offset_bounds
    return [(lo - timedelta(seconds=b), hi - timedelta(seconds=a)) for lo, hi in intervals]


def normalize(local_naive, system, offset_override=None):
    spec = SOURCES[system]
    off = offset_override or spec["offset"]
    lo, hi = clock_interval(local_naive, spec["rounding"])
    return apply_offset(utc_candidates(lo, hi, spec["tz"]), off)
```

```python
SLA = {
    "S1": ("Alert_Received", "Investigation_Start", 15 * 60),
    "S2": ("Investigation_Start", "Containment_Complete", 45 * 60),
    "S3": ("Containment_Complete", "Close_Incident", 36 * 3600),
}
PROFILE = {
    ("Alert_Received", "Investigation_Start"): (600, 180),
    ("Investigation_Start", "Containment_Complete"): (2100, 600),
    ("Alert_Received", "Containment_Complete"): (2700, 720),
    ("Containment_Complete", "Close_Incident"): (86400, 21600),
}

def feasible_durations(case, act_from, act_to):
    """Union of feasible duration intervals act_to - act_from (seconds).
    Disjoint components are preserved (no blind hull)."""
    A = feasible.get((case, act_from))
    B = feasible.get((case, act_to))
    if not A or not B:
        return None
    durs = []
    for (alo, ahi), (blo, bhi) in itertools.product(A, B):
        durs.append(((blo - ahi).total_seconds(), (bhi - alo).total_seconds()))
    return durs

def classify_sla(durs, limit_s):
    if durs is None:
        return "INDETERMINATE"
    if all(hi <= limit_s for _, hi in durs):
        return "CERTAIN_PASS"
    if all(lo > limit_s for lo, _ in durs):
        return "CERTAIN_VIOLATION"
    return "POSSIBLE_VIOLATION"

def classify_profile(durs, mu, sigma, zeta):
    if durs is None:
        return "INDETERMINATE"
    lo_b, hi_b = mu - zeta * sigma, mu + zeta * sigma
    if all(lo >= lo_b and hi <= hi_b for lo, hi in durs):
        return "CERTAIN_PASS"
    if all(hi < lo_b or lo > hi_b for lo, hi in durs):
        return "CERTAIN_VIOLATION"
    return "POSSIBLE_VIOLATION"
```
