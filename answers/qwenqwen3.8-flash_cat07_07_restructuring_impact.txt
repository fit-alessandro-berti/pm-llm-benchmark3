```python
def add_handover_features(log):
    df = to_df(log).sort_values([CASE, TIME]).copy()
    df['prev_resource'] = df.groupby(CASE)[RES].shift()
    df['prev_activity'] = df.groupby(CASE)[ACT].shift()
    df['prev_time'] = df.groupby(CASE)[TIME].shift()
    df['prev_team'] = df.groupby(CASE)[TEAM].shift()
    df['is_handover'] = (df[RES] != df['prev_resource']) & df['prev_resource'].notna()
    df['handover_wait_h'] = (df[TIME] - df['prev_time']).dt.total_seconds()/3600
    df['from_role'] = df['prev_resource'].apply(resource_role)
    df['to_role'] = df[RES].apply(resource_role)
    df['handover_type'] = df['from_role'] + ' -> ' + df['to_role']
    return df
```

```python
def case_metrics(log, sla_days_by_segment=None):
    df = to_df(log).sort_values([CASE, TIME])
    cases = df.groupby(CASE).agg(
        start= (TIME,'min'), end=(TIME,'max'),
        team=(TEAM,'first'), complexity=(COMPLEXITY,'first'), specialty=(SPECIALTY,'first'),
        segment=(SEGMENT,'first'), rework=(REWORK,'max'), outcome=(OUTCOME,'first'),
        reassignments=(REASSIGN,'max'), local_expertise=(LOCAL_EXPERTISE,'max')
    ).reset_index()
    cases['cycle_days'] = (cases['end']-cases['start']).dt.total_seconds()/86400
    if sla_days_by_segment:
       cases['sla_days'] = cases.apply(lambda r: sla_days_by_segment.get((r.team,r.complexity), sla_days_by_segment.get('default',7)), axis=1)
    else: cases['sla_days']=7
    cases['sla_miss'] = cases['cycle_days'] > cases['sla_days']
    return cases
```

```python
def add_handover_features(log):
    df = as_df(log).sort_values([CASE, TIME]).copy()
    df["prev_resource"] = df.groupby(CASE)[RES].shift()
    df["prev_activity"] = df.groupby(CASE)[ACT].shift()
    df["prev_time"] = df.groupby(CASE)[TIME].shift()
    df["is_handover"] = (
        (df[RES] != df["prev_resource"]) & df["prev_resource"].notna()
    )
    df["handover_wait_h"] = (
        (df[TIME] - df["prev_time"]).dt.total_seconds() / 3600
    )
    df["from_role"] = df["prev_resource"].apply(resource_role)
    df["to_role"] = df[RES].apply(resource_role)
    df["handover_type"] = df["from_role"] + " -> " + df["to_role"]

    h["rework_after_handover"] = h[REWORK].astype(bool)
    return (
        h.groupby("handover_type")
        .agg(
            frequency_per_100_cases=("handover_type", lambda x: len(x) / cases * 100),
            median_wait_h=("handover_wait_h", "median"),

            p95_wait_h=("handover_wait_h", lambda x: x.quantile(0.95)),
            rework_after_handover_pct=("rework_after_handover", lambda x: x.mean() * 100),
        )
        .reset_index()
    )

def case_metrics(log, default_sla_days=7):
    df = as_df(log).sort_values([CASE, TIME]).copy()
    cases = (
        df.groupby(CASE)
        .agg(
            start=(TIME, "min"),
            end=(TIME, "max"),
            team=(TEAM, "first"),
            complexity=(COMPLEXITY, "first"),
            specialty=(SPECIALTY, "first"),
            segment=(SEGMENT, "first"),
            rework=(REWORK, "max"),
            outcome=(OUTCOME, "first"),
            reassignments=(REASSIGN, "max"),
            local_expertise=(LOCAL_EXPERTISE, "max"),
        )
        .reset_index()
    )
    cases["cycle_days"] = (cases["end"] - cases["start"]).dt.total_seconds() / 86400
    cases["sla_days"] = default_sla_days
    cases["sla_miss"] = cases["cycle_days"] > cases["sla_days"]
    return cases
```

# 1. Mechanism map

The restructuring changes the organization from **local, embedded specialist loops** to **centralized queues and shared services**. The key process-mining implication is that “handovers” may not disappear; they may change form: informal local loops are replaced by formal central queue handovers, pull handovers, reassignments, and approval loops.

| Restructuring action | Mechanism | Expected process-mining signal | Countervailing mechanism / risk |
|---|---|---|---|
| Merge N_Tech, S_Tech, E_Tech into a Technical Shared Service | Reduces local caseworker  embedded tech loops and concentrates technical work in one queue | `discover_handover_of_work_network`: fewer edges like `local_caseworker -> embedded_tech` and `embedded_tech -> local_caseworker`; more edges involving central technical resources | Nominal handover reduction may be offset by new queue handovers: `local_caseworker -> Technical Shared Service`, `Technical Shared Service -> local_caseworker`, and repeated back-and-forth if advice is incomplete |
| Centralize Specialist Review, Legal Review, Risk Review, and difficult Quality Check | Creates shared-service queues for scarce expertise | `insert_case_service_waiting_time`, `get_service_time`, performance DFG: longer waiting edges before specialist activities; more reassignment or hold variants | Legal/Risk specialists are already near capacity; centralizing triage and difficult QC may create hidden overload and tail delays |
| Route all high-complexity cases to Technical Shared Service | Increases specialist utilization and standardization | Activity counts and resource similarity: central specialists appear in more cases; `discover_activity_based_resource_similarity` shows central technical resources becoming more similar across teams | High-complexity cases may wait longer in queue even if average cycle time improves; local informal advice that previously avoided referrals disappears |
| Introduce shared approval queue and remove one local supervisor | Standardizes approvals and reduces local supervisor loops | `discover_organizational_roles`: approver role becomes centralized; `discover_subcontracting_network`: more approval activities performed by shared resources | Ownership may become unclear; `reassignment_count` and handover reasons such as “clarify ownership” may rise |
| Keep Intake and simple Eligibility Review local | Preserves front-line customer contact and low-complexity speed | DFG variants: short simple variants may remain fast; `Bot_Intake -> local intake` and `local intake -> local caseworker` remain common | If local staff lose technical skills, they may escalate more cases than intended, increasing specialist handovers |
| Shared queue pull model for completed specialist advice | Decouples request and response; may reduce synchronous interruptions | Working-together network: local caseworkers and central specialists co-occur in cases, but not necessarily consecutively; handover waits may appear as long gaps | If advice lacks local context, rework may increase later, especially for regional/language or complex customer segments |
| Centralize difficult Quality Check | Standardizes quality decisions and catches issues centrally | `get_rework_cases_per_activity`: Quality Check may show more detected defects; variants with Rework may increase or shift later | Central QC backlog can delay closure and create rework loops after implementation |
| Remove embedded local technical specialists from day-to-day availability | Reduces local expertise and informal learning | `local_expertise_flag` declines; local caseworkers perform fewer technical activities; rework rises for cases where local expertise was previously high | North/South may lose regional knowledge; East may gain because E_Tech was overloaded, but learning effects can still be negative |
| Create central queues for technical, legal/risk, and approval work | Increases specialist utilization but also queue sensitivity | Service waiting time, queue depth, p90/p95 handover wait, SLA miss tail | Utilization above roughly 80–85% can cause nonlinear waiting; absence of central staff may affect all teams |
| Standardize routing by complexity and specialty | Reduces ad hoc cross-team handovers | Cross-team handover frequency decreases; handover_reason shifts from “absence cover” to “routing rule” | If routing rules are too broad, simple cases may be over-routed to specialists, increasing queue load |

Important nuance: the management estimate does not clearly show fewer total handovers. Current local technical loops are 36 + 31 = 67 per 100 cases; proposed central technical loops are 41 + 36 = 77 per 100 cases. Approval loops also increase from 42 + 39 = 81 to 49 + 43 = 92. Legal/Risk handovers increase from 9 to 12. Cross-team handovers decrease from 6 to 3. So the restructuring may reduce some informal or local loops while increasing formal central queue loops and waiting.

# 2. Scenario forecasts

## Optimistic scenario

This scenario assumes the shared service is sized correctly, routing thresholds are narrow, local informal triage is preserved, and legal/risk capacity is not over-committed.

- **Handover pattern**
  - Local caseworker  embedded tech loops fall because technical advice is centralized and more consistent.
  - Central technical handovers increase, but they are efficient: one request, one structured response, minimal rework.
  - Cross-team ad hoc handovers fall because shared services replace absence cover.
  - Handover_reason shifts toward “specialist advice” and “standardized approval,” with fewer “absence cover” or “unclear ownership” reasons.

- **Queue/waiting-time pattern**
  - Median handover waits stay close to management estimates.
  - p90 and p95 waits do not deteriorate materially.
  - Technical Shared Service utilization rises toward the target, but remains below saturation, for example 75–85%.
  - Legal/Risk queues do not worsen because broader detection is matched with added capacity or stricter triage.

- **Cycle-time distribution**
  - Median cycle time improves, especially for East.
  - p90 cycle time is flat or improves because high-complexity cases receive better specialist support.
  - High-complexity cases do not suffer large tail delays because the central service has enough capacity and clear service-level rules.

- **Rework and quality pattern**
  - Overall rework rate falls because specialist advice is more standardized and supervisor approval is consistent.
  - Quality Check detects more issues before closure, but does not create a backlog.
  - Rework after handover decreases for technical and approval loops.
  - Customer satisfaction improves, especially where local variability previously caused errors.

- **Role clarity and accountability**
  - Each case still has a named owner: local caseworker owns the case, shared service owns the advice, approval queue owns the decision.
  - `discover_organizational_roles` shows cleaner role separation rather than blurred responsibilities.

- **Fairness implications**
  - East benefits from relief of E_Tech overload.
  - North and South do not lose service quality because local knowledge is retained through informal triage and case ownership.

- **Early warning signs**
  - Minimal warning signs if central queues are monitored.
  - Still watch: p90 handover wait, high-complexity SLA misses, rework after central specialist advice, and staff reports that informal technical questions are harder to answer.

## Central scenario

This is the most likely mixed outcome: some benefits appear, but average metrics hide distributional damage.

- **Handover pattern**
  - Nominal local technical loops decrease, but total handovers may stay flat or increase because central queue handovers replace them.
  - `local caseworker -> Technical Shared Service` and `Technical Shared Service -> local caseworker` become the dominant technical handover edges.
  - Shared approval handovers increase because medium/high-value cases move to a queue.

  - Median waits may look acceptable.

- **Cycle-time distribution**
  - Average or median cycle time may improve because East improves and simple cases are automated or standardized.
  - p90 cycle time may be flat or worse.
  - High-complexity cases may experience longer total time even if they receive better specialist advice.
  - The distribution becomes more bimodal: simple cases are fast, complex cases wait in central queues.

- **Rework and quality pattern**
  - Overall rework may be stable or slightly better.
  - Rework increases for cases that require local context, regional conventions, language accommodation, or nuanced customer handling.
  - Central Quality Check may find more issues, but some are found later, creating rework after implementation.

- **Role clarity and accountability**
  - Role clarity becomes ambiguous in some variants.

- **Early warning signs**

## Adverse scenario

This scenario assumes the shared service is under-sized, routing rules are too broad, local informal expertise disappears, and legal/risk capacity is not increased.

- **Handover pattern**
  - Handovers increase rather than decrease.
  - Local caseworkers escalate more cases because they no longer have embedded technical support.

  - Repeated handovers occur: caseworker  Technical Shared Service  caseworker  Technical Shared Service.
  - Cross-team handovers may rise again if teams try to bypass central queues.

- **Queue/waiting-time pattern**

- **Fairness implications**
  - North and South may lose the most because they previously had strong local expertise.
  - East may improve only if the central service has enough capacity; otherwise East may still suffer because high-complexity cases queue behind other teams.
  - Regional, language, and high-complexity customers are disadvantaged.
  - Fairness gaps widen across teams, customer segments, and complexity levels.
  - Aggregate metrics may hide these harms.

- **Early warning signs**
  - p95 handover wait increases sharply.
  - Queue depth grows for two consecutive weeks.
  - Reassignments increase.
  - Rework after central specialist advice rises.
  - Staff report inability to get informal technical help.
  - Customer complaints increase for language or regional accommodation cases.
  - Absence events cause disproportionate SLA misses.

# 4. Risks invisible in average cycle time

1. **Tail latency**
   - Average cycle time can remain stable or improve while p90, p95, and p99 cases get much worse.
   - Central queues often create long tails because a few cases wait behind many others.
   - PM measure: compare p90/p95 cycle time and p90/p95 handover wait before and after.

2. **High-complexity cases**
   - High-complexity cases may be routed to the central service and wait longer even if simple cases become faster.
   - Management may see an average improvement because simple cases dominate volume.
   - PM measure: segment by `case_complexity`, `priority`, and `required_specialty`; monitor high-complexity p90 cycle and SLA miss.

3. **Local knowledge loss**
   - Local teams currently know regional customer conventions and can prevent unnecessary specialist referrals.
   - After centralization, local staff may initially perform normally because residual knowledge remains, so average metrics may look fine.
   - Over time, informal advice disappears, escalation increases, and rework rises for context-sensitive cases.
   - PM measure: `local_expertise_flag`, local technical activity counts, rework for cases with low local expertise, handover frequency for low-complexity cases.

4. **Fairness across teams**
   - East may improve because E_Tech was overloaded.
   - North and South may worsen because they lose strong local embedded specialists.
   - The aggregate average can hide this redistribution.
   - PM measure: team-level p90 cycle time, SLA miss, rework, and satisfaction gaps.

5. **Fairness across customer segments**
   - Customers needing language or regional accommodation may depend on local knowledge.
   - Centralized technical and approval work may treat them as standard cases, increasing rework and handovers.
   - PM measure: segmented cycle time, rework, handover_reason, and satisfaction by `customer_segment`.

6. **Learning and skill atrophy**
   - If all hard cases are routed away, local caseworkers stop developing technical judgment.
   - Short-term cycle time may improve because specialists handle the hard cases, but long-term resilience and informal triage decline.
   - PM measure: increasing handover rate for low/medium complexity cases, increasing rework after local staff attempts, decreasing local technical activity counts.

7. **Accountability and role clarity**
   - Shared queues can standardize decisions but make ownership unclear.
   - Average cycle time does not show whether cases are delayed because no one knows who should act.
   - PM measure: `reassignment_count`, handover_reason “clarify ownership,” subcontracting edges, role signature drift, repeated handover loops.

8. **Exception handling and resilience to absence**
   - Local teams may currently absorb absence through embedded specialists or informal cover.
   - Centralization can create a single point of failure: if the central service is slow or absent, all teams wait.
   - PM measure: conditional handover wait, SLA miss, and reassignment during resource absence; network centrality and working-together density.

9. **Delayed quality debt**
   - Central Quality Check may catch more issues, but if difficult QC is queued, issues may be found later and create rework after implementation.
   - Average cycle time may not worsen immediately if closure time is shortened by deferring difficult checks.
   - PM measure: rework after Quality Check, rework after Implementation, variants containing Rework, outcome quality by segment.

# 5. Pilot design

## Pilot scope

Use a **phased, conditional pilot**, not full rollout.

- **Phase 0: Baseline**
  - Observe current process for 8–12 weeks.
  - Build baseline handover networks, working-together networks, subcontracting networks, role signatures, DFG variants, service waiting times, rework rates, and segmented cycle-time distributions.

- **Phase 1: Limited Technical Shared Service**
  - Centralize technical advice only for **high-complexity cases** and a narrow set of clearly defined technical questions.
  - Do not route all legal/risk/security cases to a new central queue unless capacity is added.
  - Keep local embedded technical triage for low/medium complexity cases.
  - Preserve informal local advice channels and log them if possible.

- **Phase 2: Shared Approval Pilot**
  - Pilot shared approval only for **routine medium-value cases**.
  - Keep local supervisor approval for high-risk, high-complexity, or customer-sensitive cases.
  - Require a named case owner for every case, even when approval is shared.

- **Phase 3: Difficult Quality Check Pilot**
  - Centralize difficult Quality Check only for cases with prior rework, high complexity, or regulatory risk.
  - Monitor whether central QC creates a new backlog.

- **Phase 4: Legal/Risk Centralization Only If Capacity Exists**
  - Do not broaden legal/risk routing until Legal/Risk utilization is below a safe threshold, for example 80–85%, or until capacity is added.
  - If utilization remains above 90%, keep legal/risk triage local and use central specialists only for formal review.

## Comparison group

Use multiple comparison layers:

- **Team-level control:** East as pilot for central technical service while North and South remain local, if East is the most overloaded team.
- **Complexity-level control:** Within North or South, compare high-complexity routed cases with low/medium-complexity locally handled cases.
- **Specialty-level control:** Compare cases with `required_specialty` in `{legal, risk, security}` against cases with no specialty requirement.
- **Customer-segment control:** Compare language/regional accommodation customers against standard customers.
- **Difference-in-differences:** Compare before/after changes in pilot segments against before/after changes in control segments.

## Pre/post observation window

- Baseline: 8–12 weeks.
- Pilot: 8–12 weeks.
- Stabilization: 4 weeks after each scope change.
- Weekly cohort reporting.
- Avoid interpreting the first 2–3 weeks as final because learning and queue buildup take time.

## Segmentation

Analyze every metric by:

- `team`: North, South, East.
- `case_complexity`: low, medium, high.
- `required_specialty`: none, legal, risk, security, other.
- `customer_segment`: standard, language_accommodation, regional_accommodation, high_value, other.
- `priority`: low, medium, high.
- `handover_reason`: absence cover, specialist advice, queue waiting, clarify ownership, insufficient advice, standard routing.
- `reassignment_count`: 0, 1, 2+.
- `local_expertise_flag`: true/false.

## Guardrail metrics

| Guardrail | Metric | Warning threshold | Stop/rollback threshold |
|---|---|---|---|
| Tail latency | p90 cycle time by segment | p90 worsens by >10% | p90 worsens by >20% for 2 weeks |
| Extreme tail | p95 cycle time by segment | p95 worsens by >15% | p95 worsens by >30% or high-complexity p95 > baseline by 50% |
| High-complexity service | SLA miss for high complexity | +2 percentage points | +5 percentage points |
| Central queue wait | p90 handover wait into Technical Shared Service | p90 > 8 hours | p90 > 16 hours or queue depth grows 3 weeks |
| Legal/Risk saturation | Legal/Risk utilization or wait | utilization >85% | utilization >92% or wait p90 > 20 hours |
| Rework | Overall rework rate | +1 percentage point | +3 percentage points |
| Context-sensitive rework | Rework for language/regional or high-complexity cases | +2 percentage points | +5 percentage points |
| Ownership clarity | reassignment_count | mean +10% | mean +25% or “clarify ownership” handovers double |
| Fairness gap | p90 cycle gap between East and North/South | gap increases >10% | gap increases >25% |
| Customer fairness | p90 cycle gap across customer segments | gap increases >10% | gap increases >25% |
| Resilience | SLA miss during absence of central resource | spike > baseline | central absence causes >50% SLA miss |
| Staff feedback | Role clarity / informal advice availability | declining trend | sustained low score or complaints about inability to get help |

## Staff feedback collection

- Weekly 2-minute pulse survey:
  - “Can you get informal technical help when needed?”
  - “Is it clear who owns the case?”
  - “Are central queues delaying customer work?”
  - “Are you able to handle more complex cases yourself?”
  - “Are regional/language customer needs being met?”
- Monthly focus groups with North, South, East, central specialists, and approvers.
- Incident log for customer complaints related to handovers, waiting, or rework.
- Pair staff feedback with event-log metrics: for example, if staff report unclear ownership, check `reassignment_count`, handover_reason, and subcontracting edges.

## Rollback or adjustment triggers

Adjust rather than rollback if:

- p90 handover wait rises but median remains acceptable.
- Utilization is high but not saturated.
- Rework rises only in a narrow segment.
- Staff report moderate role ambiguity.

Adjustments include:

- Raise routing threshold so fewer low/medium cases go to central service.
- Add capacity to the specific bottleneck queue.
- Restore local embedded triage for low/medium complexity cases.
- Create a named case owner field.
- Separate legal/risk queues from technical queues.
- Add language/regional customer routing rules.
- Reduce difficult Quality Check centralization if QC becomes a bottleneck.

Rollback if:

- p95 cycle time worsens by more than 30% for 2 consecutive weeks.
- High-complexity SLA miss increases by more than 5 percentage points.
- Central queue p90 wait exceeds 16 hours.
- Legal/Risk utilization remains above 92% without capacity relief.
- Rework increases by more than 3 percentage points overall or more than 5 points for context-sensitive segments.
- Reassignment_count increases by more than 25%.
- Absence of a central resource causes system-wide failure.
- Staff survey shows sustained loss of role clarity or inability to serve regional/language customers.

# 6. Recommendation

**Recommendation: pilot first and modify the design. Do not proceed to full rollout on the strength of expected average cycle-time improvement.**

The restructuring has a plausible benefit: East is overloaded, E_Tech utilization is high, cycle times are slower, and centralized technical capacity could reduce local bottlenecks. Shared services may also standardize approvals and reduce ad hoc cross-team cover. However, the management estimate already suggests more central handovers and longer median waits for technical, approval, and legal/risk loops. The main risk is not necessarily that average cycle time will worsen; it is that average cycle time may improve or remain stable while tail latency, fairness, learning, resilience, and accountability deteriorate.

A better design would be:

1. **Centralize technical advice only for high-complexity or clearly defined cases**, not all technical questions.
2. **Preserve local informal triage** for low/medium complexity cases so local expertise is not lost.
3. **Do not centralize Legal/Risk immediately** because those specialists are already at 91–93% utilization. Add capacity or tighten triage first.
4. **Pilot shared approval for routine medium-value cases only**, while keeping accountable case ownership explicit.
5. **Keep difficult Quality Check local or hybrid initially**, unless central QC capacity is proven.
6. **Monitor p90/p95 cycle time, service waiting time, rework by segment, reassignments, role clarity, and fairness gaps**, not just averages.

Management should not infer from early averages that the restructuring is working. Early improvements may come from:

- East benefiting while North/South lose.
- Simple cases becoming faster while high-complexity cases wait longer.
- Automation reducing front-end time while central queues create back-end delays.
- Residual local expertise temporarily masking skill loss.
- Staff adapting informally in ways that are not captured in the event log.
- Averages hiding fairness problems for language, regional, high-complexity, or high-risk customers.

Proceed only if the pilot shows that central queues do not create harmful tails, that rework does not increase for context-sensitive cases, that role clarity remains stable, and that fairness gaps do not widen. If those conditions are met, expand gradually. If not, revert to local embedded specialists for the affected segments and redesign the shared service.

# 7. PM4Py analysis sketch

The following is a PM4Py-style sketch. It assumes the event log contains the listed columns and that attributes such as `team`, `case_complexity`, `required_specialty`, and `customer_segment` are present on events or joined at case level. Adapt function names and return formats to the PM4Py version in use.

```python
import pandas as pd
import pm4py

CASE = "case:concept:name"
ACT = "concept:name"
TIME = "time:timestamp"
RES = "org:resource"
TEAM = "team"
COMPLEXITY = "case_complexity"
SPECIALTY = "required_specialty"
SEGMENT = "customer_segment"
HANDOVER_REASON = "handover_reason"
REASSIGN = "reassignment_count"
LOCAL_EXPERTISE = "local_expertise_flag"
OUTCOME = "outcome"
REWORK = "rework_flag"


def format_log(log):
    return pm4py.format_dataframe(
        log,
        case_id=CASE,
        activity_key=ACT,
        timestamp_key=TIME,
        resource_key=RES,
    )


def as_df(log):
    return log.copy() if isinstance(log, pd.DataFrame) else pd.DataFrame(log)


def filter_attr(log, col, values):
    if values is None:
        return log
    df = as_df(log)
    return df[df[col].isin(values)]


def resource_role(resource):
    r = str(resource)
    if "Bot" in r:
        return "automation"
    if "Intake" in r:
        return "local_intake"
    if "Case" in r:
        return "local_caseworker"
    if "_Tech" in r or "TSS" in r:
        return "technical_specialist"
    if "Legal" in r or "Risk" in r:
        return "legal_risk_specialist"
    if "Supervisor" in r or "Approval" in r:
        return "approver"
    if "Quality" in r:
        return "quality_check"
    return "other"


def add_handover_features(log):
    df = as_df(log).sort_values([CASE, TIME]).copy()

    df["prev_resource"] = df.groupby(CASE)[RES].shift()
    df["prev_activity"] = df.groupby(CASE)[ACT].shift()
    df["prev_time"] = df.groupby(CASE)[TIME].shift()

    df["is_handover"] = (
        (df[RES] != df["prev_resource"]) & df["prev_resource"].notna()
    )

    df["handover_wait_h"] = (
        (df[TIME] - df["prev_time"]).dt.total_seconds() / 3600
    )

    df["from_role"] = df["prev_resource"].apply(resource_role)
    df["to_role"] = df[RES].apply(resource_role)
    df["handover_type"] = df["from_role"] + " -> " + df["to_role"]

    return df


def summarize_handovers(log):
    df = add_handover_features(log)
    cases = df[CASE].nunique()
    h = df[df["is_handover"]].copy()

    if h.empty:
        return pd.DataFrame()

    h["rework_after_handover"] = h[REWORK].astype(bool)

    return (
        h.groupby("handover_type")
        .agg(
            frequency_per_100_cases=(
                "handover_type",
                lambda x: len(x) / cases * 100,
            ),
            median_wait_h=("handover_wait_h", "median"),
            p90_wait_h=("handover_wait_h", lambda x: x.quantile(0.90)),
            p95_wait_h=("handover_wait_h", lambda x: x.quantile(0.95)),
            rework_after_handover_pct=(
                "rework_after_handover",
                lambda x: x.mean() * 100,
            ),
        )
        .reset_index()
    )


def case_metrics(log, default_sla_days=7):
    df = as_df(log).sort_values([CASE, TIME]).copy()

    cases = (
        df.groupby(CASE)
        .agg(
            start=(TIME, "min"),
            end=(TIME, "max"),
            team=(TEAM, "first"),
            complexity=(COMPLEXITY, "first"),
            specialty=(SPECIALTY, "first"),
            segment=(SEGMENT, "first"),
            rework=(REWORK, "max"),
            outcome=(OUTCOME, "first"),
            reassignments=(REASSIGN, "max"),
            local_expertise=(LOCAL_EXPERTISE, "max"),
        )
        .reset_index()
    )

    cases["cycle_days"] = (
        (cases["end"] - cases["start"]).dt.total_seconds() / 86400
    )

    cases["sla_days"] = default_sla_days
    cases["sla_miss"] = cases["cycle_days"] > cases["sla_days"]

    return cases


def distribution_summary(cases, label):
    return {
        "label": label,
        "cases": len(cases),
        "median_cycle_days": cases["cycle_days"].median(),
        "p90_cycle_days": cases["cycle_days"].quantile(0.90),
        "p95_cycle_days": cases["cycle_days"].quantile(0.95),
        "sla_miss_pct": cases["sla_miss"].mean() * 100,
        "rework_pct": cases["rework"].astype(bool).mean() * 100,
        "mean_reassignments": cases["reassignments"].mean(),
        "local_expertise_pct": cases["local_expertise"].astype(bool).mean() * 100,
    }


def fairness_gap(cases, group, metric="cycle_days", quantile=0.90):
    stats = cases.groupby(group)[metric].quantile(quantile)
    return stats.max() - stats.min()


def graph_to_df(graph):
    if hasattr(graph, "edges"):
        rows = []
        for u, v, data in graph.edges(data=True):
            rows.append(
                {
                    "source": u,
                    "target": v,
                    "weight": data.get("weight", 1),
                }
            )
        return pd.DataFrame(rows)

    return pd.DataFrame(graph)

        "temporal": {
            "before": temporal_metrics(lb),
```
