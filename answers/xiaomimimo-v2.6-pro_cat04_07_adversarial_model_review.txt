# model_third.py -- self-contained
import pm4py
from pm4py.objects.powl.obj import StrictPartialOrder, OperatorPOWL, Transition, SilentTransition
from pm4py.objects.powl.operator import Operator


def supplier_review_node(case_type: str):
    """case_type in {'external', 'internal', 'unknown'} (P3)."""
    if case_type == 'external':
        return Transition(label='Supplier Review')          # mandatory
    if case_type == 'internal':
        return SilentTransition()                            # must not occur
    return OperatorPOWL(operator=Operator.XOR,
                        children=[Transition(label='Supplier Review'), SilentTransition()])


def build_model_third(case_type='unknown'):
    reg = Transition(label='Register Dossier')
    cc, tr = Transition(label='Credit Check'), Transition(label='Technical Review')
    cd = Transition(label='Correct Dossier')
    er = Transition(label='Expedite Review')
    ma, aa = Transition(label='Manager Approval'), Transition(label='Auto Approval')
    rel = Transition(label='Release Order')

    pass1 = StrictPartialOrder(nodes=[cc, tr])                    # P2: both, any order
    recheck = OperatorPOWL(operator=Operator.XOR, children=[
        Transition(label='Credit Check'), # correction affected CC only
        Transition(label='Technical Review'),                     # ... TR only
        StrictPartialOrder(nodes=[Transition(label='Credit Check'),
                                  Transition(label='Technical Review')])])   # ... both
    round_ = StrictPartialOrder(nodes=[cd, recheck])
    round_.order.add_edge(cd, recheck)                            # P7: repeats only after a correction
    corrections = OperatorPOWL(operator=Operator.LOOP,
                               children=[SilentTransition(), round_])   # (CD ; recheck)*

    er_opt = OperatorPOWL(operator=Operator.XOR, children=[er, SilentTransition()])   # P4 optional
    sr_node = supplier_review_node(case_type)
    approval = OperatorPOWL(operator=Operator.XOR, children=[ma, aa])                 # P5

    root = StrictPartialOrder(nodes=[reg, sr_node, pass1, corrections, er_opt, approval, rel])
    o = root.order
    o.add_edge(reg, sr_node); o.add_edge(reg, pass1); o.add_edge(reg, er_opt)
    o.add_edge(pass1, corrections)                                 # rework inside the check region
    o.add_edge(pass1, er_opt); o.add_edge(corrections, er_opt)     # P4: ER after the checks
    o.add_edge(sr_node, approval)                                  # SR before approval; SR vs ER free
    o.add_edge(pass1, approval); o.add_edge(corrections, approval); o.add_edge(er_opt, approval)
    o.add_edge(approval, rel)
    return root


model_third = build_model_third('unknown')
model_third_external = build_model_third('external')
model_third_internal = build_model_third('internal')

```python
CHECKS = ('Credit Check', 'Technical Review')
APPROVALS = ('Manager Approval', 'Auto Approval')
KNOWN = {'Register Dossier', 'Credit Check', 'Technical Review', 'Correct Dossier',
         'Supplier Review', 'Expedite Review', 'Manager Approval', 'Auto Approval',
         'Release Order'}


def policy_check(trace, attrs=None):
    """Authoritative 4-valued policy verdict for one trace.

    CONFORMING        : all clauses provable from the trace (+attrs)
    NONCONFORMING     : at least one positively observed violation
    SUSPECT-DATA      : the only problem is *absent* mandatory events AND an
                        ingestion defect is evidenced (attrs['ingestion_defect_ticket'])
    UNDECIDABLE-DATA  : structurally plausible, but a clause needs an attribute
                        that the log does not carry (is_external / affected_checks)
    """
    attrs = dict(attrs or {})
    out = {}
    k = len(trace)
    appr = [i for i, a in enumerate(trace) if a in APPROVALS]
    if len(appr) == 1:
        k = appr[0] # index of the approval

    out['P0'] = 'pass' if all(a in KNOWN for a in trace) else 'fail'
    out['P1'] = 'pass' if len(trace) >= 2 and trace[0] == 'Register Dossier' and trace[-1] == 'Release Order' else 'fail'
    out['P5'] = 'pass' if (len(appr) == 1 and trace[-1] == 'Release Order') else 'fail'

    # P2 -- mandatory checks, before the approval
    missing = [c for c in CHECKS if not any(a == c for a in trace[:k])]

def supplier_review_node(case_type='unknown'):
    if case_type == 'external':
        return Transition(label=SR)                       # P3: mandatory
    if case_type == 'internal':
        return SilentTransition()                         # P3: must not occur
    return OperatorPOWL(operator=Operator.XOR,
                        children=[Transition(label=SR), SilentTransition()])


def build_model_third(case_type='unknown'):
    reg, cd, rel = Transition(label=REG), Transition(label=CD), Transition(label=LO)

    # ---- pass 1: BOTH mandatory checks, order free (P2) --------------
    pass1 = StrictPartialOrder(nodes=[Transition(label=CC), Transition(label=TR)])

    # ---- one correction round: Correct Dossier ; exactly the affected checks
    recheck = OperatorPOWL(operator=Operator.XOR, children=[

Neither model is scored into a single number (see §2, last paragraph), because a weighted mean would let a convenience score (comprehension, runtime) cancel a safety failure (an unreviewed release is admitted at zero cost).

## 1.2 Use cases distinguished

| Use case | What "good" means | Acceptable fitness/precision trade-off | Primary failure mode |
|---|---|---|---|
| **Exploration / discovery** | recall of observed and plausible behaviour; hypothesis generation | Low precision acceptable (F7 is legitimate *here*); fitness should be high but is not decisive | Presenting an exploratory envelope as if it were the policy |
| **Conformance enforcement (compliance)** | the model must **not admit** forbidden behaviour; deviations must be *diagnosed*, not silently absorbed | Precision against a **negative** suite must be maximal; positive fitness may be sacrificed if the sacrifice is an explicit, explained deviation | Silent admission of forbidden traces (B3, B4); false accusation of noncompliance on legitimate new combinations (F2) |
| **Simulation / forecasting** | generated behaviour must be both *possible* and *permitted* | Needs constrained generalization: a model that generates everything is useless; a model that generates 8 variants cannot simulate rework or new combinations | Garbage scenarios; volume forecasts distorted by unconstrained loops |
| **Stakeholder communication** | users answer policy questions about the model **correctly**, not just quickly | Comprehension accuracy matters more than preference | A comprehensible but wrong diagram that "explains" a control that is not actually enforced |
| **Audit evidence** | every claim traceable to a case, a rule and a reproducible computation | Reproducibility outranks all four numeric dimensions | Undocumented metrics ("33"), undocumented splits, positive-only evaluation (F5) |

Concretely: **Model S's profile (fitness  0.99, generalization 0.91, precision 0.58) is the correct profile for row 1 and the disqualifying profile for rows 2–3.** **Model C's profile (precision 0.94 on training, generalization 0.41, holdout 0.74) is the profile of a *variant catalogue*: defensible as evidence of *observed practice*, indefensible as a normative model.** The trade-off is therefore not a tuning parameter; it is chosen by the use case.

## 1.3 Assumptions (stated so they can be attacked)

- **A1 (notation).** `->` = sequence, `+` = parallel/interleaving (each branch exactly once), `X` = exclusive choice (exactly one child), `tau` = silent/skip, therefore `X(a, tau)` = *optional* `a`, and `L(A,B)` = `A (B A)*`. These are the standard PM4Py semantics; if the teams used different semantics the numbers are not comparable at all.
- **A2 (authority).** P1–P7 are the specification. The B1–B8 probes are **test cases, not ground truth** (as instructed) and where they are marked "policy-dependent" (B6) I refuse to resolve the dependency without data.
- **A3 (data gaps).** `external/internal` and `affected check` are genuinely absent from the log; F4 states some `Supplier Review` events may be missing. Both facts are treated as *hard limits on what any model can certify*, not as nuisances to be optionalized.
- **A4 (models as published).** Model C's tree contains exactly the 8 branches printed; Model S's parallel block contains exactly the 4 optional activities printed. Metrics were supposedly computed on these artefacts — this is unverifiable (no code, no seeds).
- **A5 (costs).** Where I derive alignment costs by hand I use the standard cost model sync = 0, log move = 1, model move = 1, and I label the numbers as *derived*, to be confirmed by the harness in §9. **No new metric results are reported anywhere in this document.**
- **A6 (no aggregation).** I do not average the eight measures.

# 3. Adversarial Review of Model S

## 3.1 The strongest case **for** Model S

Model S is the only artefact in the dossier that gets the *shape* of the policy right at a glance and it satisfies two clauses exactly: **P1** (Register first, Release last) and **P5** (`X('Manager Approval','Auto Approval')` is literally "exactly one approval"). Its parallel block correctly expresses that Credit Check and Technical Review have **no** required relative order (P2), which Model C freezes into two hand-written orders. It is the only model that admits the F4 holdout shapes (Supplier Review + Auto Approval + alternate check order) without modification, so it is the only model that will not generate false accusations of noncompliance against legitimate new combinations. Its runtime (0.018 s/case) and comprehension (4.4) make it the only candidate for interactive what-if and for onboarding diagrams. Under F7's framing — an exploration artefact, never an enforcement artefact — most of the objections below are simply out of scope, and "robust to missing events" is a genuine virtue when the log is known to be lossy (F4). In that reading S is not a process model at all; it is a **behaviour envelope**, and an envelope should be permissive.

## 3.2 The strongest case **against** Model S

The permissiveness of Model S is **not** noise tolerance. Noise tolerance is a mechanism that *absorbs* log defects while preserving the control structure and **raising a flag**. Model S has no deviation channel at all: a case in which Technical Review never happened replays at **cost 0**, i.e. it is numerically indistinguishable from a perfect case. The claim "robust to missing events" is therefore a euphemism for "cannot see missing events", and F1 is right. The parallel block also destroys a constraint that *is* stated in the policy (P4), and `Correct Dossier` is absent from the model entirely (P6/P7 unrepresentable). Model S is thus simultaneously too permissive (controls) and too restrictive (rework) — the two failures come from the same act of semantic deletion.

### Undesirable or policy-ambiguous behaviours admitted at zero cost

| # | Trace (abbreviated) | Why it is bad | Alignment cost in S (derived) |
|---|---|---|---|
| U1 (=B3) | `Register Dossier  Manager Approval  Release Order` | **Both** mandatory checks skipped; an unreviewed dossier is released | **0** (accepted) |
| U2 | `Register Dossier  Credit Check  Auto Approval  Release Order` | Technical Review skipped **and** auto-approved — no human ever saw it | **0** |
| U3 | `Register Dossier  Expedite Review  Auto Approval  Release Order` | No checks at all; Expedite Review precedes the checks it is supposed to follow (P4) | **0** |
| U4 (=B4) | `Register Dossier  Expedite Review  Credit Check  Technical Review  Manager Approval  Release Order` | P4 violation: Expedite Review before both mandatory checks | **0** |
| U5 | `Register Dossier  Supplier Review  Expedite Review  Manager Approval  Release Order` | Zero checks, superfluous Supplier Review (P3) on what may be an internal order | **0** |
| U6 | `Register Dossier  Technical Review  Supplier Review  Manager Approval  Release Order` | Credit Check missing; if the order is external the Supplier Review is present but the credit control is gone | **0** |
| U7 (ambiguous) | `Register Dossier  Supplier Review  Credit Check  Technical Review  Manager Approval  Release Order` | Supplier Review before the checks. P3 gives no position constraint, so this is **policy-ambiguous** — but S admits it *silently*, so a reviewer cannot tell permissive-by-design from permissive-by-accident | **0** |
| U8 (=A6) | `Register Dossier  Credit Check  Supplier Review  Manager Approval  Release Order` | Technical Review absent — the classic ingestion-defect shape (F4). S accepts it as a **perfect** case | **0** |

### The blind spot on the other side

Because `Correct Dossier` and any loop are missing, S **rejects all legitimate rework**: B6 (`… Correct Dossier  Technical Review  …`) costs **2** (derived), and A3/A4 (single-check recheck, two correction rounds) cost 2 and 5. So S is not "robust"; it is blind in one direction and soft in the other.

### Intentional tolerance vs accidental flaw

- **Intentional (defensible) tolerance** would look like: same backbone as `model_third`, plus alignments that charge logomotives on mandatory labels and emit `SUSPECT-DATA`. The *language* stays strict; only the *verdict* is softened, and only with evidence.
- **Accidental semantic flaws** in S: (i) `X('Credit Check',tau)` and `X('Technical Review',tau)` make P2 optional — this is a semantic hole, not tolerance; (ii) the parallel block leaves `Expedite Review` unconstrained — a stated policy constraint simply dropped; (iii) `Correct Dossier` missing — a whole clause of the policy (P6, P7) invisible to the model; (iv) no rework loop — legitimate behaviour rejected; (v) `Supplier Review` optional and positionless with no data hook for P3 — an undecidable clause presented as if it were handled.

# 4. Adversarial Review of Model C

## 4.1 The strongest case **for** Model C

Model C is *honest about what has been observed*. Every branch can be tied to concrete cases, which is exactly what an auditor asks for ("show me where this behaviour occurs"). It satisfies P1, P2 (both checks, both orders), P4's *ordering* (Expedite Review after the checks, before approval), P5 (exactly one approval per branch), and — within its eight branches — P7 (the only repeated checks follow a `Correct Dossier`). Its errors are overwhelmingly **false negatives**, i.e. the conservative direction: it refuses to certify behaviour it has not seen. That is precisely F8's posture, and for a compliance artefact a refusal is a *safe* failure — the case goes to a human, nothing is released wrongly. Alignment precision 0.941 on the training log means it invents almost nothing. At 0.44 s/case it is fast enough for offline audit batches. Its low simplicity score is a *stylistic* complaint, not a correctness one.

## 4.2 The strongest case **against** Model C

**(a) It is a variant table with a process-tree costume.** The policy space for one correction round is: check order (2) × Supplier Review present (2) × Expedite Review present (2) × Supplier/Expedite relative order (2, when both present) × approval type (2) × correction pattern (4)  **80 policy-plausible combinations** (the exact generator is in §9). Model C encodes **8** of them — about 10% — and **0%** of the multi-round correction space. Every holdout case that recombines *known* dimensions in a *new* way is classified as noncompliant. B2 (a valid case: either check order × Auto Approval) is rejected. A1 and A2 (below) are rejected. This is not conservatism; it is a **false-accusation machine** with an appeal load attached.

**(b) High training precision is probably a memorization artefact — and it is not even reproducible.** A model whose language is exactly the observed variant set should score  1.0 alignment precision against a log containing those variants. The reported 0.941 implies either that at least one of the eight branches is *not* covered by the training split (so the model contains unobserved behaviour — and the 80/20 split then decides the number), or that hidden transitions in the 8-way XOR distort the computation. Either way, 0.941 is a property of the split and the metric implementation, not of the model.

**(c) Holdout degradation is structural, not accidental.** 0.973  0.744 (0.229) is exactly what a memorizing model does when the test set contains recombined behaviour. F2 and F4 independently confirm it. Worse: some holdout cases contain *missing* `Supplier Review` events (F4), and the alignment will charge those ingestion defects to the model, so 0.744 mixes two very different causes.

**(d) Correction handling contradicts the policy it claims to encode.** Branch 7 hard-codes *one* correction round with **both** checks repeating. P6 says only the **affected** checks repeat; F6 says the affected set can be `{Credit Check}`, `{Technical Review}` or both. Consequences:
- B6/A3 (correct single-check recheck) are **rejected** — the model punishes the *policy-correct* behaviour;
- multi-round corrections (A4) are rejected;
- the derived alignment cost of B6 is **1** in C versus **2** in S: **the "precise" model fits the policy-ideal correction trace better than the "robust" one.** Fitness ranking and policy correctness do not even agree in sign here;
- the cost-1 alignment for A3 explains a *correct* recheck as "a missing Technical Review" (a model move). The diagnosis is actively misleading.

**(e) Duplicate labels.** 48 visible occurrences over 9 labels (39 redundant occurrences under the standard definition; the reported 33 is unexplained). Three consequences: ownership and impact analysis is impossible (F3 — which team owns the `Credit Check` that lives in branch 7 position 2 versus branch 7 position 5?); token-based generalization and simplicity are distorted by multiplicities; and a policy change must be applied in 8–9 places with no named sub-structure to refactor. Adding one binary policy dimension (e.g. "Regional Review for non-domestic orders") **doubles** the tree.

**(f) Branch enumeration scales multiplicatively and freezes accidental detail.** P4 explicitly allows Expedite Review *before or after* Supplier Review. C encodes only `Supplier Review  Expedite Review` (branch 6): it turns a policy *permission* into a policy *constraint*. The same happens with the check order (only two of the possible orders of the whole middle section are enumerated).

**(g) Unstable diagnoses from alignment ties.** For B4 (`… Expedite Review  Credit Check  …`) at least two cost-2 explanations exist: "Expedite Review is misplaced" (branch 5) and "Expedite Review is spurious **and** Supplier Review is missing" (branch 3). For B8 the minimal-cost explanation is "an extra Credit Check" (cost 1 via branch 1), while branch 7 would explain it as "you forgot to log `Correct Dossier` **and** one `Technical Review`" (cost 2). Which story a reviewer sees is a tie-breaking artefact of the aligner. **Minimal alignment cost is not a policy diagnosis.**

**(h) Maintainability and "audit-ready".** An audit-ready model must be *reviewable against the rules*. C contains no element that corresponds to "Supplier Review is required for external orders" (the attribute does not even exist) and no element corresponding to "only the affected checks repeat". It is auditable as a *record of practice*, not as an *implementation of policy*.

# 7. Third Alternative

## 7.1 Strategy chosen

**`model_third` = block-structured/partial-order procedural backbone (POWL) + explicit data constraints + an authoritative 4-valued policy checker.**

Rationale: the two inputs fail in opposite directions because each tries to encode everything in the control-flow language. The policy P1–P7 splits cleanly into (i) **ordering/structure** facts the trace alone can decide, and (ii) **data** facts (`is_external`, `affected_check`) that no control-flow model can decide (P3, P6). The third alternative therefore:

- encodes (i) **strictly** in the model — mandatory checks stay mandatory everywhere, no silent optionalization;
- encodes (ii) as **explicit tri-state data constraints** with *withheld certification* — never as `X(a, tau)`;
- adds a **deterministic checker** that is authoritative over alignment costs, because §5/§6 show that minimal-cost alignments misdiagnose both forbidden (B8) and permitted (A3) behaviour.

Formally: `model_third = (T_struct, D, check)` where `T_struct` is the POWL model below over  = {Register Dossier, Credit Check, Technical Review, Correct Dossier, Supplier Review, Expedite Review, Manager Approval, Auto Approval, Release Order}, `D = {D1: Supplier Review  is_external, D2: recheck set = affected_check set}` and `check : Trace × Attrs  {CONFORMING, NONCONFORMING, SUSPECT-DATA, UNDECIDABLE-DATA}`.

**Language of `T_struct`** (all statements are about visible events):

1. `Register Dossier` first, `Release Order` last; exactly one of `Manager Approval` / `Auto Approval` before release (P1, P5).
2. **Pass 1** = `Credit Check` and `Technical Review`, each exactly once, **in either order**, strictly before the approval (P2). They are mandatory in *every* model variant.
3. **Correction region** = `('Correct Dossier' ; recheck)*`, where `recheck` is *exactly one of* `{Credit Check}`, `{Technical Review}`, `{Credit Check, Technical Review}` (each non-empty, order free). Hence: zero or more correction rounds; every repetition of a check is preceded by a `Correct Dossier` after that check's previous occurrence (P7 by construction); after a correction **exactly a non-empty subset** of checks repeats (P6 by construction).
4. `Expedite Review` optional, strictly after pass 1 (i.e. after both mandatory checks) and strictly before the approval (P4); it is **unordered with respect to `Supplier Review` and with respect to correction rounds**, which is the literal reading of P4 ("may occur before or after Supplier Review").
5. `Supplier Review` at most once, after `Register Dossier`, before the approval, unordered with respect to everything else — because P3 constrains its *presence*, not its *position*.

# =====================================================================
import pm4py
from pm4py.objects.powl.obj import (StrictPartialOrder, OperatorPOWL,
                                    Transition, SilentTransition)
from pm4py.objects.powl.operator import Operator

REG, CC, TR, CD = 'Register Dossier', 'Credit Check', 'Technical Review', 'Correct Dossier'
SR, ER, MA, AA, LO = 'Supplier Review', 'Expedite Review', 'Manager Approval', 'Auto Approval', 'Release Order'


def supplier_review_node(case_type='unknown'):
    """P3 is a DATA rule: presence of Supplier Review depends on is_external.
       'unknown' must NOT be resolved by silently making anything else optional."""
    if case_type == 'external':
        return Transition(label=SR)                     # mandatory for external orders
    if case_type == 'internal':
        return SilentTransition()                       # must not occur for internal orders
    return OperatorPOWL(operator=Operator.XOR,
                        children=[Transition(label=SR), SilentTransition()])


def build_model_third(case_type='unknown'):
    reg, cd, rel = Transition(label=REG), Transition(label=CD), Transition(label=LO)

    # --- pass 1: BOTH mandatory checks, order free (P2) ---------------
    pass1 = StrictPartialOrder(nodes=[Transition(label=CC), Transition(label=TR)])

    # --- one correction round: Correct Dossier ; the affected checks ---
    recheck = OperatorPOWL(operator=Operator.XOR, children=[
        Transition(label=CC),                                        # affected = {CC}
        Transition(label=TR),                                        # affected = {TR}
        StrictPartialOrder(nodes=[Transition(label=CC), Transition(label=TR)])])  # affected = {CC,TR}
    round_ = StrictPartialOrder(nodes=[cd, recheck])
    round_.order.add_edge(cd, recheck)                               # P7: repeats only after a correction
    # PM4Py LOOP(children=[A, B]) denotes A (B A)* ; with A = tau this is exactly ('Correct Dossier' ; recheck)*
    corrections = OperatorPOWL(operator=Operator.LOOP,
                               children=[SilentTransition(), round_])

    er_opt = OperatorPOWL(operator=Operator.XOR,
                          children=[Transition(label=ER), SilentTransition()])   # P4: optional
    approval = OperatorPOWL(operator=Operator.XOR,
                            children=[Transition(label=MA), Transition(label=AA)])  # P5
    sr_node = supplier_review_node(case_type)

    root = StrictPartialOrder(nodes=[reg, sr_node, pass1, corrections, er_opt, approval, rel])
    root.order.add_edge(reg, sr_node)          # SR after registration
    root.order.add_edge(reg, pass1)            # P1
    root.order.add_edge(pass1, corrections)    # rework lives inside the check region
    root.order.add_edge(pass1, er_opt)         # P4: ER follows BOTH mandatory checks
    root.order.add_edge(corrections, approval) # no control work after the approval decision
    root.order.add_edge(er_opt, approval)      # P4: ER precedes approval
    root.order.add_edge(sr_node, approval)     # SR before approval (position otherwise free)
    root.order.add_edge(approval, rel)         # P1
    return root


model_third          = build_model_third('unknown')    # <-- main model
model_third_external = build_model_third('external')   # attribute-conditioned variant
model_third_internal = build_model_third('internal')   # attribute-conditioned variant

# quick structural sanity check (results must be recorded in the evaluation report)
net, im, fm = pm4py.convert_to_petri_net(model_third)
if __name__ == '__main__':
    print(pm4py.check_soundness(net, im, fm, print_diagnostics=True))
```

*Note on API portability:* if the installed PM4Py build names the POWL operators differently, only the `Operator.XOR` / `Operator.LOOP` enum members need adapting; the semantics are stated in §7.1. A block-structured process-tree equivalent exists (`->(Register, X(Supplier Review, tau), L(+(CC,TR), ->(Correct Dossier, X(...))), X(Expedite Review, tau), X(MA, AA), Release)`) and must produce the same language — a cross-check is included in §9 (G3).

## 7.3 Declarative dictionary and external checks

```python
# --- declarative / data layer -----------------------------------------
DECLARE_DICT = {
    # clause: (scope, rule text, enforcement point, decidable from the trace alone?)
    'P1': ('trace',  'Register Dossier first, Release Order last',                    'structure(T)', True),
    'P2': ('trace',  'Credit Check AND Technical Review, either order, before approval','structure(T)', True),
    'P3': ('case',   'Supplier Review present IFF is_external == True',               'data D1',      False),
    'P4': ('trace',  'Expedite Review optional, after both checks, before approval, '
                     'free w.r.t. Supplier Review',                                   'structure(T)', True),
    'P5': ('trace',  'exactly one of Manager/Auto Approval before release',           'structure(T)', True),
    'P6': ('round',  'after Correct Dossier, exactly the AFFECTED checks repeat '
                     '(non-empty subset, either order)',                              'structure(T)+data D2', False),
    'P7': ('round',  'a repeated check requires an intervening Correct Dossier',      'structure(T)', True),
}

CHECKS, APPROVALS = ('Credit Check', 'Technical Review'), ('Manager Approval', 'Auto Approval')
KNOWN = {REG, CC, TR, CD, SR, ER, MA, AA, LO}


def policy_check(trace, attrs=None):
    """Authoritative 4-valued verdict. The checker overrides alignment-cost stories:
    CONFORMING       all clauses provable from trace (+attrs)
    NONCONFORMING    at least one positively observed violation (P0/P1/P3/P4/P5/P7, or P2/P6)
    SUSPECT-DATA     only absent mandatory events AND an evidenced ingestion defect
    UNDECIDABLE-DATA structurally plausible, but P3/P6 needs an attribute the log lacks
    """
    attrs = dict(attrs or {})
    out = {}
    appr = [i for i, a in enumerate(trace) if a in APPROVALS]
    k = appr[0] if len(appr) == 1 else len(trace)          # index of the approval

    out['P0'] = 'pass' if all(a in KNOWN for a in trace) else 'fail'
    out['P1'] = 'pass' if (len(trace) >= 2 and trace[0] == REG and trace[-1] == LO) else 'fail'
    out['P5'] = 'pass' if (len(appr) == 1 and trace[-1] == LO) else 'fail'

    # P2 -- mandatory checks before the approval
    missing = [c for c in CHECKS if not any(a == c for a in trace[:k])]
    out['P2'] = 'fail' if missing else 'pass'
    out['_missing_mandatory'] = missing

    # P7 -- every non-first occurrence of a check needs a Correct Dossier since its previous one
    last, hits = {c: None for c in CHECKS}, []
    for i, a in enumerate(trace[:k]):
        if a in CHECKS:
            if last[a] is not None and CD not in trace[last[a] + 1:i]:
                hits.append({'check': a, 'repeat_index': i, 'previous_index': last[a]})
            last[a] = i
    out['P7'] = 'pass' if not hits else 'fail'
    out['_P7_hits'] = hits

    # P6 -- each correction followed by a non-empty recheck set == affected set (data D2)
    corr = [i for i, a in enumerate(trace[:k]) if a == CD]
    cuts = [0] + [i + 1 for i in corr] + [k]
    segs = [[a for a in trace[lo:hi] if a in CHECKS] for lo, hi in zip(cuts[:-1], cuts[1:])]
    rounds = [set(s) for s in segs[1:]]                    # segs[0] is pass 1
    out['_recheck_sets'] = [sorted(r) for r in rounds]
    if not corr:
        out['P6'] = 'pass'
    elif any(len(r) == 0 for r in rounds):
        out['P6'] = 'fail'                                 # correction with no recheck at all
    elif 'affected_checks' not in attrs:
        out['P6'] = 'undecidable'                          # attribute missing -> cannot certify
    else:
        want = attrs['affected_checks']
        want = [set(want)] * len(rounds) if isinstance(want, (set, frozenset)) else [set(w) for w in want]
        out['P6'] = 'pass' if want == rounds else 'fail'

    # P3 -- data D1
    sr = [i for i, a in enumerate(trace) if a == SR]
    ext = attrs.get('is_external', None)
    if ext is None:
        out['P3'] = 'undecidable'
    elif ext:
        out['P3'] = 'pass' if (len(sr) == 1 and sr[0] < k) else 'fail'
    else:
        out['P3'] = 'fail' if sr else 'pass'
    out['_P3_note'] = 'is_external absent: P3 cannot be certified' if ext is None else ''

    # P4 -- Expedite Review optional, after both checks, before approval
    er = [i for i, a in enumerate(trace) if a == ER]
    if not er:
        out['P4'] = 'pass'
    else:
        out['P4'] = 'pass' if (len(er) == 1 and er[0] < k and
                               all(trace.index(c) < er[0] for c in CHECKS if c in trace)) else 'fail'
    out['_er_inside_round'] = any(c < e < r for c in corr for e in er
                                  for r in ([min([i for i in range(c + 1, k) if trace[i] in CHECKS] or [k])]))

    clauses = ['P0', 'P1', 'P2', 'P3', 'P4', 'P5', 'P6', 'P7']
    failed = [c for c in clauses if out[c] == 'fail']
    undec = [c for c in clauses if out[c] == 'undecidable']
    positive = [c for c in failed if c != 'P2']            # violations we can actually OBSERVE
    if positive:
        verdict = 'NONCONFORMING'
    elif failed:                                           # only absent mandatory events
        verdict = 'SUSPECT-DATA' if attrs.get('ingestion_defect_ticket') else 'NONCONFORMING'
    elif undec:
        verdict = 'UNDECIDABLE-DATA'
    else:
        verdict = 'CONFORMING'
    out['verdict'], out['failed_clauses'], out['undecidable_clauses'] = verdict, failed, undec
    return out
```

## 7.4 Explicit semantics for the five required points

| Requirement | Where it is enforced | Exact semantics of `model_third` |
|---|---|---|
| **Mandatory checks (P2)** | `pass1 = StrictPartialOrder([CC, TR])` with **no** edge | Both occur exactly once, in **either** order, before the approval. They are mandatory in all three model variants. The **only** place a check may be skipped is a recheck round, which is exactly what P6 mandates. There is no `X(CC, tau)` anywhere in the model — deliberate, per the instruction not to solve missing data by optionalizing activities |
| **Optional Supplier Review (P3)** | `supplier_review_node(case_type)` + data rule D1 | Tri-state, not optional-by-default: `external`  mandatory; `internal`  excluded; `unknown`  `X(SR, tau)` for *language* queries only, with the verdict `UNDECIDABLE-DATA` and **certification withheld**. Absence of `is_external` is never resolved by assuming "optional" |
| **Expedite Review ordering (P4)** | `er_opt = X(ER, tau)` with edges `pass1  er_opt  approval` | Optional; after **both** mandatory checks (pass 1); before the approval; **unordered** with respect to `Supplier Review` (both orders admitted) and with respect to correction rounds. A trace placing `Expedite Review` *between* a `Correct Dossier` and its rechecks is admitted but flagged (`_er_inside_round`) for a governance ruling |
| **Exactly-one approval (P5)** | `approval = X(MA, AA)` before `Release Order` | Exactly one of the two approvals, once, before release |
| **Correction / recheck (P6, P7)** | `corrections = LOOP(tau, (CD ; recheck))`, `recheck = X(CC, TR, PO{CC,TR})` | Zero or more rounds; each round is `Correct Dossier` followed by **exactly one** non-empty subset of checks in either order. P7 holds by construction (a repeat can only occur inside a round, and every round starts with `Correct Dossier`). P6's *identity* of the affected set is checked by D2 and is **undecidable** while `affected_checks` is missing. A `Correct Dossier` with no recheck at all is rejected (assumption: a correction must affect at least one check; the alternative reading is a one-line variant) |

## 7.5 Limitations caused by the missing attributes

1. **P3 is undecidable per case.** Until `is_external` is logged, no model can be certified as P3-conformant. `model_third` therefore (a) keeps the superset language so that legitimate external cases (B5, B7, A1, A2) are not rejected, and (b) withholds `CONFORMING` for every trace where P3 is unresolved. Compliance sign-off must be worded "*structurally conformant, P3 unverified*". A **proxy** (e.g. supplier identifier present in the case record) may be used only if it is labelled a proxy and its error rate is measured on the hand-labelled gold set (§9).
2. **P6 is undecidable per correction round.** Any trace containing `Correct Dossier` yields `UNDECIDABLE-DATA` unless `affected_checks` is annotated. Operationally this means **no corrected case may be auto-released**; it goes to a human who confirms the recheck set. This is intentional: F6's requirement cannot be checked from a control-flow log and pretending otherwise is exactly the Model S error.
3. **Log completeness is assumed for P2.** A missing `Technical Review` is indistinguishable from a skipped one without an ingestion audit. `model_third` refuses to accept such a case (verdict `NONCONFORMING`), but will reclassify to `SUSPECT-DATA` **only** on evidence (`ingestion_defect_ticket`). It never classifies it `CONFORMING`.
4. **No time, resource or data dimensions.** "Expedite" is modelled as an ordering constraint only; `model_third` cannot say whether a case *should* have been expedited (that needs a service-level attribute), and the correction loop is unbounded (no liveness/ageing constraint) — see the dissent in §11.
5. **Backfill requirement.** Both attributes must be added to the logging standard *with* a backfill plan for historical cases; until then, evaluation of P3/P6 error rates relies on the hand-labelled gold set and is therefore sample-limited.

# 8. Comparative Behavior Audit

**Own adversarial traces** (designed to separate the three artefacts):

- **A1** `Register Dossier  Credit Check  Technical Review  Expedite Review  Supplier Review  Manager Approval  Release Order` — Expedite Review **before** Supplier Review, which P4 explicitly permits.
- **A2** `Register Dossier  Technical Review  Credit Check  Supplier Review  Auto Approval  Release Order` — external + Auto Approval + reversed check order (the F4 holdout shape).
- **A3** `Register Dossier  Credit Check  Technical Review  Correct Dossier  Credit Check  Manager Approval  Release Order` — correction affecting **Credit Check only** (F6).
- **A4** `Register Dossier  Credit Check  Technical Review  Correct Dossier  Credit Check  Technical Review  Correct Dossier  Technical Review  Manager Approval  Release Order` — **two** correction rounds with different affected sets.
- **A5** `Register Dossier  Credit Check  Technical Review  Expedite Review  Correct Dossier  Credit Check  Manager Approval  Release Order` — correction **after** an Expedite Review (a region the policy does not speak about).
- **A6** `Register Dossier  Credit Check  Supplier Review  Manager Approval  Release Order` — `Technical Review` absent (ingestion-defect shape).
- **A7** `Register Dossier  Credit Check  Technical Review  Correct Dossier  Technical Review  Technical Review  Manager Approval  Release Order` — a duplicated recheck inside one round (P7 in the second round).

Abbreviations: `R, CC, TR, CD, SR, ER, MA, AA, RO` as in §5.

| Behavior or requirement | Model S | Model C | Third alternative | Best-supported interpretation |
|---|---|---|---|---|
| **P1** bookends | Satisfied | Satisfied | Satisfied | Uncontested |
| **P2** both checks mandatory, either order, before approval | **Violated** (checks optional: U1–U6 at cost 0) | Satisfied within its 8 branches | Satisfied (pass 1 mandatory in every variant) | A check is a *control*, so it must be unskippable; order must stay free (P2 says so) |
| **P3** Supplier Review iff external | **Violated** (optional and positionless; no data hook) | **Violated** (present in 3 of 8 branches as a frequency proxy) | Enforced by data rule D1; **certification withheld** while `is_external` is missing | The clause is **undecidable** from the control-flow log; any model claiming to enforce it is over-claiming |
| **P4** Expedite Review after both checks, before approval, free w.r.t. Supplier Review | **Violated** (ordering unconstrained: B4/U3/U4) | Ordering satisfied, but the permission "before *or after* Supplier Review" is frozen to one order (branch 6) | Satisfied, both SR/ER orders admitted | S drops a stated constraint; C converts a permission into a constraint |
| **P5** exactly one approval | Satisfied | Satisfied | Satisfied | Uncontested |
| **P6** only affected checks repeat | **Not representable** (no `Correct Dossier`, no loop) | **Violated** (branch 7 hard-codes *both* checks, one round only) | Structurally enforced for all three subsets; identity of the subset checked by D2  `UNDECIDABLE-DATA` without the attribute | The rule is data-dependent; the honest output is "plausible, unverified", not "compliant" |
| **P7** repeat without correction is suspicious | Vacuously "satisfied" (rework unrepresentable) — wrong reason | Satisfied in the 8 branches; but its cheapest explanation of B8 is "an extra Credit Check" and its branch-7 story is "you forgot to log `Correct Dossier`" | Enforced by construction **and** named explicitly by the P7 checker (`repeat_index`) | Correct rejection must come with a correct *diagnosis*; B8 shows both inputs fail that test |
| **B1** happy path | Accept (0) | Accept (0) | Accept (0) | Valid; non-discriminating |
| **B2** valid cross-product | Accept (0) | **Reject (2)** | Accept (0) | Valid; C's rejection is a false accusation |
| **B3** no checks | **Accept (0)** | Reject (2) | Reject (2)  `NONCONFORMING (P2)` | Forbidden; S's acceptance is disqualifying |
| **B4** Expedite first | **Accept (0)** | Reject (2), two contradictory cost-2 stories | Reject (1) + named P4 defect | Forbidden; S structurally wrong, C diagnostically unstable |
| **B5** external/urgent, Auto Approval | Accept (0) | **Reject (3)** | Accept (0) | Plausible valid; C invents three violations |
| **B6** TR-only recheck | Reject (2) | Reject (1: "missing Credit Check") | **Accept (0)**; P6 `UNDECIDABLE-DATA` | Policy-dependent; only meaningful with `affected_check` |
| **B7** external, Manager Approval | Accept (0) | Accept (0) | Accept (0) | Valid external case |
| **B8** repeat without correction | Reject (1), unrepresentable rework | Reject (1), misleading story | Reject (1) + **explicit P7 diagnosis** (tie in cost resolved by the checker) | Forbidden/suspicious; rejection alone is not evidence of P7 semantics |
| **A1** Expedite before Supplier | Accept (0) | **Reject (2)** | Accept (0) | Valid (P4 permits it); only the third model honours the permission |
| **A2** external + Auto Approval + reversed checks | Accept (0) | **Reject (2)** | Accept (0) | Valid; the exact F4 holdout shape |
| **A3** CC-only recheck | Reject (2) | Reject (1: "missing Technical Review") | **Accept (0)** | Valid iff `affected_check = {Credit Check}`; C's diagnosis is inverted |
| **A4** two correction rounds | Reject (5) | Reject (2) | **Accept (0)** | Valid under P6 with per-round affected sets; neither input can express it |
| **A5** correction after Expedite Review | Reject (2) | Reject (2) | **Accept (0)** + `_er_inside_round = False`; P6 `UNDECIDABLE-DATA` | **Policy-silent.** Needs a ruling on whether a correction invalidates an already-performed Expedite Review; neither input notices the gap |
| **A6** missing Technical Review | **Accept (0)** | Reject | Reject (1)  `NONCONFORMING (P2)`, reclassifiable to `SUSPECT-DATA` only with an ingestion-defect ticket | Data-quality shape; must be blocked and triaged, never accepted |
| **A7** duplicated recheck in one round | Reject (unrepresentable) | Reject (branch 7 tolerates only one) | Reject (1) + **explicit P7 diagnosis** (`repeat_index` in round 1) | Forbidden under P7 generalised to rounds |

**Reading of the table.** The third alternative is the only artefact that (i) rejects everything the policy forbids among these 22 rows *with a named clause*, (ii) accepts everything the policy permits among these rows, and (iii) returns `UNDECIDABLE-DATA` — instead of a confident wrong answer — on the two clauses the log cannot decide.

## 9.1 Environment pinning and reproducible splitting

```python
# requirements.txt:  pm4py==<PINNED BUILD>   numpy==...   pandas==...
# record: python version, OS, CPU, thread settings, and the resolved PM4Py commit
import json, random
import numpy as np, pm4py
from pm4py.objects.log.obj import EventLog, Trace, Event

PIN = {"pm4py": "<pinned>", "python": "<pinned>", "seed": 20240517}
SPLIT_SEEDS = [11, 23, 37, 53, 71]          # 5 independent 80/20 splits, never one lucky split
CHECKS = ('Credit Check', 'Technical Review')

def preprocess(log):
    """Deterministic, PUBLISHED preprocessing. Every rule and every count goes to preprocess.json."""
    steps = []
    log = pm4py.filter_case_size(log, 2, 60)                 # rule + retained/removed counts
    steps.append(("filter_case_size", 2, 60, len(log)))
    json.dump({"pin": PIN, "steps": steps}, open("preprocess.json", "w"), indent=2)
    return log

def stratum(tr):
    """Attribute-stratified key. Missing attributes form their OWN bucket and are reported."""
    return (tr.attributes.get("is_external", "<missing>"),
            'has_correction' if any(e['concept:name'] == 'Correct Dossier' for e in tr) else 'plain')

def split(log, seed, ratio=0.8):
    rng, buckets = random.Random(seed), {}
    for tr in log:
        buckets.setdefault(stratum(tr), []).append(tr)
    train, test = EventLog(), EventLog()
    for key in sorted(buckets, key=str):
        g = list(buckets[key]); rng.shuffle(g)
        cut = int(round(ratio * len(g)))
        train += g[:cut]; test += g[cut:]
        assert len(g) >= 5, f"stratum {key} too small for a stratified split"
    # variant-level leakage audit: report variants present in BOTH splits
    return train, test
```

Publish: the case-id list of every split, the bucket sizes, and the variant-overlap table between train and test. Repeat over `SPLIT_SEEDS` and report mean ± sd **per metric**, never a pooled score.

## 9.2 Metrics reported separately (never averaged)

```python
def metrics(model, log):
    net, im, fm = pm4py.convert_to_petri_net(model)
    is_sound, diag = pm4py.check_soundness(net, im, fm, print_diagnostics=True)
    return {
        "soundness": (is_sound, diag),
        "fitness":   pm4py.fitness_alignments(log, net, im, fm),       # train and holdout separately
        "precision": pm4py.precision_alignments(log, net, im, fm),     # positive log only (flagged)
        "generalization": pm4py.generalization_tbr(log, net, im, fm),
        "simplicity": pm4py.simplicity_petri_net(net, im, fm, variant="arc_degree"),
    }
```

Reported artefacts per model and per split: training fitness, holdout fitness, training precision, holdout precision, generalization, simplicity, soundness, and the four values of `policy_check`'s verdict distribution on the holdout. **The verdict distribution is the headline metric**, not fitness.

## 9.3 Curated negative trace suite (the missing measurement)

 30 traces, each with a clause tag and a compliance-officer signature. Eight seeds:

```python
NEGATIVES = {
 'N1_no_checks':            [REG, MA, LO],
