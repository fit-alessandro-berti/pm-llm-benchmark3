```python
def get_visible_traces(net, im, fm, max_visible_len=15, max_tau_steps=50):
    from collections import deque
    if im is None or fm is None:
        return set()
    results = set()
    # queue items: (marking, tau_count, visible_trace)
    queue = deque([(im, 0, ())])
    seen_states = set()
    
    while queue:
        mark, tau_count, trace = queue.popleft()
        # normalize marking for hashing
        mark_frozen = frozenset((p, c) for p, c in mark.items() if c > 0)
        if (mark_frozen, len(trace)) in seen_states:
            continue
        seen_states.add((mark_frozen, len(trace)))
        
        if mark_frozen == frozenset((p, c) for p, c in fm.items() if c > 0):
            if trace:
                results.add(trace)
            continue
        
        if len(trace) >= max_visible_len:
            continue
            
        for t in pm4py.get_enabled_transitions(net, mark):
            if t.label is None:
                if tau_count + 1 > max_tau_steps:
                    continue
                new_mark = fire(net, mark, t)
                queue.append((new_mark, tau_count + 1, trace))
            else:
                new_mark = fire(net, mark, t)
                queue.append((new_mark, 0, trace + (t.label,)))
    return results
```

```python
# Helper to fire a transition on a marking (standard Petri net math, not a pm4py conversion)
def fire_transition(net, marking, t):
    m = marking.copy()
    for arc in t.in_arcs:
        m[arc.source] -= arc.weight
    for arc in t.out_arcs:
        m[arc.target] += arc.weight
    return m

# Bounded visible-trace exploration
def get_visible_traces(net, im, fm, max_visible_len=15, max_tau_steps=50):
    from collections import deque
    if im is None or fm is None:
        return set()
    
    results = set()
    queue = deque([(im, 0, ())])
    seen_states = set()
    
    while queue:
        mark, tau_count, trace = queue.popleft()
        mark_frozen = frozenset((p, c) for p, c in mark.items() if c > 0)
        if (mark_frozen, len(trace)) in seen_states:
            continue
        seen_states.add((mark_frozen, len(trace)))
        
        # Exact final-marking equality check
        if mark_frozen == frozenset((p, c) for p, c in fm.items() if c > 0):
            if trace:
                results.add(trace)
            continue
        
        if len(trace) >= max_visible_len:
            continue
            
        for t in pm4py.get_enabled_transitions(net, mark):
            if t.label is None:  # hide tau

                queue.append((new_mark, tau_count + 1, trace))
            else:
                new_mark = fire_transition(net, mark, t)
                queue.append((new_mark, 0, trace + (t.label,)))
    return results
```
