```python
import traceback
import pm4py
from pm4py.objects.powl.obj import (
    StrictPartialOrder,
    OperatorPOWL,
    Transition,
    SilentTransition,
)
from pm4py.objects.process_tree.obj import Operator


def build_source_powl():
    register = Transition(label="Register Change")
    assess_risk = Transition(label="Assess Risk")
    prepare_evidence = Transition(label="Prepare Evidence")

    legal_review = Transition(label="Legal Review")
    skip_legal = SilentTransition()
    optional_legal = OperatorPOWL(
        operator=Operator.XOR,
        children=[legal_review, skip_legal],
    )

    configure_trial = Transition(label="Configure Trial")
    inspect_trial = Transition(label="Inspect Trial")
    trial_body = StrictPartialOrder(nodes=[configure_trial, inspect_trial])
    trial_body.order.add_edge(configure_trial, inspect_trial)

    correct_defect = Transition(label="Correct Defect")
    verification_loop = OperatorPOWL(
        operator=Operator.LOOP,
        children=[trial_body, correct_defect],
    )

    release_change = Transition(label="Release Change")
    archive_change = Transition(label="Archive Change")

    powl = StrictPartialOrder(
        nodes=[
            register,
            assess_risk,
            prepare_evidence,
            optional_legal,
            verification_loop,
            release_change,
            archive_change,
        ]
    )

    powl.order.add_edge(register, assess_risk)
    powl.order.add_edge(register, prepare_evidence)
    powl.order.add_edge(assess_risk, optional_legal)
    powl.order.add_edge(assess_risk, verification_loop)
    powl.order.add_edge(prepare_evidence, verification_loop)
    powl.order.add_edge(optional_legal, release_change)
    powl.order.add_edge(verification_loop, release_change)
    powl.order.add_edge(release_change, archive_change)
    return powl


def has_valid_accepting_net(net, im, fm):
    return (
        net is not None
        and im is not None
        and fm is not None
        and len(im) > 0
        and len(fm) > 0
    )


def diagnose_net(name, net, im, fm):
    print(f"\n--- Diagnostics for {name} ---")
    if not has_valid_accepting_net(net, im, fm):
        print(
            f"[FAIL] {name}: missing net or null/empty initial/final marking "
            f"(net={net is not None}, im={im}, fm={fm})."
        )
        return {"valid_markings": False, "is_wf_net": False, "is_sound": False}

    is_wf = False
    is_sound = False
    try:
        is_wf = bool(pm4py.check_is_workflow_net(net))
        print(f"Workflow net: {is_wf}")
    except Exception as exc:
        print(f"[ERROR] check_is_workflow_net failed for {name}: {exc}")

    try:
        sound_res = pm4py.check_soundness(net, im, fm, print_diagnostics=True)
        is_sound = bool(sound_res[0]) if isinstance(sound_res, tuple) else bool(sound_res)
        print(f"Soundness: {is_sound}")
    except Exception as exc:
        print(f"[ERROR] check_soundness failed for {name}: {exc}")

    return {"valid_markings": True, "is_wf_net": is_wf, "is_sound": is_sound}


def safe_save_vis(fn, *args, description="visualization"):
    try:
        fn(*args)
        print(f"[SAVED] {description} -> {args[-1]}")
    except Exception as exc:
        print(f"[WARN] Could not save {description} ({args[-1]}): {exc}")


def run_conversions(source_powl):
    artifacts = {
        "source_powl": source_powl,
        "nets": {},
        "diagnostics": {},
        "bpmn_from_net": None,
        "tree_from_net": None,
        "bpmn_from_tree": None,
        "tree_conversion_error": None,
    }

    safe_save_vis(
        pm4py.save_vis_powl,
        source_powl,
        "01_source_powl.png",
        description="source POWL",
    )

    # 1. POWL -> accepting Petri net
    net_from_powl, im_from_powl, fm_from_powl = pm4py.convert_to_petri_net(source_powl)
    artifacts["nets"]["net_from_powl"] = (net_from_powl, im_from_powl, fm_from_powl)
    artifacts["diagnostics"]["net_from_powl"] = diagnose_net(
        "net_from_powl", net_from_powl, im_from_powl, fm_from_powl
    )
    if has_valid_accepting_net(net_from_powl, im_from_powl, fm_from_powl):
        safe_save_vis(
            pm4py.save_vis_petri_net,
            net_from_powl,
            im_from_powl,
            fm_from_powl,
            "02_net_from_powl.png",
            description="accepting Petri net from POWL",
        )

    # 2. Pipeline 1: accepting Petri net -> BPMN
    if has_valid_accepting_net(net_from_powl, im_from_powl, fm_from_powl):
        try:
            bpmn_from_net = pm4py.convert_to_bpmn(
                net_from_powl, im_from_powl, fm_from_powl
            )
            artifacts["bpmn_from_net"] = bpmn_from_net
            safe_save_vis(
                pm4py.save_vis_bpmn,
                bpmn_from_net,
                "03_bpmn_from_net.png",
                description="BPMN from accepting Petri net",
            )
        except Exception as exc:
            print(f"[FAIL] Petri net -> BPMN conversion failed: {exc}")
            traceback.print_exc()

    # 3. Pipeline 2: accepting Petri net -> process tree (explicitly guarded)
    if has_valid_accepting_net(net_from_powl, im_from_powl, fm_from_powl):
        try:
            tree_from_net = pm4py.convert_to_process_tree(
                net_from_powl, im_from_powl, fm_from_powl
            )
            if tree_from_net is None:
                raise ValueError("convert_to_process_tree returned None.")
            artifacts["tree_from_net"] = tree_from_net
            print(f"\n[INFO] Process tree conversion succeeded: {tree_from_net}")
            safe_save_vis(
                pm4py.save_vis_process_tree,
                tree_from_net,
                "04_tree_from_net.png",
                description="process tree from Petri net",
            )

            # 4. If process tree succeeded, convert tree -> BPMN
            try:
                bpmn_from_tree = pm4py.convert_to_bpmn(tree_from_net)
                artifacts["bpmn_from_tree"] = bpmn_from_tree
                safe_save_vis(
                    pm4py.save_vis_bpmn,
                    bpmn_from_tree,
                    "05_bpmn_from_tree.png",
                    description="BPMN from process tree",
                )
            except Exception as exc:
                print(f"[FAIL] Process tree -> BPMN conversion failed: {exc}")
                traceback.print_exc()

        except Exception as exc:
            artifacts["tree_conversion_error"] = exc
            print(
                "\n[EXPECTED / INTERPRETED FAILURE] Petri net -> Process Tree failed: "
                f"{type(exc).__name__}: {exc}"
            )
            print(
                "Interpretation: source_powl contains a non-series-parallel dependency "
                "(Assess Risk enables optional_legal, while verification_loop waits for "
                "both Assess Risk and Prepare Evidence). Exact block-structured folding "
                "into a process tree is therefore blocked unless concurrency is altered "
                "or transitions are duplicated."
            )

    # 5. Round-trip each produced BPMN back to an accepting Petri net
    for label, bpmn_obj, out_img in [
        ("rt_net_from_bpmn_net", artifacts["bpmn_from_net"], "06_rt_net_from_bpmn_net.png"),
        ("rt_net_from_bpmn_tree", artifacts["bpmn_from_tree"], "07_rt_net_from_bpmn_tree.png"),
    ]:
        if bpmn_obj is None:
            print(f"\n[SKIP] {label}: upstream BPMN model was not produced.")
            continue
        try:
            rt_res = pm4py.convert_to_petri_net(bpmn_obj)
            if not isinstance(rt_res, tuple) or len(rt_res) != 3:
                raise ValueError(f"Unexpected return structure from convert_to_petri_net: {rt_res}")
            rt_net, rt_im, rt_fm = rt_res
            artifacts["nets"][label] = (rt_net, rt_im, rt_fm)
            artifacts["diagnostics"][label] = diagnose_net(label, rt_net, rt_im, rt_fm)
            if has_valid_accepting_net(rt_net, rt_im, rt_fm):
                safe_save_vis(
                    pm4py.save_vis_petri_net,
                    rt_net,
                    rt_im,
                    rt_fm,
                    out_img,
                    description=f"round-trip accepting Petri net ({label})",
                )
        except Exception as exc:
            print(f"[FAIL] BPMN -> Petri net failed for {label}: {exc}")
            traceback.print_exc()

    return artifacts
```

# 4. Preservation and Loss Audit

Because conversion outcomes depend on exact structural translation and runtime round-trip checks (executed by Section 5), entries below distinguish between structural expectation (`expected preserved`), structural ambiguity (`ambiguous`), expected block-structuring failure (`conversion failed`), and unexecuted branches (`not tested`).

| Source invariant | POWLPetri | PetriBPMN | PetriTree | BPMN round trip | Evidence or uncertainty |
| :--- | :--- | :--- | :--- | :--- | :--- |
| `INV-01` (Register first) | `expected preserved` | `expected preserved` | `conversion failed` | `expected preserved` | Sequential place/arc from initial marking preserves `Register Change` first; `PetriTree` fails globally on the downstream N-pattern (marked `conversion failed`, making its downstream BPMN `not tested` if exception occurs). |
| `INV-02` (Assess/Prepare concurrency) | `expected preserved` | `expected preserved` | `conversion failed` | `expected preserved` | `Register Change` splits to two concurrent places; Petri-to-BPMN synthesizes a parallel split gateway, though layout may visually stack them vertically. |
| `INV-03` (Optional Legal after Assess) | `expected preserved` | `ambiguous` | `conversion failed` | `expected preserved` | POWL `XOR(Legal Review, tau)` produces a visible branch and a silent `tau` branch. In BPMN, the silent branch is an unmarked bypass edge or unlabeled gateway path that readers can misread; round-trip Petri net restores an equivalent silent path unless gateway reduction drops/merges markings. |
| `INV-04` (Verification waits on Assess + Prepare) | `expected preserved` | `expected preserved` | `conversion failed` | `expected preserved` | Petri net places a join transition/place requiring tokens from both `Assess Risk` and `Prepare Evidence`; `PetriTree` fails specifically because `Assess Risk` also independently enables `optional_legal`. |
| `INV-05` (Legal–verification concurrency) | `expected preserved` | `ambiguous` | `conversion failed` | `expected preserved` | Asymmetric split from `Assess Risk` (`optional_legal` vs `verification_loop` precondition) can cause `PetriBPMN` to emit nested/chained AND-gateways that visually suggest a barrier or dependency with `Prepare Evidence`; behavioral preservation requires Section 5 trace verification. |
| `INV-06` (Configure before Inspect) | `expected preserved` | `expected preserved` | `conversion failed` | `expected preserved` | Strict edge inside `trial_body` maps directly to a sequential place between `Configure Trial` and `Inspect Trial`. |
| `INV-07` (Binary loop with Correct Defect) | `expected preserved` | `ambiguous` | `conversion failed` | `expected preserved` | `OperatorPOWL(LOOP, [trial_body, correct_defect])` maps to a structured do-then-choice-redo net, but `PetriBPMN` renders cycles using backward sequence flows through XOR gateways whose entry/exit conditions are visually unlabeled. |
| `INV-08` (Minimum 1 iteration) | `expected preserved` | `expected preserved` | `conversion failed` | `expected preserved` | Loop entry runs `trial_body` before reaching the exit/redo XOR choice; however, misdrawn XOR loop gateways in BPMN can visually suggest 0-iteration bypasses. |
| `INV-09` (AND-join before Release) | `expected preserved` | `expected preserved` | `conversion failed` | `expected preserved` | Both `optional_legal` completion and `verification_loop` exit feed an AND-join before `Release Change`; round-trip soundness must confirm no leftover tokens remain in either branch. |
| `INV-10` (Archive finality & exact fm) | `expected preserved` | `expected preserved` | `conversion failed` | `expected preserved` | Single sink place in `fm_from_powl` maps to a BPMN End Event; round-trip must confirm exact `fm` reachability and absence of deadlocks or improper completion. |

## Detailed Risk Discussion

- **True concurrency versus arbitrary interleaving**: `source_powl` defines partial-order concurrency (`Assess Risk || Prepare Evidence`, `optional_legal || verification_loop`), meaning concurrent events are causally independent rather than merely ordered both ways. Accepting Petri nets represent concurrency via independent token-enabled transitions (diamond interleavings in the reachability graph), whereas BPMN represents it explicitly with parallel (`+`) gateways. If a converter (such as a fallback tree/miner pipeline) replaced concurrency with an explicit choice of sequences, trace-level interleavings might appear identical while true concurrency and state-space structure would be lost.
- **Absence of an edge between Optional Legal Review and verification**: Because `Assess Risk -> optional_legal`, `Assess Risk -> verification_loop`, and `Prepare Evidence -> verification_loop` form an asymmetric ("N-shaped") partial order, `optional_legal` is concurrent with both `Prepare Evidence` and `verification_loop` once `Assess Risk` finishes. This breaks single-entry/single-exit block nesting, explaining why `pm4py.convert_to_process_tree` fails on `net_from_powl`. Any forced tree conversion that succeeded by introducing sequence edges (e.g., placing `optional_legal` strictly before or after `verification_loop`) would silently lose valid interleavings (`INV-05`).
- **Silent optional behavior**: `SilentTransition()` inside `optional_legal` becomes a `tau` (`label=None`) transition in `net_from_powl`. In BPMN, silent transitions may become direct gateway-to-gateway bypass edges or unlabeled/empty routing nodes. If BPMN-to-Petri-net round-tripping eliminates or merges a gateway incorrectly, the bypass can disappear (making `Legal Review` mandatory) or leak a token at the downstream join.
- **Binary-loop semantics**: `OperatorPOWL(LOOP, [trial_body, correct_defect])` enforces do-body (`Configure Trial -> Inspect Trial`) execution at least once, followed by an XOR decision either to exit or to run `Correct Defect` and re-enter `trial_body`. Converting this Petri-net cycle to BPMN and back risks either inverting the loop condition (allowing 0 iterations) or breaking the synchronization place at loop exit if `tau` transitions around the loop boundary are misrouted.
- **The join before Release Change**: `Release Change` requires synchronization of two concurrent paths (`optional_legal` and `verification_loop`). In a Petri net, this is an explicit transition consuming tokens from both branches; in BPMN, it is a Parallel Join gateway. If `PetriBPMN` or `BPMNPetri` emits an Exclusive (`XOR`) join instead of a Parallel (`AND`) join, `Release Change` could fire prematurely as soon as the first branch finishes (`INV-09` violation) while leaving an unconsumed token in the other branch (unsoundness).
- **Activity-label uniqueness**: All nine visible transitions in `source_powl` have distinct labels. This avoids false footprint merges or ambiguous visible trace matching. However, all silent transitions share `label is None`, so round-trip validation cannot match silent transitions by name and must instead use `tau`-closure and exact final-marking equality.
- **Initial and final markings**: `convert_to_petri_net` on `source_powl` produces an accepting Petri net with a single initial token in `im` and a single sink token in `fm`. Round-tripping from BPMN relies on start and end events to reconstruct `im` and `fm`. A missing end-event mapping or multi-sink net would yield `fm is None` or `len(fm) == 0`, invalidating completion checks.
- **Gateway or routing-node ambiguity in BPMN visualization**: Petri nets separate places and transitions, whereas BPMN represents routing via gateways. When a Petri net has silent transitions adjacent to splits/joins (e.g., `Assess Risk` splitting to both an XOR-split for `optional_legal` and an AND-join for `verification_loop`), `convert_to_bpmn` can emit chained unlabeled gateways or mixed implicit/explicit joins that look indistinguishable from choice or sequential gating on a diagram.

# 5. Round-Trip Validation Code

The code below validates `net_from_powl` against every successfully generated round-tripped net using soundness checks, footprint comparison, reachability-graph size/structure comparison, bounded `tau`-aware visible-trace exploration, positive/negative witness replay, and optional PM4Py playout.

```python
from collections import deque
import pm4py
from pm4py.objects.petri_net.obj import Marking


def normalize_marking(marking):
    return Marking({p: c for p, c in marking.items() if c > 0})


def marking_key(marking):
    return tuple(sorted((p.name, c) for p, c in marking.items() if c > 0))


def fire_transition(marking, transition):
    new_m = dict(marking)
    for arc in transition.in_arcs:
        new_m[arc.source] = new_m.get(arc.source, 0) - arc.weight
        if new_m[arc.source] <= 0:
            del new_m[arc.source]
    for arc in transition.out_arcs:
        new_m[arc.target] = new_m.get(arc.target, 0) + arc.weight
    return Marking(new_m)


def tau_closure(net, start_marking, max_total_tokens=20, max_closure_states=2000):
    """
    Computes all markings reachable from start_marking via 0 or more silent
    (label is None) transitions, preventing infinite silent cycles via visited tracking.
    """
    start_norm = normalize_marking(start_marking)
    visited = {marking_key(start_norm): start_norm}
    queue = deque([start_norm])
    truncated = False

    while queue:
        if len(visited) >= max_closure_states:
            truncated = True
            break
        curr = queue.popleft()
        for t in pm4py.get_enabled_transitions(net, curr):
            if t.label is None:
                nxt = fire_transition(curr, t)
                if sum(nxt.values()) > max_total_tokens:
                    truncated = True
                    continue
                k = marking_key(nxt)
                if k not in visited:
                    visited[k] = nxt
                    queue.append(nxt)

    return list(visited.values()), truncated


def explore_bounded_visible_language(
    net,
    im,
    fm,
    max_visible_len=14,
    max_total_tokens=20,
    max_explored_states=50000,
):
    """
    Explores all completed visible traces of length <= max_visible_len.
    - Hides transitions where t.label is None via tau_closure (avoiding silent cycles).
    - Requires exact final-marking equality (m == fm_norm) for completed traces.
    - Enforces visible-length bound (covering up to 3 verification iterations:
      7/8 steps for 1 iteration, 10/11 for 2, 13/14 for 3).
    """
    im_norm = normalize_marking(im)
    fm_norm = normalize_marking(fm)

    completed_traces = set()
    deadlock_traces = set()
    seen_states = {(marking_key(im_norm), ())}
    queue = deque([(im_norm, ())])
    truncated = False

    while queue:
        if len(seen_states) >= max_explored_states:
            truncated = True
            break

        curr_m, prefix = queue.popleft()
        closure_markings, tau_trunc = tau_closure(
            net, curr_m, max_total_tokens=max_total_tokens
        )
        if tau_trunc:
            truncated = True

        for m_tau in closure_markings:
            if m_tau == fm_norm:
                completed_traces.add(prefix)

            enabled = pm4py.get_enabled_transitions(net, m_tau)
            if not enabled and m_tau != fm_norm:
                deadlock_traces.add(prefix)

            if len(prefix) >= max_visible_len:
                continue

            for t in enabled:
                if t.label is not None:
                    m_next = fire_transition(m_tau, t)
                    if sum(m_next.values()) > max_total_tokens:
                        truncated = True
                        continue
                    next_prefix = prefix + (t.label,)
                    state_k = (marking_key(m_next), next_prefix)
                    if state_k not in seen_states:
                        seen_states.add(state_k)
                        queue.append((m_next, next_prefix))

    return {
        "completed_traces": completed_traces,
        "deadlock_traces": deadlock_traces,
        "exhaustive_within_bound": not truncated,
        "explored_states": len(seen_states),
    }


def accepts_visible_trace(net, im, fm, trace, max_total_tokens=20):
    """
    Exact replay check for a visible trace: allows arbitrary finite silent
    moves before/between/after visible labels and requires exact fm equality.
    """
    fm_norm = normalize_marking(fm)
    frontier_list, trunc = tau_closure(net, im, max_total_tokens=max_total_tokens)
    frontier = {marking_key(m): m for m in frontier_list}

    for label in trace:
        next_frontier = {}
        for m in frontier.values():
            for t in pm4py.get_enabled_transitions(net, m):
                if t.label == label:
                    m_after = fire_transition(m, t)
                    c_list, c_trunc = tau_closure(
                        net, m_after, max_total_tokens=max_total_tokens
                    )
                    trunc = trunc or c_trunc
                    for cm in c_list:
                        next_frontier[marking_key(cm)] = cm
        frontier = next_frontier
        if not frontier:
            return False

    return any(m == fm_norm for m in frontier.values())

def compare_footprints(name_a, fp_a, name_b, fp_b):
    print(f"\n[Footprints] Comparing {name_a} vs {name_b}")
    if isinstance(fp_a, dict) and isinstance(fp_b, dict):
        for key in ("activities", "sequence", "parallel"):
            val_a = set(fp_a.get(key, set()))
            val_b = set(fp_b.get(key, set()))
            only_a = val_a - val_b
            only_b = val_b - val_a
            print(f"  {key}: match={val_a == val_b}, only_in_{name_a}={only_a}, only_in_{name_b}={only_b}")
    else:
        print(f"  Footprint structure differs or is non-dict: {type(fp_a)} vs {type(fp_b)}")


def validate_round_trips(artifacts, max_visible_len=14):
    ref_tuple = artifacts["nets"].get("net_from_powl")
    if ref_tuple is None or not all(ref_tuple):
        raise RuntimeError("Reference net_from_powl is missing or has null markings.")

    ref_net, ref_im, ref_fm = ref_tuple

    # 1. Soundness & Footprints & Reachability graph for reference net
    ref_sound = pm4py.check_soundness(ref_net, ref_im, ref_fm, print_diagnostics=False)
    ref_fp = pm4py.discover_footprints(ref_net, ref_im, ref_fm)
    ref_rg = pm4py.convert_to_reachability_graph(ref_net, ref_im, ref_fm)
    print(
        f"[Reference] net_from_powl: sound={ref_sound[0] if isinstance(ref_sound, tuple) else ref_sound}, "
        f"RG states={len(ref_rg.states)}, RG transitions={len(ref_rg.transitions)}"
    )

    # 2. Bounded visible-trace exploration for reference net
    ref_lang = explore_bounded_visible_language(
        ref_net, ref_im, ref_fm, max_visible_len=max_visible_len
    )
    print(
        f"[Reference] Bounded exploration (max_visible_len={max_visible_len}): "
        f"traces={len(ref_lang['completed_traces'])}, "
        f"deadlocks={len(ref_lang['deadlock_traces'])}, "
        f"exhaustive_within_bound={ref_lang['exhaustive_within_bound']}"
    )

    # 3. Verify reference net satisfies all positive/negative witnesses
    for w_name, trace in POSITIVE_WITNESSES:
        assert accepts_visible_trace(ref_net, ref_im, ref_fm, trace), (
            f"Reference net unexpectedly rejected {w_name}"
        )
    for w_name, trace in NEGATIVE_WITNESSES:
        assert not accepts_visible_trace(ref_net, ref_im, ref_fm, trace), (
            f"Reference net unexpectedly accepted {w_name}"
        )
    print("[Reference] All positive and negative witnesses passed on net_from_powl.")

    # 4. Optional playout check on reference net if supported
    if hasattr(pm4py, "play_out"):
        try:
            sim_log = pm4py.play_out(ref_net, ref_im, ref_fm)
            print(f"[Reference] play_out generated {len(sim_log)} traces.")
        except Exception as exc:
            print(f"[Reference] play_out skipped/failed: {exc}")

    # 5. Compare every round-tripped net against reference
    for net_name, net_tuple in artifacts["nets"].items():
        if net_name == "net_from_powl":
            continue
        rt_net, rt_im, rt_fm = net_tuple
        print(f"\n================ Validating {net_name} ================")
        if rt_net is None or rt_im is None or rt_fm is None or len(rt_im) == 0 or len(rt_fm) == 0:
            print(f"[REJECT] {net_name}: null or empty initial/final marking.")
            continue

        rt_sound = pm4py.check_soundness(rt_net, rt_im, rt_fm, print_diagnostics=True)
        is_rt_sound = rt_sound[0] if isinstance(rt_sound, tuple) else bool(rt_sound)
        print(f"Soundness: {is_rt_sound}")

        try:
            rt_fp = pm4py.discover_footprints(rt_net, rt_im, rt_fm)
            compare_footprints("net_from_powl", ref_fp, net_name, rt_fp)
        except Exception as exc:
            print(f"[WARN] discover_footprints failed for {net_name}: {exc}")

        try:
            rt_rg = pm4py.convert_to_reachability_graph(rt_net, rt_im, rt_fm)
            print(
                f"[Reachability Graph] {net_name}: "
                f"states={len(rt_rg.states)} (ref={len(ref_rg.states)}), "
                f"transitions={len(rt_rg.transitions)} (ref={len(ref_rg.transitions)})"
            )
        except Exception as exc:
            print(f"[WARN] convert_to_reachability_graph failed for {net_name}: {exc}")

        rt_lang = explore_bounded_visible_language(
            rt_net, rt_im, rt_fm, max_visible_len=max_visible_len
        )
        missing_traces = ref_lang["completed_traces"] - rt_lang["completed_traces"]
        extra_traces = rt_lang["completed_traces"] - ref_lang["completed_traces"]
        print(
            f"[Bounded Language <= {max_visible_len}] "
            f"exhaustive_within_bound={rt_lang['exhaustive_within_bound']}, "
            f"ref_count={len(ref_lang['completed_traces'])}, "
            f"rt_count={len(rt_lang['completed_traces'])}, "
            f"missing_in_rt={len(missing_traces)}, "
            f"extra_in_rt={len(extra_traces)}, "
            f"deadlocks={len(rt_lang['deadlock_traces'])}"
        )
        if missing_traces:
            print("  Sample missing trace:", next(iter(missing_traces)))
        if extra_traces:

        # Witness testing
        pos_failures = [
            w_name for w_name, tr in POSITIVE_WITNESSES
            if not accepts_visible_trace(rt_net, rt_im, rt_fm, tr)
        ]
        neg_failures = [
            w_name for w_name, tr in NEGATIVE_WITNESSES
            if accepts_visible_trace(rt_net, rt_im, rt_fm, tr)
        ]
        print(f"[Witnesses] positive_failures={pos_failures}, negative_failures={neg_failures}")

        # Playout cross-check: replay playout traces from rt_net onto ref_net
        if hasattr(pm4py, "play_out"):
            try:
                rt_log = pm4py.play_out(rt_net, rt_im, rt_fm)
                unaccepted = 0
                for trace in rt_log:

                    if not accepts_visible_trace(ref_net, ref_im, ref_fm, vis_seq):
                        unaccepted += 1
                print(
                    f"[Playout cross-check] {len(rt_log)} simulated traces from {net_name}; "
                    f"rejected by ref_net={unaccepted}"
                )
            except Exception as exc:
                print(f"[Playout cross-check] skipped/failed for {net_name}: {exc}")
```

# 6. Visual-Semantics Review

Even when `rt_net_from_bpmn_net` passes every soundness and trace-equivalence test, the generated BPMN diagram can easily be misread by engineering, compliance, and audit stakeholders:

1. **Parallel gateways (`+`)**: Because `Assess Risk` enables `optional_legal` independently of `Prepare Evidence`, while `verification_loop` requires both `Assess Risk` and `Prepare Evidence`, `convert_to_bpmn` can generate two chained or asymmetric parallel splits/joins. Readers often misinterpret an intermediate `+` gateway as a global synchronization barrier—incorrectly assuming `Legal Review` must wait for `Prepare Evidence`, or that `Assess Risk` and `Prepare Evidence` must both finish before `Legal Review` begins.
2. **Exclusive gateways (`×`)**: Two structurally different `XOR` constructs appear in the process: the optionality choice for `Legal Review` (`XOR(Legal Review, tau)`) and the loop exit/re-entry choice after `Inspect Trial`. In auto-generated BPMN, both render as bare `×` diamonds without guard labels, making it unclear which outgoing branch skips `Legal Review`, which branch exits verification, and which branch routes to `Correct Defect`.
3. **Loop visualization**: The `Correct Defect` back-edge forms a visual cycle connecting the post-`Inspect Trial` XOR split back to an XOR merge before `Configure Trial`. Without explicit labeling, stakeholders frequently confuse *do-while* (`>= 1` iteration) with *while-do* (`>= 0` iterations) and assume verification can be bypassed completely.
4. **Silent routing nodes**: `SilentTransition()` in `optional_legal` and silent plumbing transitions from Petri-net loop/join synthesis may render as empty sequence flows bypassing `Legal Review`, chained gateway-to-gateway hops, or unlabeled nodes. Stakeholders often mistake an empty bypass arc for a drawing error or an unmodeled exception path.
5. **Apparent ordering from graph layout**: Left-to-right hierarchical layout engines place concurrent branches vertically above/below one another or stagger horizontal X-coordinates based on edge count. Readers routinely infer false causality from geometry—for instance, reading `Assess Risk` (if placed higher or slightly further left) as preceding `Prepare Evidence`, or reading `Legal Review` as strictly preceding `Configure Trial` when it happens to align vertically to the left of the loop box.

### Recommended Visual Annotations and Documentation

- **Gateway labels**: Label the `optional_legal` split `Legal Review Required?` (`[Yes] -> Legal Review`, `[No / Bypass] -> join`) and the loop-exit split `Trial Passed?` (`[Yes] -> join`, `[No] -> Correct Defect`).
- **BPMN Groups & Text Annotations**: Enclose `Configure Trial -> Inspect Trial -> Correct Defect` in a BPMN Group labeled `Verification Loop (min 1 iteration; Correct Defect triggers re-test)` and attach an explicit Text Annotation to the `Assess Risk` / `Prepare Evidence` region stating: `Assess Risk and Prepare Evidence are concurrent. Legal Review depends only on Assess Risk and may run concurrently with Prepare Evidence and Verification.`
- **Bypass legend**: Add a diagram legend noting that unlabeled gateway-to-gateway edges represent intentional silent skips (`tau`), not missing tasks.

# 7. Rejection Criteria

A converted artifact must be **rejected or quarantined** if any of the following conditions occur:

1. **Null, empty, or non-workflow-net structure**: `net`, `im`, or `fm` is `None`, `im` or `fm` has zero tokens, or `pm4py.check_is_workflow_net` returns `False`.
2. **Unsoundness**: `pm4py.check_soundness` returns `False`, indicating dead transitions, unreachable final markings, improper completion (tokens remaining in `optional_legal` or loop places when `fm` is reached), or deadlocks.
3. **Premature release or missing synchronization**: Any negative witness testing synchronization fails—specifically, if `Release Change` is admitted before `Inspect Trial` or `Legal Review` (`NEG-5`, `NEG-6`), or if `Configure Trial` is admitted before `Prepare Evidence` (`NEG-3`).
4. **Loss of required concurrency / interleaving**: Bounded language comparison or positive witnesses show that valid interleavings are rejected (e.g., rejecting `Prepare Evidence` before `Assess Risk` in `POS-2`, or rejecting `Legal Review` before `Prepare Evidence` in `POS-3`, or rejecting `Legal Review` inside the verification loop in `POS-4`).
5. **Collapse of optionality or loop cardinality**: `Legal Review` becomes mandatory (rejecting `POS-1`/`POS-2`), `0` verification iterations are admitted (`NEG-4`), `Correct Defect` can exit directly to `Release Change` (`NEG-7`), or `Inspect Trial` can precede `Configure Trial` (`NEG-9`).
6. **Failed exact-final-marking or termination check**: A prefix missing `Archive Change` (`NEG-8`) matches `fm`, or `deadlock_traces` is non-empty during bounded exploration.
7. **Symmetric bounded-language or footprint divergence**: `missing_in_rt > 0` or `extra_in_rt > 0` under bounded exploration (`max_visible_len = 14`), or `discover_footprints` detects mismatched `sequence` / `parallel` relations on visible activities.
8. **Silent livelock or unbounded token growth**: `tau_closure` or `explore_bounded_visible_language` triggers token-cap or state-cap truncation (`exhaustive_within_bound == False`), indicating a gateway-induced silent cycle or unbounded place.

# 8. Canonical Representation Recommendation

- **Authoritative canonical representation (storage & editing)**: **`source_powl` (the POWL model stored in version-controlled Python/JSON-canonical form)**. POWL natively captures partial-order concurrency, asymmetric dependencies, optional XOR, and binary LOOP scopes without the low-level place proliferation of Petri nets or the visual-gateway ambiguities of BPMN.
- **Derivative representations**:
  - `net_from_powl` (`(net, im, fm)`): primary behavioral/verification artifact used for soundness checking, simulation, and conformance checking.
  - `bpmn_from_net` (annotated): stakeholder visualization artifact generated exclusively from `net_from_powl`.
  - Round-trip nets (`rt_net_from_bpmn_net`), reachability graphs, and footprint matrices: automated QA artifacts.
- **Required validation evidence per derivative**:
  - Every `net_from_powl` build must include `check_is_workflow_net == True`, `check_soundness == True`, and passing positive/negative witness checks.
  - Every `bpmn_from_net` diagram must be accompanied by its round-tripped `rt_net_from_bpmn_net` passing soundness, exact-`fm` witness checks, footprint equality, and `0` missing/extra traces in bounded visible-language comparison (`max_visible_len >= 14`).
- **Policy on process-tree intermediates**: **Disallowed as a required or automatic intermediate.** As established by the N-shaped dependency between `Assess Risk`, `Prepare Evidence`, `optional_legal`, and `verification_loop`, `source_powl` is not block-structured. A process-tree intermediate may only be permitted if a future model happens to be block-structured *and* the resulting BPMN round-trips with zero bounded-language or footprint divergence.
- **Preventing round-trip drift**: Enforce a strict **one-way compilation workflow**. All model changes must be authored in `source_powl`, reviewed in code/model diff, and compiled downstream in CI/CD to regenerate Petri nets, run Section 5 validation gates, and emit read-only BPMN diagrams. Direct edits to generated BPMN XML or Petri-net files must be prohibited (or blocked by checksum verification), preventing unverified layout tweaks from silently altering gateway types or synchronization semantics.

# 9. Residual Uncertainty

Even if every diagnostic and validation check in Sections 3 and 5 passes completely, the following claims remain unestablished:

1. **Behavior beyond the bounded loop horizon (`> 3` verification iterations)**: Exhaustive bounded exploration up to `max_visible_len = 14` proves language equality only for `0`, `1`, and `2` executions of `Correct Defect`. It cannot rule out structural defects in a converted net that only manifest after `k >= 4` iterations (for example, an unintended capacity/accumulator place that blocks or leaks on the fourth cycle).
2. **Branching-time equivalence (weak bisimulation) and timing of silent (`tau`) choices**: Comparing completed visible trace sets and hiding `label is None` transitions validates *trace* equivalence, not weak bisimulation. It cannot distinguish *when* an internal silent decision occurs (e.g., deciding to skip `Legal Review` immediately after `Assess Risk` versus deferring the silent skip until just before `Release Change`), even though those internal states differ in which future continuations or monitoring hooks are active.
3. **True partial-order concurrency vs. interleaving equivalence**: Reachability graphs, footprints, and visible trace sets evaluate interleaving semantics (sequentialized firing orders). They cannot prove whether two concurrent activities (`Assess Risk` and `Prepare Evidence`, or `Legal Review` and `Configure Trial`) retain true causal independence (pomset/step semantics) versus being serialized in two separate branches that happen to yield the same linear trace set.
4. **Stakeholder comprehension of the rendered BPMN**: Passing mathematical round-trip checks proves behavioral equivalence of the BPMN semantics, not human readability. Automated layout can still mislead stakeholders about concurrency, optionality, or loop minimums unless reviewed with the visual annotations in Section 6.
5. **Data guards, defect-severity rules, and timing/resource constraints**: Because `source_powl` models pure control flow, passing validation says nothing about whether external business rules (e.g., what risk score mandates `Legal Review`, or what defect threshold forces `Correct Defect`) are preserved in downstream execution engines.
