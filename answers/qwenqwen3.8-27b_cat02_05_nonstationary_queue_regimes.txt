def mean_shift_score(x):
    # x pd.Series with index 0..n-1, values
    scores=[]
    n=len(x)
    for i in range(1,n-1):
       left=x.iloc[:i]; right=x.iloc[i:]
       if left.std(ddof=1)==0 or right.std(ddof=1)==0:
           denom=0
       else:
           denom=(left.std(ddof=1)**2 + right.std(ddof=1)**2)**0.5
       score=(right.mean()-left.mean())/denom if denom>0 else 0
       scores.append((x.index[i], score))
    return pd.DataFrame(scores, columns=['split_before','score']).sort_values('score', ascending=False)
```

def top_mean_shift(df, metric, top=3):
    x = df[metric].reset_index(drop=True)
    dates = df['date'].reset_index(drop=True)
    rows = []
    for i in range(1, len(x) - 1):
        left = x.iloc[:i]
        right = x.iloc[i:]
        if len(left) < 2 or len(right) < 2:
            score = 0.0
        else:
            sl = left.std(ddof=1)
            sr = right.std(ddof=1)
            denom = (sl ** 2 + sr ** 2) ** 0.5
            score = (right.mean() - left.mean()) / denom if denom > 0 else 0.0
        rows.append({
            'split_before': dates.iloc[i],
            'split_after': dates.iloc[i + 1],
            'left_mean': left.mean(),
            'right_mean': right.mean(),
            'standardized_shift': score
        })
    return pd.DataFrame(rows).sort_values('standardized_shift', ascending=False).head(top)

wd = daily[~daily['is_weekend']].reset_index(drop=True)
cp_frames = []
for metric in ['complex_pct', 'median_service_min', 'mean_wip', 'staff_fte']:
    cp = top_mean_shift(wd, metric, top=2)
    cp.insert(0, 'metric', metric)
    cp_frames.append(cp)
change_points = pd.concat(cp_frames, ignore_index=True)
