  ```python
  def seq(*nodes):
      """Total order: n1 -> n2 -> ... -> nk (POWL sequence)."""
      po = StrictPartialOrder(list(nodes))
      for i in range(len(nodes) - 1):
          po.order.add_edge(nodes[i], nodes[i + 1])
      return po


  def xor(*nodes):
      return OperatorPOWL(Operator.XOR, list(nodes))

      """Zero or more repetitions of `repeat_block`.
      LOOP(do=tau, redo=repeat_block) has language  tau (repeat_block tau)*

      return OperatorPOWL(Operator.LOOP, [SilentTransition(), repeat_block])

      return StrictPartialOrder(list(nodes))
  ```

  ```python
  def moves(align_result):
      for (lg, md) in align_result["alignment"]:
          if lg != ">>" and md != ">>": yield "sync"
          elif lg != ">>": yield "log"
          else: yield "model"   # note: model moves on invisible transitions may appear as ('>>', None) or ('>>', 'skip')
  def custom_fitness(diags, w_log=1.0, w_model=1.0):
      # normalized: 1 - cost / worst-case, mirroring standard trace-fitness normalization
      vals = []
      for d in diags:
          m = list(moves(d))
          cost = w_log*m.count("log") + w_model*m.count("model")
          worst = w_log*len([x for x in m if x!="model"]) + w_model*len([x for x in m if x!="log"])  # rough upper bound
          vals.append(1 - cost/max(worst,1))
  ```

  ```python
  def perturb(traces, rng, drop=None, drop_p=0.0, dup=None, dup_p=0.0):
      out = []
      for acts in traces:
          new = []
          for a in acts:
              if drop is not None and a == drop and rng.random() < drop_p:
                  continue                     # simulate a missing event
              new.append(a)
              if dup is not None and a == dup and rng.random() < dup_p:
                  new.append(a)                # simulate a duplicated event
          out.append(new)
      return out
  ```

  ```python
  """
  model_third — compact block-structured POWL model with tiered declarative policy checks.
  Requires pm4py >= 2.7.x (POWL support). A process-tree string with identical language
  is provided at the bottom as a fallback / cross-check.
  """
  import pm4py
  from pm4py.objects.powl.obj import (
      Transition, SilentTransition, OperatorPOWL, StrictPartialOrder,
  )
  from pm4py.objects.process_tree.obj import Operator


  # --------------------------------------------------------------------------
  # POWL construction helpers
  # --------------------------------------------------------------------------
  def seq(*nodes):
      """Total order: n1 -> n2 -> ... -> nk."""
      po = StrictPartialOrder(list(nodes))
      for i in range(len(nodes) - 1):
          po.order.add_edge(nodes[i], nodes[i + 1])
      return po


  def xor(*nodes):
      """Exclusive choice over the children."""
      return OperatorPOWL(Operator.XOR, list(nodes))


  def opt(node):
      """Optional execution: XOR(node, tau)."""
      return xor(node, SilentTransition())


  def concur(*nodes):
      """Full concurrency: strict partial order with no edges (= AND block)."""
      return StrictPartialOrder(list(nodes))


  def star(repeat):
      """Zero or more repetitions of `repeat`:
      LOOP(do=tau, redo=repeat) has language  tau (repeat tau)* = repeat*."""
      return OperatorPOWL(Operator.LOOP, [SilentTransition(), repeat])


  # --------------------------------------------------------------------------
  # Activities (fresh Transition objects per position; identical labels are fine
  # and are documented: 'Credit Check' x3, 'Technical Review' x3).
  # --------------------------------------------------------------------------
  REG = Transition("Register Dossier")

  CC_1 = Transition("Credit Check")        # first execution
  TR_1 = Transition("Technical Review")    # first execution
  CD   = Transition("Correct Dossier")

  CC_2 = Transition("Credit Check")        # recheck option: CC only
  TR_2 = Transition("Technical Review")    # recheck option: TR only
  CC_3 = Transition("Credit Check")        # recheck option: both
  TR_3 = Transition("Technical Review")    # recheck option: both

  SR   = Transition("Supplier Review")
  ER   = Transition("Expedite Review")
  MA   = Transition("Manager Approval")
  AA   = Transition("Auto Approval")
  RO   = Transition("Release Order")


  # --------------------------------------------------------------------------
  # Structure
  # --------------------------------------------------------------------------
  # P2: both mandatory checks, either order, before approval. No tau: skipping
  # either check is a *deviation* (classified CRITICAL), never a silent skip.
  mandatory_checks = concur(CC_1, TR_1)

  # P6: each Correct Dossier is followed by re-execution of exactly the affected
  # checks: none / CC only / TR only / both in either order. The scope is an
  # explicit choice point (answers F6: the decision is now visible in the model).
  recheck_scope = xor(SilentTransition(), CC_2, TR_2, concur(CC_3, TR_3))
  correction_cycle = seq(CD, recheck_scope)
  correction_zone = star(correction_cycle)          # (CD, scope)*  — corrections may repeat

  # P3/P4: optional Supplier Review and optional Expedite Review, either order
  # relative to each other AND relative to correction cycles; the whole zone
  # sits after both mandatory checks and before approval.
  #   - SR optionality is NOT silent: rule R1 below demands the sourcing
  #     attribute and escalates when it is absent.
  #   - ER must follow both (initial) checks and precede approval — enforced by
  #     zone placement. ER vs. rechecks is policy-ambiguous; if the board rules
  #     "ER after all rechecks", add one edge: zone.order.add_edge(correction_zone, er_opt).
  sr_opt = opt(SR)
  er_opt = opt(ER)
  review_zone = StrictPartialOrder([correction_zone, sr_opt, er_opt])
  # no edges: correction cycles, SR, ER fully concurrent
  # (governance toggle) review_zone.order.add_edge(correction_zone, er_opt)

  # P5: exactly one approval (XOR without tau: zero or two approvals = deviation).
  approval = xor(MA, AA)

  # P1: register first, release last.
  model_third = seq(REG, mandatory_checks, review_zone, approval, RO)


  # --------------------------------------------------------------------------
  # Conversions / sanity checks (run these; do not trust, verify)
  # --------------------------------------------------------------------------
  net_third, im_third, fm_third = pm4py.convert_to_petri_net(model_third)
  ```

  ```python
  MANDATORY = ("Credit Check", "Technical Review")
  APPROVALS = ("Manager Approval", "Auto Approval")


  def post_hoc_policy_scan(activities, sourcing=None):
      """Deterministic policy scan for one case.
      `activities`: ordered activity labels of the case.
      `sourcing`: 'external' / 'internal' / None (attribute missing).
      Returns a list of (rule, severity, message). Severities:
        CRITICAL   -> blocking (compliance)
        SUSPICIOUS -> flagged for investigation (P7)
        VERIFY     -> manual confirmation needed (missing attribute)
        UNVERIFIABLE -> cannot be checked with current data; must not silently pass
      """
      f = []

      # R6: register first / release last
      if activities and activities[0] != "Register Dossier":
          f.append(("R6", "CRITICAL", "case does not start with Register Dossier"))
      if activities and activities[-1] != "Release Order":
          f.append(("R6", "CRITICAL", "case does not end with Release Order"))

      # R4: exactly one approval
      n_appr = sum(a in APPROVALS for a in activities)
      if n_appr != 1:
          f.append(("R4", "CRITICAL", f"expected exactly one approval, found {n_appr}"))

      # R2: repeated mandatory check without intervening Correct Dossier (P7)
      cd_count = 0
      cd_at_last_exec = {}
      for i, a in enumerate(activities):
          if a == "Correct Dossier":
              cd_count += 1
          elif a in MANDATORY:
              if a in cd_at_last_exec and cd_at_last_exec[a] == cd_count:
                  f.append(("R2", "SUSPICIOUS",
                            f"{a} at position {i} repeats without intervening Correct Dossier"))
              cd_at_last_exec[a] = cd_count

      # R5: Expedite Review after both initial mandatory checks and before approval
      idx = {a: [i for i, x in enumerate(activities) if x == a] for a in set(activities)}
      if "Expedite Review" in idx:
          er_i = idx["Expedite Review"][0]
          if not ("Credit Check" in idx and idx["Credit Check"][0] < er_i
                  and "Technical Review" in idx and idx["Technical Review"][0] < er_i):
              f.append(("R5", "CRITICAL", "Expedite Review before both mandatory checks"))
          appr_i = [i for i, x in enumerate(activities) if x in APPROVALS]
          if appr_i and er_i > appr_i[0]:
              f.append(("R5", "CRITICAL", "Expedite Review after approval"))

      # R1: Supplier Review required for external orders (P3)
      has_sr = "Supplier Review" in idx
      if sourcing == "external" and not has_sr:
          f.append(("R1", "CRITICAL", "external order without Supplier Review"))
      elif sourcing is None and not has_sr:
          f.append(("R1", "UNVERIFIABLE",
                    "sourcing attribute absent: SR requirement cannot be verified (do NOT silently pass)"))

      # R3: recheck scope after each correction needs the affected-check attribute
      # (flag for manual confirmation; see limitations in the dossier)
      if "Correct Dossier" in idx:
          f.append(("R3", "VERIFY",
                    "verify re-executed checks match the correction's affected checks (attribute missing)"))

      return f
  ```

  ```python
  def move_types(align):
      for lg, md in align["alignment"]:
          if lg != ">>" and md != ">>":
              yield "sync"
          elif lg != ">>":
              yield "log"
          else:
              yield "model"   # includes invisible model moves; split if needed

  def reweighted_fitness(diags, w_log=1.0, w_model=1.0):
      """Recompute a normalized trace fitness from move counts under alternative
      cost weights. Report the spread across (w_log, w_model) in {(1,1),(1,2),(2,1),(1,5)}."""
      vals = []
      for d in diags:
          mt = list(move_types(d))
          n_log, n_model = mt.count("log"), mt.count("model")
          n = len(mt)
          cost = w_log * n_log + w_model * n_model
          worst = max(n * max(w_log, w_model), 1)
          vals.append(1.0 - cost / worst)
      return float(np.mean(vals))
  ```

# 1. Review Scope and Assumptions

**Scope.** I review Model S and Model C as candidate artifacts for a dossier-ordering process, treating the reported metrics as *evidence to be audited*, not as ground truth. The decision before the board is not "which model has the best aggregate score" but "which model may be used, for which purpose, under which conditions."

**Assumptions I make explicit.**

- Operator semantics are standard process-tree semantics: `->` sequence, `X` exclusive choice (exactly one branch), `+` parallel (each child once, any interleaving), `tau` silent/no-op, `*` loop (do, redo, implicit silent exit).
- "A model accepts a trace" means the trace is in the model's *language* (a perfect, cost-0 replay exists). Near-fit behavior under fitness thresholds is flagged separately, because threshold-based "acceptance" is where silent pathology hides.
- The probes B1–B8 are business-policy tests, not an exhaustive ground truth.
- The absence of the external/internal attribute (P3) and the affected-check attribute (P6) is a *first-class modeling constraint*, not something to be wished away by making activities optional.
- Both teams' claims of "same 80/20 split, same PM4Py installation" are unverifiable as submitted (no seeds, no preprocessing code, no cost settings, no holdout case IDs). All cross-model numbers are therefore provisional.

**Use cases I distinguish, and why the trade-off changes with each.**

| Use case | Fitness floor | Precision target | Data-awareness | Comprehension |
|---|---|---|---|---|
| Exploration / hypothesis generation | High (do not mislead analysts) | Low acceptable — false leads are cheap and filtered by humans | Nice to have | High |
| Conformance enforcement | High (false alarms destroy adoption) | Near-perfect on *curated forbidden behavior*; also high recall on ratified valid variants | **Required** (P3, P6 are data rules) | High for auditors |

4. **`Register Dossier, Supplier Review, Manager Approval, Release Order`** and **`Register Dossier, Expedite Review, Manager Approval, Release Order`.** In S's language. Reviews with no checks at all; Supplier Review on what may be an internal order. Policy-ambiguous at best (P3 is unverifiable without the attribute). *Data blindness masquerading as flexibility.*
5. **External order with a missing Supplier Review event** (F4): `Register Dossier, Credit Check, Technical Review, Manager Approval, Release Order` fits S perfectly and is indistinguishable from a genuine internal order. P3 is silently unenforceable. *Missing-event bias built into the model.*
6. **B8 — `Register Dossier, Credit Check, Technical Review, Credit Check, Manager Approval, Release Order`.** Strictly *not* in S's language (a leaf fires at most once), but it achieves near-perfect alignment fitness (one log move). Under any threshold-based acceptance S effectively passes it and is blind to P7. *Operational blindness, not language admission.*
7. **`Register Dossier, Credit Check, Technical Review, Manager Approval, Auto Approval, Release Order`.** Strictly not in the language (XOR picks one), near-fit; S cannot reliably enforce exactly-one approval under noise. Partial credit: the XOR at least makes double approval a *visible* deviation.
8. **Any trace containing `Correct Dossier`** (e.g., B6). Not in S's language — the activity is absent from its alphabet. The most policy-critical loop (P6/P7) is erased. Given S's 0.992 training fitness, either corrections are rare in training or they were filtered in preprocessing — which is undocumented. *This is the strongest single reason to distrust S's fitness number.*

**What S gets right:** Register first, Release last, exactly one approval *event* in the model, and soundness. That is roughly three of nine activities' worth of policy.

**The classification principle.** Intentional noise tolerance (arguable for Supplier Review given P3's missing attribute, defensible for logged-but-lost events) and accidental semantic flaws (mandatory checks optional; Expedite Review unanchored; corrections erased) are indistinguishable *inside* S. That is the core design error: tolerance should live in the conformance layer as classified deviations, not in the model as optionality.

# 4. Adversarial Review of Model C

**Strongest case for Model C.** C is the only submission that enforces P2's mandatoriness and P4's ordering: it rejects B3 and B4 outright, and it rejects B8 (repeated check without correction). It is sound by construction, every branch is an auditable statement of observed behavior, and its precision on training is high. F8's posture — unseen behavior is noncompliant until approved — is a legitimate, even conservative, compliance philosophy: block first, review exceptions. On a pure control-flow negative suite (see N1–N8 in §9), C passes cleanly, and no other submitted model except the third alternative does.

**Strongest case against Model C.** C is not a process model; it is a memorized variant list with a process-model costume. Its language is exactly 8 traces. The costs:

- **Unseen combinations of known behavior.** B2 (`TR, CC, Auto Approval`) and B5 (`TR, CC, SR, ER, Auto Approval`) are policy-valid and rejected — precisely the F4 holdout population, explaining the 0.744. C also encodes *spurious* constraints never stated in policy: Auto Approval appears only in branch 8, so *no* auto-approved case may have a Supplier Review or Expedite Review, and no branch allows Expedite Review before Supplier Review although P4 explicitly permits it. C has smuggled training-set correlations in as norms.
- **Correction handling.** Variant 7 is the only correction branch: exactly one correction, rechecking *both* checks, in the *fixed* order `CC, TR`. So a legitimate both-scope recheck in the other order is rejected; a TR-only recheck (B6) is rejected; multiple corrections (P6: "may repeat") are rejected; a correction after Supplier Review is rejected; a correction affecting no check is rejected. F6's complaint is structural, not cosmetic.
- **Holdout degradation (0.973  0.744).** Consistent with overfitting, but also possibly amplified by split composition or missing Supplier Review events. Without seeds, the *size* of the drop is not interpretable — only its *direction*, corroborated by B2/B5, is.
- **Precision as memorization artifact.** The variants were selected *from* the training log; precision was then measured *on* the same log. A model that is literally the training variant set trivially maximizes that number. The informative quantities — holdout precision and precision against curated negatives (F5) — were not computed.
- **Duplicate labels.** ~40 same-label transitions by naive count (the "33" is not reproducible under any obvious definition). Consequences: multiple equal-cost optimal alignments (diagnostics become solver-dependent), unstable precision, and the F3 impact-analysis burden — a single change to "Supplier Review" semantics touches four branches.
- **Maintainability.** The full policy space is (check order 2) × (SR yes/no 2) × (ER yes/no 2) × (approval type 2) × correction patterns ( 5)  80+ variants. Enumeration does not scale; each new observed pattern is a new branch (F2), and every branch is another place to introduce an inconsistency.
- **The audit-readiness claim is overstated.** C cannot enforce P3 either: an external order whose Supplier Review event is missing replays *perfectly* on branch 1, 2, or 8. No alarm. C is data-blind in exactly the way the auditor should care most about, and it rejects-then-cannot-explain the cases where it does fire (all-or-nothing conformance, no deviation classification for B6-type scope questions).

C's real value is as a **variant catalog and regression fixture** — documentation of what was observed — not as a normative model.

# 6. Likely Metric Pathologies

Ranked by likely decision impact on the governance question (which model, for which use).

1. **Positive-log-only precision (F5).** The board's central question is compliance fitness, and precision is the decision-critical metric — yet it was computed only against observed positive traces, and for C quasi-circularly against the very log from which its variants were mined. Neither 0.584 nor 0.941 says anything about the rate at which each model admits *forbidden* behavior. Until a curated negative suite exists, no enforcement decision can rest on these numbers. (Mitigation: §9.3.)
2. **Fitness rewarded by optionalizing mandatory behavior.** S's 0.992/0.981 are partly artifacts: `tau` branches make violations free silent skips, and the alphabet omits Correct Dossier. If the board reads "high fitness" as "accurately describes the controlled process," it will certify a model that cannot represent the controls at all. (Mitigation: negative suite + per-requirement checks.)
3. **Missing-event bias (F4).** Confirmed contamination: missing Supplier Review events inflate S's fitness, depress C's, and — decisively — make P3 silently unenforceable in *both* models (an external order without SR replays cleanly on C's branches 1/2/8 and on S). This pathology biases model selection toward permissive models for the wrong reason. (Mitigation: §9.4 perturbation tests + attribute join.)
4. **Train/holdout leakage and split sensitivity.** No seeds, no stratification, no published holdout IDs; whole-log preprocessing before splitting would be leakage; and if the teams' preprocessing differed (e.g., S filtering corrections), the two holdout sets are different logs and *every* cross-model comparison fails. The 0.744's magnitude is uninterpretable; only its direction, corroborated by B2/B5, is usable. (Mitigation: §9.1–9.2.)
5. **Alignment-cost sensitivity.** Move costs, silent-move pricing, and normalization are undocumented; two "same PM4Py" installations with different parameters produce incomparable fitness values. Silent moves priced at zero systematically favor permissive models. (Mitigation: §9.6 reweighting.)
6. **Duplicated-label effects.** C's ~40 same-label transitions produce alignment ties (solver-dependent diagnostics), unstable align-based precision, and F3's impact-analysis burden. The reported count of 33 is not reproducible under the obvious definition (48 labeled transitions  9 labels = 39). (Mitigation: publish definition; cap duplicates; positional disambiguation.)
7. **Generalization rewarded by excessive permissiveness / metric mislabeling.** The token-based "generalization" score measures element-usage frequency, not the ability to accept valid unseen behavior. S's 0.907 is a small-model artifact; C's 0.412 is partly a size penalty. Reading the column as "S generalizes, C doesn't" gets the semantic conclusion right for the wrong, non-evidentiary reason — the real evidence is holdout fitness plus B2/B5. (Mitigation: §9.5 unseen-combination suite.)
8. **Silent-transition effects.** S's four `tau` XORs let alignments teleport past mandatory behavior at zero cost. Silent transitions are not inherently bad (the third alternative uses them deliberately for optionality and loop exits), but they must be *inventory-controlled and documented*; in S they are the mechanism of pathology #2.
9. **Stakeholder-score methodology.** Informal 1–5, undocumented rubric and raters, likely advocacy bias; measures likability and apparent simplicity, not correctness. A "comprehensible" wrong model is a governance hazard. (Mitigation: §9.9 task-based protocol.)
10. **Runtime confounding by model size.** The 24× gap is real operationally but is not a validity signal: C's runtime is inflated by its own state space, duplicate-label tie exploration, and by the extra search its misfit causes. Report p95, same hardware, same settings. (Mitigation: §9.8.)

# 7. Third Alternative

**Design principle.** Encode the policy's *hard control structure* in a small block-structured model; encode *data-dependent and unverifiable rules* as explicit declarative checks with severity tiers (CRITICAL / SUSPICIOUS / VERIFY / UNVERIFIABLE); never buy fitness by weakening the model. Noise tolerance moves to the conformance layer, where it is visible and auditable.

### 7.1 Compact formal description

```
model_third = SEQ(
    Register,
    AND(CC, TR),                                    -- P2: both mandatory, either order
    PO{ CORR, OPT(SR), OPT(ER) },                   -- P3/P4/P6: review zone
    XOR(MA, AA),                                    -- P5: exactly one approval
    Release                                         -- P1
)
CORR  = LOOP(do = , redo = SEQ(CD, SCOPE))    (CD · SCOPE)*
SCOPE = XOR(, CC, TR, AND(CC, TR))                -- P6: re-run exactly the affected checks
OPT(a) = XOR(a, )
PO{...} = strict partial order with no edges (full concurrency)
```

Language: `Register · interleavings{CC, TR} · interleavings{ (CD·s)*, SR?, ER? } · (MA | AA) · Release`, where `s  {, CC, TR, CCTR}`.

Size: 13 visible transitions over 9 labels (4 duplicate-labeled: CC ×3, TR ×3, all positionally disambiguated — "first run" vs. "recheck"), versus C's 48 visible transitions / ~39 duplicates and S's 8 transitions / 0 policy content. Block-structured  sound by construction (verified anyway in §9).

### 7.2 Self-contained PM4Py code

```python
"""
model_third — block-structured POWL model + tiered declarative policy checks.
Requires pm4py >= 2.7.x (POWL support). An equivalent process-tree string is
given at the bottom as a cross-check / fallback.
"""
import pm4py
from pm4py.objects.powl.obj import (
    Transition, SilentTransition, OperatorPOWL, StrictPartialOrder,
)
from pm4py.objects.process_tree.obj import Operator


# ------------------------------------------------------------------ helpers
def seq(*nodes):
    """Total order: n1 -> n2 -> ... -> nk."""
    po = StrictPartialOrder(list(nodes))
    for i in range(len(nodes) - 1):
        po.order.add_edge(nodes[i], nodes[i + 1])
    return po

def xor(*nodes):
    return OperatorPOWL(Operator.XOR, list(nodes))

def opt(node):
    """Optional execution: XOR(node, tau)."""
    return xor(node, SilentTransition())

def concur(*nodes):
    """Full concurrency: strict partial order with no edges (= AND block)."""
    return StrictPartialOrder(list(nodes))

def star(repeat):
    """Zero or more repetitions: LOOP(do=tau, redo=repeat) has language repeat*."""
    return OperatorPOWL(Operator.LOOP, [SilentTransition(), repeat])


# ------------------------------------------------------------------ activities
REG  = Transition("Register Dossier")
CC_1 = Transition("Credit Check")          # first execution
TR_1 = Transition("Technical Review")      # first execution
CD   = Transition("Correct Dossier")
CC_2 = Transition("Credit Check")          # recheck scope: CC only
TR_2 = Transition("Technical Review")      # recheck scope: TR only
CC_3 = Transition("Credit Check")          # recheck scope: both
TR_3 = Transition("Technical Review")      # recheck scope: both
SR   = Transition("Supplier Review")
ER   = Transition("Expedite Review")
MA   = Transition("Manager Approval")
AA   = Transition("Auto Approval")
RO   = Transition("Release Order")


# ------------------------------------------------------------------ structure
# P2: both mandatory checks, either order, before approval.
# No tau here: skipping a mandatory check is a DEVIATION (classified CRITICAL
# downstream), never a free silent skip. This is the key safety difference from S.
mandatory_checks = concur(CC_1, TR_1)

# P6: each Correct Dossier is followed by re-execution of exactly the affected
# checks: none / CC only / TR only / both in either order. The scope is an
# explicit, readable choice point (answers F6).
recheck_scope = xor(SilentTransition(), CC_2, TR_2, concur(CC_3, TR_3))
correction_cycle = seq(CD, recheck_scope)
correction_zone = star(correction_cycle)            # (CD, scope)*  — corrections may repeat

# P3/P4: optional SR and optional ER, either order relative to each other AND
# relative to correction cycles. The whole zone sits after both mandatory
# checks and before approval.
sr_opt = opt(SR)
er_opt = opt(ER)
review_zone = StrictPartialOrder([correction_zone, sr_opt, er_opt])
# no edges: full concurrency
# Governance toggle: if the board rules that Expedite Review must follow ALL
# rechecks, uncomment the next line (one-edge change, no restructuring):
# review_zone.order.add_edge(correction_zone, er_opt)

# P5: exactly one approval (XOR without tau: zero or two approvals = deviation).
approval = xor(MA, AA)

# P1: register first, release last.
model_third = seq(REG, mandatory_checks, review_zone, approval, RO)


# ------------------------------------------------------------------ checks
net_third, im_third, fm_third = pm4py.convert_to_petri_net(model_third)
# pm4py.check_soundness(net_third, im_third, fm_third, print_diagnostics=True)
# sample = pm4py.play_out(model_third)   # eyeball / archive the language


# Equivalent process-tree string (same language; use as fallback or cross-check;
# language equivalence can be verified via play-out sampling + probe acceptance):
MODEL_THIRD_TREE = (
    "->( 'Register Dossier', "
    "  +( 'Credit Check', 'Technical Review' ), "
    "  +( "
    "    *( tau, ->( 'Correct Dossier', "
    "         X( tau, 'Credit Check', 'Technical Review', "
    "           +( 'Credit Check', 'Technical Review' ) ) ) ), "
    "    X( 'Supplier Review', tau ), "
    "    X( 'Expedite Review', tau ) "
    "  ), "
    "  X( 'Manager Approval', 'Auto Approval' ), "
    "  'Release Order' )"
)
# model_third_alt = pm4py.parse_process_tree(MODEL_THIRD_TREE)
```

### 7.3 Declarative dictionaries and external checks

```python
POLICY_RULES = {
    "R1_supplier_review_external": {
        "severity": "CRITICAL (external & no SR) / UNVERIFIABLE (attribute absent)",
        "rule": "external(order) -> Supplier Review occurs before approval",
        "attribute_needed": "case:sourcing in {external, internal}",
        "note": "if the attribute is absent the check ESCALATES; it never silently passes",
    },
    "R2_repeated_check_needs_correction": {
        "severity": "SUSPICIOUS",
        "rule": "each mandatory-check occurrence after its first must be preceded by "
                "a Correct Dossier executed after that check's previous occurrence (P7)",
        "note": "structurally excluded by the correction zone; residual log moves are "
                "classified SUSPICIOUS, not CRITICAL — P7 says suspicious, not forbidden",
    },
    "R3_recheck_scope": {
        "severity": "VERIFY",
        "rule": "after each Correct Dossier, exactly the affected checks re-run (P6)",
        "attribute_needed": "corrected_dossier.affected_checks",
        "note": "without the attribute, partial-scope and empty-scope cycles are "
                "flagged for manual confirmation, not silently accepted",
    },
    "R4_exactly_one_approval": {"severity": "CRITICAL",
                                "rule": "exactly one of {Manager, Auto} Approval (P5)"},
    "R5_expedite_window": {"severity": "CRITICAL",
                           "rule": "Expedite Review after both initial mandatory checks "
                                   "and before approval (P4)"},
    "R6_frame": {"severity": "CRITICAL",
                 "rule": "Register Dossier first, Release Order last (P1)"},
}

MANDATORY = ("Credit Check", "Technical Review")
APPROVALS = ("Manager Approval", "Auto Approval")


def post_hoc_policy_scan(activities, sourcing=None):
    """Deterministic policy scan for one case. `sourcing` is 'external'/'internal'
    or None when the attribute is missing. Returns [(rule, severity, message)]."""
    f = []
    if activities and activities[0] != "Register Dossier":
        f.append(("R6", "CRITICAL", "case does not start with Register Dossier"))
    if activities and activities[-1] != "Release Order":
        f.append(("R6", "CRITICAL", "case does not end with Release Order"))

    n_appr = sum(a in APPROVALS for a in activities)
    if n_appr != 1:
        f.append(("R4", "CRITICAL", f"expected exactly one approval, found {n_appr}"))

    # R2 (P7): repeat without intervening correction
    cd_count, cd_at_last_exec = 0, {}
    for i, a in enumerate(activities):
        if a == "Correct Dossier":
            cd_count += 1
        elif a in MANDATORY:
            if a in cd_at_last_exec and cd_at_last_exec[a] == cd_count:
                f.append(("R2", "SUSPICIOUS",
                          f"{a} at position {i} repeats without intervening Correct Dossier"))
            cd_at_last_exec[a] = cd_count

    # R5 (P4): expedite window
    pos = {a: i for i, a in enumerate(activities)}
    if "Expedite Review" in pos:
        er_i = pos["Expedite Review"]
        if not ("Credit Check" in pos and pos["Credit Check"] < er_i
                and "Technical Review" in pos and pos["Technical Review"] < er_i):
            f.append(("R5", "CRITICAL", "Expedite Review before both mandatory checks"))
        appr_i = [i for i, a in enumerate(activities) if a in APPROVALS]
        if appr_i and er_i > appr_i[0]:
            f.append(("R5", "CRITICAL", "Expedite Review after approval"))

    # R1 (P3): supplier review for external orders — data-dependent
    has_sr = "Supplier Review" in pos
    if sourcing == "external" and not has_sr:
        f.append(("R1", "CRITICAL", "external order without Supplier Review"))
    elif sourcing is None and not has_sr:
        f.append(("R1", "UNVERIFIABLE",
                  "sourcing attribute absent: SR requirement cannot be verified (never silently pass)"))

    # R3 (P6): recheck scope — needs the affected-check attribute
    scopes, current = [], None
    for a in activities:
        if a == "Correct Dossier":
            if current is not None:
                scopes.append(current)
            current = []
        elif current is not None and a in MANDATORY:
            current.append(a)
    if current is not None:
        scopes.append(current)
    for k, sc in enumerate(scopes):
        if len(sc) < 2:
            f.append(("R3", "VERIFY",
                      f"correction {k+1} re-ran {sc or 'no checks'}; "
                      f"confirm against affected-check record (attribute missing)"))
    return f


def classify_alignment(align_result):
    """Map model_third alignment moves to policy findings (skeleton).
    Model moves on mandatory activities are CRITICAL skips — the exact move
    class that Model S prices at zero via tau."""
    findings = []
    for (lg, md) in align_result["alignment"]:
        if lg == ">>" and md in ("Credit Check", "Technical Review"):
            findings.append(("R2", "CRITICAL", f"mandatory check skipped: {md}"))
        elif lg == ">>" and md in APPROVALS:
            findings.append(("R4", "CRITICAL", "missing approval"))
        elif lg == ">>" and md == "Expedite Review":
            pass  # ER is optional; skipping it is legitimate
        elif md == ">>" and lg in MANDATORY:
            findings.append(("R2", "SUSPICIOUS", f"repeated {lg}; verify preceding correction"))
        elif md == ">>" and lg in APPROVALS:
            findings.append(("R4", "CRITICAL", "second approval event"))
        elif md == ">>" and lg == "Expedite Review":
            findings.append(("R5", "CRITICAL", "Expedite Review outside allowed window"))
        elif md == ">>" and lg == "Correct Dossier":
            findings.append(("R3", "VERIFY", "Correct Dossier in unexpected position"))
    return findings
```

### 7.4 Explicit semantics

- **Mandatory checks (P2):** Credit Check and Technical Review each execute exactly once, in either interleaving, before anything in the review zone and before approval. There is no `tau` escape: a skipped check is a *model move* in the alignment and is classified CRITICAL. Missing-event robustness is handled by the classification layer (a skipped check may be downgraded to VERIFY if evidence of a logging gap exists), not by the model.
- **Optional Supplier Review (P3):** optional in *control flow only*. The normative rule "external  Supplier Review" is rule R1 with the sourcing attribute; when the attribute is missing, the outcome is UNVERIFIABLE and escalates — it never silently passes. This is deliberately *not* the Model S solution.
- **Expedite Review ordering (P4):** optional; enabled only after both initial mandatory checks (zone placement); must precede approval; unordered relative to Supplier Review (P4 explicitly allows both orders) and relative to correction cycles. The one ambiguity — whether ER must also follow *rechecks* — is documented and resolvable with a single partial-order edge (toggle in the code).
- **Exactly-one approval (P5):** XOR without `tau`. Zero or two approvals are structural deviations, classified CRITICAL (R4). Which approval fires is a data question (auto-approval thresholds), not a control-flow question; the model stays neutral, avoiding C's spurious "no SR/ER with Auto Approval" constraint.
- **Correction/recheck behavior (P6, P7):** zero or more cycles of (Correct Dossier  re-execute exactly the affected checks), with the scope an explicit four-way choice: none / CC / TR / both (either order). A repeated check without a preceding correction is *outside the language* and surfaces as a SUSPICIOUS deviation (R2) — flagged, not silently passed and not silently blocked, matching P7's wording.

**On the instruction "do not solve missing data by silently treating every activity as optional":** the only control-flow optionality in model_third is (a) Supplier Review — paired with an escalating data rule, not silent; (b) Expedite Review — genuinely optional per P4; and (c) the zero-iteration case of the correction loop — semantically correct, since not every case has a correction. Mandatory checks and approval are hard requirements.

### 7.5 Limitations caused by the missing attributes

1. **No sourcing attribute (P3):** R1 degrades to UNVERIFIABLE escalations for every SR-less case. The P3 wording ("was not included in the control-flow discovery log") suggests the attribute exists in the order system; a join is a precondition for *blocking* enforcement of P3 — for any model, including C.
2. **No affected-check attribute (P6):** R3 cannot machine-verify recheck scope; partial- and empty-scope cycles are accepted-and-flagged. B6-type cases are therefore "plausibly valid, pending confirmation," which is the honest representation of the evidence.
3. **Documented assumptions with toggles:** corrections are modeled after the initial checks (P6's wording and variant 7 support this; an early-correction block could be added if the business confirms pre-check corrections); Supplier Review is placed after the checks (all observed variants, policy silent — a partial-order relaxation is a one-line change); ER-vs-recheck order is ambiguous (toggle provided).
4. **Residual duplicate labels (4):** CC and TR each appear three times; ambiguity is positionally resolved (rechecks are reachable only after CD) and is far below C's regime.
5. **Runtime/simplicity** are expected to be in S's class given the net size, but this is an expectation to be measured in §9, not a claim.
