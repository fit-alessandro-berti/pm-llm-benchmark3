```python
O, A, C, P, R, X = ('Open Case', 'Analyze Revision', 'Check Revision',
                    'Approve Revision', 'Rework Case', 'Close Case')
MAX_VISIBLE, MAX_ANALYZE, PRINT_LIMIT = 12, 3, None   # PRINT_LIMIT=None prints all

MODEL_A = """->('Open Case',
  *(->(+('Analyze Revision','Check Revision'), X('Approve Revision', tau)), 'Rework Case'),
  'Close Case')"""
MODEL_B = """->('Open Case',
  +('Check Revision', *(->('Analyze Revision', X('Approve Revision', tau)), 'Rework Case')),
  'Close Case')"""

def load(s):
    tree = pm4py.parse_process_tree(" ".join(s.split()))
    net, im, fm = pm4py.convert_to_petri_net(tree)
    res = pm4py.check_soundness(net, im, fm, print_diagnostics=False)
    sound = res[0] if isinstance(res, tuple) else bool(res)
    return tree, net, im, fm, sound

def mkey(m):                       # hashable marking key (id is stable within one net)
    return tuple(sorted((id(p), c) for p, c in m.items() if c > 0))

def bounded_language(net, im, fm, max_visible=MAX_VISIBLE, max_analyze=MAX_ANALYZE):
    """EXHAUSTIVE bounded exploration of complete visible traces (tau hidden)."""
    sem, fkey = ClassicSemantics(), mkey(fm)
    seen = {(mkey(im), ())}
    queue, complete = deque([(im, ())]), set()
    while queue:
        m, tr = queue.popleft()
        if mkey(m) == fkey:                       # final-marking equality, explicit
            complete.add(tr)
        for t in pm4py.get_enabled_transitions(net, m):
            m2 = sem.execute(t, net, m)
            if t.label is None:                   # silent: trace unchanged
                tr2 = tr
            else:
                tr2 = tr + (t.label,)
                if len(tr2) > max_visible or tr2.count(A) > max_analyze:
                    continue                      # visible-length / analyze bound
            st = (mkey(m2), tr2)
            if st not in seen:                    # repeated (marking, trace) states cut
                seen.add(st)
                queue.append((m2, tr2))
    return complete

# ---- independent oracle from the closed-form definitions -------------------
def oracle_A():
    blocks = [order + ((P,) if ap else ()) for order in ((A, C), (C, A)) for ap in (0, 1)]
    out = set()
    for n in range(1, MAX_ANALYZE + 1):
        for combo in itertools.product(blocks, repeat=n):
            t = (O,)
            for i, b in enumerate(combo):
                t += ((R,) if i else ()) + b
            out.add(t + (X,))
    return {t for t in out if len(t) <= MAX_VISIBLE}

def oracle_B():
    units, out = [(A,), (A, P)], set()
    for n in range(1, MAX_ANALYZE + 1):
        for combo in itertools.product(units, repeat=n):
            w = ()
            for i, u in enumerate(combo):
                w += ((R,) if i else ()) + u
            for pos in range(len(w) + 1):
                out.add((O,) + w[:pos] + (C,) + w[pos:] + (X,))
    return {t for t in out if len(t) <= MAX_VISIBLE}

# ---- P1 / P2 predicates ------------------------------------------------------
def p1_compliant(t):
    segs = [[]]
    for e in t[1:-1]:
        if e == R: segs.append([])
        else: segs[-1].append(e)
    return all(s.count(A) == 1 and s.count(C) == 1 for s in segs)

def p2_abs(t):
    return (t[0] == O, t[-1] == X, t.count(A), t.count(C) > 0, t.count(R) > 0)

def show(title, traces):
    print(f"\n{title}: {len(traces)}")
    for t in sorted(traces, key=lambda x: (len(x), x))[:PRINT_LIMIT]:
        print("  ", " , ".join(t))

if __name__ == "__main__":
    tA, nA, imA, fmA, soundA = load(MODEL_A)
    tB, nB, imB, fmB, soundB = load(MODEL_B)
    print("Sound A:", soundA, " Sound B:", soundB)

    LA = bounded_language(nA, imA, fmA)       # exhaustive within the bound
    LB = bounded_language(nB, imB, fmB)
    assert LA == oracle_A() and LB == oracle_B(), "BFS disagrees with closed-form oracle"

    print(f"|L<=(A)|={len(LA)} |L<=(B)|={len(LB)} |common|={len(LA & LB)}")
    print("Bounded equal (P3):", LA == LB)
    show("Unique to A", LA - LB)
    show("Unique to B", LB - LA)

    print("\nP1: A compliant:", sum(map(p1_compliant, LA)), "/", len(LA),
          "| B compliant:", sum(map(p1_compliant, LB)), "/", len(LB))
    print("P1: (A  Comp) == (B  Comp):",
          {t for t in LA if p1_compliant(t)} == {t for t in LB if p1_compliant(t)})

    piA = {p2_abs(t) for t in LA}; piB = {p2_abs(t) for t in LB}
    print("P2 abstraction equal within bound:", piA == piB, sorted(piA))

# ---- HEURISTIC, NON-EXHAUSTIVE: random play-out ------------------------------
def heuristic_playout(tree, n=500):
    """Random sampling: can find counterexamples, can never prove equivalence;
    rare traces (e.g. 3 attempts with all approvals) may be missed."""
    log = pm4py.play_out(tree, parameters={"num_traces": n})
    if hasattr(log, "groupby"):   # DataFrame return in some versions
        return {tuple(g["concept:name"]) for _, g in log.groupby("case:concept:name")}
    return {tuple(e["concept:name"] for e in tr) for tr in log}
```

# 6. Stochastic and Footprint Comparison

**Footprints (`pm4py.discover_footprints`).** Footprints record the directly-follows relation, classified as causal or parallel, plus start and end activities and minimum trace length. They can refute equivalence but cannot establish it.

- **Difference including P.** At language level the pair (P, C) is causal "CP" in A, since P never directly precedes C. In B both "C P" (T0-style with P after C) and "P C" (T5) occur, so the pair is parallel. A correct footprint comparison should therefore differ on (P, C). This is hand-derived, and pm4py's tree-based footprint may approximate it.
- **Footprints agree after hiding P, yet the languages differ.** Project P away (the P2 view). Then both models give these directly-follows pairs:
  - OA, OC, AC, CA
  - AR, RA, CR, RC
  - AX, CX

  Both have start set {O}, end set {X}, and minimum length 4. The footprints are identical, but T1 (two checks) and T4 (one check) separate the languages. Footprints cannot count occurrences of C against A.

```python
tA, tB = pm4py.parse_process_tree(" ".join(MODEL_A.split())), pm4py.parse_process_tree(" ".join(MODEL_B.split()))
fpA, fpB = pm4py.discover_footprints(tA), pm4py.discover_footprints(tB)
def norm(v): return {tuple(x) if isinstance(x, (list, tuple)) else x for x in v}
for key in ("sequence", "parallel", "start_activities", "end_activities", "activities"):
    a, b = norm(fpA.get(key, [])), norm(fpB.get(key, []))
    print(key, "equal" if a == b else f"differ: only A={a-b}, only B={b-a}")
```

**Stochastic language and EMD.**
- `pm4py.get_stochastic_language` returns {trace: probability} for a log, or for a net using transition weights.
- I do not rely on pm4py's default weights. I **declare** the following distributions and truncate to n  3 and  12 events.
  - Loop continuation probability q = 0.4, so P(n)  (1q)q¹.
  - Approve probability 0.5 per attempt.
  - In A, each block order is 50/50.
  - In B, the position of C is uniform over the insertion points.
- The probabilities are assumptions, not data. The result depends entirely on them.

```python
# run after the Section 5 script (reuses O,A,C,P,R,X)
Q, PAPP = 0.4, 0.5
def dist_A():
    d = {}
    for n in range(1, 4):
        for combo in itertools.product(list(itertools.product((0, 1), repeat=2)), repeat=n):
            # each block: (order_flag, approve_flag)
            pr, t = (1 - Q) * Q ** (n - 1), (O,)
            for i, (o, ap) in enumerate(combo):
                t += ((R,) if i else ()) + ((A, C) if o == 0 else (C, A)) + ((P,) if ap else ())
                pr *= 0.5 * (PAPP if ap else 1 - PAPP)
            if len(t) + 1 <= MAX_VISIBLE: d[t + (X,)] = d.get(t + (X,), 0) + pr
    z = sum(d.values()); return {k: v / z for k, v in d.items()}
def dist_B():
    d = {}
    for n in range(1, 4):
        for aps in itertools.product((0, 1), repeat=n):
            w, pr = (), (1 - Q) * Q ** (n - 1)
            for i, ap in enumerate(aps):
                w += ((R,) if i else ()) + (A,) + ((P,) if ap else ())
                pr *= PAPP if ap else 1 - PAPP
            for pos in range(len(w) + 1):
                t = (O,) + w[:pos] + (C,) + w[pos:] + (X,)
                if len(t) <= MAX_VISIBLE: d[t] = d.get(t, 0) + pr / (len(w) + 1)
    z = sum(d.values()); return {k: v / z for k, v in d.items()}
emd = pm4py.compute_emd(dist_A(), dist_B())     # Levenshtein ground distance
print("EMD under declared weights:", emd)
# Sampled alternative (noisy): pm4py.get_stochastic_language(play_out_log)
```

I report no numeric EMD because I did not run it.

**Why these never prove trace equivalence.**
- **Footprints** abstract to pairwise relations and lose counts. Equal footprints (after hiding P) coexist with unequal languages (T1, T4).
- **A low or even zero EMD** depends on invented probabilities, on truncation and renormalisation, and on a Levenshtein ground distance. The counterexamples differ by one edit (one missing C), so they sit close in that metric. A trace with probability 0 under the declared weights, or a rare trace in a sampled log, is invisible to the measure. Only a trace-level inclusion check (Section 4) can prove inequivalence, and only a proof over the unbounded languages could establish equivalence.

```python
from collections import Counter
def tree_stats(t):
    st = {"nodes": 0, "ops": 0, "vis": 0, "sil": 0, "depth": 0}
    labels = Counter()
    def rec(n, d):
        st["nodes"] += 1; st["depth"] = max(st["depth"], d)
        if n.children:
            st["ops"] += 1
            for c in n.children: rec(c, d + 1)
        elif n.label is None: st["sil"] += 1
        else: st["vis"] += 1; labels[n.label] += 1
    rec(t, 0)
    st["dup_labels"] = sum(c - 1 for c in labels.values() if c > 1)
    return st

for name, s in (("A", MODEL_A), ("B", MODEL_B)):
    tree, net, im, fm, sound = load(s)
    print(name, tree_stats(tree),
          "places/trans/arcs:", len(net.places), len(net.transitions), len(net.arcs),
          "simplicity:", pm4py.simplicity_petri_net(net, im, fm))
```

# 8. Simplified Model

**Target: P2 (executive dashboard).**
- For P1 there is no smaller exact model. A already has 5 operators and a single , and optional approval needs a  or skip. P1 should therefore keep A unchanged.
- P3 requires visible P, so any model that drops P loses the bounded-language match.
- P2 is the only view where a real reduction is legitimate.

**Simplified model S:**
```
->('Open Case', +('Check Revision', *('Analyze Revision','Rework Case')), 'Close Case')
```
S is B with the approval XOR removed, which is B projected away from P. Its structure compared with both originals:

| Metric | S | A | B |
|---|---|---|---|
| Nodes | 8 | 12 | 12 |
| Operators | 3 | 5 | 5 |
| Visible leaves | 5 | 6 | 6 |
| Silent leaves | 0 | 1 | 1 |
| Depth (edges) | 3 | 4 | 5 |

```python
import pm4py
from collections import deque
from pm4py.objects.petri_net.semantics import ClassicSemantics

simplified_model = pm4py.parse_process_tree(
    "->('Open Case', +('Check Revision', *('Analyze Revision','Rework Case')), 'Close Case')")
net_s, im_s, fm_s = pm4py.convert_to_petri_net(simplified_model)
res = pm4py.check_soundness(net_s, im_s, fm_s, print_diagnostics=False)
print("S sound:", res[0] if isinstance(res, tuple) else res)

def _k(m): return tuple(sorted((id(p), c) for p, c in m.items() if c > 0))

def accepts(net, im, fm, trace):            # exact complete-trace membership test
    sem, fk, seen, stack = ClassicSemantics(), _k(fm), set(), [(im, 0)]
    while stack:
        m, i = stack.pop()
        if (_k(m), i) in seen: continue
        seen.add((_k(m), i))
        if i == len(trace) and _k(m) == fk: return True
        for t in pm4py.get_enabled_transitions(net, m):
            if t.label is None: stack.append((sem.execute(t, net, m), i))
            elif i < len(trace) and t.label == trace[i]:
                stack.append((sem.execute(t, net, m), i + 1))
    return False

O, A, C, R, X = 'Open Case', 'Analyze Revision', 'Check Revision', 'Rework Case', 'Close Case'
positive = [(O, A, C, X),
            (O, C, A, R, A, X),
            (O, A, R, C, A, X),
            (O, A, R, A, R, A, C, X)]
negative = [(O, A, X),                    # no check
            (O, C, X),                    # no analysis
            (A, O, C, X),                 # Open not first
            (O, A, C, 'Approve Revision', X),   # Approve is outside S's alphabet
            (O, A, C, R, A, C, X)]        # raw A trace with two checks (needs preprocessing)
for t in positive: assert accepts(net_s, im_s, fm_s, t), t
for t in negative: assert not accepts(net_s, im_s, fm_s, t), t

# Bounded pi-equivalence check against the originals (reuses Section 5 code if desired):
# {p2_abs(t) for t in bounded_language(S)} == {p2_abs(t) for t in LA} == {... LB}
# Expected: {(True,True,1,True,False), (True,True,2,True,True), (True,True,3,True,True)}
```

**Preservation table (relative to A and B, under the P2 observation ).**

| Property | Status in S | Note |
|---|---|---|
| O first, X last | exact | |
| At least one A, number n of A | exact | The loop gives every n  1. |
| Any Rework occurred | exact | R occurs iff n  2. |
| At least one C occurred | abstracted | The Boolean is exact, but S has exactly one C. |
| Number of checks per attempt | lost | S accepts n = 2 with one check, which violates P1. |
| Check-vs-analysis interleaving | abstracted | Hidden by P2. S allows C anywhere. |
| P (Approve) and  | lost (removed) | P2 hides them. P1 and P3 need them. |
| P1 compliance | lost | S does not enforce "own check per attempt." |
| P3 bounded language equality with A or B | lost | The alphabets differ. S equals B only after projecting P away. |
| Unbounded iteration | exact | |

**Regression traces (S accepts):**
- O, A, C, X
- O, C, A, R, A, X
- O, A, R, C, A, X
- O, A, R, A, R, A, C, X

**Negative traces (S rejects):**
- O, A, X (no check)
- O, C, X (no analysis)
- A, O, C, X (Open not first)
- O, A, C, P, X (P outside S's alphabet)
- O, A, C, R, A, C, X (a raw A trace with two checks, which is out of S's alphabet-and-count contract)

The last negative means S is a dashboard specification, not a conformance model for raw logs. Logs need the P2 preprocessing (drop P, collapse C to one occurrence) first. An even smaller alternative is `->(O, *(A,R), C, X)` (7 nodes, no concurrency). It is also -equivalent, but it rejects the valid ordering "C before A".

# 9. Final Recommendation

- **P1 (operational compliance):** Use **Model A**. Every trace has one check per analysis attempt, and P is placed after the check. B is unsuitable because it admits attempts with no check (T4) and permits "A P C" (T5).
- **P2 (dashboard):** A and B are **equivalent only under the stated **. Either can feed the dashboard, and S is the smallest faithful specification. S should replace A and B **only** for dashboard use, and should never be used to validate compliance.
- **P3 (bounded testing):** A and B are **not equivalent**. The witnesses T1 to T6 lie inside the bound. Expect |L(A)| = 76, |L(B)| = 85 and 4 shared traces.
- **Replacement:** S must not replace A for compliance checking, and P1 has no smaller exact model.

**Limits of bounded comparison.**
- A bounded match, had one occurred, would be evidence only. It says nothing about traces with more than three analyses or more than 12 events. For example, it would not cover the A-traces excluded by the length bound (n = 3 with three approvals).
- Bounded search gives a conclusive result only in the negative direction, because each witness is a real trace.
- The proofs here rest on the invariants #C = #A for A and #C = 1 for B.
- Random play-out, footprints and EMD can refute equivalence or corroborate -level equivalence. None of them proves trace equivalence.
