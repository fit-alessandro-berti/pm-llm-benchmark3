import os

# Source POWL model (provided)
register = Transition(label="Register Change")
assess_risk = Transition(label="Assess Risk")
prepare_evidence = Transition(label="Prepare Evidence")

legal_review = Transition(label="Legal Review")
skip_legal = SilentTransition()
optional_legal = OperatorPOWL(
    operator=Operator.XOR,
    children=[legal_review, skip_legal],
)

configure_trial = Transition(label="Configure Trial")
inspect_trial = Transition(label="Inspect Trial")
trial_body = StrictPartialOrder(nodes=[configure_trial, inspect_trial])
trial_body.order.add_edge(configure_trial, inspect_trial)

correct_defect = Transition(label="Correct Defect")
verification_loop = OperatorPOWL(
    operator=Operator.LOOP,
    children=[trial_body, correct_defect],
)

release_change = Transition(label="Release Change")
archive_change = Transition(label="Archive Change")

source_powl = StrictPartialOrder(
    nodes=[
        register,
        assess_risk,
        prepare_evidence,
        optional_legal,
        verification_loop,
        release_change,
        archive_change,
    ]
)

source_powl.order.add_edge(register, assess_risk)
source_powl.order.add_edge(register, prepare_evidence)
source_powl.order.add_edge(assess_risk, optional_legal)
source_powl.order.add_edge(assess_risk, verification_loop)
source_powl.order.add_edge(prepare_evidence, verification_loop)
source_powl.order.add_edge(optional_legal, release_change)
source_powl.order.add_edge(verification_loop, release_change)
source_powl.order.add_edge(release_change, archive_change)

# Conversion Pipeline 1: POWL  Petri  BPMN
print("=== Pipeline 1: POWL  Petri  BPMN ===")
try:
    net_from_powl, im_from_powl, fm_from_powl = pm4py.convert_to_petri_net(source_powl)
    print(" Petri net conversion successful")

    # Check workflow net properties

    print(f" Workflow net: {is_wf_net}")

    # Check soundness
    soundness = pm4py.check_soundness(net_from_powl, im_from_powl, fm_from_powl, print_diagnostics=True)
    print(f" Soundness: {soundness}")

    # Convert to BPMN
    bpmn_from_net = pm4py.convert_to_bpmn(net_from_powl, im_from_powl, fm_from_powl)
    print(" BPMN conversion successful")

    # Save visualizations
    os.makedirs("output", exist_ok=True)
    pm4py.save_vis_petri_net(net_from_powl, im_from_powl, fm_from_powl, "output/powl_to_petri.png")
    pm4py.save_vis_bpmn(bpmn_from_net, "output/petri_to_bpmn.png")

except Exception as e:

# Conversion Pipeline 2: POWL  Petri  Process Tree  BPMN

def compare_nets(net1, im1, fm1, net2, im2, fm2, max_visible_length=10, max_iterations=5):
    """
    Compare two accepting Petri nets using bounded trace exploration.
    Returns True if all bounded traces match exactly.
    """
    print("\n=== Round-Trip Validation ===")

    # Check basic properties
    if not pm4py.check_is_workflow_net(net1) or not pm4py.check_is_workflow_net(net2):
        print(" Not workflow nets")
        return False

    sound1 = pm4py.check_soundness(net1, im1, fm1)
    sound2 = pm4py.check_soundness(net2, im2, fm2)

    if not sound1 or not sound2:
        print(" Unsound nets detected")
        return False

    # Get footprints for structural comparison
    fp1 = pm4py.discover_footprints(net1, im1, fm1)
    fp2 = pm4py.discover_footprints(net2, im1, fm2)

    if fp1 != fp2:
        print(" Footprints differ")
        return False

    # Bounded trace exploration
    def explore_traces(net, im, fm, max_len, max_iter):
        traces = set()
        current = [(im, [])]

        for _ in range(max_iter):
            next_gen = []
            for marking, path in current:
                if marking == fm:
                    traces.add(tuple(path))
                    continue

                enabled = pm4py.get_enabled_transitions(net, marking)
                for trans in enabled:
                    if trans.label is None:
                        continue  # Skip silent transitions

                    new_marking = pm4py.petri_semantics(net, marking, trans)
                    new_path = path + [trans.label]
                    if len(new_path) <= max_len:
                        next_gen.append((new_marking, new_path))

            if not next_gen:
                break
            current = next_gen

        return traces

    traces1 = explore_traces(net_from_powl, im_from_powl, fm_from_powl, max_visible_length, max_iterations)

    if traces1 != traces2:
        print(" Trace sets differ")
        print(f"Original traces: {traces1}")
        print(f"Round-trip traces: {traces2}")
        return False

    print(" All bounded traces match")
    return True

# 6. Visual-Semantics Review

**Potential Misinterpretations in BPMN Rendering:**

1. **Parallel Gateways**: The unordered relationship between Assess Risk/Prepare Evidence and Legal Review/Verification may be rendered as parallel gateways, suggesting they can execute simultaneously when the source only allows arbitrary interleaving.

2. **Exclusive Gateways**: The optional Legal Review (XOR with silent transition) might be visualized as an exclusive gateway with a default path, potentially obscuring the true optionality.

3. **Loop Visualization**: The verification loop might be shown as a simple loop gateway rather than a more complex structure that properly represents the Correct Defect activity between iterations.

4. **Silent Routing Nodes**: The silent transition in the Legal Review branch might be omitted or shown as a regular activity, losing the semantic distinction between optional and mandatory paths.

5. **Graph Layout**: The spatial arrangement might suggest false dependencies (e.g., placing Legal Review before verification when they are actually unordered).

**Proposed Annotations:**
- Add explicit labels to gateways indicating "unordered" where true concurrency exists
- Use dashed lines or color coding to distinguish silent transitions
- Include a legend explaining loop semantics and optional paths
- Add textual annotations near the Release Change activity explaining the synchronization requirement
- Document that the BPMN visualization is a behavioral approximation and not a direct semantic mapping

# 7. Rejection Criteria

A conversion result should be rejected or quarantined if any of the following conditions are met:

1. **Unsoundness**: The Petri net fails soundness checks (no proper completion, deadlocks, or improper termination)
2. **Lost Optionality**: The optional Legal Review path is collapsed into a mandatory path
3. **Premature Release**: The Release Change activity can fire before both the verification loop and optional legal review complete
4. **Missing Verification**: The model allows traces without any verification iterations
5. **False Ordering**: The BPMN visualization shows artificial ordering between unordered activities (Assess Risk/Prepare Evidence or Legal Review/Verification)
6. **Silent Transition Loss**: Silent transitions are converted to visible activities or omitted entirely
7. **Final Marking Violation**: The final marking can be reached with incomplete execution (e.g., archive before release)
8. **Label Collisions**: Activity labels are not preserved or become ambiguous in the target model
9. **Loop Semantics Violation**: The verification loop doesn't enforce ConfigureInspectCorrect Defect ordering
10. **Initial Marking Issues**: The initial marking doesn't properly isolate the Register Change activity
