Function normalize_log(log, map_labels=True, remove_queue=True):
    new_log = EventLog()
    for i, trace in enumerate(log):
        new_trace = Trace()
        new_trace.attributes["concept:name"] = trace.attributes.get("concept:name", str(i))
        for event in trace:
            act = event["concept:name"]
            if map_labels and act == "Risk Screen":
                act = "Classify"
            if remove_queue and act == "Queue Routed":
                continue
            new_event = Event()
            for k,v in event.items(): new_event[k]=v
            new_event["concept:name"] = act
            new_trace.append(new_event)
        if len(new_trace)>0:
            new_log.append(new_trace)
    return new_log

```
def normalize_log(log, map_risk_screen=True, remove_queue_routed=True):
    new_log = EventLog()
    for i, trace in enumerate(log):
        new_trace = Trace()
        new_trace.attributes["concept:name"] = trace.attributes.get("concept:name", f"case_{i}")
        for event in trace:
            act = event["concept:name"]
            if map_risk_screen and act == "Risk Screen":
                act = "Classify"
            if remove_queue_routed and act == "Queue Routed":
                continue
            new_event = Event()
            for k, v in event.items():
                new_event[k] = v
            new_event["concept:name"] = act
            new_trace.append(new_event)
        new_log.append(new_trace)
    return new_log
```

```
def order_by_source_sequence(log, seq_key="source:sequence"):
    new_log = EventLog()
    for i, trace in enumerate(log):
        new_trace = Trace()
        new_trace.attributes.update(trace.attributes)
        events = list(trace)
        if events and all(seq_key in e for e in events):
            events.sort(key=lambda e: e[seq_key])
            for e in events: new_trace.append(e)
        else:
            for e in trace: new_trace.append(e)
        new_log.append(new_trace)
    return new_log
```

```
def alignment_diagnostics(log, net, im, fm):
    aligned = alignments.apply_log(log, net, im, fm)
    log_moves = model_moves = sync_moves = 0
    costs = []
    for a in aligned:

            if op == "log": log_moves += 1
```

```python
def copy_event(event):

    for k, v in event.items():
        new_event[k] = v
    return new_event


def normalize_log(log, map_risk_screen=True, remove_queue_routed=True):
    """Transparent normalization: map label and optionally remove technical event."""
    new_log = EventLog()
    for i, trace in enumerate(log):
        new_trace = Trace()
        new_trace.attributes["concept:name"] = trace.attributes.get("concept:name", f"case_{i}")
        for event in trace:
            activity = event["concept:name"]

            if map_risk_screen and activity == "Risk Screen":
                activity = "Classify"

            if remove_queue_routed and activity == "Queue Routed":
                continue

            new_event = copy_event(event)
            new_event["concept:name"] = activity
            new_trace.append(new_event)

        if len(new_trace) > 0:
            new_log.append(new_trace)
    return new_log


def order_by_source_sequence(log, seq_key="source:sequence"):
    """
    Reorder events when source sequence is present. If absent, returns the log
    unchanged but makes the dependency explicit.
    """
    new_log = EventLog()
    for i, trace in enumerate(log):
        new_trace = Trace()
        for k, v in trace.attributes.items():
            new_trace.attributes[k] = v

        events = list(trace)
        if events and all(seq_key in ev for ev in events):
            events = sorted(events, key=lambda ev: ev[seq_key])

        for ev in events:
            new_trace.append(ev)
        new_log.append(new_trace)
    return new_log


def dfg_summary(log, top_n=8):
    dfg, start_activities, end_activities = pm4py.discover_dfg(log)
    return Counter(dfg).most_common(top_n), start_activities, end_activities


def alignment_diagnostics(log, net, im, fm):
    aligned = alignment_algorithm.apply_log(log, net, im, fm)
    log_moves = 0
    model_moves = 0
    sync_moves = 0
    costs = []

    for aligned_trace in aligned:
        costs.append(aligned_trace.get("cost", 0))
        for move in aligned_trace.get("alignment", []):
            # PM4Py move tuples start with 'sync', 'log', or 'model'.
            op = move[0] if isinstance(move, tuple) and len(move) > 0 else None
            if op == "log":
                log_moves += 1
            elif op == "model":
                model_moves += 1
            elif op == "sync":
                sync_moves += 1

    return {
        "avg_alignment_cost": (sum(costs) / len(costs)) if costs else None,
        "log_moves": log_moves,
        "model_moves": model_moves,
        "sync_moves": sync_moves,
        "num_cases": len(aligned),
    }


def alignment_fitness_safe(log, net, im, fm):
    try:
        return pm4py.fitness_alignments(log, net, im, fm)
    except Exception as e:
        print(f"[WARN] pm4py.fitness_alignments failed: {e}")
        try:
            from pm4py.algo.evaluation.replay_fitness import algorithm as replay_fitness
            return replay_fitness.apply(log, net, im, fm)
        except Exception as e2:
            print(f"[WARN] fallback fitness failed: {e2}")
            return None


def precision_alignments_safe(log, net, im, fm):
    try:
        return pm4py.precision_alignments(log, net, im, fm)
    except Exception as e:
        print(f"[WARN] top-level precision_alignments failed: {e}")
        try:
            from pm4py.algo.evaluation.precision import algorithm as precision_eval
            return precision_eval.apply(log, net, im, fm)
        except Exception as e2:
            print(f"[WARN] fallback precision failed: {e2}")
            return None


def make_leaf(label):
    return ProcessTree(label=label)


def make_operator(op, children):
    tree = ProcessTree(operator=op)
    tree.children = children
    for child in children:
        try:
            child.parent = tree
        except Exception:
            pass
    return tree


def build_optional_signoff_tree():
    """
    Candidate tree:
    ->(
        'Receive',
        'Classify',
        X( *('Standard Review','Request Info'),
           ->('Enhanced Review','Compliance Check') ),
        X(tau, 'Supervisor Sign-off'),
        'Decision',
        'Notify'
      )

    The empty ProcessTree() represents tau/silent behavior.
    This permits sign-off after either review path, but it overgeneralizes:
    it does not enforce the EUR 50,000 condition.
    """
    optional_signoff = make_operator(
        ProcessTreeOperator.XOR,
        [ProcessTree(), make_leaf("Supervisor Sign-off")]
    )

    review_choice = make_operator(
        ProcessTreeOperator.XOR,
        [
            make_operator(
                ProcessTreeOperator.LOOP,
                [make_leaf("Standard Review"), make_leaf("Request Info")]
            ),
            make_operator(
                ProcessTreeOperator.SEQUENCE,
                [make_leaf("Enhanced Review"), make_leaf("Compliance Check")]
            ),
        ],
    )

    root = make_operator(
        ProcessTreeOperator.SEQUENCE,
        [
            make_leaf("Receive"),
            make_leaf("Classify"),
            review_choice,
            optional_signoff,
            make_leaf("Decision"),
            make_leaf("Notify"),
        ],
    )
    return root
```

## 7. Provisional conclusion

The primary explanation for the measured conformance collapse is a compound measurement and model-governance failure. The Jun-Jul drop is most consistent with the UI label change from `Classify` to `Risk Screen`, the introduction of the technical event `Queue Routed`, and the ETL change that used warehouse-ingestion time instead of source sequence for `Decision` and `Notify`. The staged improvements after label mapping, technical-event removal, and source ordering are unusually diagnostic: each repair removes a different class of alignment moves. Aug-Sep adds a real policy-driven change, `Supervisor Sign-off` for high-value claims, but the normative model was not updated; optional sign-off improves fitness yet overgeneralizes because it ignores the EUR 50,000 condition. Confidence that recording/model artifacts explain most of the *measured* conformance drop is high, roughly 90-95%.

A separate question is whether a real compliance problem remains. It may. The 25 Jun-Jul and 30 Aug-Sep enhanced-review traces without a recorded compliance check have not been audited against source evidence. They could be true bypasses, missing events, or model/extraction defects. The stable appeal and approval rates are reassuring but weak because incident detection is explicitly incomplete and the high-value attribute was omitted from the aggregate alignment. Confidence that a material, confirmed compliance problem exists is currently moderate-to-low, around 30-45%; confidence that it cannot yet be ruled out is high.

Immediate actions: IT should implement source-sequence ordering and event-semantic mappings, provide high-value attributes, and produce an event dictionary distinguishing business from technical events. Process owners should version the normative model and add a data-aware sign-off rule for high-value claims only. Compliance should audit missing compliance checks, low-value sign-offs, and `Manual Override` cases, with reviewer-level analysis. No one should claim employees are noncompliant, that the optional sign-off model proves conformity, that stable outcomes prove no hidden issue, or that `Queue Routed`/`Manual Override` are purely technical until provenance is confirmed.

## 8. Model-governance recommendation

1. **Version models and label mappings as governed artifacts.** Every normative model, process-tree version, label crosswalk, and preprocessing rule must have an owner, effective date, version hash, and approval record; conformance reports must state exactly which model and mapping were used.

2. **Create event-semantic contracts for every activity label.** Each business event must include a stable semantic identifier, human label, source system, UI/version applicability, definition, allowed predecessors/successors, and approval status; renames such as `Classify` to `Risk Screen` must update the contract before release.

3. **Classify technical events explicitly and enforce their handling.** Gateway, queue, monitoring, extraction, and system-generated events must be tagged as technical, excluded or handled separately by conformance pipelines, and distinguished from real user actions such as `Manual Override`.

4. **Preserve and use timestamp provenance and source sequence.** Store source-system event time, ingestion time, sequence number, and clock assumptions; conformance and ordering-sensitive analyses must prefer validated source sequence over warehouse-ingestion time when ordering is disputed.

5. **Model data-aware constraints alongside control flow.** High-value thresholds, resource authorizations, mandatory sign-offs, and other policy conditions must be encoded as model guards/variants or separately tested constraints, with stratified reporting by attributes such as high-value status.

6. **Run post-release validation and monitor model fit, precision, and data quality.** After UI, gateway, ETL, or policy changes, compare raw and normalized conformance, check variant/DFG shifts, examine alignment moves, verify missing-resource and missing-event rates, and require manual provenance checks before attributing deviations to employee behavior.
