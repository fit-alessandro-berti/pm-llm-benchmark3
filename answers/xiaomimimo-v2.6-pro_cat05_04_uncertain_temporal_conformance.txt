```python
def reading_interval(local: datetime, rounding: str, meas: float) -> Tuple[datetime, datetime]:
    if rounding == "exact":
        return local - timedelta(seconds=meas), local + timedelta(seconds=meas)
    if rounding == "floor60":
        m = local.replace(second=0, microsecond=0)
        return m, m + timedelta(minutes=1)                  # half-open
    if rounding == "round10":
        s = int(round(local.second / 10.0) * 10) % 60
        t = local.replace(second=s, microsecond=0)
        return t - timedelta(seconds=5), t + timedelta(seconds=5)
    if rounding == "round60":
        m = local.replace(second=0, microsecond=0)
        return m - timedelta(seconds=30), m + timedelta(seconds=30)
    raise ValueError(rounding)
```

def local_to_utc_alternatives(lo: datetime, hi: datetime, tz: ZoneInfo) -> List[Tuple[Interval, int]]:
    """Wall-clock local interval [lo, hi)  ->  list of (utc interval, fold)."""
    out = []

            if ub > ua:
