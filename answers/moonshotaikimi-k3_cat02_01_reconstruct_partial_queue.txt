Status-aware parse:

```python
def parse_cell(x):
    """Return (Timestamp|None, status). MISSING/OPEN/blank are distinct statuses, never silently NaT-equated."""
    x = x.strip()
    if x == "":
        return None, "absent"      # genuinely no such event (e.g., S2 never started)
    if x == "MISSING":
        return None, "missing"     # event happened, timestamp not recorded
    if x == "OPEN":
        return None, "open"        # event started, not completed by 14:00
    return pd.Timestamp(f"{DAY} {x}"), "recorded"
```

def feasible_start_bounds(a, acts):

def impute(acts, mode):

def snapshots(a, marks):
    ev = build_events(a)  # (time, prio, station, dq, ds)
    ev.sort(key=lambda e: (e[0], e[1]))

# Precision-repair two-station process: queue reconstruction
# Requires: pandas, numpy, pm4py (>=2.7 for the case-level insert_* helpers)

import io
import numpy as np
import pandas as pd
import pm4py

DAY = "2026-05-12"
W0, W1 = pd.Timestamp(DAY + " 08:00"), pd.Timestamp(DAY + " 14:00")
OUTAGE_C2 = (pd.Timestamp(DAY + " 10:55"), pd.Timestamp(DAY + " 11:10"))
GRID = pd.Timedelta(minutes=5)      # timestamp rounding grid -> durations carry +/-5 min
ROUND_TOL = pd.Timedelta(minutes=5) # worst-case error of a difference of two rounded stamps

RAW = """..."""  # as above
cases = pd.read_csv(io.StringIO(RAW), dtype=str, keep_default_na=False)

def parse_cell(x):
    """Status-aware parse: 'MISSING' (occurred, no start logged), 'OPEN' (started, no end
    by 14:00) and '' (event has not occurred / queueing) are three DIFFERENT statuses."""
    x = x.strip()
    if x == "":        return None, "absent"
    if x == "MISSING": return None, "missing"
    if x == "OPEN":    return None, "open"
    return pd.Timestamp(f"{DAY} {x}"), "recorded"

# ---- long (activity-instance) table; one row per station visit ----
recs = []
for _, r in cases.iterrows():
    arr, _ = parse_cell(r["arrival_s1"])
    s1s, s1ss = parse_cell(r["s1_start"]); s1e, s1es = parse_cell(r["s1_end"])
    s2s, s2ss = parse_cell(r["s2_start"]); s2e, s2es = parse_cell(r["s2_end"])
    recs.append(dict(case=r["case"], station="S1", act="S1: Incoming Inspection",
                     resource=r["s1_resource"] or None, queue_entry=arr,
                     start=s1s, start_status=s1ss, end=s1e, end_status=s1es))
    recs.append(dict(case=r["case"], station="S2", act="S2: Calibration",
                     resource=r["s2_resource"] or None, queue_entry=s1e,   # enters S2 queue at S1 completion
                     start=s2s, start_status=s2ss, end=s2e, end_status=s2es))
act = pd.DataFrame(recs)

def feasible_start_bounds(row, acts):
    """[lo, hi] for a MISSING start: queue entry, same-resource predecessor end,
    C2 outage, and nonnegative service (start <= own end)."""
    lo = row["queue_entry"]
    prev = acts[(acts["resource"] == row["resource"]) & (acts["case"] != row["case"])
                & acts["end"].notna() & (acts["end"] <= row["end"])]["end"]
    if len(prev):
        lo = max(lo, prev.max()) if lo is not None else prev.max()
    if row["resource"] == "C2" and lo is not None and OUTAGE_C2[0] <= lo < OUTAGE_C2[1]:
        lo = OUTAGE_C2[1]
    return lo, row["end"]   # hi = completion (service >= 0)

def impute(acts, mode):
    """mode: 'early' (work-conserving FIFO, our central), 'mid' (grid midpoint), 'late' (degenerate latest)."""
    a = acts.copy()
    for i, r in a[a["start_status"] == "missing"].iterrows():
        lo, hi = feasible_start_bounds(r, acts)
        s = {"early": lo, "late": hi}.get(mode)
        if mode == "mid":
            s = (lo + (hi - lo) / 2).round("5min")
        a.loc[i, ["start", "start_lo", "start_hi"]] = [s, lo, hi]
        a.loc[i, "start_status"] = f"imputed_{mode}"
    return a

def enrich(a):
    a = a.copy()
    # censored OPEN ends contribute elapsed service up to the window edge (lower bound)
    a["end_eff"] = a["end"].where(a["end_status"] != "open", W1)
    a["censored_end"] = a["end_status"] == "open"
    a["wait_min"] = (a["start"] - a["queue_entry"]).dt.total_seconds() / 60
    a["svc_min"] = (a["end_eff"] - a["start"]).dt.total_seconds() / 60
    return a

# ---- event sweep: queue/WIP curves; tie convention: service-end (1) < start (2) < queue-entry (3) ----
def sweep(a):
    ev = []
    for _, r in a.iterrows():
        if r["queue_entry"] is not None and r["start"] is None:
            ev.append((r["queue_entry"], 3, r["station"], 1, 0))      # still waiting: entry only
        if r["start"] is not None:
            ev += [(r["queue_entry"], 3, r["station"], 1, 0),
                   (r["start"], 2, r["station"], -1, 1),
                   (r["end_eff"], 1, r["station"], 0, -1)]
    ev.sort(key=lambda e: (e[0], e[1]))
    return ev

def station_curves(a, step_min=30):
    ev = sweep(a)
    marks = pd.date_range(W0, W1, freq=f"{step_min}min")
    q = {"S1": 0, "S2": 0}; s = {"S1": 0, "S2": 0}
    integ = {"S1q": 0.0, "S1s": 0.0, "S2q": 0.0, "S2s": 0.0}
    rows, i, prev_t = [], 0, W0
    peaks = {"q1": 0, "q2": 0, "wip": 0}
    # integrate continuously first
    t_last, qL, sL = W0, {"S1":0,"S2":0}, {"S1":0,"S2":0}
    for (t, _, st, dq, ds) in ev:
        d = (t - t_last).total_seconds()/60
        for k, qq, ss in (("S1", qL, sL),): pass
        integ["S1q"] += d*qL["S1"]; integ["S1s"] += d*sL["S1"]
        integ["S2q"] += d*qL["S2"]; integ["S2s"] += d*sL["S2"]
        qL[st]+=dq; sL[st]+=ds; t_last=t
        peaks["q1"]=max(peaks["q1"],qL["S1"]); peaks["q2"]=max(peaks["q2"],qL["S2"])
        peaks["wip"]=max(peaks["wip"], qL["S1"]+sL["S1"]+qL["S2"]+sL["S2"])
    d=(W1-t_last).total_seconds()/60
    integ["S1q"]+=d*qL["S1"]; integ["S1s"]+=d*sL["S1"]; integ["S2q"]+=d*qL["S2"]; integ["S2s"]+=d*sL["S2"]
    # snapshots with the same convention
    i=0; q={"S1":0,"S2":0}; s={"S1":0,"S2":0}
    for m in marks:
        while i < len(ev) and ev[i][0] <= m:
            _,_,st,dq,ds = ev[i]; q[st]+=dq; s[st]+=ds; i+=1
        rows.append(dict(time=m.strftime("%H:%M"), S1_queue=q["S1"], S1_in_svc=s["S1"],
                         S2_queue=q["S2"], S2_in_svc=s["S2"],
                         WIP_total=q["S1"]+s["S1"]+q["S2"]+s["S2"]))
    snap = pd.DataFrame(rows)
    means = dict(S1_queue=integ["S1q"]/360, S1_svc=integ["S1s"]/360,
                 S2_queue=integ["S2q"]/360, S2_svc=integ["S2s"]/360,
                 WIP_total=(integ["S1q"]+integ["S1s"]+integ["S2q"]+integ["S2s"])/360)
    return snap, means, peaks

def little_check(a, station, t0, t1):
    d = a[(a.station==station)]
    ent = d[["case","queue_entry","start"]].dropna(subset=["queue_entry"])
    # integrate queue length
    ev = []
    for _, r in ent.iterrows():
        ev.append((r.queue_entry, 3, 1))
        if r.start is not None: ev.append((r.start, 2, -1))
    ev.sort(key=lambda e:(e[0],e[1]))
    t_last, q, integ = t0, 0, 0.0
    # initialise q at t0
    for _, r in ent.iterrows():
        if r.queue_entry <= t0 and (r.start is None or r.start > t0): q += 1
    i=0
    for (t,_,dq) in ev:
        if t<=t0: continue
        if t>t1: break
        integ += (t-t_last).total_seconds()/60 * q
        q += dq; t_last = t
    integ += (t1-t_last).total_seconds()/60 * q
    Lbar = integ/((t1-t0).total_seconds()/60)
    dep = ent[(ent.start.notna()) & (ent.start>t0) & (ent.start<=t1)]
    lam = len(dep)/((t1-t0).total_seconds()/60)
    Wbar = ((dep.start-dep.queue_entry).dt.total_seconds()/60).mean()
    pre  = ((t0 - ent[(ent.queue_entry<=t0)&((ent.start.isna())|(ent.start>t0))].queue_entry).dt.total_seconds()/60).sum()
    post = ((t1 - ent[(ent.queue_entry<=t1)&((ent.start.isna())|(ent.start>t1))].queue_entry).dt.total_seconds()/60).sum()
    depwait_sum = ((dep.start-dep.queue_entry).dt.total_seconds()/60).sum()
    corrected = depwait_sum + post - pre
    print(f"[{station} {t0:%H:%M}-{t1:%H:%M}] mean queue L={Lbar:.2f}; naive lam*W={lam*Wbar:.2f} "
          f"(lambda={lam*60:.2f}/h, W={Wbar:.1f} min); boundary-corrected integral "
          f"{corrected:.0f} vs exact {integ:.0f} case-min")

# 1. Assumptions and notation

**Notation.** For case *i*: `a_i` = external arrival (S1 queue entry); `s1b_i, s1e_i` = S1 start/end; `s2b_i, s2e_i` = S2 start/end. Per the scenario, the S2 queue entry time is `e2_i = s1e_i` (no separate transfer timestamp). Window `T = [08:00, 14:00]`, length 360 min. Times are sometimes written as minutes after 08:00 (08:00 = 0, 14:00 = 360).

- **A1 — Simultaneous-event convention.** All timestamps share a 5-min grid, so distinct instants collide. At equal reported times I process events in the order: *activity-end (departure)  activity-start (queue exit + service begin)  queue-entry (arrival / S1 completion)*. Justification: this preserves single-resource feasibility and work conservation at rounding collisions (e.g., P07's imputed start 09:25 = P05's I1 end). Effect bounded: any queue count changes by 1 and any peak WIP by 1 under the reverse convention.
- **A2 — Rounding treatment.** Reported times are treated as exact for *central* estimates; every reported duration of length d has true value in [d5, d+5] min (two independently rounded endpoints). Count-based rates (arrivals/hour) are rounding-robust; durations and utilizations are not. Rounding uncertainty is reported separately from imputation uncertainty and never silently mixed.
- **A3 — Capacity.** S1 = {I1, I2}, S2 = {C1, C2}, all single-task; no bypassing; no preemption assumed. C2 is *unavailable* (not "busy") 10:55–11:10; utilization for C2 is reported both per window (360 min) and per available time (345 min).
- **A4 — Behavioral model for the central scenario.** Each station is a pooled FIFO queue with work-conserving dispatch (an idle resource takes the longest-waiting case). This is *not* assumed blindly: it is consistent with 18 of 20 observable dispatches (every recorded start occurs 0–5 min after some resource frees, in arrival order); the three violations (§5, E5) are flagged as anomalies rather than smoothed over.
- **A5 — Missing starts.** A MISSING start `s_i` is bounded, never point-imputed silently: `L_i = max(queue_entry_i, latest recorded end of the same resource before completion_i, outage end if C2)  s_i  end_i = U_i` (nonnegative service). Three scenarios: **early** = work-conserving earliest feasible start (central, justified by A4); **mid** = feasible-interval midpoint rounded to the grid; **late** = latest feasible start (degenerate zero-service bound). Note: the late bound is a mathematical extreme, not a physical claim — under it some services are 0 min, which contradicts the 25–50 min observed service distribution; it is used only as a conservative bracket.
- **A6 — Censoring.** OPEN activities contribute *elapsed* service to 14:00 as a right-censored lower bound (P15-S2  30, P17-S2  5, P21-S1  30 min); elapsed  final service. Cases still waiting at 14:00 contribute censored lower bounds on waiting time (P16  85, P18  50, P19  35, P20  15, P22  110 min). Censored cases are included in WIP, queue integrals, and busy time (as lower bounds); they are excluded from completed-service means with the exclusion stated.
- **A7 — Boundary conditions.** The system is assumed empty at 08:00 (first arrival exactly 08:00 supports this; labeled assumption, unverifiable). Everything at 14:00 is right-censored.
- **A8 — Cross-system timestamps.** The external arrival clock is assumed synchronized with the operational clock (given same rounding). A clock skew of a few minutes is a labeled alternative hypothesis (H5, §5) for small negative/positive gaps; no reported wait is negative, so no correction is applied.
- **A9 — Minor gaps.** The 10:50–10:55 C2 idle gap (queue non-empty, outage pending) is treated as changeover/rounding slack, not additional capacity loss.

# 2. Reconstruction method and formulas

**Case-level quantities** (all in minutes, reported-grid values; ±5 rounding tolerance applies to each):
- S1 wait `w1_i = s1b_i  a_i`; S1 service `s1_i = s1e_i  s1b_i`; S1 sojourn `w1_i + s1_i = s1e_i  a_i`.
- S2 wait `w2_i = s2b_i  e2_i`; S2 service `s2_i = s2e_i  s2b_i`; S2 sojourn `= s2e_i  e2_i`.
- Total sojourn (completed) `= s2e_i  a_i`. For censored cases, `W1  a_i` is the lower bound.
- Waiting and service are kept strictly separate: queue time accumulates only while not in service; for OPEN cases, elapsed service is reported as `14:00  start` with a `` flag.

**Station-level quantities** (event-sweep integrals under A1):
- `Q1(t) =  1[a_i  t < s1b_i]` (waiting only); `WIP1(t) =  1[a_i  t < s1e_i]`; analogously `Q2(t), WIP2(t)` with `e2_i, s2b_i, s2e_i`. Total `WIP(t) = #(a_i  t)  #(s2e_i  t)`.
- Mean queue/WIP `= ·dt / 360`; peaks are post-event values under A1.
- Busy time of resource r `=  (end_eff  start)` clipped to the window (`end_eff = end` or 14:00 if OPEN). Utilization = busy / 360; availability-adjusted = busy / 345 for C2; *input-conditional* = busy / (window  initial starved period before the station's first possible input), reported because raw utilization of a starved downstream station understates saturation.
- Rates per subperiod = event counts / subperiod length (counts are unaffected by rounding within ±2.5 min unless an event straddles a bin edge; noted where relevant).
