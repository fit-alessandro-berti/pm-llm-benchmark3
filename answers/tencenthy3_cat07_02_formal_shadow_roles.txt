```python
import pandas as pd
import pm4py

# 1. Load and format
df = pd.read_csv("event_log.csv")
df = pm4py.format_dataframe(
    df,
    case_id='case:concept:name',
    activity_key='concept:name',
    timestamp_key='time:timestamp',
    resource_key='org:resource'
)

# 2. Organizational roles (activity-frequency / similarity based)
roles = pm4py.discover_organizational_roles(
    df,
    resource_key='org:resource',
    activity_key='concept:name',
    case_id_key='case:concept:name',
    timestamp_key='time:timestamp'
)
# roles is a list of (role_name, resources, activities)

# 3. Activity-based similarity
sim_matrix = pm4py.discover_activity_based_resource_similarity(
    df,
    resource_key='org:resource',
    activity_key='concept:name'
)

# 4. Handover, working-together, subcontracting
handover = pm4py.discover_handover_of_work_network(
    df, resource_key='org:resource', activity_key='concept:name'
)
together = pm4py.discover_working_together_network(
    df, resource_key='org:resource', activity_key='concept:name'
)
subcontract = pm4py.discover_subcontracting_network(
    df, resource_key='org:resource', activity_key='concept:name'
)

# 5. Join with case complexity and substitution indicators
# avoid duplicate case-level attributes
case_meta = df[['case:concept:name','case_complexity','temporary_substitution',
                'escalation_level','customer_segment','product_line']].drop_duplicates()
event_res = df[['case:concept:name','org:resource','concept:name','time:timestamp']]

# median complexity per resource (weight by unique cases)
res_complex = (
    event_res.merge(case_meta, on='case:concept:name')
    .groupby('org:resource')
    .agg(median_complexity=('case_complexity','median'),
         pct_substitution_cases=('temporary_substitution', lambda s: (s==True).mean()),
         critical_cases=('escalation_level', lambda s: (s=='critical').sum()))
    .reset_index()
)

# Map roles to resources and then to complexity
role_rows = []
for role_name, resources, activities in roles:
    for r in resources:
        role_rows.append({'role':role_name,'resource':r,'role_activities':activities})
role_df = pd.DataFrame(role_rows).merge(res_complex, left_on='resource', right_on='org:resource')

# 6. Avoid over-interpreting frequency clusters as authority:
# e.g., inspect who actually performs A9 Supervisor Approval vs only appears in same cluster
approval_events = df[df['concept:name']=='A9 Supervisor Approval']
approval_auth = approval_events['org:resource'].value_counts()
print("A9 approval authority distribution:", approval_auth)

# Also filter substitution episodes to see if temps/backups diverge in quality
df_sub = pm4py.filter_event_attribute_values(
    df, 'temporary_substitution', [True], level='case'
)
# compare reopen rates etc.
```

# 5. PM4Py role-discovery and similarity code

```python
import pandas as pd
import pm4py

# -------------------------------------------------
# 1. Load and format the event dataframe
# -------------------------------------------------
df = pd.read_csv("order_to_resolution_event_log.csv")

df = pm4py.format_dataframe(
    df,
    case_id="case:concept:name",
    activity_key="concept:name",
    timestamp_key="time:timestamp",
    resource_key="org:resource"
)

# -------------------------------------------------
# 2. Discover organizational roles (frequency/similarity based)
# -------------------------------------------------
roles = pm4py.discover_organizational_roles(
    df,
    resource_key="org:resource",
    activity_key="concept:name",
    case_id_key="case:concept:name",
    timestamp_key="time:timestamp"
)
# `roles` is a list of (role_name, [resources], [activities])

# -------------------------------------------------
# 3. Activity-based resource similarity
# -------------------------------------------------
sim = pm4py.discover_activity_based_resource_similarity(
    df,
    resource_key="org:resource",
    activity_key="concept:name"
)

# -------------------------------------------------
# 4. Handover, working-together, subcontracting networks
# -------------------------------------------------
handover = pm4py.discover_handover_of_work_network(
    df, resource_key="org:resource", activity_key="concept:name"
)
together = pm4py.discover_working_together_network(
    df, resource_key="org:resource", activity_key="concept:name"
)
subcontract = pm4py.discover_subcontracting_network(
    df, resource_key="org:resource", activity_key="concept:name"
)

# -------------------------------------------------
# 5. Join role output with case_complexity and
#    temporary_substitution indicators
# -------------------------------------------------
# Case-level attributes are repeated on events; deduplicate first
case_meta = df[[
    "case:concept:name", "case_complexity", "temporary_substitution",
    "escalation_level", "customer_segment", "product_line"
]].drop_duplicates()

event_res = df[["case:concept:name", "org:resource", "concept:name"]]

# Median complexity and substitution exposure per resource
res_profile = (
    event_res.merge(case_meta, on="case:concept:name")
    .groupby("org:resource")
    .agg(
        median_case_complexity=("case_complexity", "median"),
        pct_substitution_cases=("temporary_substitution", lambda s: (s == True).mean()),
        critical_case_count=("escalation_level", lambda s: (s == "critical").sum())
    )
    .reset_index()
)

# Flatten role discovery output
role_rows = []
for role_name, resources, activities in roles:
    for r in resources:
        role_rows.append({
            "mined_role": role_name,
            "resource": r,
            "mined_activities": activities
        })
role_df = pd.DataFrame(role_rows).merge(
    res_profile, left_on="resource", right_on="org:resource"
)

# -------------------------------------------------
# 6. Avoid over-interpreting frequency clusters as authority
# -------------------------------------------------
# Authority must be checked from the activity itself, not cluster size.
approval_dist = (
    df[df["concept:name"] == "A9 Supervisor Approval"]
    .groupby("org:resource")
    .size()
    .sort_values(ascending=False)
)
print("A9 approval authority distribution:")
print(approval_dist)

# Inspect temporary substitution episodes explicitly
df_sub = pm4py.filter_event_attribute_values(
    df, "temporary_substitution", [True], level="case"
)
reopen_during_sub = df_sub[df_sub["concept:name"] == "A13 Reopen Request"]
print("Reopens during substitution episodes:", len(reopen_during_sub))

# Example: separate Chen and Dina despite high similarity
print(sim.loc[["R03_Chen", "R04_Dina"]])
# Then use res_profile to show Chen's higher median_case_complexity / critical count
```

The code intentionally separates **mined activity clusters** from **case-complexity and authority checks** so that similar profiles (e.g., Chen/Dina, Hana/Mateo, Fatima/Giorgio) are not automatically treated as identical roles.

- **Staffing:** Retain dedicated commercial capacity (Fatima) rather than assuming Giorgio’s shadow activity makes her redundant. Keep temporary agents explicitly bounded to low-complexity intake; do not backfill permanent front-office roles with temps based only on activity overlap.
- **Training:** Use the quality gatekeepers (Ivan/Julia) to train temps on complete entitlement notes, addressing the 17 reopens. Develop Dina toward Chen’s critical-case handling instead of treating them as interchangeable.

**Cautions about changing staffing based only on mined roles:**

3. **Do not eliminate shadow or temporary roles simply because they appear redundant in frequency clusters.** Giorgio’s pricing work is less frequent than Fatima’s but prevents backlog during absence; temps absorb volume but caused reopens when misused—both require governance, not removal based on counts alone.
