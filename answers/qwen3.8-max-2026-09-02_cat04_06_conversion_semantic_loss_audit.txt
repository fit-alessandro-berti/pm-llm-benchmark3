def marking_items(m):

            tokens = int(tokens)

        if tokens > 0:

def fire_transition(net, marking, trans):
    try:
        new_m = marking.copy()
    except Exception:
        new_m = Marking(marking)
    if Marking is not dict and not isinstance(new_m, Marking):
        try: new_m = Marking(new_m)
        except Exception: pass
    for arc in getattr(trans, "in_arcs", []):
        p = arc.source
        w = int(getattr(arc, "weight", 1) or 1)
        current = int(new_m.get(p, 0))
        if current < w:
            return None
        if current == w:
            try: del new_m[p]
            except Exception: new_m[p] = 0
        else:
            new_m[p] = current - w
    for arc in getattr(trans, "out_arcs", []):
        p = arc.target
        w = int(getattr(arc, "weight", 1) or 1)
        new_m[p] = int(new_m.get(p, 0)) + w
    return new_m

def tau_closure(net, marking, max_states=200, max_depth=50):
    if marking is None: return [], True
    results = {}
    start_key = marking_key(marking)
    results[start_key] = marking
    frontier = deque([(marking, 0)])
    truncated = False
    while frontier:
        m, depth = frontier.popleft()
        if depth >= max_depth:
            truncated = True
            continue
        try:

                    continue
                results[k] = nm
                frontier.append((nm, depth+1))

def bounded_visible_language(net, im, fm, max_visible_length=10, max_tau_states=200, max_tau_depth=50, max_search_states=20000):
    language = set()
    if net is None or im is None or fm is None:
        return language, True
    truncated = False
    start_closure, trunc = tau_closure(net, im, max_tau_states, max_tau_depth)
    truncated = truncated or trunc
    frontier = deque()
    visited = set()
    for m in start_closure:
        if marking_equal(m, fm):
            language.add(())
        if len(()) < max_visible_length:
            k = ((), marking_key(m))
            if k not in visited:
                visited.add(k)
                frontier.append(((), m))
    search_states = 0
    while frontier and search_states < max_search_states:
        trace, m = frontier.popleft()
        search_states += 1
        if len(trace) >= max_visible_length:
            continue
        if marking_equal(m, fm):
            # completed trace; do not expand past final marking
            continue
        try:
            enabled = pm4py.get_enabled_transitions(net, m)
        except Exception:
            truncated = True
            continue
        for t in enabled:
            if is_silent(t):
                continue
            nm = fire_transition(net, m, t)
            if nm is None:
                continue
            new_trace = trace + (str(getattr(t, "label", "")),)
            closure, trunc = tau_closure(net, nm, max_tau_states, max_tau_depth)
            truncated = truncated or trunc
            for cm in closure:
                if marking_equal(cm, fm):
                    language.add(new_trace)
                if len(new_trace) < max_visible_length:
                    k = (new_trace, marking_key(cm))
                    if k not in visited and len(visited) < max_search_states:
                        visited.add(k)
                        frontier.append((new_trace, cm))
                    elif k not in visited:
                        truncated = True
    if search_states >= max_search_states or len(visited) >= max_search_states:
        truncated = True
    return language, truncated

# 1. Source-Semantics Baseline

```python
def as_accepting_net(result):
    """
    Normalize conversion results.

    PM4Py conversion functions usually return (net, im, fm), but defensive
    handling is useful because BPMN/tree conversions can vary by version.
    """
    if result is None:
        return None, None, None
    if isinstance(result, tuple):
        if len(result) == 3:
            return result[0], result[1], result[2]
        if len(result) == 2:
            return result[0], result[1], None
        if len(result) == 1:
            return result[0], None, None
    return result, None, None


def safe_save(artifact_name, save_fn, *args, **kwargs):
    """Save a visualization without allowing visualization errors to hide conversion semantics."""
    try:
        save_fn(*args, **kwargs)
    except Exception as exc:
        save_errors.append({"artifact": artifact_name, "error": repr(exc)})


def state_count(obj):
    """Best-effort count of states in a reachability graph / transition system."""
    for attr in ("states", "nodes", "vertices"):
        val = getattr(obj, attr, None)
        if val is not None:
            try:
                return len(val)
            except Exception:
                pass
    if isinstance(obj, (list, tuple, set, frozenset)):
        return len(obj)
    return None


def footprint_signature(net, im, fm):
    """

    Footprint structures can be version-dependent, so we serialize defensively.
    """
    if net is None or im is None or fm is None:
        return "missing-markings"
    try:

        payload = json.dumps(fp, sort_keys=True, default=str)
        return hashlib.sha256(payload.encode("utf-8")).hexdigest()
    except Exception as exc:
        return f"ERROR:{repr(exc)}"

        report["reachability_error"] = repr(exc)

    return report
```

# 4. Preservation and Loss Audit

The following audit assumes the validation code in Section 5 is executed. Where a property cannot be fully proven by trace-based validation, I mark it as approximated or ambiguous rather than validated preserved. The process-tree column is marked `conversion failed` because the source behavior is expected to be non-block-structured; if a particular PM4Py version nevertheless returns a tree, that tree must still pass the Section 5 checks before being trusted.

| Source invariant | POWLPetri | PetriBPMN | PetriTree | BPMN round trip | Evidence or uncertainty |
|---|---|---|---|---|---|
| S1 `Register Change` first | validated preserved | validated preserved | conversion failed | validated preserved | Positive/negative witness tests; initial marking check. |
| S2 `Assess Risk` / `Prepare Evidence` unordered | validated preserved | approximated | conversion failed | approximated | Bounded language can check that both orders exist, but BPMN gateway layout may still suggest a fixed order. |
| S3 Optional `Legal Review` with silent alternative | validated preserved | approximated | conversion failed | ambiguous | Presence and absence of `Legal Review` are testable; the silent skip itself may become hidden routing in BPMN. |
| S4 Verification requires both `Assess Risk` and `Prepare Evidence` | validated preserved | validated preserved | conversion failed | validated preserved | Negative witness: `Configure Trial` before `Prepare Evidence` must be rejected. |
| S5 Optional legal and verification unordered / may overlap | expected preserved | approximated | conversion failed | approximated | Positive witnesses place `Legal Review` before, during, and after verification. Full overlap space is large; bounded tests only sample it. |
| S6 `Configure Trial` before `Inspect Trial` in each iteration | validated preserved | validated preserved | conversion failed | validated preserved | Direct positive/negative witness tests. |
| S7 `Correct Defect` separates repeated iterations | validated preserved | expected preserved | conversion failed | expected preserved | Witness with two iterations and missing `Correct Defect` negative witness. BPMN loop visualization may obscure the redo branch. |
| S8 At least one verification iteration | validated preserved | expected preserved | conversion failed | expected preserved | Negative witness: release without any `Configure Trial` / `Inspect Trial`. |
| S9 Release waits for legal branch and verification loop | validated preserved | expected preserved | conversion failed | expected preserved | Negative witness: release before legal when legal appears later; release before verification. |
| S10 `Archive Change` final after release | validated preserved | expected preserved | conversion failed | expected preserved | Exact final-marking equality after completed visible trace. |
| S11 No extra visible labels / label identity preserved | expected preserved | ambiguous | conversion failed | ambiguous | BPMN may add start/end events, silent gateways, or renamed routing nodes. Visible label set should be checked explicitly. |

Additional audit notes:

- **True concurrency versus arbitrary interleaving.**  
  The POWL model explicitly represents absence of order. A Petri net can represent concurrent structure, but trace-based validation observes interleavings. If both `Assess Risk, Prepare Evidence` and `Prepare Evidence, Assess Risk` are accepted, that validates the interleaving consequence of concurrency, but it does not by itself prove that the converted BPMN preserves a true concurrent execution semantics rather than one serialized implementation.

- **Lack of an edge between Optional Legal Review and verification.**  
  This is one of the highest-risk conversion points. The source says legal review and verification are unordered except for the common predecessor `Assess Risk` and common successor `Release Change`. A converter that serializes them, for example by forcing legal review before verification or after verification, may still look plausible in BPMN but loses a source invariant.

- **Silent optional behavior.**  
  The XOR with a `SilentTransition` means the skip is not a visible activity. BPMN cannot directly show a silent transition in a business-friendly way. It may become an implicit path, an unlabeled gateway flow, or a hidden routing node. Round-trip validation must hide `None` labels but still require that the no-legal visible trace completes exactly.

- **Binary-loop semantics.**  
  The POWL loop is binary: first child is the loop body, second child is the redo fragment. The intended meaning is body, then zero or more repetitions of redo followed by body. Therefore `Correct Defect` is not an arbitrary activity inside the loop; it is the separator between iterations. BPMN loop markers or visual back-edges can obscure this.

- **Join before Release Change.**  
  Release must wait for both the optional legal branch and the verification loop. In a Petri net this is naturally represented by a place or synchronization structure. In BPMN it may become a parallel join, an inclusive join, or a collection of gateway nodes. If the join is rendered as a simple merge, stakeholders may misread it as non-synchronizing.

- **Activity-label uniqueness.**  
  The visible source labels are unique. However, converters may introduce duplicate silent labels, empty labels, or generated labels such as `tau`, `start`, `end`, or gateway identifiers. Trace alignment should use visible labels carefully and should not assume that every BPMN object corresponds to a source activity.

- **Initial and final markings.**  
  BPMN models usually do not expose Petri markings. The round-tripped Petri net must be checked for non-null initial and final markings, workflow-net shape, soundness, and exact final-marking equality for completed traces. A BPMN diagram that looks correct but produces a missing final marking is not semantically acceptable.

- **Gateway or routing-node ambiguity in BPMN visualization.**  
  BPMN is a visual object-model, not merely a language object. Parallel gateways can be misread as synchronous barriers, exclusive gateways can be misread as business choices rather than silent optionality, and layout can imply ordering where none exists. Successful serialization or visualization is not evidence of semantic fidelity.

```python
# ----------------------------------------------------------------------------
# Round-trip validation driver
# Requires the conversion code from Section 3 to have been executed.
# ----------------------------------------------------------------------------
import json
import hashlib
from collections import deque

import pm4py

try:
    from pm4py.objects.petri_net.obj import Marking
except Exception:
    Marking = dict

if "net_from_powl" not in globals():
    raise RuntimeError("net_from_powl is not defined. Run the Section 3 conversion code first.")
if "roundtrip_nets" not in globals():
    raise RuntimeError("roundtrip_nets is not defined. Run the Section 3 conversion code first.")

# ----------------------------------------------------------------------------
# Silent-transition handling and marking utilities
# ----------------------------------------------------------------------------
SILENT_LABELS = {
    "",
    "tau",
    "silent",
    "start",
    "end",
    "start event",
    "end event",
}


def is_silent(t):
    label = getattr(t, "label", None)
    if label is None:
        return True
    return str(label).strip().lower() in SILENT_LABELS


def copy_marking(m):
    try:
        c = m.copy()
    except Exception:
        c = Marking(m)

    if Marking is not dict and not isinstance(c, Marking):
        try:
            c = Marking(c)
        except Exception:
            pass

    return c


def marking_items(m):
    if m is None:
        return
    try:
        items = m.items()
    except Exception:
        items = {}.items()

    for p, tokens in items:
        try:
            tokens = int(tokens)
        except Exception:
            tokens = 1
        if tokens > 0:
            yield p, tokens


def marking_key(m):
    return tuple(
        sorted(
            (id(p), getattr(p, "name", ""), tokens)
            for p, tokens in marking_items(m)
        )
    )


def marking_equal(a, b):
    if a is b:
        return True
    if a is None or b is None:
        return False

    da = {p: tokens for p, tokens in marking_items(a)}
    db = {p: tokens for p, tokens in marking_items(b)}
    return da == db


def fire_transition(net, marking, trans):
    new_m = copy_marking(marking)

    for arc in getattr(trans, "in_arcs", []):
        p = arc.source
        w = int(getattr(arc, "weight", 1) or 1)
        current = int(new_m.get(p, 0))

        if current < w:
            return None

        if current == w:
            try:
                del new_m[p]
            except Exception:
                new_m[p] = 0
        else:
            new_m[p] = current - w

    for arc in getattr(trans, "out_arcs", []):
        p = arc.target
        w = int(getattr(arc, "weight", 1) or 1)
        new_m[p] = int(new_m.get(p, 0)) + w

    return new_m


def tau_closure(net, marking, max_states=300, max_depth=80):
    """
    Explore silent transitions only.

    - Hides transitions whose label is None or silent-like.
    - Avoids infinite silent cycling using visited marking keys.
    - Bounds exploration by max_states and max_depth.
    """
    if marking is None:
        return [], True

    results = {}
    start_key = marking_key(marking)
    results[start_key] = marking

    frontier = deque([(marking, 0)])
    truncated = False

    while frontier:
        m, depth = frontier.popleft()

        if depth >= max_depth:
            truncated = True
            continue

        try:
            enabled = pm4py.get_enabled_transitions(net, m)
        except Exception:
            truncated = True
            continue

        for t in enabled:
            if not is_silent(t):
                continue

            nm = fire_transition(net, m, t)
            if nm is None:
                continue

            k = marking_key(nm)
            if k not in results:
                if len(results) >= max_states:
                    truncated = True
                    continue
                results[k] = nm
                frontier.append((nm, depth + 1))

    return list(results.values()), truncated


def accepts_visible_trace(
    net,
    im,
    fm,
    visible_trace,
    max_visible_length=12,
    max_tau_states=300,
    max_tau_depth=80,
):
    """
    Accept a visible trace if, after hiding silent transitions, the trace can fire
    and the exact final marking fm is reached.
    """
    if net is None or im is None or fm is None:
        return False

    if len(visible_trace) > max_visible_length:
        return False

    current, _ = tau_closure(net, im, max_tau_states, max_tau_depth)
    if not current:
        return False

    for wanted in visible_trace:
        next_marks = {}

        for m in current:
            try:
                enabled = pm4py.get_enabled_transitions(net, m)
            except Exception:
                continue

            for t in enabled:
                if is_silent(t):
                    continue

                label = getattr(t, "label", None)
                if label == wanted or str(label) == str(wanted):
                    nm = fire_transition(net, m, t)
                    if nm is None:
                        continue

                    closure, _ = tau_closure(net, nm, max_tau_states, max_tau_depth)
                    for cm in closure:
                        k = marking_key(cm)
                        if k not in next_marks:
                            next_marks[k] = cm

        if not next_marks:
            return False

        current = list(next_marks.values())

    for m in current:
        if marking_equal(m, fm):
            return True

    return False


def bounded_visible_language(
    net,
    im,
    fm,
    max_visible_length=11,
    max_tau_states=300,
    max_tau_depth=80,
    max_search_states=50000,
):
    """
    Enumerate completed visible traces up to a bound.

    This is exhaustive only if truncated remains False under the chosen bounds.
    It hides silent transitions and requires exact final-marking equality for
    completed traces.
    """
    language = set()

    if net is None or im is None or fm is None:
        return language, True

    truncated = False

    start_closure, start_trunc = tau_closure(net, im, max_tau_states, max_tau_depth)
    truncated = truncated or start_trunc

    frontier = deque()
    visited = set()

    for m in start_closure:
        if marking_equal(m, fm):
            language.add(())

        k = ((), marking_key(m))
        if k not in visited:
            visited.add(k)
            frontier.append(((), m))

    search_states = 0

    while frontier and search_states < max_search_states:
        trace, m = frontier.popleft()
        search_states += 1

        if len(trace) >= max_visible_length:
            continue

        # Do not expand completed final markings; completed traces are enough.
        if marking_equal(m, fm):
            continue

        try:
            enabled = pm4py.get_enabled_transitions(net, m)
        except Exception:
            truncated = True
            continue

        for t in enabled:
            if is_silent(t):
                continue

            nm = fire_transition(net, m, t)
            if nm is None:
                continue

            new_trace = trace + (str(getattr(t, "label", "")),)

            closure, closure_trunc = tau_closure(net, nm, max_tau_states, max_tau_depth)
            truncated = truncated or closure_trunc

            for cm in closure:
                if marking_equal(cm, fm):
                    language.add(new_trace)

                if len(new_trace) < max_visible_length:
                    k = (new_trace, marking_key(cm))
                    if k not in visited:
                        if len(visited) >= max_search_states:
                            truncated = True
                        else:
                            visited.add(k)
                            frontier.append((new_trace, cm))

    if search_states >= max_search_states or len(visited) >= max_search_states:
        truncated = True

    return language, truncated


def evaluate_witnesses(
    net,
    im,
    fm,
    witnesses,
    max_visible_length=12,
    max_tau_states=300,
    max_tau_depth=80,
):
    results = []

    for trace, expected in witnesses:
        error = None
        try:
            accepted = accepts_visible_trace(
                net,
                im,
                fm,
                trace,
                max_visible_length=max_visible_length,
                max_tau_states=max_tau_states,
                max_tau_depth=max_tau_depth,
            )
        except Exception as exc:
            accepted = False
            error = repr(exc)

        results.append(
            {
                "trace": trace,
                "expected": expected,
                "accepted": accepted,
                "pass": accepted == expected,
                "error": error,
            }
        )

    return results


def witness_summary(results):
    return {
        "total": len(results),
        "passed": sum(1 for r in results if r["pass"]),
        "failed": sum(1 for r in results if not r["pass"]),
    }

                    return len(val)
                except Exception:
                    pass
        if isinstance(rg, (list, tuple, set, frozenset)):
            return len(rg)
        return None
    except Exception as exc:
        return f"ERROR:{repr(exc)}"

        "Register Change",

        "Prepare Evidence",
        "Configure Trial",

        "Inspect Trial",
        "Release Change",
        "Archive Change",
    ],
    # Correct Defect before first iteration.
    [
        "Register Change",
        "Assess Risk",
        "Prepare Evidence",
        "Correct Defect",
        "Configure Trial",
        "Inspect Trial",
        "Release Change",
        "Archive Change",

    # Release before Legal Review when Legal Review appears later.
    [
        "Register Change",
        "Assess Risk",

    [

witnesses = (
    [(trace, True) for trace in positive_witnesses]

source_witness_results = evaluate_witnesses(

        source_language_truncated = None
        source_language_error = repr(exc)

        "soundness": source_soundness,
        "workflow_net": source_workflow,
        "footprint_signature": source_footprint,
        "reachability_state_count": source_reachability,

            r for r in source_witness_results if not r["pass"]

            "max_search_states": MAX_SEARCH_STATES,

    rt_report["has_initial_marking"] = rt_im is not None
    rt_report["has_final_marking"] = rt_fm is not None

        validation_report["roundtrips"][name] = rt_report

    rt_report["workflow_net"] = quick_workflow_net(rt_net)
    rt_report["reachability_state_count"] = quick_reachability_state_count(
        rt_net,
        rt_im,

        rt_net,
        rt_im,
        rt_fm,
        witnesses,
        max_visible_length=12,

    rt_report["witness_mismatches_against_source"] = [
        {
            "trace": src["trace"],
            "expected": src["expected"],
            "source_accepted": src["accepted"],
            "roundtrip_accepted": rt["accepted"],
            "source_pass": src["pass"],
            "roundtrip_pass": rt["pass"],
        }
        for src, rt in zip(source_witness_results, rt_witness_results)
        if src["accepted"] != rt["accepted"] or not rt["pass"]
    ]

    if RUN_BOUNDED_LANGUAGE and source_language is not None:
        try:
            rt_language, rt_language_truncated = bounded_visible_language(
                rt_net,
                rt_im,
                rt_fm,
                max_visible_length=MAX_VISIBLE_LENGTH,
                max_search_states=MAX_SEARCH_STATES,
            )

            missing_from_roundtrip = sorted(source_language - rt_language)
            extra_in_roundtrip = sorted(rt_language - source_language)

            rt_report["bounded_language"] = {
                "trace_count": len(rt_language),
                "truncated": rt_language_truncated,
                "equal_up_to_bound": source_language == rt_language,
                "missing_sample": [list(t) for t in missing_from_roundtrip[:5]],
                "extra_sample": [list(t) for t in extra_in_roundtrip[:5]],
                "missing_count": len(missing_from_roundtrip),
                "extra_count": len(extra_in_roundtrip),
            }
        except Exception as exc:
            rt_report["bounded_language"] = {"error": repr(exc)}

    validation_report["roundtrips"][name] = rt_report

print(json.dumps(validation_report, indent=2, default=str))
```

**Exhaustiveness statement.**  
The witness checks are exhaustive only for the explicitly listed positive and negative traces. The `bounded_visible_language` comparison is exhaustive for completed visible traces up to `MAX_VISIBLE_LENGTH = 11` and within `MAX_SEARCH_STATES = 50000` only when the corresponding `truncated` flag is `False`. Because the source contains an unbounded verification loop, no finite visible-length bound can prove full unbounded language equivalence. Silent transitions are hidden, silent cycling is bounded by visited marking keys and depth limits, and completed traces must reach the exact final marking.

# 6. Visual-Semantics Review

A BPMN rendering can be behaviorally acceptable at the trace level and still be semantically misleading.

1. **Parallel gateways can be misread as strict synchronous barriers.**  
   A parallel fork/join may be used to represent concurrent branches. Stakeholders may infer that all branches must always start and finish at exactly the same time, whereas the source POWL only requires absence of ordering plus synchronization before release.

2. **Exclusive gateways can hide silent optionality.**  
   The optional legal review is an XOR between `Legal Review` and a silent transition. In BPMN, the silent path may be drawn as an unlabeled flow or collapsed into gateway routing. A reader may think the process has no optional legal step, or may think the gateway represents a business decision rather than a silent skip.

3. **Loop visualization can obscure binary-loop semantics.**  
   BPMN loop markers can make it look as if `Inspect Trial` alone repeats, or as if `Correct Defect` is optional cleanup. The source loop is body-then-redo: `Configure Trial` before `Inspect Trial`, then `Correct Defect` only when another iteration follows.

4. **Silent routing nodes are not business activities.**  
   Petri-to-BPMN conversion may introduce routing nodes, start/end events, or unnamed flows. These are not source activities. If they are visually prominent, users may mistake them for required work.

5. **Graph layout can imply ordering.**  
   If `Legal Review` is drawn above or before the verification loop, readers may assume it must occur first. If `Assess Risk` is drawn left of `Prepare Evidence`, readers may assume a required sequence. In the source, several of these elements are unordered.

Recommended annotations:

- Add a note: “Horizontal/vertical layout does not imply ordering unless a sequence flow and gateway semantics require it.”
- Annotate the legal gateway: “Optional legal review; skip path is silent.”
- Annotate the release join: “Release Change requires completion of both optional legal branch and verification loop.”
- Annotate the loop: “Each iteration: Configure Trial then Inspect Trial. Correct Defect separates iterations.”
- Provide a small invariant table next to the BPMN diagram.
- Provide example valid traces and invalid traces.
- Color or style silent routing nodes differently from visible activities.
- Store the BPMN as a derivative artifact with its validation report, not as the authoritative source.

# 7. Rejection Criteria

A conversion result should be rejected or quarantined if any of the following holds:

1. **Unsound round-tripped net.**  
   `pm4py.check_soundness` reports unsoundness, or soundness cannot be established because markings are missing or the net is not a workflow net.

2. **Missing initial or final marking.**  
   Any conversion step returns a net without a usable initial or final marking. BPMN may hide markings, but the round-tripped accepting net must expose them.

3. **Positive witness rejected.**  
   A valid source trace, such as a trace without `Legal Review` or a trace with two verification iterations separated by `Correct Defect`, is not accepted by the round-tripped net with exact final-marking equality.

4. **Negative witness accepted.**  
   The round-tripped net allows release before verification, inspection before configuration, legal review before risk assessment, or archive before release.

5. **Release before verification becomes possible.**  
   Any trace reaching `Release Change` without at least one completed `Configure Trial` / `Inspect Trial` iteration is a hard semantic violation.

6. **Optional legal behavior collapses.**  
   The converter makes `Legal Review` mandatory, removes it entirely, or makes the silent skip visible as a business activity.

7. **Required interleaving is lost.**  
   The source allows both `Assess Risk, Prepare Evidence` and `Prepare Evidence, Assess Risk`. If the round-tripped model accepts only one order, the absence-of-order invariant is lost.

8. **Loop separator semantics lost.**  
   The round-tripped model allows repeated verification iterations without `Correct Defect`, or allows `Correct Defect` before the first iteration.

9. **Exact final-marking test fails.**  
   A trace reaches the visible end but leaves residual tokens, or reaches a marking that is not exactly the declared final marking.

10. **Process-tree conversion succeeds but validation shows drift.**  
   If the process-tree pipeline does not raise an exception but footprint, witness, or bounded-language checks differ from the source Petri net, the tree-derived BPMN should be rejected even though serialization succeeded.

# 8. Canonical Representation Recommendation

**Canonical authoritative storage: the source POWL model.**  
The POWL object should be the authoritative semantic representation because it directly encodes the partial order, the silent optional branch, the binary loop, and the synchronization before release. It is less ambiguous than BPMN layout and more compact than a generated Petri net.

**Derivative machine-checkable artifact: the accepting Petri net.**  
The Petri net produced by `pm4py.convert_to_petri_net(source_powl)` should be stored as a derived validation artifact. It is the best executable basis for soundness, footprints, reachability diagnostics, and round-trip checks.

**Stakeholder visualization: annotated BPMN.**  
The BPMN model produced from the primary pipeline should be treated as a derivative communication artifact. It is suitable for stakeholders but should not be the source of truth.

Recommended artifact hierarchy:

| Artifact | Role | Authority | Required validation evidence |
|---|---|---|---|
| Source POWL | Authoritative semantic model | Yes | Version control, hash, invariant table. |
| Accepting Petri net | Executable derivative for validation | No, derivative | Workflow-net check, soundness, footprints, witness tests, bounded-language report. |
| BPMN from Petri net | Stakeholder visualization | No, derivative | Round-trip Petri diagnostics, witness comparison, visual ambiguity notes. |
| Process tree | Optional intermediate | Not recommended by default | If used, must pass full witness/footprint/bounded-language comparison. |
| BPMN from process tree | Optional visualization | Not recommended unless tree validation passes | Same validation as above, plus explicit justification. |

**Process-tree intermediate policy.**  
The process-tree intermediate should not be allowed by default for this model. The source behavior is not guaranteed to be block-structured because optional legal review and verification overlap and then synchronize before release. If a process-tree conversion succeeds, it should be quarantined until the Section 5 validation demonstrates no behavioral drift.

**Avoiding round-trip drift.**  
Future edits should be made to the POWL model, not to the BPMN diagram. The safe workflow is:

1. Edit the POWL.
2. Regenerate the accepting Petri net.
3. Run soundness, footprint, witness, and bounded-language validation.
4. Regenerate BPMN only after validation passes.
5. Store hashes and validation reports with each generated artifact.
6. Never overwrite the authoritative POWL from a BPMN round trip.

If business users need to edit BPMN directly, the BPMN edit should be treated as a change proposal. The change should be re-expressed in POWL, then regenerated and validated. Direct BPMN-to-POWL round trips should not be trusted as canonical.

# 9. Residual Uncertainty

Even if every test in Sections 3 and 5 passes, the following claims remain unproven:

1. **Full unbounded language equivalence.**  
   The validation is bounded. The source loop can repeat arbitrarily many times. Passing tests up to a visible-length or state-count bound does not prove equivalence for all possible loop iterations.

2. **True concurrency preservation.**  
   Trace and footprint checks can show that certain interleavings are possible, but they cannot fully prove that the BPMN execution semantics preserves the same true concurrency, fairness, or scheduling independence as the POWL/Petri representation.

3. **Converter correctness in all cases.**  
   The audit validates observable consequences of the converted models, but it cannot prove that PM4Py’s internal conversion functions preserve every semantic nuance of POWL for all future model edits.

4. **Silent-transition intent.**  
   Hiding tau transitions is necessary for visible-trace comparison, but it can conceal silent livelocks, alternative silent routings, or ambiguous optional-skip placement that may matter in an execution engine.

5. **Human interpretation of BPMN.**  
   Validation can show that a BPMN diagram is behaviorally acceptable, but it cannot guarantee that stakeholders will not misread gateways, layout, loop markers, or silent routing nodes.
