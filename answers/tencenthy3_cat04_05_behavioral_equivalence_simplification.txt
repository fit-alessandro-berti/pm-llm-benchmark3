def tree_stats(tree):
    # recursive
    if tree.operator is None:
        # leaf
        label = tree.label
        return {'nodes':1, 'depth':1, 'vis':0 if label=='tau' else 1, 'silent':1 if label=='tau' else 0, 'labels':[label]}

        max_child_depth = 0
        for child in tree.children:
            cs = tree_stats(child)
            stats['nodes'] += cs['nodes']
            stats['vis'] += cs['vis']
            stats['silent'] += cs['silent']
            stats['labels'].extend(cs['labels'])
            max_child_depth = max(max_child_depth, cs['depth'])

import pm4py
from collections import deque
from pm4py.objects.petri_net.obj import Marking

A_str = "->( 'Open Case', *( ->( +('Analyze Revision', 'Check Revision'), X('Approve Revision', tau) ), 'Rework Case' ), 'Close Case' )"
B_str = "->( 'Open Case', +( 'Check Revision', *( ->( 'Analyze Revision', X('Approve Revision', tau) ), 'Rework Case' ) ), 'Close Case' )"

treeA = pm4py.parse_process_tree(A_str)
treeB = pm4py.parse_process_tree(B_str)

netA, imA, fmA = pm4py.convert_to_petri_net(treeA)
netB, imB, fmB = pm4py.convert_to_petri_net(treeB)

print("Soundness A:", pm4py.check_soundness(netA, imA, fmA, print_diagnostics=False))
print("Soundness B:", pm4py.check_soundness(netB, imB, fmB, print_diagnostics=False))

def mark_hash(m):
    return frozenset((p.name, m[p]) for p in m if m[p] > 0)

def fire(net, marking, trans):
    m = Marking()
    m.update(marking)
    for arc in trans.in_arcs:
        m[arc.source] -= 1
        if m[arc.source] <= 0:
            del m[arc.source]
    for arc in trans.out_arcs:
        m[arc.target] = m.get(arc.target, 0) + 1
    return m

def bfs_visible_traces(net, im, fm, max_len=12, max_analyze=3):
    fm_h = mark_hash(fm)
    results = set()
    # queue items: (marking, trace_tuple, analyze_count, silent_visited_set)
    queue = deque()
    queue.append((im, (), 0, set([mark_hash(im)])))
    while queue:
        m, trace, acount, visited = queue.popleft()
        if mark_hash(m) == fm_h:
            results.add(trace)
            continue
        for t in pm4py.get_enabled_transitions(net, m):
            silent = (t.label is None) or (isinstance(t.label, str) and t.label.lower().startswith('tau'))
            new_m = fire(net, m, t)
            new_m_h = mark_hash(new_m)
            if silent:
                if new_m_h in visited:
                    continue
                new_visited = set(visited)
                new_visited.add(new_m_h)
                queue.append((new_m, trace, acount, new_visited))
            else:
                label = t.label
                new_trace = trace + (label,)
                new_acount = acount + (1 if label == 'Analyze Revision' else 0)
                if len(new_trace) > max_len or new_acount > max_analyze:
                    continue
                queue.append((new_m, new_trace, new_acount, set([new_m_h])))
    return results

traces_A = bfs_visible_traces(netA, imA, fmA)
traces_B = bfs_visible_traces(netB, imB, fmB)

print("Unique to B (within P3 bound):")
for t in sorted(traces_B - traces_A):

# 1. Equivalence Notions

- **Visible trace equivalence after hiding `tau`**  
  Let `_v` be the set of visible activity labels (`Open Case`, `Analyze Revision`, `Check Revision`, `Approve Revision`, `Rework Case`, `Close Case`). For a trace `` in the model, `hide()` removes all `tau` events and yields a word over `_v`. Two models are visibly trace-equivalent iff  
  `{ hide() |  is a complete run }` is identical for both models.  
  Duplicate visible labels are distinguished **by occurrence position** in the word; they are not given independent token identities.

- **Bounded visible-language equivalence under P3**  
  Restrict the above visible language to complete traces `w` such that `|w|  12` and `count(w, 'Analyze Revision')  3`. Two models are P3-equivalent iff these two restricted sets are equal. Bounded equivalence is evidence only; finding no difference does not prove unbounded equivalence.

- **Business-relevant equivalence under P1**  
  The observation boundary retains the full visible trace and treats the following as relevant: every analysis revision must have its own `Check Revision` before the next `Rework Case` or `Close Case`; `Rework Case` separates attempts; `Approve Revision` is optional but its presence/placement is visible. Formally, P1-equivalence means visible trace equivalence together with the compliance predicate  
  ` revision i:  a Check in the same iteration as Analyze i, completed before Rework_i or Close`.  
  Duplicate labels are distinguished by position, as above.

- **Abstracted dashboard equivalence under P2**  
  Define abstraction `h(w) = (first=Open Case, last=Close Case, n_Analyze = count(w,'Analyze Revision'), checkOccurred = ('Check Revision'  w), reworkOccurred = ('Rework Case'  w))`.  
  `Approve Revision` and `tau` are hidden; multiple `Check Revision` occurrences are collapsed to the Boolean `checkOccurred`. P2-equivalence means equality of the sets `{ h(w) }` produced by the two models. Here, duplicate `Check Revision` labels are **not** distinguished; they are aggregated. `Analyze Revision` duplicates are counted but not individually identified.

| T1 | Open Case  Analyze Revision  Check Revision  Rework Case  Analyze Revision  Check Revision  Close Case | Yes | No | Second `Check Revision` is required by A but impossible in B (B has only one Check). | P1, P3 |
| T2 | Open Case  Analyze Revision  Check Revision  Approve Revision  Rework Case  Analyze Revision  Check Revision  Close Case | Yes | No | A emits two Checks (one per revision); B cannot, and would also lack the second Check. | P1, P3 |
| T3 | Open Case  Check Revision  Analyze Revision  Rework Case  Check Revision  Analyze Revision  Close Case | Yes | No | A allows `Check` before `Analyze` inside each iteration via `+`; B cannot repeat `Check`. | P1, P3 |
| T4 | Open Case  Analyze Revision  Rework Case  Analyze Revision  Check Revision  Close Case | No | Yes | B emits only one `Check` for two `Analyze` events; A would require a `Check` before the first `Rework Case`. | P1, P3 |
| T5 | Open Case  Check Revision  Analyze Revision  Approve Revision  Rework Case  Analyze Revision  Close Case | No | Yes | Single `Check` precedes all `Rework`; second `Analyze` has no own `Check`, so A rejects. | P1, P3 |
| T6 | Open Case  Analyze Revision  Approve Revision  Rework Case  Analyze Revision  Check Revision  Close Case | No | Yes | `Approve` occurs before the only `Check`; in A any `Approve` must follow its iteration’s `Check`. | P1, P3 |

All listed traces contain two analysis attempts, satisfying the requirement that at least one counterexample contains two attempts.

**Effect of optional `Approve Revision` and hidden `tau`:**  
The `X('Approve Revision', tau)` choice means every iteration may either show `Approve Revision` or hide it via `tau`. Because `tau` is removed from visible traces, the absence of `Approve Revision` does not by itself distinguish A from B. However, when `Approve Revision` is present, A forces it to occur after the same iteration’s `Check Revision`, whereas B may schedule `Approve Revision` before the single `Check Revision` (e.g., T6). Thus optional approval adds a second, orthogonal dimension of non-equivalence, but the primary difference remains the multiplicity and placement of `Check Revision`.

# 5. Bounded Comparison Code

```python
import pm4py
from collections import deque
from pm4py.objects.petri_net.obj import Marking

# 1. parse both process trees
A_str = "->( 'Open Case', *( ->( +('Analyze Revision', 'Check Revision'), X('Approve Revision', tau) ), 'Rework Case' ), 'Close Case' )"
B_str = "->( 'Open Case', +( 'Check Revision', *( ->( 'Analyze Revision', X('Approve Revision', tau) ), 'Rework Case' ) ), 'Close Case' )"

treeA = pm4py.parse_process_tree(A_str)
treeB = pm4py.parse_process_tree(B_str)

# 2. convert to accepting Petri nets
netA, imA, fmA = pm4py.convert_to_petri_net(treeA)
netB, imB, fmB = pm4py.convert_to_petri_net(treeB)

# 3. soundness checks
print("Soundness A:", pm4py.check_soundness(netA, imA, fmA, print_diagnostics=True))
print("Soundness B:", pm4py.check_soundness(netB, imB, fmB, print_diagnostics=True))

# helper: hash a marking ignoring zero tokens
def mark_hash(m):
    return frozenset((p.name, m[p]) for p in m if m[p] > 0)

# helper: fire a transition and return a new Marking
def fire(net, marking, trans):
    m = Marking()
    m.update(marking)
    for arc in trans.in_arcs:
        m[arc.source] -= 1
        if m[arc.source] <= 0:
            del m[arc.source]
    for arc in trans.out_arcs:
        m[arc.target] = m.get(arc.target, 0) + 1
    return m

# 4./5. exhaustive bounded BFS over visible traces
def bfs_visible_traces(net, im, fm, max_len=12, max_analyze=3):
    fm_h = mark_hash(fm)
    results = set()
    # queue item: (marking, visible_trace_tuple, analyze_count, silent_visited_set)
    queue = deque()
    queue.append((im, (), 0, set([mark_hash(im)])))
    while queue:
        m, trace, acount, visited = queue.popleft()
        if mark_hash(m) == fm_h:
            results.add(trace)
            continue
        for t in pm4py.get_enabled_transitions(net, m):
            # silent transition?
            silent = (t.label is None) or (isinstance(t.label, str) and t.label.lower().startswith('tau'))
            new_m = fire(net, m, t)
            new_m_h = mark_hash(new_m)
            if silent:
                if new_m_h in visited:
                    continue
                new_visited = set(visited)
                new_visited.add(new_m_h)
                queue.append((new_m, trace, acount, new_visited))
            else:
                label = t.label
                new_trace = trace + (label,)
                new_acount = acount + (1 if label == 'Analyze Revision' else 0)
                if len(new_trace) > max_len or new_acount > max_analyze:
                    continue
                queue.append((new_m, new_trace, new_acount, set([new_m_h])))
    return results

# 5./6. compare under P3 bounds
traces_A = bfs_visible_traces(netA, imA, fmA, max_len=12, max_analyze=3)
traces_B = bfs_visible_traces(netB, imB, fmB, max_len=12, max_analyze=3)

print("\nUnique to A (within P3 bound):")
for t in sorted(traces_A - traces_B):
    print("   ", " -> ".join(t))
print("\nUnique to B (within P3 bound):")
for t in sorted(traces_B - traces_A):
    print("   ", " -> ".join(t))
```

This is **exhaustive bounded exploration**, not `pm4py.play_out`. Every enabled transition is expanded; only silent cycles are pruned via the `silent_visited_set`, and visible expansion is pruned strictly by the P3 length and `Analyze Revision` bounds. `play_out` would be random/heuristic and would not guarantee that all bounded unique traces are found.

# 6. Stochastic and Footprint Comparison

`pm4py.discover_footprints` and `pm4py.get_stochastic_language` / `pm4py.compute_emd` can add corroborating evidence, but they cannot establish exact trace equivalence.

```python
# Footprint comparison (visible activity pairs)
fp_a = pm4py.discover_footprints(treeA)
fp_b = pm4py.discover_footprints(treeB)
print("Footprints identical?", fp_a == fp_b)
# Expected: False, because the (Check Revision, Approve Revision) pair is
# sequence (Check -> Approve only) in A, but parallel (both orders) in B.

# Stochastic language comparison (illustrative probabilities only)
# We declare an illustrative assumption: every XOR choice is equiprobable,
# including X(Approve Revision, tau). No domain probabilities are claimed.
sl_a = pm4py.get_stochastic_language(treeA)
sl_b = pm4py.get_stochastic_language(treeB)
emd = pm4py.compute_emd(sl_a, sl_b)
print("EMD between stochastic languages:", emd)
```

**Why these do not prove equivalence:**
- Footprints are a pairwise projection (directly-follows / parallel / choice). Two models can have identical footprints while differing in occurrence multiplicity or nested ordering.
- A stochastic language plus EMD is a distance between probability distributions over traces (or sampled traces). Low EMD does not imply the underlying visible trace sets are equal; rare but compliance-critical traces (e.g., two `Analyze` with one `Check`) may have small probability under assumed uniform choices and still be missed.
- Therefore, equal footprints or a low EMD can only add negative/positive evidence, never a proof of trace equivalence.

# 7. Complexity Comparison

```python
def tree_stats(tree):
    if tree.operator is None:
        label = tree.label
        return {'nodes':1, 'depth':1,
                'vis':0 if (label is None or str(label).lower().startswith('tau')) else 1,
                'silent':1 if (label is None or str(label).lower().startswith('tau')) else 0,
                'labels':[label]}
    stats = {'nodes':1, 'depth':0, 'vis':0, 'silent':0, 'labels':[]}
    max_d = 0
    for child in tree.children:
        cs = tree_stats(child)
        stats['nodes'] += cs['nodes']
        stats['vis'] += cs['vis']
        stats['silent'] += cs['silent']
        stats['labels'].extend(cs['labels'])
        max_d = max(max_d, cs['depth'])
    stats['depth'] = 1 + max_d
    return stats

for name, t in [("A", treeA), ("B", treeB)]:
    s = tree_stats(t)
    from collections import Counter
    dup = sum(v for v in Counter(s['labels']).values() if v > 1)
    print(name, "nodes:", s['nodes'], "depth(node count):", s['depth'],
          "visible leaves:", s['vis'], "silent leaves:", s['silent'],
          "duplicate-label count:", dup)
    net, im, fm = pm4py.convert_to_petri_net(t)
    print("  simplicity:", pm4py.simplicity_petri_net(net, im, fm))
```

Exact structural values derived from the trees:

| Metric | Model A | Model B |
|---|---|---|
| Process-tree node count | 12 | 12 |
| Depth (nodes / edges) | 5 / 4 | 5 / 4 |
| Visible leaves | 6 (`Open`, `Analyze`, `Check`, `Approve`, `Rework`, `Close`) | 6 |
| Silent leaves (`tau`) | 1 | 1 |
| Duplicate-label count in tree | 0 | 0 |

Both original trees contain the same operator multiset (one root sequence, one loop, one inner sequence, one parallel, one xor), so their converted WF-nets have the same node counts; `pm4py.simplicity_petri_net` returns equal scores for A and B (the script above prints the exact floats). The only structural difference is the nesting order of the loop and the parallel gateway.

# 8. Simplified Model

**Target semantics:** a clearly defined compromise — **P1 compliance with serialized intra-revision order**. We keep the mandatory per-revision `Check Revision` and the optional `Approve Revision`, but we replace the inner parallel gateway `+('Analyze Revision','Check Revision')` by the sequence `->('Analyze Revision','Check Revision')`. This removes concurrency inside a revision, making the tree smaller, while still satisfying the P1 rule that each analysis revision has its own check before the next `Rework Case` or `Close Case`.

**1. Compact process tree**

```text
->( 'Open Case', *( ->( 'Analyze Revision', 'Check Revision', X('Approve Revision', tau) ), 'Rework Case' ), 'Close Case' )
```

**2. PM4Py code**

```python
import pm4py

simplified_str = "->( 'Open Case', *( ->( 'Analyze Revision', 'Check Revision', X('Approve Revision', tau) ), 'Rework Case' ), 'Close Case' )"
simplified_model = pm4py.parse_process_tree(simplified_str)
netS, imS, fmS = pm4py.convert_to_petri_net(simplified_model)
print("Simplified sound:", pm4py.check_soundness(netS, imS, fmS, print_diagnostics=True))
```

The simplified tree has 11 nodes (vs 12 for A and B), removes the `+` operator, and therefore yields a Petri net with one fewer AND-split/AND-join.

**3. Preservation table**

| Property | Status in simplified model | Comment |
|---|---|---|
| `Open Case` first, `Close Case` last | exact | Root sequence unchanged. |
| Per-revision `Check Revision` (count = `#Analyze`) | exact | Loop still wraps the check; B’s single-check behavior is rejected. |
| `Check Revision` before next `Rework Case` / `Close Case` | exact | Sequence enforces it even more strictly than A’s parallel. |
| Optional `Approve Revision` after `Check Revision` | exact | `X('Approve Revision', tau)` retained. |
| Intra-revision concurrency (`Check` before `Analyze` allowed) | lost / weakened | Only `Analyze  Check` is now allowed; A also allowed `Check  Analyze`. |
| P2 dashboard abstraction (`#Analyze`, `Rework` flag, `Check` Boolean) | exact | Abstraction does not distinguish the lost concurrency. |
| Full P3 visible trace language | weakened | Simplified model is a strict subset of A’s visible language; not equivalent to B. |

**4. Regression and negative traces**

Regression traces (accepted by both A and the simplified model):

- R1: `Open Case  Analyze Revision  Check Revision  Close Case`
- R2: `Open Case  Analyze Revision  Check Revision  Approve Revision  Rework Case  Analyze Revision  Check Revision  Close Case`
- R3: `Open Case  Analyze Revision  Check Revision  Rework Case  Analyze Revision  Check Revision  Approve Revision  Close Case`

Negative traces (rejected by the simplified model):

- N1: `Open Case  Check Revision  Analyze Revision  Rework Case  Check Revision  Analyze Revision  Close Case`  
  Accepted by A (parallel order), rejected by S because `Check` must follow `Analyze` within a revision.
- N2: `Open Case  Analyze Revision  Rework Case  Analyze Revision  Check Revision  Close Case`  
  Accepted by B (single global check), rejected by S and by A because the second analysis revision lacks its own `Check`.

# 9. Final Recommendation

- **P1 (operational compliance):** Use **Model A**. Model B fails because it performs only one `Check Revision` for the whole loop. The simplified model in Section 8 may replace A only if the business explicitly agrees to forbid the `Check-before-Analyze` intra-revision order; otherwise it weakens P1 trace language, even though it preserves the core compliance predicate.
- **P2 (executive dashboard):** Both A and B are **equivalent only under the stated dashboard abstraction**. Either original is suitable for the dashboard. The simplified model (or an even smaller P2-only model that drops `Approve Revision` and the loop-internal parallelism) is also suitable and cheaper to maintain, because P2 already hides those details.
- **P3 (bounded testing):** The two originals are **not equivalent** even inside the bound. They must be tested as distinct models. Bounded comparison is useful but limited: here it found counterexamples, so non-equivalence is proven; however, if the bound had allowed only one `Analyze Revision`, the check would have wrongly suggested equivalence. Bounded equivalence should never be reported as proof of unbounded equivalence.
