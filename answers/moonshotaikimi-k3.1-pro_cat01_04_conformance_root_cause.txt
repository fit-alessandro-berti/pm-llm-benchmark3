from pm4py.objects.process_tree.obj import ProcessTree, Operator

def expand_variants(rows):
    """Expand every supplied variant count into individual traces."""
    lines = []
    for variant_string, count in rows:
        activities = variant_string.split("|")
        encoded_trace = " ".join(a.replace(" ", "_") for a in activities)
        lines.extend([encoded_trace] * count)

    log = pm4py.parse_event_log_string("\n".join(lines))

    for case_index, trace in enumerate(log):
        trace.attributes["concept:name"] = f"case-{case_index}"
        for event in trace:
            event["concept:name"] = event["concept:name"].replace("_", " ")
    return log


def normalize(log, drop_queue_routed=False):
    """
    Transparent diagnostic normalization:
      - always map Risk Screen -> Classify;
      - optionally remove Queue Routed.
    The input log is not modified.
    """
    result = EventLog()
    for trace in log:
        new_trace = Trace(attributes=dict(trace.attributes))
        for event in trace:
            old_name = event["concept:name"]

            if drop_queue_routed and old_name == "Queue Routed":
                continue

            new_event = Event(dict(event))
            if old_name == "Risk Screen":
                new_event["concept:name"] = "Classify"
            new_trace.append(new_event)

        result.append(new_trace)
    return result


def print_dfg(title, log, limit=10):
    dfg, starts, ends = pm4py.discover_dfg(log)
    print(f"\nDFG: {title}")
    print("start activities:", starts)
    print("end activities:", ends)
    edges = sorted(dfg.items(), key=lambda item: (-item[1], item[0]))
    for (source, target), frequency in edges[:limit]:
        print(f"  {source} -> {target}: {frequency}")


def alignment_report(title, log, petri_model, show_example=False):
    """Compute per-trace alignment diagnostics and aggregate alignment fitness."""
    net, initial_marking, final_marking = petri_model

    diagnostics = pm4py.conformance_diagnostics_alignments(
        log, net, initial_marking, final_marking
    )
    fitness = pm4py.fitness_alignments(
        log, net, initial_marking, final_marking
    )

    costs = [
        float(item["cost"])
        for item in diagnostics
        if isinstance(item, dict) and "cost" in item
    ]
    nonzero = [item for item in diagnostics if item.get("cost", 0) > 0]

    print(f"\nALIGNMENTS: {title}")
    print("  traces:", len(log))
    print("  mean alignment cost:",
          sum(costs) / len(costs) if costs else None)
    print("  nonzero-cost traces:", len(nonzero))
    print("  alignment fitness:", fitness)

    if show_example and nonzero:
        print("  example nonzero-cost diagnostic:", nonzero[0])

    return fitness


# 1-2. Encode and convert the normative process tree.
normative_tree = pm4py.parse_process_tree("""

    'Receive',
    'Classify',
    X (
        * ( 'Standard Review', 'Request Info' ),

normative_petri = pm4py.convert_to_petri_net(normative_tree)

logs = {period: expand_variants(rows) for period, rows in VARIANTS.items()}


# 6. Construct diagnostic normalization conditions.
june_label_only = normalize(logs["Jun-Jul"], drop_queue_routed=False)
june_normalized = normalize(logs["Jun-Jul"], drop_queue_routed=True)
august_normalized = normalize(logs["Aug-Sep"], drop_queue_routed=True)


# 4. Discover frequency DFGs on raw and normalized logs.
print_dfg("June-July raw", logs["Jun-Jul"])
print_dfg("June-July label mapped and Queue Routed removed", june_normalized)
print_dfg("August-September normalized", august_normalized)


# 5 and 7. Compare raw and normalized alignment diagnostics and fitness.
alignment_report(
    "April-May raw versus normative model",
    logs["Apr-May"],
    normative_petri
)
alignment_report(
    "June-July raw versus normative model",
    logs["Jun-Jul"],

alignment_report(

    june_label_only,
    normative_petri
)
alignment_report(
    "June-July label mapping plus Queue Routed removal",
    june_normalized,
    normative_petri

# Helpers for constructing the candidate tree programmatically.
def leaf(label):
    return ProcessTree(label=label)


def node(operator, *children):
    result = ProcessTree(operator=operator)
    result.children = list(children)
    for child in children:
        child.parent = result
    return result


# 8. Candidate: the original model plus an optional sign-off after the
# review branch and before Decision. An empty leaf is the process-tree tau.
standard_loop = node(
    Operator.LOOP,
    leaf("Standard Review"),
    leaf("Request Info")
)
enhanced_path = node(
    Operator.SEQUENCE,
    leaf("Enhanced Review"),
    leaf("Compliance Check")
)
review_choice = node(Operator.XOR, standard_loop, enhanced_path)

optional_signoff = node(
    Operator.XOR,
    leaf("Supervisor Sign-off"),
    ProcessTree()  # silent tau branch
)

candidate_tree = node(
    Operator.SEQUENCE,
    leaf("Receive"),
    leaf("Classify"),
    review_choice,
    optional_signoff,
    leaf("Decision"),
    leaf("Notify")
)
candidate_petri = pm4py.convert_to_petri_net(candidate_tree)


# Compare both models on exactly the same normalized August-September log.
# Source-sequence repair is deliberately not fabricated from variant strings.
for model_name, model in [
    ("normative", normative_petri),
    ("optional sign-off candidate", candidate_petri),
]:
    alignment_report(
        f"August-September normalized versus {model_name}",
        august_normalized,
        model
    )
    net, initial_marking, final_marking = model
    precision = pm4py.precision_alignments(
        august_normalized, net, initial_marking, final_marking
    )
    print(f"  alignment-based precision ({model_name}):", precision)

## 7. Provisional conclusion

**Primary explanation.** The measured collapse was predominantly caused by measurement and model drift rather than by a demonstrated sudden deterioration in reviewer behavior. The strongest evidence is the staged recovery: label mapping isolated the UI-related mismatch, technical-event removal isolated the queue-event contribution, and source-sequence ordering addressed a separate ETL ordering defect. In August, policy-required sign-offs additionally diverged from an unchanged normative tree. Confidence is high—approximately 90%—that this composite explanation dominates the reported drop, and very high that the individual label, queue, and ordering mechanisms contribute. This does not establish that every filtered or reordered event was harmless.

**Residual compliance assessment.** A real compliance problem may remain. The 55 enhanced-review cases without recorded checks survived the stated preprocessing and cannot yet be classified as bypasses, missing events, renamed activities, or eligibility errors. Overrides also require semantic and authorization review, but their use for routing correction means they must not automatically be counted as violations. Confidence that employee noncompliance explains the overall collapse is low; plausibility that it explains some residual cases is moderate, while confidence that the residual represents an unresolved control risk is high. Stable appeals, approvals, and reported incidents provide little exculpatory evidence.

**Immediate actions.** IT should preserve immutable raw logs and transformation versions, document event producers and schemas, restore source ordering without overwriting ingestion timestamps, investigate resource missingness, and independently re-extract the missing-check cases. Process owners should approve or reject the label mapping from functional evidence, determine whether queue routing belongs in the business model, and define the legitimate scope of overrides and sign-offs. Compliance should lead a blinded audit of all 55 missing-check cases, with risk-based samples of overrides and sign-offs, cross-classified by reviewer, extraction batch, value, and case complexity.

It must not yet be claimed that employees caused the collapse, that all deviations are artifacts, that all reversed traces are repaired, or that stable outcomes prove compliance. In particular, 89% fitting after preprocessing is not “89% compliant,” and 86% under an optional-sign-off model is not validation of the EUR 50,000 policy.

## 8. Model-governance recommendation

1. **Version normative models with policy effective dates.** Every conformance result must identify its model version, preprocessing version, policy basis, owner, and approval date. Policy changes must trigger a model-impact review before enforcement, with old and new models retained for effective-dated analysis.

2. **Maintain event-semantic contracts.** Business activities must have stable source function identifiers independent of screen labels, plus versioned definitions for lifecycle, case correlation, producer, and activity meaning. UI or integration releases must not silently change these semantics; mappings require documented approval and same-case validation.

3. **Govern technical events explicitly.** Technical events should be namespaced and distinguished from normative human activities at ingestion. Any conformance filter must be versioned, justified by provenance, applied to a preserved copy, and reported with before-and-after variants, DFGs, fitness, and precision.

4. **Preserve timestamp and ordering provenance.** Store source event time, ingestion time, source sequence, timezone, clock quality, and transformation lineage separately. Never overwrite source time with ingestion time; quarantine implausible orderings and use validated source ordering for conformance while documenting the time basis used for cycle-time metrics.

5. **Represent data-aware and resource-aware obligations.** Case value, policy effective date, review eligibility, and resource authorization must be completeness-checked and joined before conformance conclusions. Mandatory high-value sign-off must be implemented as a guard or eligible-stratum requirement, not approximated by an unrestricted optional activity.

6. **Require post-release process-mining validation.** UI, gateway, ETL, and logging releases must pass canary or dual-run checks comparing release-stratified variants, frequency DFGs, alignment diagnostics, fitness, precision, and data-quality rates against a baseline. IT, process owners, and compliance must jointly investigate material shifts and source-audit residual deviations before accepting either a model update or a behavioral explanation.
