@dataclass
class TimeInterval:
    """A UTC time interval [lo, hi) in seconds since epoch."""
    lo: float  # epoch seconds
    hi: float  # epoch seconds

    def midpoint(self) -> float:
        return (self.lo + self.hi) / 2

    def width(self) -> float:
        return self.hi - self.lo

    def __repr__(self):
        lo_dt = datetime.fromtimestamp(self.lo, tz=timezone.utc)
        hi_dt = datetime.fromtimestamp(self.hi, tz=timezone.utc)
        return f"[{lo_dt.isoformat()}, {hi_dt.isoformat()}]"


def parse_recorded(recorded_str: str, tz_name: Optional[str] = None,
                   force_offset_hours: Optional[int] = None) -> datetime:
    """Parse a recorded timestamp. If tz_name is given, interpret in that zone.
    If force_offset_hours is given (for DST disambiguation), use a fixed UTC offset."""
    if force_offset_hours is not None:
        naive = datetime.strptime(recorded_str, "%Y-%m-%d %H:%M:%S")
        if recorded_str.count(":") == 1:
            naive = datetime.strptime(recorded_str, "%Y-%m-%d %H:%M")
        fixed_tz = timezone(timedelta(hours=force_offset_hours))
        return naive.replace(tzinfo=fixed_tz)
    if tz_name:
        naive = datetime.strptime(recorded_str, "%Y-%m-%d %H:%M:%S")
        if recorded_str.count(":") == 1:
            naive = datetime.strptime(recorded_str, "%Y-%m-%d %H:%M")
        return naive.replace(tzinfo=ZoneInfo(tz_name))
    # HUB / SENSOR: UTC
    naive = datetime.strptime(recorded_str, "%Y-%m-%d %H:%M:%S")
    return naive.replace(tzinfo=timezone.utc)


def normalize_to_utc_intervals(recorded_str: str, system: str,
                               extra: dict) -> list[TimeInterval]:
    """
    Return a list of TimeInterval objects in UTC epoch seconds.
    For DST-ambiguous timestamps, returns multiple intervals.
    """
    meta = SOURCE_META[system]
    tz_name = extra.get("tz")
    dst_fold = extra.get("dst_fold", False)
    entry_delay_max = extra.get("entry_delay_max", 0)
    weak_offset = extra.get("weak_offset", False)

    # Determine offset bounds
    if weak_offset:
        off_lo, off_hi = -720, 720
    else:
        off_lo, off_hi = meta["off_lo"], meta["off_hi"]

    intervals = []

    if dst_fold:
        # Generate both CEST (+2) and CET (+1) branches
        for offset_hours in [2, 1]:
            dt = parse_recorded(recorded_str, force_offset_hours=offset_hours)
            dt_epoch = dt.timestamp()
            # Apply resolution spread
            clock_lo = dt_epoch + meta["res_lo"]
            clock_hi = dt_epoch + meta["res_hi"]
            # Apply entry delay (shifts occurrence earlier)
            clock_lo -= entry_delay_max
            # Apply clock offset: true = clock - offset, offset in [off_lo, off_hi]
            true_lo = clock_lo - off_hi
            true_hi = clock_hi - off_lo
            intervals.append(TimeInterval(true_lo, true_hi))
    else:
        dt = parse_recorded(recorded_str, tz_name=tz_name)
        dt_epoch = dt.timestamp()
        clock_lo = dt_epoch + meta["res_lo"]
        clock_hi = dt_epoch + meta["res_hi"]
        # Apply entry delay
        clock_lo -= entry_delay_max
        true_lo = clock_lo - off_hi
        true_hi = clock_hi - off_lo
        intervals.append(TimeInterval(true_lo, true_hi))

    return intervals

def feasible_duration(case_id: str, act_a: str, act_b: str,
                      causal: bool = True) -> Optional[tuple[float, float]]:
    """Return (min_dur, max_dur) in seconds. Returns None if events missing."""
    a_list = get_intervals(case_id, act_a)
    b_list = get_intervals(case_id, act_b)
    if not a_list or not b_list:
        return None

    a_lo = min(iv.lo for iv in a_list)
    a_hi = max(iv.hi for iv in a_list)
    b_lo = min(iv.lo for iv in b_list)
    b_hi = max(iv.hi for iv in b_list)

    dur_min = b_lo - a_hi
    dur_max = b_hi - a_lo

    if causal and dur_min < 0:
        dur_min = 0.0

    return (dur_min, dur_max)


def classify_sla(dur_min: float, dur_max: float, threshold: float) -> str:
    if dur_min > threshold:
        return "CERTAIN_VIOLATION"
    if dur_max <= threshold:
        return "CERTAIN_PASS"
    return "POSSIBLE_VIOLATION"


def classify_profile(dur_min: float, dur_max: float, avg: float, std: float,
                      zeta: float) -> str:

## 7. Governance Recommendation

### Timestamp Storage
- **Store all timestamps in UTC** at ingestion. Retain the original local-time representation and zone identifier as metadata, never as the primary timestamp.
- **Retain raw recorded timestamps** (pre-normalization) alongside the normalized UTC intervals. Any post-hoc reanalysis must be able to re-derive intervals from raw data.

### Clock-Offset Telemetry
- **Mandate NTP diagnostic logging** for all systems contributing incident timestamps. Record per-hour offset bounds, not just daily averages. T-409 demonstrates the risk of missing per-hour NTP data.
- **For external sources** (VENDOR-NY, MOBILE-LON), require a clock-synchronization attestation or flag events as "unsynchronized" with an explicit wide bound. Do not silently default to zero offset.

### Resolution Metadata
- **Record the display/rounding semantics** for every source system (floor-to-minute, round-to-10s, exact-to-second). This metadata must travel with the timestamp and be applied before offset correction.
- **Never discard sub-minute precision** by truncating to minute during ingestion; the rounding semantics are part of the evidence.

### Start/Complete Semantics
- **Distinguish start and complete events** explicitly in the schema. SENSOR completion events and missing Investigation_Start events (T-404) show that conflating lifecycle transitions creates inferential gaps.
- **Require Investigation_Start as a mandatory event**; do not accept duration-only work records as a substitute. If duration records are used, retain the full band (min, max) and the source of the estimate.

### DST Fold Handling
- **Log the DST fold indicator** (first occurrence vs. second occurrence) for every timestamp in a DST-affected zone during the fold window. T-405 shows that ingestion ordering can partially resolve ambiguity, but this is fragile and should not replace explicit logging.
- **Prefer UTC** to eliminate DST ambiguity entirely. If local time must be stored, include the UTC offset explicitly (e.g., `+02:00` or `+01:00`).

### Causal Identifiers
- **Add causal linkage identifiers** between Alert_Sent and Alert_Received, and between successive lifecycle events. T-407 demonstrates that without a receipt acknowledgment, Alert_Received must be inferred with wide bounds.
- **Log delivery latency** at the event level for cross-system notifications, not just network-level averages.

### Mobile Entry Delay Policy
- **Define whether user-entered timestamps reflect occurrence time or recording time**. T-406 requires an explicit policy: if the entered value is the user's estimate of occurrence time, entry delay widens the occurrence uncertainty; if it is the recording time, the occurrence time is unknown.
- **Recommend**: mobile timestamps should include both a `user_entered_time` and a `device_submit_time`, with the system clock of the receiving server as a cross-check.

### Interval-Overlap Enforcement Policy
- **Adopt a policy for whether POSSIBLE_VIOLATION triggers enforcement**. Options:
  - **Strict**: any feasible violation triggers a flagged review (conservative, may over-flag).
  - **Probabilistic**: require the violation fraction of the feasible interval to exceed a threshold (e.g., >50% of the interval violates).
  - **Preponderance**: flag only if the midpoint violates AND the interval does not fully contain a pass region.
- **Recommend the strict policy for S3 (Close_Incident)** because the consequence of a missed 36-hour closure is severe, and the **probabilistic policy for S1/S2** where clock-offset noise dominates.
- **Document the policy decision** and the tolerance level (0, 2, or 5 minutes) in the SLA contract. The sensitivity analysis in Section 4 shows that conclusions are highly sensitive to tolerance for marginal cases.

### Provenance and Auditability
- **Version all source metadata** (offset bounds, rounding rules, time-zone definitions). A reanalysis six months later must be able to reproduce the exact intervals.
- **Log every normalization decision** (DST branch elimination, causal constraints, inferred events) as an explicit annotation on the derived interval, not as silent code logic.
