def build_order_index(ocel, extra_edges=()):
    ev=ocel.events
    nodes = list(ev["eid"])
    edges = set()
    for a,b in extra_edges:
       if a in nodes and b in nodes: edges.add((a,b))
    # same source same original_timestamp source_seq edges
    for (src, ts), grp in ev.groupby(["source","original_timestamp"]):
       grp = grp.sort_values("source_seq")
       for a,b in zip(grp["eid"], grp["eid"].iloc[1:]): edges.add((a,b))
    # topological sort deterministic
    indeg={n:0 for n in nodes}; adj={n:[] for n in nodes}
    for a,b in edges: adj[a].append(b); indeg[b]+=1
    # priority key: original_timestamp, ingested_at, eid
    import heapq
    def key(n):
       r=ev.loc[ev.eid==n].iloc[0]
       return (str(r["original_timestamp"]), str(r.get("ingested_at","")), n)
    heap=[key(n) for n in nodes if indeg[n]==0]; heapq.heapify(heap)
    order=[]

    if len(order)!=len(nodes): raise ValueError("cycle")
    ev["order_index"] = ev["eid"].map({n:i for i,n in enumerate(order)})

```python
def _ensure_eid(ocel):

def _as_list(value):
    if value is None or (isinstance(value, float) and pd.isna(value)):
        return []
    if isinstance(value, str):
        try:
            parsed = json.loads(value)
            if isinstance(parsed, list):
                return parsed
        except Exception:
            return [value]
    if isinstance(value, (list, tuple, set)):

    return [value]

def verified_pair_merge(ocel, left_eid, right_eid, canonical_eid=None,
                        refined_activity=None, qualifier=None, rule_version="manual-verified-v1"):
    ev = ocel.events
    if canonical_eid is None:
        canonical_eid = left_eid
    if left_eid not in ev["eid"].values or right_eid not in ev["eid"].values:
        return ocel

    c_idx = ev.index[ev["eid"] == canonical_eid][0]
    d_idx = ev.index[ev["eid"] == right_eid][0]
    c = ev.loc[c_idx]
    d = ev.loc[d_idx]

    merged_from = sorted(set(_as_list(c.get("merged_from")) + _as_list(d.get("merged_from")) + [left_eid, right_eid]))
    source_keys = sorted(set(_as_list(c.get("source_event_key")) + _as_list(d.get("source_event_key"))))
    orig_acts = _as_list(c.get("original_activity")) + _as_list(d.get("original_activity"))
    orig_ts = [str(x) for x in (_as_list(c.get("original_timestamp")) + _as_list(d.get("original_timestamp")))]

    conflicts = {}
    for col in ev.columns:
        if col in PROTECTED_COLUMNS:
            continue
        a, b = c.get(col), d.get(col)
        if pd.notna(a) and pd.notna(b) and str(a) != str(b):
            conflicts[col] = {"canonical": str(a), "discarded": str(b)}

    ev.loc[c_idx, "merged_from"] = json.dumps(merged_from)
    ev.loc[c_idx, "source_event_keys"] = json.dumps(source_keys)
    ev.loc[c_idx, "original_activities"] = json.dumps(orig_acts)
    ev.loc[c_idx, "original_timestamps"] = json.dumps(orig_ts)
    ev.loc[c_idx, "provenance"] = json.dumps({"merge_rule": rule_version, "canonical": canonical_eid})
    ev.loc[c_idx, "conflicts"] = json.dumps(conflicts)
    if refined_activity is not None:
        ev.loc[c_idx, "activity"] = refined_activity
    if qualifier is not None:
        ev.loc[c_idx, "qualifier"] = qualifier

    rel = _relations_df(ocel)

        ocol = _object_col(rel)

            _set_relations(ocel, rel)

    ocel.events = ev
    return ocel

def build_order_index(ocel, extra_edges=()):
    ev = ocel.events
    nodes = list(ev["eid"])
    node_set = set(nodes)
    edges = set()
    for a, b in extra_edges:
        if a in node_set and b in node_set:
            edges.add((a, b))

    # Within-source sequence edges for same original business timestamp.
    if {"source", "original_timestamp", "source_seq", "eid"}.issubset(ev.columns):
        for _, grp in ev.groupby(["source", "original_timestamp"], dropna=False):
            grp = grp.sort_values("source_seq")
            ids = list(grp["eid"])
            for a, b in zip(ids, ids[1:]):
                edges.add((a, b))

    adj = {n: [] for n in nodes}
    indeg = {n: 0 for n in nodes}
    for a, b in edges:
        adj[a].append(b)
        indeg[b] += 1

    row_by_eid = {row["eid"]: row for _, row in ev.iterrows()}
    def heap_key(n):
        r = row_by_eid[n]
        return (str(r.get("original_timestamp", r.get("timestamp", ""))),
                str(r.get("ingested_at", "")),
                n)

    heap = [heap_key(n) for n in nodes if indeg[n] == 0]
    heapq.heapify(heap)
    order = []
    while heap:
        _, _, n = heapq.heappop(heap)
        order.append(n)
        for m in sorted(adj[n]):
            indeg[m] -= 1
            if indeg[m] == 0:
                heapq.heappush(heap, heap_key(m))
    if len(order) != len(nodes):
        raise ValueError("Ordering cycle detected; inspect semantic constraints.")

    ev["order_index"] = ev["eid"].map({n: i for i, n in enumerate(order)})
    ocel.events = ev
    return ocel
```

## 1. Duplicate-handling decision tree

Use stable provenance and object identity first; never use `activity + timestamp` alone as a duplicate key. The goal is to remove only what the evidence proves is the same technical or physical observation, while preserving abstraction-level events, object-specific concurrency, and source authority.

| Terminal category | Test / evidence | Conservative action | Aggressive alternative |
|---|---|---|---|
| Exact technical duplicate | Same `source` and same `source_event_key`, or same source-local `source_seq`, timestamp, activity, objects, and `payload_hash`. | **Drop** the redundant row, but retain its `source_event_key`, `source_seq`, `ingested_at`, and `payload_hash` in provenance. | **Drop** mechanically if the duplicate key is technical and source-local. |
| Same physical event observed by multiple systems | Verified reconciliation report, same business key, same critical object set, and compatible payloads. Example: `e03/e04` same scan; `e11/e16` same invoice. | **Merge attributes and relations** into one canonical event. Keep both source identifiers and original values. If attribute conflicts exist, do not overwrite; store conflicts. | **Merge** if a trusted business key exists. |
| Summary event derived from a detailed event | Source is an ERP/Billing/PSP summary, generated from another detailed event, and may lack lower-level objects. Example: `e07` from WMS packing; `e15` from payment messages. | Default: **retain both with refined labels or qualifiers**. Do not merge unless the summary has no unique attributes, no unique objects, and the process view only needs the physical event. | **Merge** or **drop** the summary only when provenance proves it is a pure derived duplicate. |
| Harmonization collision between semantically distinct source activities | Canonical activity is equal, but `original_activity`, source semantics, or lifecycle meaning differ. Example: `e08` warehouse release vs `e09` carrier acceptance. | **Retain both with refined labels or qualifiers**. Do not merge. If labels cannot be safely refined, **quarantine for review**. | **Merge** only if a domain rule explicitly says the two labels are equivalent. |
| Concurrent same-label events involving different objects | Same activity and timestamp, but critical object identities differ. Example: `e05` picks `I2`; `e14` captures `M2`. | **Retain both unchanged** or with object qualifiers. Do not merge based on shared `O1`, `V1`, or `P1`. | **Merge** only if the differing object is irrelevant to the analysis view and this loss is accepted. |
| Unresolved candidate | Equal activity/time/objects but conflicting payload, unknown business key, or ambiguous source semantics. | **Quarantine for review**. Keep raw event, mark `status = quarantine`, and exclude from canonical process model until reviewed. | **Merge** only after review; mechanical merging is unsafe. |

## 3. Timestamp-tie ordering rules

The pipeline must produce a deterministic total order for tools that require one, while keeping the underlying partial order visible. The total order is an analysis convention, not proof of physical sequence.

### Ordering policy

| Priority | Rule | Semantic or deterministic? | Use |
|---:|---|---|---|
| 1 | Known causal/domain constraints | Semantic | `e12` before `e13`; `e09` before `e10`; derived summary after source events where provenance is explicit |
| 2 | Lifecycle constraints on shared objects | Semantic | Picks before packs for `O1/I1/I2/P1`; package release before carrier handover; payment authorization before capture for the same `M1` |
| 3 | Source-specific semantics and refined labels | Semantic | Carrier acceptance is not warehouse dispatch; ERP summary is not the same abstraction level as PSP capture |
| 4 | Within-source sequence | Source-internal evidence | Use `source_seq` only within the same `source` and same original timestamp; cross-source `source_seq` is not comparable |
| 5 | Ingestion time | Deterministic technical tie-breaker | Use only after semantic and source-local constraints; never as primary business order |
| 6 | Stable fallback | Deterministic | Canonical `eid` lexicographic order |

Implementation pattern: preserve `original_timestamp`; add a separate `order_index`; optionally add `tie_group` or `concurrency_group`; then sort by `order_index`. If a deterministic timestamp is needed, derive it from `order_index` only after the partial order is built, and keep it separate from `original_timestamp`.

### Why adding microseconds before semantic ordering is dangerous

If `e03`, `e04`, `e05`, `e06`, and `e07` are assigned arbitrary microseconds such as `10:00:00.000`, `.001`, `.002`, `.003`, `.004`, the log may later be interpreted as:

```text
Pick I1 -> Pick I2 -> Pack detail -> Pack summary
```

with real durations and causal sequence. That story is not supported by the source evidence. It can create false rework, false precedence, false performance bottlenecks, and false object lifecycles. Microseconds should not be used to decide semantic order; they should only be a last-resort technical total order after provenance and constraints are applied.

## 4. Conservative cleaning pipeline

Minimum-change pipeline:

1. **Copy raw OCEL.** Never modify `ocel_raw` in place.
2. **Preserve original fields.** Add or copy `original_timestamp`, `original_activity`, `source`, `source_event_key`, `source_seq`, `ingested_at`, and `payload_hash`.
3. **Pre-check candidate duplicate groups.** Group by source-local keys, activity/timestamp, payload hash, and object sets. Inspect groups before any destructive operation.
4. **Remove only exact technical duplicates.** Use `pm4py.ocel_drop_duplicates` only for source-local exact duplicates, such as identical `source + source_event_key` rows.
5. **Apply verified same-physical-event merges.** Merge `e03/e04` and `e11/e16` using a custom verified-pair merge. Union non-conflicting attributes and relations; retain conflicts.
6. **Retain abstraction-level events.** Keep `e06` and `e07`; keep `e15`; refine labels or add qualifiers rather than deleting them.
7. **Refine harmonization collisions.** Keep `e08` and `e09`, but refine `e09` to `Carrier Acceptance` and `e08` to `Dispatch Package [WMS release]`.
8. **Build a partial order from semantic constraints.** Add known causal edges, object lifecycle edges, and within-source sequence edges.
9. **Produce a deterministic total order.** Topologically sort with ingestion time and `eid` as fallbacks. Store `order_index`; do not overwrite `original_timestamp`.
10. **Validate referential integrity.** Ensure all E2O relations point to existing events and objects, and all provenance columns remain present.

Expected retained event count: **15**. Raw count is 17; conservative merges remove `e04` and `e16` as canonical duplicates. Expected E2O relation count: **39**, assuming merged relations are unioned and duplicate E2O links are removed. Legitimate concurrency remains represented by distinct events and shared original timestamps; the total order is only an analysis overlay.

## 5. Aggressive alternatives

### Evidence-based semantic merging

This scenario accepts stronger business assumptions than the conservative pipeline:

- Merge `e03/e04` as one physical scan.
- Merge `e11/e16` as one invoice issuance.
- Merge `e06/e07` as one packing business transition, treating the ERP status as an abstraction-level duplicate of the WMS detail.
- Retain `e15` as an ERP payment summary, because it has no Payment object and may carry invoice-level balance attributes.

Expected retained event count: **14**. Expected E2O relation count: **37**, because `e07` relations are a subset of `e06` relations.

Most likely semantic damage: loss of the ERP packing observation as a separate event. This can hide WMS-to-ERP latency, remove an abstraction-level lifecycle step, and make the packing transition look instantaneous or single-source. If `e15` were also dropped or merged into `e13/e14`, the damage would be larger: the ERP invoice-level payment summary would disappear or partial payments would be conflated.

### Mechanically aggressive PM4Py merging

This scenario applies library duplicate merging broadly, for example:

```python
pm4py.ocel_merge_duplicates(
    ocel,
    attributes=["activity", "timestamp"],
    have_common_object=True,
    have_common_timestamp=True,
)
```

Expected retained event count: approximately **10**. Expected E2O relation count: approximately **28**.

Likely merges:

- `e03/e04/e05` into one `Pick Item` because they share `O1` and `P1`.
- `e06/e07` into one `Pack Package`.
- `e08/e09` into one `Dispatch Package`.
- `e11/e16` into one `Issue Invoice`.
- `e13/e14/e15` into one `Capture Payment` because they share `O1` and `V1`.

Most likely semantic damage: collapse of distinct physical and financial events. `e05` is a real pick of `I2`; `e14` is a real capture of `M2`; `e15` is an ERP summary with no Payment object. Mechanical merging destroys item-level and payment-level granularity.

`have_common_object=True` is not sufficient protection because `O1` and `V1` are global identifiers shared by many distinct events. `e13` and `e14` both touch `O1` and `V1`, but their critical distinguishing object is `Payment`: `M1` vs `M2`. Common order or invoice objects prove association, not identity. A safe merge rule must require equality of the event’s critical object type and identifier, or a verified business key.

```python
import copy
import json
import heapq
import warnings
import pandas as pd
import pm4py

PROTECTED_COLUMNS = {
    "eid", "original_timestamp", "original_activity", "source", "source_event_key",
    "source_seq", "ingested_at", "payload_hash", "merged_from", "source_event_keys",
    "original_activities", "original_timestamps", "provenance", "conflicts",
    "qualifier", "order_index"
}

def _ensure_eid(ocel):
    ev = ocel.events
    if "eid" not in ev.columns:
        ev = ev.copy()
        ev["eid"] = [str(i) for i in ev.index]
        ocel.events = ev
    return ocel

def ensure_preserved(ocel):
    ocel = _ensure_eid(ocel)
    ev = ocel.events
    if "original_timestamp" not in ev.columns:
        ev["original_timestamp"] = ev["timestamp"]
    if "original_activity" not in ev.columns:
        ev["original_activity"] = ev["activity"]
    for col in ["source", "source_event_key"]:
        if col not in ev.columns:
            ev[col] = None
    ocel.events = ev
    return ocel

def _relations_df(ocel):
    for attr in ("relations", "events_objects", "object_relations"):
        if hasattr(ocel, attr):
            df = getattr(ocel, attr)
            if isinstance(df, pd.DataFrame):
                return df
    return None

def _set_relations(ocel, df):
    for attr in ("relations", "events_objects", "object_relations"):
        if hasattr(ocel, attr):
            setattr(ocel, attr, df)
            return

def _event_col(rel):
    for c in ("event_id", "eid", "event"):
        if c in rel.columns:
            return c
    return None

def _object_col(rel):
    for c in ("object_id", "oid", "object"):
        if c in rel.columns:
            return c
    return None

def candidate_duplicate_groups(ocel):
    ev = ocel.events
    groups = {}
    if {"source", "source_event_key"}.issubset(ev.columns):
        groups["exact_source_key"] = ev.groupby(
            ["source", "source_event_key"], dropna=False
        ).size().to_dict()
    if {"activity", "timestamp"}.issubset(ev.columns):
        groups["activity_timestamp"] = ev.groupby(
            ["activity", "timestamp"], dropna=False
        ).size().to_dict()
    if {"activity", "timestamp", "payload_hash"}.issubset(ev.columns):
        groups["same_physical_candidate"] = ev.groupby(
            ["activity", "timestamp", "payload_hash"], dropna=False
        ).size().to_dict()
    return groups

def _as_list(value):
    if value is None or (isinstance(value, float) and pd.isna(value)):
        return []
    if isinstance(value, str):
        try:
            parsed = json.loads(value)
            if isinstance(parsed, list):
                return parsed
        except Exception:
            return [value]
    if isinstance(value, (list, tuple, set)):
        return list(value)
    return [value]

def verified_pair_merge(
    ocel,
    left_eid,
    right_eid,
    canonical_eid=None,
    refined_activity=None,
    qualifier=None,
    rule_version="manual-verified-v1"
):
    """
    Merge two verified events into one canonical event.

    This helper is intentionally conservative:
    - it preserves original source metadata;
    - it stores conflicts instead of silently overwriting them;
    - it unions E2O relations and removes duplicate relation rows.
    """
    ev = ocel.events
    if canonical_eid is None:
        canonical_eid = left_eid

    if left_eid not in ev["eid"].values or right_eid not in ev["eid"].values:
        return ocel

    c_idx = ev.index[ev["eid"] == canonical_eid][0]
    d_idx = ev.index[ev["eid"] == right_eid][0]
    c = ev.loc[c_idx]
    d = ev.loc[d_idx]

    merged_from = sorted(set(
        _as_list(c.get("merged_from")) +
        _as_list(d.get("merged_from")) +
        [left_eid, right_eid]
    ))
    source_keys = sorted(set(
        _as_list(c.get("source_event_key")) +
        _as_list(d.get("source_event_key"))
    ))
    orig_acts = _as_list(c.get("original_activity")) + _as_list(d.get("original_activity"))
    orig_ts = [
        str(x)
        for x in (_as_list(c.get("original_timestamp")) + _as_list(d.get("original_timestamp")))
    ]

    conflicts = {}
    for col in ev.columns:
        if col in PROTECTED_COLUMNS:
            continue
        a, b = c.get(col), d.get(col)
        if pd.notna(a) and pd.notna(b) and str(a) != str(b):
            conflicts[col] = {"canonical": str(a), "discarded": str(b)}

    ev.loc[c_idx, "merged_from"] = json.dumps(merged_from)
    ev.loc[c_idx, "source_event_keys"] = json.dumps(source_keys)
    ev.loc[c_idx, "original_activities"] = json.dumps(orig_acts)
    ev.loc[c_idx, "original_timestamps"] = json.dumps(orig_ts)
    ev.loc[c_idx, "provenance"] = json.dumps({
        "merge_rule": rule_version,
        "canonical": canonical_eid
    })
    ev.loc[c_idx, "conflicts"] = json.dumps(conflicts)

    if refined_activity is not None:
        ev.loc[c_idx, "activity"] = refined_activity
    if qualifier is not None:
        ev.loc[c_idx, "qualifier"] = qualifier

    rel = _relations_df(ocel)
    if rel is not None:
        rel = rel.copy()
        ecol = _event_col(rel)
        ocol = _object_col(rel)
        if ecol and ocol:
            rel.loc[rel[ecol] == right_eid, ecol] = canonical_eid
            rel = rel.drop_duplicates(subset=[ecol, ocol])
            _set_relations(ocel, rel)

    ev = ev.drop(index=d_idx).copy()
    ocel.events = ev
    return ocel

def drop_exact_technical_duplicates(ocel):
    """
    Justified only for exact technical duplicates, e.g. same source-local key.
    """
    groups = candidate_duplicate_groups(ocel)
    exact = groups.get("exact_source_key", {})
    if any(v > 1 for v in exact.values()):
        try:
            ocel = pm4py.ocel_drop_duplicates(
                ocel,
                attributes=["source", "source_event_key", "timestamp", "activity", "payload_hash"]
            )
        except TypeError:
            ev = ocel.events
            ev = ev.drop_duplicates(subset=["source", "source_event_key"], keep="first").copy()
            ocel.events = ev
    return ocel

def build_order_index(ocel, extra_edges=()):
    """
    Build a deterministic total order from a semantic partial order.
    The original business timestamp is never overwritten.
    """
    ev = ocel.events
    nodes = list(ev["eid"])
    node_set = set(nodes)
    edges = set()

    for a, b in extra_edges:
        if a in node_set and b in node_set:
            edges.add((a, b))

    # Within-source sequence is used only within the same source and timestamp.
    if {"source", "original_timestamp", "source_seq", "eid"}.issubset(ev.columns):
        for _, grp in ev.groupby(["source", "original_timestamp"], dropna=False):
            grp = grp.sort_values("source_seq")
            ids = list(grp["eid"])
            for a, b in zip(ids, ids[1:]):
                edges.add((a, b))

    adj = {n: [] for n in nodes}
    indeg = {n: 0 for n in nodes}
    for a, b in edges:
        adj[a].append(b)
        indeg[b] += 1

    row_by_eid = {row["eid"]: row for _, row in ev.iterrows()}

    def heap_key(n):
        r = row_by_eid[n]
        return (
            str(r.get("original_timestamp", r.get("timestamp", ""))),
            str(r.get("ingested_at", "")),
            n
        )

    heap = [heap_key(n) for n in nodes if indeg[n] == 0]
    heapq.heapify(heap)
    order = []

    while heap:
        _, _, n = heapq.heappop(heap)
        order.append(n)
        for m in sorted(adj[n]):
            indeg[m] -= 1
            if indeg[m] == 0:
                heapq.heappush(heap, heap_key(m))

    if len(order) != len(nodes):
        raise ValueError("Ordering cycle detected; inspect semantic constraints.")

    ev["order_index"] = ev["eid"].map({n: i for i, n in enumerate(order)})
    ocel.events = ev
    return ocel

def validate_referential_integrity(ocel):
    ev = ocel.events
    for col in ["original_timestamp", "original_activity", "source", "source_event_key"]:
        if col not in ev.columns:
            raise AssertionError(f"Missing preserved column: {col}")

    rel = _relations_df(ocel)
    if rel is None:
        return

    ecol = _event_col(rel)
    ocol = _object_col(rel)

    if ecol:
        missing = set(rel[ecol]) - set(ev["eid"])
        if missing:
            raise AssertionError(f"Relations point to missing events: {missing}")

    if hasattr(ocel, "objects") and isinstance(ocel.objects, pd.DataFrame) and ocol:
        obj_col = _object_col(ocel.objects)
        if obj_col:
            missing_obj = set(rel[ocol]) - set(ocel.objects[obj_col])
            if missing_obj:
                raise AssertionError(f"Relations point to missing objects: {missing_obj}")

def make_conservative(ocel_raw):
    ocel = copy.deepcopy(ocel_raw)
    ensure_preserved(ocel)

    # Pre-check candidate groups before any destructive action.
    _ = candidate_duplicate_groups(ocel)

    # Remove only exact technical duplicates.
    ocel = drop_exact_technical_duplicates(ocel)

    # Strongly supported same-physical-event merges only.
    ocel = verified_pair_merge(
        ocel, "e03", "e04",
        canonical_eid="e03",
        refined_activity="Pick Item",
        qualifier="verified same physical scan; WMS location enriched"
    )
    ocel = verified_pair_merge(
        ocel, "e11", "e16",
        canonical_eid="e11",
        refined_activity="Issue Invoice",
        qualifier="same invoice V1; ERP + Billing merge"
    )

    # Refine labels; do not delete abstraction-level or harmonization-collision events.
    ev = ocel.events
    label_refinements = {
        "e07": ("Pack Package [ERP summary]", "ERP status update from WMS"),
        "e08": ("Dispatch Package [WMS release]", "released from warehouse"),
        "e09": ("Carrier Acceptance", "original carrier acceptance harmonized to Dispatch Package"),
        "e15": ("Capture Payment [ERP summary]", "ERP invoice-level balance update derived from PSP captures"),
    }
    for eid, (activity, qual) in label_refinements.items():
        mask = ev["eid"] == eid
        ev.loc[mask, "activity"] = activity
        ev.loc[mask, "qualifier"] = qual
    ocel.events = ev

    extra_edges = [
        ("e03", "e06"), ("e03", "e07"),
        ("e05", "e06"), ("e05", "e07"),
        ("e06", "e07"),
        ("e09", "e10"),
        ("e12", "e13"),
        ("e13", "e15"), ("e14", "e15"),
    ]
    ocel = build_order_index(ocel, extra_edges=extra_edges)
    ocel = pm4py.ocel_sort_by_additional_column(ocel, "order_index")
    ocel = pm4py.ocel_add_index_based_timedelta(ocel)
    validate_referential_integrity(ocel)
    return ocel

def make_evidence_based(ocel_raw):
    ocel = make_conservative(ocel_raw)

    # Evidence-based scenario: merge packing detail and ERP status as one transition.
    ocel = verified_pair_merge(
        ocel, "e06", "e07",
        canonical_eid="e06",
        refined_activity="Pack Package",
        qualifier="evidence-based merge of WMS detail and ERP status as one packing transition"
    )

    extra_edges = [
        ("e03", "e06"), ("e05", "e06"),
        ("e09", "e10"),
        ("e12", "e13"),
        ("e13", "e15"), ("e14", "e15"),
    ]
    ocel = build_order_index(ocel, extra_edges=extra_edges)
    ocel = pm4py.ocel_sort_by_additional_column(ocel, "order_index")
    ocel = pm4py.ocel_add_index_based_timedelta(ocel)
    validate_referential_integrity(ocel)
    return ocel

def make_aggressive_pm4py(ocel_raw):
    warnings.warn(
        "Aggressive PM4Py merging is for sensitivity testing only. "
        "have_common_object=True can collapse distinct partial payments and item picks.",
        RuntimeWarning
    )

    ocel = copy.deepcopy(ocel_raw)
    ensure_preserved(ocel)
    _ = candidate_duplicate_groups(ocel)

    # Mechanical merge: equal activity + equal timestamp + any common object.
    ocel = pm4py.ocel_merge_duplicates(
        ocel,
        attributes=["activity", "timestamp"],
        have_common_object=True,
        have_common_timestamp=True,
    )

    ocel = build_order_index(ocel, extra_edges=())
    ocel = pm4py.ocel_sort_by_additional_column(ocel, "order_index")
    ocel = pm4py.ocel_add_index_based_timedelta(ocel)
    validate_referential_integrity(ocel)
    return ocel
```

## 7. Sensitivity analysis

Assumed counts for this dataset:

- Raw: 17 events, 44 E2O relations.
- Conservative: 15 events, 39 E2O relations.
- Evidence-based merge: 14 events, 37 E2O relations.
- Aggressive PM4Py merge: approximately 10 events, approximately 28 E2O relations.

| Aspect | Raw | Conservative | Evidence-based merge | Aggressive PM4Py merge |
|---|---|---|---|---|
| Event count and E2O relation count | 17 / 44 | 15 / 39 | 14 / 37 | ~10 / ~28 |
| Order-flattened DFG | Many same-label nodes and possible self-loop candidates; tie order is ambiguous | Refined labels separate WMS, ERP, scanner, carrier, PSP, Billing; stable `order_index` | ERP packing summary removed; one packing node remains | Pick, Pack, Dispatch, Invoice, and Capture nodes may collapse; fewer nodes and false direct edges |
| Item-flattened DFG | `e03/e04` duplicate may create a false `Pick Item -> Pick Item` loop on `I1`; `e05` shows `I2` | `I1` and `I2` remain distinct picks; no false self-loop from `e03/e04` | Same as conservative for items | `e03/e04/e05` may merge; item-specific pick sequence for `I2` can disappear |
| Payment-flattened DFG | `M1` and `M2` captures share label; ERP summary can appear as another capture loop | `M1` and `M2` remain distinct; ERP summary has no Payment object | Same as conservative for payments | `e13/e14/e15` may merge; partial payments and ERP summary collapse into one capture event |
| OC-DFG edges and frequency metrics | Frequencies inflated by duplicate scans, duplicate invoice issuance, duplicate dispatch, duplicate captures | Frequencies reflect verified physical events and abstraction labels | Packing frequency reduced by one; ERP abstraction edge removed | Frequencies distorted by unioned relations; false co-occurrences can appear |
| ET–OT weights | `Capture Payment` high to `Payment` and `Invoice`; `Dispatch Package` high to `Delivery` due duplicate carrier/WMS records | Weights split by refined labels; `Carrier Acceptance` appears as its own event type | `Pack Package` weight to `Item` and `Package` comes only from WMS detail | `Capture Payment` may be unioned across `M1`, `M2`, and `V1`; `Carrier Acceptance` type may disappear |
| OTG relations | Many object-type co-occurrences are overrepresented by duplicates | Preserves `Item-Package`, `Payment-Invoice`, `Delivery-Package`, and order-level links | Preserves most object types but loses ERP packing observation | May create false object-type associations, such as one capture event linked to both `M1` and `M2` |
| Apparent rework or loops | False loops: `Pick Item -> Pick Item`, `Pack Package -> Pack Package`, `Dispatch Package -> Dispatch Package`, `Issue Invoice -> Issue Invoice`, `Capture Payment -> Capture Payment` | Loops mostly removed; summary edges are labeled as abstraction transitions, not rework | Fewer loops; may hide true integration latency | Loops removed, but true concurrency and distinct object events are also removed |
| Lifecycle and performance measurements | Same-second ties are ambiguous; ingestion order can mislead if used | Original timestamps preserved; `order_index` is deterministic but not treated as physical proof | Packing latency between WMS and ERP is hidden | Partial payment latency, ERP summary latency, and carrier acceptance latency are compressed or lost |

Concrete edges or patterns whose interpretation changes:

1. `Pick Item -> Pick Item`: raw may look like rework; conservative resolves it into `e03/e04` merge plus distinct `e05`; aggressive may erase `I2`.
2. `Pack Package -> Pack Package`: raw may look like duplicate packing; conservative refines it into detail vs summary; evidence-based merge removes the edge.
3. `Dispatch Package -> Dispatch Package`: raw may look like duplicate dispatch; conservative refines it into warehouse release vs carrier acceptance.
4. `Issue Invoice -> Issue Invoice`: raw may look like duplicate invoicing; conservative merges ERP and Billing as one invoice issuance.
5. `Capture Payment -> Capture Payment`: raw may look like repeated payment capture; conservative preserves `M1`, `M2`, and ERP summary as distinct roles.
6. `Carrier Acceptance -> Handover to Carrier`: conservative makes this meaningful; aggressive may lose `Carrier Acceptance` by merging it into `Dispatch Package`.
7. `Authorize Payment -> Capture Payment`: preserved in conservative and evidence-based; can be distorted in aggressive if capture events are merged across payment objects.

## 8. Audit trail and reversibility

A provenance schema should be separate from the cleaned OCEL but linked by canonical event IDs.

| Field | Meaning |
|---|---|
| `canonical_event_id` | Stable ID used in the cleaned OCEL, e.g. `e03` after merging `e04` |
| `source_event_ids` | All raw event IDs contributing to the canonical event, e.g. `["e03", "e04"]` |
| `merge_rule` | Rule ID, e.g. `verified_same_physical_scan_v1` or `same_invoice_v1` |
| `reviewer_or_rule_version` | Human reviewer, reconciliation report ID, or rule version |
| `original_activity` | Raw activity per source event |
| `refined_activity` | Cleaned or qualified activity used in the canonical view |
| `original_timestamp` | Raw UTC timestamp per source event |
| `adjusted_timestamp` | Synthetic or index-based timestamp, if used; never replaces original |
| `order_index` | Deterministic total-order position |
| `attributes_retained_from_each_source` | Source-to-attribute map, e.g. WMS location, Billing invoice attributes |
| `discarded_conflicts` | Conflicting values not written into the canonical row, with reason |
| `confidence` | High, medium, low |
| `reason` | Human-readable evidence, e.g. “reconciliation report says same physical scan” |

Example JSON record:

```json
{
  "canonical_event_id": "e03",
  "source_event_ids": ["e03", "e04"],
  "merge_rule": "verified_same_physical_scan_v1",
  "reviewer_or_rule_version": "reconciliation-2026-05-20",
  "original_activity": {"e03": "Pick Item", "e04": "Pick Item"},
  "refined_activity": "Pick Item",
  "original_timestamp": {"e03": "2026-05-18T10:00:00Z", "e04": "2026-05-18T10:00:00Z"},
  "adjusted_timestamp": null,
  "order_index": 2,
  "attributes_retained_from_each_source": {
    "e03": {"payload_hash": "h-pick-i1", "location": "WMS-enriched"},
    "e04": {"payload_hash": "h-pick-i1-alt", "scanner_id": "S-901"}
  },
  "discarded_conflicts": {
    "payload_hash": {
      "canonical": "h-pick-i1",
      "discarded": "h-pick-i1-alt",
      "reason": "same physical scan; payloads differ due enrichment"
    }
  },
  "confidence": "high",
  "reason": "Reconciliation report confirms e03 and e04 are the same physical scan."
}
```

An analyst can reconstruct the raw view by expanding each canonical event into its `source_event_ids`, restoring each source event’s `original_activity`, `original_timestamp`, `source`, `source_event_key`, and `payload_hash`, and restoring E2O relations from the per-source provenance. Dropped or quarantined events should remain in a separate table with status, reason, and original identifiers. Because the raw OCEL is never modified in place, reconstruction is always possible from raw plus audit records.

## 9. Failure modes and safeguards

1. **Collapsing distinct partial payments**  
   Failure: `e13` and `e14` are merged because both are `Capture Payment` at the same time and share `O1/V1`.  
   Safeguard: require equality of the critical object type and identifier, here `Payment` (`M1` vs `M2`), before merging.

2. **Converting abstraction-level updates into false rework**  
   Failure: `e06` and `e07`, or `e13` and `e15`, appear as repeated activities and are interpreted as rework.  
   Safeguard: refine labels or add qualifiers; track `derived_from`; do not treat summary events as physical duplicates.

3. **Ordering by ingestion time**  
   Failure: `ingested_at` becomes the primary business order, making ERP summaries appear before source events.  
   Safeguard: use `original_timestamp` and semantic constraints first; use ingestion time only as a deterministic fallback.

4. **Treating shared objects as proof of duplication**  
   Failure: `have_common_object=True` merges unrelated events that share `O1`, `V1`, or `P1`.  
   Safeguard: require a verified business key or same critical object identity; common global objects are insufficient.

5. **Unioning incompatible E2O relations**  
   Failure: merging `e15` with `e13/e14` adds `M1` or `M2` to an ERP summary that intentionally has no Payment object.  
   Safeguard: only union relations for verified same-physical events; otherwise keep relation sets separate and record derivation.

6. **Changing durations through synthetic microseconds**  
   Failure: arbitrary microsecond ordering creates false durations and false causal edges.  
   Safeguard: preserve `original_timestamp`; use `order_index` for total order; do not feed synthetic timestamps into performance analysis unless explicitly marked.

7. **Losing source-authoritative attributes**  
   Failure: Billing invoice attributes are overwritten by ERP values during `e11/e16` merge.  
   Safeguard: store per-source attributes and conflicts; use a source-preference policy only when the business key and attribute authority are known.

8. **Applying duplicate functions to the whole OCEL without inspecting candidate groups**  
   Failure: `ocel_merge_duplicates` silently removes real events such as `e05` or `e14`.  
   Safeguard: run candidate-group pre-checks, require verified pair lists, execute on deep copies, and compare sensitivity metrics before accepting a pipeline.

9. **Creating ordering cycles from overconfident lifecycle rules**  
   Failure: adding too many semantic precedence constraints makes topological sorting impossible.  
   Safeguard: separate mandatory domain constraints from optional assumptions; quarantine conflicting edges for review.

10. **Hiding legitimate concurrency**  
   Failure: a deterministic total order is presented as if all events were strictly sequential.  
   Safeguard: retain original timestamps, add concurrency groups or tie groups, and state that total order is only an analysis convention.
