state_cols = ["amount", "risk_level", "jurisdiction", "customer_type", "exemption_id"]

df["prev_amount"] = df.groupby("case:concept:name")["amount"].shift(1)

    df["prev_amount"].notna() & df["amount"].notna() & (df["prev_amount"] > 0),
    (df["amount"] / df["prev_amount"] - 1.0) * 100.0,
    np.nan,
)

df["is_material_amount_increase"] = df["is_update_amount"] & df["amount_increase_pct"].gt(20)
df["is_jurisdiction_update"] = df["concept:name"].eq("Update_Jurisdiction")
df["is_customer_update"] = df["concept:name"].eq("Update_Customer_Type")
df["is_risk_update"] = df["concept:name"].eq("Risk_Reclassify")
df["is_material_change"] = (
    df["is_material_amount_increase"]
    | df["is_jurisdiction_update"]
    | df["is_customer_update"]
    | df["is_risk_update"]
)

def add_latest_ts(df, activity, col):
    df[col] = pd.Series(pd.NaT, index=df.index, dtype="datetime64[ns, UTC]")
    df.loc[df["concept:name"].eq(activity), col] = df["time:timestamp"]
    df[col] = df.groupby("case:concept:name")[col].ffill()
    return df

def add_latest_order(df, condition, col):
    df[col] = np.nan
    df.loc[condition, col] = df["event_order"]
    df[col] = df.groupby("case:concept:name")[col].ffill()
    return df

def add_latest_value(df, condition, col, value_col):
    df[col] = np.nan
    df.loc[condition, col] = df[value_col]
    df[col] = df.groupby("case:concept:name")[col].ffill()
    return df

for activity, col in [
