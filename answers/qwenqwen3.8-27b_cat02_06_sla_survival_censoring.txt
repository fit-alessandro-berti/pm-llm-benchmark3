def cumulative_incidence(data, horizon, t="t_an", event="event", comp="comp", censor="censor"):
    d = data[data[t].notna()].copy()
    d = d[d[t] <= horizon]
    times = sorted(d[t].unique())
    F=0.0; S_total=1.0; curve=[]
    for s in times:
        n = int(((d[t] >= s)).sum())
        if n == 0: continue
        d_event = int(((d[t] == s) & (d[event] == 1)).sum())
        d_total = int(((d[t] == s) & ((d[event] == 1) | (d[comp] == 1))).sum())
        if d_event > 0 and S_total > 0:
            F += (d_event / n) * S_total
        if d_total > 0:
            S_total *= (1.0 - d_total / n)
        curve.append((s, F, S_total, n))
    # find F at horizon (last <= horizon)
    if curve:
        F_horizon = max([c[1] for c in curve if c[0] <= horizon], default=0.0)
    else: F_horizon=0.0
    return F_horizon, pd.DataFrame(curve, columns=["time","CIF","S_total","n_at_risk"])

def build_pm4py_features(df):
    if PM4PY_AVAILABLE:
        try:
            try:
                EventLog = pm4py.OOP.EventLog
                Event = pm4py.OOP.Event
            except AttributeError:
                EventLog = pm4py.EventLog
                Event = pm4py.Event
            events = []
            for _, r in df.iterrows():
                for act in r["prefix"].split(">"):
                    act = act.strip()
                    if act:
                        events.append(Event(case_id=r["case"], activity=act))
            log = EventLog()
            for ev in events:
                log.append(ev)
            variants = pm4py.get_variants(log)
            variant_counts = pd.Series(variants).sort_values(ascending=False)
            used_pm4py = True
        except Exception as exc:
            print("PM4Py variant extraction failed; using fallback:", exc)
            variant_counts = None
            used_pm4py = False
    else:
        variant_counts = None
        used_pm4py = False

    variant_series = df["prefix"].apply(lambda p: tuple(a.strip() for a in p.split(">") if a.strip()))
    if variant_counts is None:
        variant_counts = variant_series.value_counts()
    out = df.copy()
    out["variant"] = variant_series
    out["variant_count"] = variant_series.map(lambda v: int(variant_counts.get(v, 0)))
    out["path_length"] = variant_series.apply(len)
    out["has_autocheck"] = variant_series.apply(lambda v: "AutoCheck" in v)
    out["has_escalate"] = variant_series.apply(lambda v: "Escalate" in v)
    out["has_rework"] = variant_series.apply(lambda v: "Rework" in v)
    out["review_count"] = variant_series.apply(lambda v: v.count("Review"))
    out["last_activity"] = variant_series.apply(lambda v: v[-1] if v else "")
    return out, variant_counts, used_pm4py

def encode_late(df, scenario):

def cumulative_incidence(data, horizon, t="t_an", event="event", comp="comp", censor="censor"):
    d = data[data[t].notna()].copy()
    d = d[d[t] <= horizon]
    times = sorted(d[t].unique())
    F = 0.0
    S_total = 1.0
    curve = []
    for s in times:
        n = int((d[t] >= s).sum())
        if n == 0:
            continue
        d_event = int(((d[t] == s) & (d[event] == 1)).sum())
        d_total = int(((d[t] == s) & ((d[event] == 1) | (d[comp] == 1))).sum())
        if d_event > 0 and S_total > 0:
            F += (d_event / n) * S_total
        if d_total > 0:
            S_total *= (1.0 - d_total / n)
        curve.append((s, F, S_total, n))
    vals = [c[1] for c in curve if c[0] <= horizon]
    F_horizon = vals[-1] if vals else 0.0
    return F_horizon, pd.DataFrame(curve, columns=["time", "CIF", "S_total", "n_at_risk"])

def km_survival(data, horizon, t="t_an", event="event", censor="censor"):
    d = data[data[t].notna()].copy()
    d = d[d[t] <= horizon]
    times = sorted(d[t].unique())
    S = 1.0
    curve = []
    for s in times:
        n = int((d[t] >= s).sum())
        if n == 0: continue
        d_event = int(((d[t] == s) & (d[event] == 1)).sum())
        if d_event > 0:
            S *= (1.0 - d_event / n)
        curve.append((s, S, n))
    vals = [c[1] for c in curve if c[0] <= horizon]
    S_horizon = vals[-1] if vals else 1.0
    return S_horizon, pd.DataFrame(curve, columns=["time", "KM_S", "n_at_risk"])

# 1. SLA estimands and time definitions

**Time origin and horizon.**  
Let `t = 0` at `Submit`. The SLA horizon is `H = 72` hours. For terminal rows, `duration_or_age_h` is the observed time from `Submit` to the terminal outcome. For `open` rows, `duration_or_age_h` is the age at the observation cutoff and is a right-censoring time, not a final outcome. For at-72-hour estimands, cases without the relevant event by 72 hours are administratively censored at 72 hours. This does not mean they resolved; it means the at-72-hour outcome is known.

**Events, censoring, and competing risks.**  
- **Event of interest** depends on the estimand: timely resolution, late resolution, or first internal terminal disposition.  
- **Right censoring** occurs for `open` cases at their age at cutoff. It may also be used for `transferred` cases under an external-censoring assumption. Administrative truncation at 72 hours is also a form of censoring for at-72 estimands.  
- **Competing events** are terminal outcomes that prevent the event of interest. For resolution, `Reject`, `Withdraw`, and possibly `Transfer` are competing non-resolution outcomes.  
- **Already-observed breaches** include cases resolved after 72 hours, such as S03 at 76 hours and S13 at 110 hours, and cases still open after 72 hours, such as S09 at 92 hours and S26 at 104 hours. These are observed failures for an at-72 resolution SLA, even if their final outcome is not yet known.

## Estimand 1: Timely resolution by 72 hours

Define:

`E1 = P(T_resolve <= 72)`

where `T_resolve` is the resolution time and non-resolution outcomes are treated as competing risks. The complement is:

`P(not resolved by 72) = 1 - E1`

This estimand is best for the operational question: “What share of cases meet the SLA by reaching `Resolve` before 72 hours?” It is appropriate for SLA governance, cohort comparison, and customer-facing success metrics. It should be estimated with a cumulative-incidence method, not ordinary Kaplan-Meier, because rejection, withdrawal, and transfer can prevent later resolution.

## Estimand 2: No terminal disposition by 72 hours

Define:

`E2 = P(T_terminal > 72)`

where `T_terminal` is the first internal terminal disposition, for example `Resolve`, `Reject`, `Withdraw`, and possibly `Transfer` if treated as an internal terminal state. This estimand answers: “What share of cases are still unresolved / undisposed at the SLA deadline?” It is best for workload, backlog, and operational escalation decisions. It is distinct from `E1` because a case can be not resolved by 72 hours either because it is still open or because it was rejected/withdrawn/transferred before 72 hours.

## Estimand 3: Cumulative incidence of late resolution after 72 hours

Define:

`E3 = P(72 < T_resolve <= H_late)`

where `H_late` can be the maximum observed follow-up, here 110 hours, or a chosen operational horizon. This estimand answers: “Among cases that do not meet the SLA, what share later resolve?” It is useful for tail-risk recovery, customer impact, and deciding whether late resolution is an acceptable operational outcome. It requires a competing-risk cumulative-incidence model because timely resolution, rejection, withdrawal, and transfer can prevent late resolution.

## Decision matching

| Operational decision | Best estimand |
|---|---|
| Did the case meet the SLA? | `E1`, timely resolution by 72 hours |
| How many cases are still undisposed at the SLA deadline? | `E2`, no terminal disposition by 72 hours |
| How many SLA-breaching cases later resolve? | `E3`, cumulative incidence of late resolution |
| Which process states or policy cohorts are associated with tail risk? | Cause-specific hazards, landmarking, or multi-state transition probabilities |

The three transfer assumptions are intentionally different:

1. **Competing internal terminal:** transfer is an observed internal terminal state and a competing non-resolution outcome.  
2. **Informative external censoring:** transfer stops internal observation; the case may still resolve externally, so transfer is a censoring time, possibly informative.  
3. **Hybrid operational failure:** transfer before 72 hours is treated as a terminal non-resolution/failure, while transfer after 72 hours is treated as censoring for internal late-resolution estimation because the SLA breach has already occurred and the internal system no longer observes the case.

# 3. Why naive completed-case analysis is biased

A completed-case analysis that keeps only `resolved` rows would retain S01, S02, S03, S04, S07, S13 in cohort A and S16, S17, S18, S19, S20, S27, S28 in cohort B. It would exclude rejected, withdrawn, transferred, and open cases. That is biased for several reasons.

**Survivor bias.**  
Resolved-only analysis keeps cases that survived rejection, withdrawal, transfer, and long open periods. For example, S09 is open at 92 hours, S11 is open at 81 hours, S25 is open at 79 hours, and S26 is open at 104 hours. These are exactly the tail-risk cases that matter for SLA governance, but they are absent from a resolved-only sample. If some of them later resolve, resolved-only analysis underestimates late resolution. If many never resolve, resolved-only analysis overstates operational success.

**Informative censoring.**  
Open cases with large `last_event_gap_h` are suspicious for informative censoring. S11 has gap 55 hours at age 81, S25 has gap 48 hours at age 79, S26 has gap 72 hours at age 104, and S09 has gap 31 hours at age 92. If long inactivity is associated with lower probability of later resolution, treating these cases as non-informative censored observations can bias tail-risk estimates. Transfer cases S08, S21, and S32 may also be harder cases; treating transfer as ordinary censoring without discussion can hide a high-risk pathway.

**Competing risks.**  
Cases such as S05 rejected at 9 hours, S12 withdrawn at 15 hours, S23 rejected at 11 hours, S22 withdrawn at 44 hours, and S30 withdrawn at 70 hours cannot later resolve. If they are treated as censored in a resolution-only Kaplan-Meier analysis, the estimated probability of resolution is biased upward. They must be treated as competing events, or their censoring must be explicitly modeled as informative.

**Cohort imbalance and non-randomized policy rollout.**  
Cohort A has 6 resolved, 2 rejected, 2 withdrawn, 1 transferred, and 4 open cases. Cohort B has 7 resolved, 1 rejected, 2 withdrawn, 2 transferred, and 5 open cases. Cohort B also contains more P1 cases and more transfers. A resolved-only cohort comparison may make cohort B look faster or better because it excludes B’s transferred and open P1 cases. Conversely, if transfers are treated as failures, cohort B may look worse. Because assignment was not randomized and rollout compliance was incomplete, any cohort difference is observational and must be treated as associational unless stronger design information is available.

**Active open cases below versus above 72 hours.**  
Open cases below 72 hours, such as S10 at 47 hours, S15 at 35 hours, S24 at 67 hours, S29 at 51 hours, and S31 at 23 hours, have not yet crossed the SLA horizon. They are at risk of future breach, but they are not yet observed breaches. Open cases above 72 hours, such as S09 at 92 hours, S11 at 81 hours, S25 at 79 hours, and S26 at 104 hours, have already crossed the SLA horizon without a terminal disposition. Treating all open cases identically as “censored at age” without distinguishing horizon crossing misclassifies tail risk.

# 4. Statistical analysis plan

## Kaplan-Meier

Kaplan-Meier is appropriate for a single event, such as first internal terminal disposition, and can estimate `E2 = P(no terminal by 72 hours)`. It is not appropriate as the primary method for resolution probability if rejection, withdrawal, and transfer are treated as ordinary censoring, because that overstates the probability of eventual resolution.

## Aalen-Johansen / cumulative incidence

Aalen-Johansen cumulative incidence is the preferred primary method for:

- `E1`: timely resolution by 72 hours, with non-resolution outcomes as competing events;
- `E3`: late resolution after 72 hours, with timely resolution and non-resolution outcomes as competing events.

It gives absolute probabilities rather than hazards and naturally handles multiple terminal causes. For this 32-case extract, it is more defensible than a parametric model because the event count is small.

## Cause-specific hazards

Cause-specific Cox or accelerated failure time models can estimate how cohort, priority, and time-varying process features affect the instantaneous resolution hazard, treating competing outcomes as censored. This is useful for understanding drivers, but hazard ratios are not the same as marginal SLA probabilities. With only 32 cases, a cause-specific model should be simple, regularized, or used only as a sensitivity analysis.

## Subdistribution hazards

Fine-Gray subdistribution hazard models estimate the effect of covariates on the marginal cumulative incidence of resolution while accounting for competing risks. They are useful if rejection/withdrawal/transfer are truly competing and the goal is marginal resolution probability. However, if transfer is informative censoring rather than a true competing event, Fine-Gray may be misleading. Use it as a secondary sensitivity analysis, not the primary estimator.

## Restricted mean time

Restricted mean time or restricted mean cumulative incidence can summarize the area under the cumulative-incidence curve up to 72 or 110 hours. It is useful for governance discussions about average time loss or average delay, but it is less directly interpretable than the at-72 breach probability. With small data, report it with bootstrap uncertainty if used.

## Landmarking

A 72-hour landmark analysis is valuable. Among cases with no terminal disposition before 72 hours, estimate the probability of later resolution versus competing outcomes. This avoids immortal-time bias because all cases in the landmark stratum have survived to 72 hours. It also allows pre-72 process information, such as whether the case reached `Escalate` or `Rework`, to be used as a baseline variable at the landmark without leaking future information.

## Multi-state model

A multi-state representation can model transitions among `Submit`, `Screen`, `AutoCheck`, `Review`, `Escalate`, `Rework`, and terminal states. With exact timestamps, one could estimate state-specific transition hazards and time-dependent probabilities of reaching `Resolve`. With the current extract, exact state durations are not available, so a full parametric multi-state model is not justified. A nonparametric path/variant analysis and landmark state-specific summaries are more appropriate.

## Recommended primary and secondary approaches

**Primary:**  
Use Aalen-Johansen cumulative incidence for timely resolution by 72 hours and late resolution after 72 hours, with transfer treated under at least three scenarios. Use Kaplan-Meier for no-terminal-by-72 hours. Stratify by cohort and priority.

**Secondary:**  
Use a 72-hour landmark analysis and a simple cause-specific hazard model with time-varying process features. Use a multi-state or variant-based descriptive analysis to identify high-risk paths, especially repeated `Review`, `Rework`, late `Escalate`, and long inactivity gaps.

## Immortal-time and leakage risks

Treating `escalated = yes` as a baseline covariate at `Submit` is unsafe. A case must survive long enough to reach `Escalate` to be classified as escalated. Quick resolutions such as S01 at 18 hours and S18 at 12 hours never reach escalation, so an “ever escalated” group is selected on survival to escalation. This creates immortal-time bias.

Similarly, using the full prefix, `rework_count`, or terminal status to predict SLA breach from `Submit` leaks future information. These variables should be used as time-varying covariates, landmark covariates, or descriptive process-mining features, not as baseline predictors.

# 6. Sensitivity analysis

At least three analyses should be compared.

## Analysis 1: Transfer as competing internal terminal

- `Reject` and `Withdraw` are competing non-resolution events.
- `Transfer` is a terminal internal state and competing non-resolution event.
- Open cases are right-censored at cutoff.
- This is the most conservative operational interpretation if transfer means the internal case is closed as non-resolved.

## Analysis 2: Transfer as informative external censoring

- `Reject` and `Withdraw` remain competing non-resolution events.
- `Transfer` is right-censored at transfer time.
- This assumes transferred cases may still resolve externally and that the internal system simply loses observation.
- If transferred cases are systematically harder, this assumption can bias estimates. In the extract, moving transfers from competing to censored raises the timely-resolution estimate from about 0.30 to about 0.31 and raises the no-terminal-by-72 estimate from about 0.40 to about 0.44.

## Analysis 3: Hybrid operational failure

- Pre-72 transfer is treated as terminal non-resolution/failure.
- Post-72 transfer is treated as censoring for internal late-resolution estimation.
- This is operationally plausible because a pre-72 transfer is already a failure to resolve internally, while a post-72 transfer occurs after the SLA breach and may no longer inform internal late-resolution risk.
- This scenario differs from Analysis 1 mainly for late-resolution estimation, especially for S32 transferred at 96 hours.

## Additional sensitivity: withdrawal/rejection as informative censoring

If withdrawal or rejection is driven by case difficulty and would otherwise have resolved later, treating them as competing events may understate potential resolution. If they are true terminal non-resolution outcomes, treating them as censored would overstate resolution. This sensitivity can reverse conclusions about cohort effect if withdrawal/rejection rates differ by cohort or priority.

## Additional sensitivity: informative open-case censoring

Long `last_event_gap_h` values, especially S26 with 72 hours, S11 with 55 hours, and S25 with 48 hours, suggest that open-case censoring may be informative. If long gaps predict lower later resolution, standard censoring overestimates late resolution. If long gaps merely reflect administrative batching, the effect may be smaller. This can reverse the conclusion that open cases beyond 72 hours are likely to recover.

## Conclusions that could reverse

- **Cohort B improves SLA performance:** Could reverse if transfers in cohort B are treated as failures rather than censored, or if cohort B’s higher P1 mix is not adjusted for.
- **Cohort B worsens SLA performance:** Could reverse if transfers are informative censoring and cohort B’s transferred cases would have resolved externally.
- **Long inactivity predicts breach:** Could reverse if long gaps are administrative artifacts rather than true stagnation.
- **Escalation reduces tail risk:** Could reverse if escalation is a post-treatment marker of difficult cases rather than a protective action.
- **Open cases beyond 72 hours are likely to resolve later:** Could reverse if open cases with long gaps are less likely to resolve and are being censored non-informatively.

# 7. Operational interpretation

Survival outputs should be used for monitoring and triage, not as causal claims about policy effects.

## Alerts

1. **SLA breach alert:** Any case with age >= 72 hours and not `Resolve` should trigger a breach review. In the extract, this includes S09, S11, S25, and S26. S32 is also relevant because it is transferred after 72 hours.
2. **Pre-breach risk alert:** Cases between 48 and 60 hours that are in `Review` or `Escalate`, have repeated `Review`, have `Rework`, or have a large `last_event_gap_h` should be prioritized.
3. **Stalled-case alert:** Open cases with large inactivity gaps, such as S26 with 72 hours, S11 with 55 hours, and S25 with 48 hours, should receive manual review.
4. **Transfer follow-up alert:** Transferred cases S08, S21, and S32 should be tracked for external outcome if available.

## Workload actions

- Monitor stage occupancy near the SLA deadline, especially `Review` and `Escalate`.
- If many cases are in `Review` at 48-60 hours, increase review capacity or reassign cases.
- If `Escalate` cases accumulate, prioritize P1 escalated cases and review escalation routing.
- If `Rework` loops are common, investigate whether rework is a symptom of upstream quality issues.

## SLA governance

Report at least:

- Cumulative incidence of timely resolution by 72 hours.
- Cumulative incidence of late resolution after 72 hours.
- Probability of no terminal disposition by 72 hours.
- Transfer rate and transfer-treatment sensitivity.
- Calibration of predicted risk scores by priority and cohort.

Do not present a cohort or priority risk score as a causal effect of the policy change. Cohort B began after a policy change, but assignment was not randomized and rollout compliance was incomplete. The appropriate language is associational: “Cohort B is associated with lower/higher estimated breach risk under assumption X,” not “The policy caused lower/higher breach risk.”

## Calibration and fairness checks

- **Calibration:** Compare predicted probabilities to observed outcomes using time-based validation or bootstrap. Use calibration curves, Brier score, or calibration intercept/slope. Recalibrate if predictions are systematically over- or under-confident.
- **Priority fairness:** Check that P1 cases are not systematically under-alerted or over-penalized. Stratify outcomes and alert rates by P1, P2, and P3.
- **Cohort fairness:** Check that cohort B is not disadvantaged by incomplete rollout or different case mix. Adjust or stratify by priority and, if available, case complexity.
- **Process fairness:** Ensure that path-based risk scores do not penalize cases for legitimate process variations, such as `AutoCheck` order changes, without accounting for the policy context.

def encode_success(df, scenario):
    """
    Estimand E1: cumulative incidence of timely resolution by 72 hours.

    event = 1  : resolved at t <= 72
    comp  = 1  : non-resolution terminal before 72 under the scenario
    censor = 1 : open cases, transfer under censoring scenario, or terminal after 72 truncated at 72
    """

    d["t_an"] = d["duration_or_age_h"].clip(upper=SLA)
    d["event"] = 0

    # Open cases: right-censored at min(age, 72).
    d.loc[d["observed_status"] == "open", "censor"] = 1

    return d


def encode_open72(df, scenario):

    d["t_an"] = d["duration_or_age_h"].clip(upper=SLA)
    d["event"] = 0
    d["censor"] = 0

    term = d["observed_status"].isin(TERMINAL)
    m = term & (d["duration_or_age_h"] <= SLA)

    d.loc[m, "event"] = 1
    d.loc[m, "censor"] = 0

    if scenario == "transfer_censored":
        d.loc[m & (d["observed_status"] == "transferred"), "event"] = 0
        d.loc[m & (d["observed_status"] == "transferred"), "censor"] = 1

    # Terminal after 72: no terminal by 72, so censor at 72.
    d.loc[term & (d["duration_or_age_h"] > SLA), "event"] = 0
    d.loc[term & (d["duration_or_age_h"] > SLA), "censor"] = 1

    # Open cases: right-censored at min(age, 72).
    d.loc[d["observed_status"] == "open", "event"] = 0
    d.loc[d["observed_status"] == "open", "censor"] = 1

    return d


def encode_late(df, scenario):
    """
    Estimand E3: cumulative incidence of late resolution after 72 hours.

    event = 1  : resolved at t > 72
    comp  = 1  : any other terminal outcome that prevents late resolution, subject to scenario
    censor = 1 : open cases and transfer under censoring/hybrid scenarios
    """
    d = df.copy()
    d["t_an"] = d["duration_or_age_h"]
    d["event"] = 0
    d["comp"] = 0
    d["censor"] = 0

    term = d["observed_status"].isin(TERMINAL)
    late_res = term & (d["observed_status"] == "resolved") & (d["duration_or_age_h"] > SLA)
    d.loc[late_res, "event"] = 1

    other_term = term & ~late_res
    d.loc[other_term, "comp"] = 1

    if scenario == "transfer_censored":
        d.loc[other_term & (d["observed_status"] == "transferred"), "comp"] = 0
        d.loc[other_term & (d["observed_status"] == "transferred"), "censor"] = 1

    elif scenario == "transfer_hybrid_failure":
        # Pre-72 transfers remain competing; post-72 transfers are censored for internal late-resolution.
        post = other_term & (d["observed_status"] == "transferred") & (d["duration_or_age_h"] > SLA)
        d.loc[post, "comp"] = 0
        d.loc[post, "censor"] = 1

    d.loc[d["observed_status"] == "open", "censor"] = 1

    return d


def cumulative_incidence(data, horizon, t="t_an", event="event", comp="comp", censor="censor"):
    """
    Manual Aalen-Johansen cumulative incidence for one cause of interest.

    event = cause of interest
    comp  = competing events
    censor= right-censored observations
    """
    d = data[data[t].notna()].copy()
    d = d[d[t] <= horizon]
    times = sorted(d[t].unique())

    F = 0.0
    S_total = 1.0
    curve = []

    for s in times:
        n = int((d[t] >= s).sum())
        if n == 0:
            continue

        d_event = int(((d[t] == s) & (d[event] == 1)).sum())
        d_total = int(((d[t] == s) & ((d[event] == 1) | (d[comp] == 1))).sum())

        if d_event > 0 and S_total > 0:
            F += (d_event / n) * S_total

        if d_total > 0:
            S_total *= (1.0 - d_total / n)

        curve.append((s, F, S_total, n))

    vals = [c[1] for c in curve if c[0] <= horizon]
    F_horizon = vals[-1] if vals else 0.0
    return F_horizon, pd.DataFrame(curve, columns=["time", "CIF", "S_total", "n_at_risk"])


def km_survival(data, horizon, t="t_an", event="event", censor="censor"):
    """
    Manual Kaplan-Meier survival estimate for a single event type.
    """
    d = data[data[t].notna()].copy()
    d = d[d[t] <= horizon]
    times = sorted(d[t].unique())

    S = 1.0
    curve = []

    for s in times:
        n = int((d[t] >= s).sum())
        if n == 0:
            continue

        d_event = int(((d[t] == s) & (d[event] == 1)).sum())
        if d_event > 0:
            S *= (1.0 - d_event / n)

        curve.append((s, S, n))

    vals = [c[1] for c in curve if c[0] <= horizon]
    S_horizon = vals[-1] if vals else 1.0
    return S_horizon, pd.DataFrame(curve, columns=["time", "KM_S", "n_at_risk"])


def estimate(df, scenario, estimand):
    if estimand == "success_by_72":
        enc = encode_success(df, scenario)
        val, curve = cumulative_incidence(enc, SLA)
    elif estimand == "open_at_72":
        enc = encode_open72(df, scenario)
        val, curve = km_survival(enc, SLA)
    elif estimand == "late_resolution":
        enc = encode_late(df, scenario)
        val, curve = cumulative_incidence(enc, H_LATE)
    else:
        raise ValueError("Unknown estimand")

    return val, curve, enc


def summarize_stratified(df, scenario, estimand, by):
    rows = []
    for name, sub in df.groupby(by):
        val, curve, enc = estimate(sub, scenario, estimand)
        rows.append(
            {
                "stratum": name,
                "n": len(sub),
                "estimate": round(val, 4),
                "events": int(enc["event"].sum()),
                "competing": int(enc["comp"].sum()) if "comp" in enc.columns else 0,
                "censored": int(enc["censor"].sum()),
            }
        )
    return pd.DataFrame(rows)

# 4. PM4Py-compatible control-flow feature derivation.
#    No timestamps are added; PM4Py is used only for sequence/variant semantics.
# ----------------------------------------------------------------------
def build_pm4py_features(df):
    used_pm4py = False
    variant_counts = None

    if PM4PY_AVAILABLE:
        try:
            try:
                EventLog = pm4py.OOP.EventLog
                Event = pm4py.OOP.Event
            except AttributeError:
                EventLog = pm4py.EventLog
                Event = pm4py.Event

            events = []
            for _, r in df.iterrows():
                for act in r["prefix"].split(">"):
                    act = act.strip()
                    if act:
                        events.append(Event(case_id=r["case"], activity=act))

            try:
                log = EventLog(events)
            except TypeError:
                log = EventLog()
                for ev in events:
                    log.append(ev)

            variants = pm4py.get_variants(log)
            variant_counts = pd.Series(
                {tuple(k): int(v) for k, v in variants.items()}
            ).sort_values(ascending=False)
            used_pm4py = True

        except Exception as exc:
            print("PM4Py variant extraction failed; using fallback:", exc)

    # Fallback / manual variant representation, also used to attach features.
    variant_series = df["prefix"].apply(
        lambda p: tuple(a.strip() for a in p.split(">") if a.strip())
    )

    if variant_counts is None:
        variant_counts = variant_series.value_counts()

    out = df.copy()
    out["variant"] = variant_series
    out["variant_count"] = variant_series.map(lambda v: int(variant_counts.get(v, 0)))
    out["path_length"] = variant_series.apply(len)
    out["has_autocheck"] = variant_series.apply(lambda v: "AutoCheck" in v)
    out["has_escalate"] = variant_series.apply(lambda v: "Escalate" in v)
    out["has_rework"] = variant_series.apply(lambda v: "Rework" in v)
    out["review_count"] = variant_series.apply(lambda v: v.count("Review"))
    out["last_activity"] = variant_series.apply(lambda v: v[-1] if v else "")

    return out, variant_counts, used_pm4py

# 9. Limitations and additional data

The strongest limitations are missing event-level timing, unknown external transfer outcomes, small sample size, and non-randomized policy rollout.

## Highest-priority missing data

1. **Exact event timestamps for each activity.**  
   Needed to estimate time-varying covariates, stage-specific hazards, escalation timing, inactivity over time, and multi-state transition probabilities. The current extract provides only total duration/age and final inactivity gap.

2. **External outcomes after transfer.**  
   Needed to decide whether transfer is a competing terminal event, informative censoring, or a delayed resolution. Without external resolution status and time, transfer assumptions remain speculative.

3. **Cutoff and start dates.**  
   Needed to align cohort A and cohort B, account for administrative censoring, and understand whether open cases are censored because follow-up ended or because the case was genuinely inactive.

4. **Policy rollout compliance and assignment mechanism.**  
   Needed to know whether cohort B actually received the automated-screening and escalation-policy change, and whether assignment was random, quasi-random, or driven by case characteristics.

5. **Time-varying workload and staffing.**  
   Needed to separate case-level risk from system-level capacity constraints. SLA tail risk may be driven by reviewer availability, escalation queue length, or transfer processing time.

## Additional useful data

6. **Reason codes for withdrawal, rejection, and transfer.**  
   These would help determine whether competing events are true terminal non-resolution or informative administrative decisions.

7. **Case complexity and customer characteristics.**  
   Needed for adjustment, fairness checks, and to avoid confounding by case mix.

8. **Longer follow-up for open cases.**  
   Needed to reduce right censoring and better estimate late-resolution tail risk.

9. **Multiple observations per case or full event log.**  
   The prefix field provides sequence but not timing. A full event log would allow proper survival and multi-state modeling.

10. **Operational definitions of SLA breach and transfer.**  
   Whether transfer before 72 hours is considered an internal failure, a competing outcome, or a censored external case materially changes the estimand and the operational interpretation.
