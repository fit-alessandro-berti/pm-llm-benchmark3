```python
"""
Revised normative POWL model M1 for supplier onboarding.
Requires pm4py >= 2.7 (POWL objects).
No external files are read.
"""

from pm4py.objects.powl.obj import (
    Transition,
    SilentTransition,
    StrictPartialOrder,
    OperatorPOWL,
)
from pm4py.objects.process_tree.obj import Operator


def T(label: str) -> Transition:
    """Fresh labeled transition (never share node objects between branches)."""
    return Transition(label=label)


def zero_or_more(label: str) -> OperatorPOWL:
    """
    Zero or more repetitions of `label`.
    LOOP(do, redo) plays as: do (redo do)*  -- process-tree semantics.
    With redo = tau,  LOOP(x, tau)  == x+   (one or more).
    XOR(tau, x+)                          == x*   (zero or more).
    """
    at_least_once = OperatorPOWL(
        operator=Operator.LOOP,
        children=[T(label), SilentTransition()],
    )
    return OperatorPOWL(
        operator=Operator.XOR,
        children=[SilentTransition(), at_least_once],
    )


def seq(*nodes):
    """Strict partial order with a chain of edges n0 -> n1 -> ... -> nk."""
    po = StrictPartialOrder(nodes=list(nodes))
    for a, b in zip(nodes, nodes[1:]):
        po.order.add_edge(a, b)
    return po
```

```python
"""
M1 — revised normative POWL model for supplier onboarding / reactivation.
Requires: pm4py >= 2.7 (powl objects). Reads no external files.

Semantics notes
---------------
* StrictPartialOrder: nodes + order edges; unordered nodes are concurrent.
* OperatorPOWL(XOR, ...): exclusive choice among children.
* OperatorPOWL(LOOP, [do, redo]): do (redo do)*  — process-tree loop semantics.
  zero_or_more(x) = XOR(tau, LOOP(x, tau))  == x*   (avoids do/redo parity quirk).
* The XOR that selects the route is *free* in POWL: the model says which
  orderings are legal, not which route a given case may take. Route
  authorization (risk, country, group, dates, caps) is enforced by the
  data/policy-overlay layer (see overlay table). Conformance verdict =
  (trace fits an authorized branch) AND (overlay authorizes that branch
  for that case at that time).
* Transitions with the same label in different branches are distinct node
  instances of the same business activity; alignment is by label.
"""

from pm4py.objects.powl.obj import (
    Transition,
    SilentTransition,
    StrictPartialOrder,
    OperatorPOWL,
)
from pm4py.objects.process_tree.obj import Operator


def T(label):
    return Transition(label=label)


def zero_or_more(label):
    at_least_once = OperatorPOWL(
        operator=Operator.LOOP, children=[T(label), SilentTransition()]
    )
    return OperatorPOWL(
        operator=Operator.XOR, children=[SilentTransition(), at_least_once]
    )


def chain(*nodes):
    po = StrictPartialOrder(nodes=list(nodes))
    for a, b in zip(nodes, nodes[1:]):
        po.order.add_edge(a, b)
    return po


def standard_route():
    """CS-9 baseline: sanctions || quality, then manual approval, then activation.
    Request_More_Info cycles run concurrently with the checks and must close
    before the approval decision."""
    sanctions = T("Sanctions_Check")
    quality = T("Quality_Review")
    checks = StrictPartialOrder(nodes=[sanctions, quality])  # concurrent

    rmi = zero_or_more("Request_More_Info")
    manual = T("Manual_Procurement_Approval")
    activate = T("Activate_Supplier")

    route = StrictPartialOrder(nodes=[checks, rmi, manual, activate])
    route.order.add_edge(checks, manual)
    route.order.add_edge(rmi, manual)
    route.order.add_edge(manual, activate)
    return route


def auto_route():
    """Low-risk pilot/permanent route: sanctions || master-data validation,
    Auto_Approve, activation, deferred Quality_Review (deadline = data rule)."""
    sanctions = T("Sanctions_Check")
    mdv = T("Master_Data_Validation")
    auto = T("Auto_Approve")
    activate = T("Activate_Supplier")
    quality = T("Quality_Review")

    route = StrictPartialOrder(nodes=[sanctions, mdv, auto, activate, quality])
    route.order.add_edge(sanctions, auto)
    route.order.add_edge(mdv, auto)
    route.order.add_edge(auto, activate)
    route.order.add_edge(activate, quality)
    return route


def emergency_route():
    """FD-7-style provisional activation: sanctions before activation;
    quality, retrospective approval, and daily monitoring afterwards
    (concurrent). Caps/deadlines are data rules."""
    sanctions = T("Sanctions_Check")
    activate = T("Activate_Supplier")
    quality = T("Quality_Review")
    manual = T("Manual_Procurement_Approval")
    monitoring = zero_or_more("Transaction_Monitoring")

    route = StrictPartialOrder(
        nodes=[sanctions, activate, quality, manual, monitoring]
    )
    route.order.add_edge(sanctions, activate)
    route.order.add_edge(activate, quality)
    route.order.add_edge(activate, manual)
    route.order.add_edge(activate, monitoring)
    return route


def exit_branch():
    """Pre-screen rejection (no quality review required — CS-9 requires quality
    only before *activation*) or requester withdrawal."""
    return OperatorPOWL(
        operator=Operator.XOR,
        children=[T("Reject_Supplier"), T("Withdraw_Supplier")],
    )


def onboarding_flow():
    register = T("Register_Supplier")
    decision = OperatorPOWL(
        operator=Operator.XOR,
        children=[
            standard_route(),
            auto_route(),
            emergency_route(),
            exit_branch(),
        ],
    )
    notify = T("Notify_Requester")  # CRM-mirrored; stable notification_id (data rule)
    close = T("Close_Case")
    flow = StrictPartialOrder(nodes=[register, decision, notify, close])
    flow.order.add_edge(register, decision)
    flow.order.add_edge(decision, notify)
    flow.order.add_edge(notify, close)
    return flow


def reactivation_flow():
    """Separate case class: prior certification re-validated at assessment;
    sanctions re-check and manual approval mandatory; quality only if the
    certificate is stale (data rule decides the optional branch)."""
    assess = T("Reactivation_Assessment")
    sanctions = T("Sanctions_Check")
    quality_opt = OperatorPOWL(
        operator=Operator.XOR,
        children=[T("Quality_Review"), SilentTransition()],
    )
    manual = T("Manual_Procurement_Approval")
    activate = T("Activate_Supplier")
    notify = T("Notify_Requester")
    close = T("Close_Case")
    return chain(assess, sanctions, quality_opt, manual, activate, notify, close)


M1 = OperatorPOWL(
    operator=Operator.XOR, children=[onboarding_flow(), reactivation_flow()]
)


if __name__ == "__main__":
    print("M1 constructed.")
    try:
        import pm4py
        net, im, fm = pm4py.convert_to_petri_net(M1)
        print(
            "Petri-net conversion OK:",
            len(net.places),
            "places,",
            len(net.transitions),
            "transitions.",
        )
    except Exception as exc:  # version-dependent API
        print("Petri-net conversion unavailable in this pm4py version:", exc)
```

## 2. Prosecution Brief

**A. Formal authority, and why it matters.** CS-9 is the only controlled, formally approved instrument governing onboarding (N1). PD-3 is minutes and an email; it was never incorporated into CS-9 or M0, and the designated compliance voice later stated she had not approved weakening CS-9 (N2). **B. Hard-control breaches.** SUP-703: a HIGH-risk new foreign supplier — ineligible on every PD-3 criterion — was activated via Auto_Approve with no manual procurement approval and quality review only after activation, followed by a first purchase of EUR 180,000 (SUP-703; Audit flag). **D. ## 3. Defense Brief

**A. Fitness is not compliance — decompose the deviation rate.** Of M0's deviations: 289 in-window eligible pilot cases and the FD-7 cases were *authorized*; 47 Request_More_Info cases reflect a *beneficial practice M0 cannot express*, with zero premature activations (Aggregate); notification "missingness" is 29% by construction after the CRM migration (N4); one case is a provable duplicate (SUP-711); one is pure label drift (SUP-713); two are scope artifacts (SUP-709, 710). What remains — SUP-703, 705, 708, 712, 714, 715 — is the *entire* genuinely contested set. Measuring a stale model against degraded logs and calling the result "noncompliance" is measurement error, not audit.

**B. PD-3 was a legitimate, monitored exception.** The steering committee is a competent operational organ; the compliance observer was present and contemporaneously wrote "pilot may proceed as a monitored exception" (N2). That is acquiescence on which operations reasonably relied; her later statement limits the pilot's *future* scope, it cannot retroactively void relied-upon conduct. SUP-702 is the paradigm: eligible supplier, in window, quality review next business day, CRM notification evidenced by a **stable notification_id**. To condemn it is to punish exactly the behavior the organization authorized and monitored.

**C. FD-7 and proportionality.** SUP-704 satisfied every FD-7 condition (sanctions first, EUR 18,000 under the cap, daily monitoring, next-day quality and approval). SUP-705 occurred one day after automatic expiry amid a *factually continuing* flood recovery, with EUR 12,000 spend and full next-day remediation. The real failure is institutional: no extension or renewal mechanism existed. Legalistic expiry-by-hours should not convert prudent urgent action into "material noncompliance"; if it does, the lesson taught to operations is that compliance is a trap, not a tool.

**D. Proven model defects, proven by the cases themselves.** (i) *Over-constraint*: M0's `checks  decision` forces Quality_Review before Reject_Supplier; CS-9 requires quality only before *activation* and expressly contemplates an abbreviated pre-screen for rejected leads (N1). (iii) *No reactivation semantics*: SUP-710 performed the risk-critical controls (sanctions re-check, manual approval) on a valid certification; demanding a fresh full onboarding is a category error. (iv) *Control-flow blindness*: SUP-702 and SUP-703 have **identical traces** and opposite compliance — proof that control-flow-only conformance is the wrong instrument and that the "deviation rate" aggregates incommensurable things. (v) *Concurrency is right but inert*: M0's sanctions  quality matches CS-9, yet without data awareness the model cannot distinguish a compliant pilot case from a EUR 180,000 high-risk bypass.

**E. Trace deviation vs. misconduct.** SUP-708: quality review *did* precede activation; the sanctions slip was 40 minutes; **zero transactions** occurred before clearance; no pilot or emergency was claimed, so no one is alleging concealment — this is a sequencing slip with no exposure, not "severe." SUP-706: a CRM message exists at +3 hours; the *organization's* migration made the link mutable; blaming operations for tooling it did not choose is perverse. SUP-711: one source transaction ID — a duplicate, full stop.

**Defense's acknowledgments (genuine risk and limits).** SUP-703 is indefensible: an ineligible HIGH-risk supplier, no manual approval, deferred quality, EUR 180,000 — real noncompliance with material risk, and the UI's willingness to offer "pilot" mitigates culpability, not the fact. SUP-708's sanctions inversion is a genuine hard-control failure; luck is not a control. SUP-712's segregation breach is serious, and aliasing may indicate concealment. Informal practice cannot substitute for amendment: SUP-714's "imminent approval" and SUP-705's post-expiry continuation crossed the line where reasonable reliance ends. The defense's remedy is a revised model, explicit scope rules, fixed data collection, and a governed overlay process — not blanket absolution, and not enforcement of a demonstrably wrong artifact.

### 5.2 System-level judgment

**Operational compliance: materially noncompliant, but in a bounded and identifiable subset.** The hard controls that exist in every version of the norm — sanctions before activation, approval before activation, segregation of duties, conflict-of-interest disclosure — were violated in at least six named cases, at least 14 ineligible in-window pilot uses, six post-pilot uses, plus nine unresolved eligibility cases. Two audit-flagged severe cases (SUP-703, SUP-708) are confirmed. Sanctions-control failures on HIGH-risk suppliers are reportable-grade events. At the same time, the *rate* of M0 fitness deviation is not a valid compliance metric: the majority of deviations are authorized overlays (290 cases), model-expressiveness gaps (47 RMI cases; reject-path over-constraint; reactivation), or data artifacts (29% notification missingness; duplicates; label drift). Both of these statements are true simultaneously, which is why neither side's headline survives intact.

**Model validity: M0 is invalid as the sole normative instrument**, on four grounds — expressiveness (no loop, no auto/emergency branches), scope (rejected leads, reactivations, withdrawals undefined), fidelity (stricter than CS-9 on the reject path; silent on CS-9's resource rule), and data-awareness (cannot represent risk, amounts, deadlines, identity, or authority windows, and its event inputs are degraded). The invalidity of the *artifact* does not invalidate the *norm*: CS-9 stands and remains enforceable through layered, data-aware checks. M0 should be replaced by M1 (Section 7), not dissolved.

**Enforcement posture.** Enforce hard controls immediately with data-aware conformance; suspend fitness-based enforcement of M0's over-constraints (notably quality-before-rejection); remediate the notification pipeline before drawing evidentiary conclusions from missing Notify_Requester events; formalize or terminate low-risk auto-approval (absent an amendment, SUP-714-type cases will recur); institute an emergency-extension protocol; re-audit the 14 + 9 + 6 aggregate cases.

### 5.3 Where I reject each side

**Rejecting the prosecution:**
1. *PD-3 as void ab initio.* A documented steering-committee decision, executed openly, with the compliance observer contemporaneously permitting it "as a monitored exception," is a legitimate time-boxed exception within ordinary committee competence. Condemning 290 condition-compliant monitored cases on formalism would punish reasonable reliance; the observer's later statement limits the pilot's future (no permanent weakening), it does not reach backward. SUP-702 stands as justified.
2. *Raw-log fitness as a compliance metric.* With 29% notification missingness, an unmodeled beneficial loop, taxonomy drift, and duplicates, M0 fitness systematically overstates noncompliance.
3. *Enforcing M0 where it exceeds CS-9.* SUP-709 shows M0 demanding Quality_Review before rejection — a rule CS-9 does not contain. Conformance must be measured against the norm, not against a provably over-expressive proxy.

**Rejecting the defense:**
1. *Obsolescence as exculpation.* SUP-703, 705, 708, 712, 714 are caused by no model gap; each violates a rule present in every version of the norm. An imperfect model never licenses choosing an unauthorized branch.
2. *Necessity and belief as authority.* Sunset clauses are deliberate governance devices. Urgency (SUP-705) and expected amendment (SUP-714) create a duty to obtain fast authorization; they do not self-authorize. Proportionality affects sanctioning, not classification.
3. *No-harm reasoning.* SUP-708's "no transactions before clearance" is fortune, not control. Preventive controls are judged on execution.
4. *Practice as norm.* That RMI is desirable (SUP-707) or auto-approval normalized (SUP-714) does not amend CS-9. Practice proposes; governance disposes.

## 6. Revised Normative Architecture

**Layer 0 — Case taxonomy and scope rules (governance document + case-class attribute).**
- `ONBOARDING_NEW`: governed from Register_Supplier; includes rejection and withdrawal outcomes. Rejected leads take the exit branch — sanctions pre-screen, no Quality_Review required (cures SUP-709). Withdrawal before a decision closes the case without compliance implication.
- `ONBOARDING_REACTIVATION`: governed from Reactivation_Assessment; sanctions re-check and manual approval always mandatory; Quality_Review required iff prior certification is invalid or older than 12 months (cures SUP-710).
- Overlays (pilots, emergencies) never edit the base model; they are versioned data artifacts with hard expiry.

**Layer 1 — Control-flow core (POWL, model M1 in Section 7).** Belongs in POWL: existence and ordering of activities per authorized route; concurrency (sanctions  quality on the standard route; sanctions  master-data validation on the auto route); the Request_More_Info loop; provisional activation ordering on the emergency route; optional quality on reactivation; notification before closure. POWL deliberately does **not** encode risk class, country, amounts, deadlines, identities, or authority windows — the XOR route choice is structurally free; *which route a case may take* is a Layer-2 decision.

**Layer 2 — Data-aware eligibility, amount, time, and authority rules (pandas/SQL over the enriched log).** Route-authorization matrix evaluated on frozen eligibility snapshots (risk, country, corporate group, MDV result) against overlay windows; EUR 20,000 emergency cap until retrospective approval; "quality within two business days" deadline checks; notification mirror reconciliation; deduplication on (source_system, source_transaction_id).

**Layer 3 — Declarative constraints (DECLARE / log-skeleton style).** Cross-branch guards that POWL cannot safely express: `Sanctions_Check precedes Activate_Supplier` globally (belt-and-braces); exactly-one outcome notification before Close_Case (CRM-completed events count); Request_More_Info only before the approval decision; existence of Transaction_Monitoring on the emergency route; segregation of duties as `canonical(requester)  canonical(approver)`; label-mapping validity at event time.

**Layer 4 — Resource and organizational controls.** IAM alias registry with canonical actor IDs resolved at write time; delegation register with authority evidence; conflict-of-interest disclosure workflow (relationship declarations plus recorded waiver decisions — cures SUP-715); SoD hard-blocked in the approval UI (cures SUP-712); emergency-extension protocol with pre-authorized criteria and a decision SLA (cures SUP-705's governance gap).

**Layer 5 — Policy-overlay governance lifecycle.** Every pilot/emergency overlay must carry: rule_id and version, authority document, recorded compliance sign-off, scope predicates, monitoring metrics, effective/recorded timestamps, and machine-enforced expiry. Extensions require new authority *before* expiry; nothing is ever extended by belief or factual necessity alone.

**Mapping summary.** Ordering/concurrency/loops  POWL. Existence, cardinality, cross-branch ordering, actor inequality  DECLARE. Windows, eligibility, caps, deadlines, snapshots, reconciliation  pandas/data checks. Disclosures, delegations, sign-offs, extension protocol  organizational controls.

## 7. PM4Py Model and Data Proposal

**Design decisions and stated limitations.**
1. **Route authorization is not in the POWL.** M1 enumerates the *orderings* that are legal per route; whether a given case may take a route is decided by the overlay/data layer. Conformance = (trace fits an authorized branch) AND (the overlay authorizes that branch for that case at that time). This is why SUP-702 and SUP-703 — identical traces — receive opposite verdicts.
2. **Repeatable Request_More_Info** is modeled on the standard route, concurrent with the checks and closing before the approval decision (covers SUP-707 and the 47 aggregate cases). RMI on other routes or after approval is not represented; a DECLARE constraint can loosen this if practice requires.
3. **Zero-or-more loops** are built as `XOR(, LOOP(x, ))` to avoid the process-tree `do/redo` parity quirk.
4. **Master_Data_Validation** is a required activity on the auto route (PD-3 requires it). Legacy pilot logs did not record it as a discrete event; a documented legacy allowance (DECLARE: MDV may be silent for pre-M1 cases) is the defensible approximation rather than hiding the gap.
5. **Reactivation_Assessment** is a new required activity; legacy reactivations (SUP-710) are evaluated under a mapped legacy variant.
6. **The emergency route models FD-7's permissive ordering** (activation first, retrospective quality/approval). An emergency case that completes quality *before* activation fits no branch and is flagged for review — a deliberately safe direction.
7. **Notify_Requester is the final outcome notification** (after decision-branch completion, before Close_Case), consistent with all observed traces; interim communications are out of model scope.
8. Same-label transitions in different branches are distinct node instances of the same business activity; alignment is by label.

```python
"""
M1 -- Revised normative POWL model for supplier onboarding / reactivation.
Requires pm4py >= 2.7 (POWL objects). Reads no external files.

Layer 1 of the revised architecture: control flow only.
Route authorization (risk/country/group/date/cap) is Layer 2 (see POLICY_OVERLAY).
"""

from pm4py.objects.powl.obj import (
    Transition,
    SilentTransition,
    StrictPartialOrder,
    OperatorPOWL,
)
from pm4py.objects.process_tree.obj import Operator


def T(label):
    """Fresh labeled transition. Never share node instances between branches."""
    return Transition(label=label)


def zero_or_more(label):
    """x* == XOR(tau, LOOP(x, tau)). LOOP(do, redo) plays as do (redo do)*;
    with redo = tau this is x+ (one or more), and XOR with tau makes it optional."""
    at_least_once = OperatorPOWL(
        operator=Operator.LOOP, children=[T(label), SilentTransition()]
    )
    return OperatorPOWL(
        operator=Operator.XOR, children=[SilentTransition(), at_least_once]
    )


def chain(*nodes):
    """Linear sequence as a strict partial order (transitive-reduction edges)."""
    po = StrictPartialOrder(nodes=list(nodes))
    for a, b in zip(nodes, nodes[1:]):
        po.order.add_edge(a, b)
    return po


def standard_route():
    """CS-9 baseline: Sanctions_Check || Quality_Review, then manual approval,
    then activation. Request_More_Info cycles run concurrently with the checks
    and must close before the approval decision."""
    sanctions = T("Sanctions_Check")
    quality = T("Quality_Review")
    checks = StrictPartialOrder(nodes=[sanctions, quality])  # concurrent, no edges

    rmi = zero_or_more("Request_More_Info")
    manual = T("Manual_Procurement_Approval")
    activate = T("Activate_Supplier")

    route = StrictPartialOrder(nodes=[checks, rmi, manual, activate])
    route.order.add_edge(checks, manual)   # both checks precede approval
    route.order.add_edge(rmi, manual)      # info cycles close before approval
    route.order.add_edge(manual, activate)  # CS-9: approval before activation
    return route


def auto_route():
    """Low-risk auto route (PD-3 pattern, future permanent variant):
    Sanctions_Check || Master_Data_Validation, then Auto_Approve, activation,
    deferred Quality_Review. The 2-business-day deadline is a data rule (R-PD3-002)."""
    sanctions = T("Sanctions_Check")
    mdv = T("Master_Data_Validation")
    auto = T("Auto_Approve")
    activate = T("Activate_Supplier")
    quality = T("Quality_Review")

    route = StrictPartialOrder(nodes=[sanctions, mdv, auto, activate, quality])
    route.order.add_edge(sanctions, auto)   # sanctions || master-data validation
    route.order.add_edge(mdv, auto)
    route.order.add_edge(auto, activate)
    route.order.add_edge(activate, quality)  # deferred quality is legal here ONLY
    return route


def emergency_route():
    """FD-7 pattern: Sanctions_Check strictly before (provisional) activation;
    Quality_Review, retrospective Manual_Procurement_Approval, and daily
    Transaction_Monitoring afterwards, concurrently. Cap/deadline = data rules."""
    sanctions = T("Sanctions_Check")
    activate = T("Activate_Supplier")
    quality = T("Quality_Review")
    manual = T("Manual_Procurement_Approval")
    monitoring = zero_or_more("Transaction_Monitoring")

    route = StrictPartialOrder(
        nodes=[sanctions, activate, quality, manual, monitoring]
    )
    route.order.add_edge(sanctions, activate)
    route.order.add_edge(activate, quality)
    route.order.add_edge(activate, manual)
    route.order.add_edge(activate, monitoring)
    return route


def exit_branch():
    """Rejected leads: optional sanctions pre-screen, then Reject or Withdraw.
    No Quality_Review required -- CS-9 requires quality only before ACTIVATION
    (repairs M0's over-constraint exposed by SUP-709)."""
    pre_screen = OperatorPOWL(
        operator=Operator.XOR, children=[T("Sanctions_Check"), SilentTransition()]
    )
    outcome = OperatorPOWL(
        operator=Operator.XOR,
        children=[T("Reject_Supplier"), T("Withdraw_Supplier")],
    )
    branch = StrictPartialOrder(nodes=[pre_screen, outcome])
    branch.order.add_edge(pre_screen, outcome)
    return branch


def onboarding_flow():
    register = T("Register_Supplier")
    decision = OperatorPOWL(
        operator=Operator.XOR,
        children=[standard_route(), auto_route(), emergency_route(), exit_branch()],
    )
    notify = T("Notify_Requester")  # outcome notification; CRM mirror w/ stable ID
    close = T("Close_Case")
    flow = StrictPartialOrder(nodes=[register, decision, notify, close])
    flow.order.add_edge(register, decision)
    flow.order.add_edge(decision, notify)
    flow.order.add_edge(notify, close)
    return flow


def reactivation_flow():
    """Separate case class: certification re-validated at assessment; sanctions
    re-check and manual approval mandatory; Quality_Review only if the prior
    certificate is stale/invalid (the data layer decides the optional branch)."""
    return chain(
        T("Reactivation_Assessment"),
        T("Sanctions_Check"),
        OperatorPOWL(
            operator=Operator.XOR,
            children=[T("Quality_Review"), SilentTransition()],
        ),
        T("Manual_Procurement_Approval"),
        T("Activate_Supplier"),
        T("Notify_Requester"),
        T("Close_Case"),
    )


M1 = OperatorPOWL(
    operator=Operator.XOR, children=[onboarding_flow(), reactivation_flow()]
)
```

**Event- and data-collection proposal.**

*Event record (append-only).* `event_id` (UUID); `case_id`; `case_class`; `activity` (canonical label); `label_version` and `mapping_rule_id`; `ts_effective_utc` (immutable business time) and `ts_recorded_utc` (immutable ingestion time — distinct columns, never overwritten); `source_system`; `source_transaction_id` (idempotency/dedup key); `actor_id_canonical`, `actor_alias_used`, `delegation_ref` (who acted for whom, with authority evidence); `rule_context` (list of rule_id@version in force at ts_effective, establishing pilot/emergency authority); `eligibility_snapshot` (frozen JSON at decision time: risk_class, country, corporate_group, group_approval_id, mdv_result, cert_id, cert_valid_until, spend_to_date); `overlay_refs` (pilot_id/emergency_id + authority document ID + sign-off record); `notification_id` (stable UUID generated by the onboarding system, echoed in the CRM payload, immutably binding case  message); amounts/currencies for purchase events; `schema_version`.

*Collection governance.* CRM must mirror notifications synchronously or via a reconciliation job keyed on `notification_id`; free-text case references are non-evidence. The IAM alias registry canonicalizes actors at write time, with audited alias merges — SoD checks (R-CS9-004) run on canonical IDs, closing the SUP-712 blind spot. Automated decisions (Master_Data_Validation, Auto_Approve) must log events with rule/version IDs — no silent system decisions. Eligibility snapshots are frozen at decision time; retrospective mutation is prohibited. The overlay registry is a separate, versioned table with authority documents, compliance sign-off, and effective/record timestamps; the conformance engine refuses route authorization outside an overlay's window, making SUP-705/SUP-714-style drift mechanically impossible.

*Conformance pipeline.* (1) Deduplicate on (source_system, source_transaction_id); (2) apply versioned label mapping; (3) complete events by joining CRM on notification_id; (4) evaluate route authorization from overlay + snapshot; (5) align the trace to M1 for its case class; (6) evaluate DECLARE constraints (global sanctions-before-activation, notification cardinality, RMI placement, monitoring existence, SoD); (7) run data checks (caps, deadlines, CoI registry join). Output: a layered verdict per case — which is precisely the instrument whose absence turned a stale POWL model, two expired authorities, and a broken log into an unresolvable dispute.
