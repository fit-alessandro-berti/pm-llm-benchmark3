- Delayed labels: external feed 7-35 days after horizon. Use `label_available_at`; only label if <= analysis cutoff. Do not impute default for immature. Use delayed-outcome modeling or censor at feed availability. For training, exclude immature from binary label but include as censored in survival.
- Selection through activation: default label only for activated. Activation is not random; risk factors influence both activation and default. Primary multi-state includes activation as part of process; conditional model requires selection adjustment (IPW by activation probability, stratification, or two-step). Testable: compare covariate distribution of activated vs non-activated; positivity. Untestable: whether default risk among rejected would have been lower/higher.
- Transfer: competing event (leaves local) or missing follow-up. Primary treat unlinked transfer as terminal competing/censored; linked transfers followed if activation/outcome observed. - Permanently missing outcome feeds (partner-originated): some activated accounts never get default_90d. EXTRACTION_TIME = pd.Timestamp("2026-02-15")
PRED_EVENT_COUNT = 3
PRED_MAX_DAYS = 14
FEED_LAG_MAX_DAYS = 35
HORIZON_DAYS = 90

CANONICAL_ACTIVITY = {
    "Credit Review": "CreditReview",
    "Credit Review (new)": "CreditReview",
    "Request Documents": "DocumentRequest",
    "Documents Requested": "DocumentRequest",
    "Account Activated": "AccountActivated",
    "Account activated": "AccountActivated",
}
TERMINAL_ACTIVITIES = {"Rejected", "Withdrawn", "Transferred"}
ACTIVATION_ACTIVITY = "AccountActivated"
DEFAULT_ACTIVITY = "DefaultRecorded"
FORBIDDEN_FEATURE_COLUMNS = {
    "final_limit", "decision_reason", "collection_status", "default_90d",
    "label_available_at", "data_extract_time", "concept:name", "time:timestamp"

def load_events(path="applications_events.csv"):
    df = pd.read_csv(path)
    for col in ["time:timestamp", "label_available_at", "data_extract_time"]:
        if col in df.columns:
            df[col] = pd.to_datetime(df[col], errors="coerce")
    # Canonicalize activity names after migration; keep original for audit.
    df["activity_canonical"] = df["concept:name"].map(CANONICAL_ACTIVITY).fillna(df["concept:name"])
    # Remove exact duplicates created by migration (same case, canonical activity, time, resource).
    before = len(df)
    df = df.drop_duplicates(subset=["case:concept:name", "activity_canonical", "time:timestamp", "org:resource"])
    df.attrs["duplicates_removed"] = before - len(df)
    return df.sort_values(["case:concept:name", "time:timestamp"]).reset_index(drop=True)

def build_case_table(events):
    rows = []
    for cid, g in events.groupby("case:concept:name", sort=False):
        start_time = g["time:timestamp"].min()
        activation = g.loc[g["activity_canonical"] == ACTIVATION_ACTIVITY, "time:timestamp"].min()
        terminal_time = g.loc[g["activity_canonical"].isin(TERMINAL_ACTIVITIES), "time:timestamp"].min()
        terminal_activity = None
        if pd.notna(terminal_time):
            terminal_activity = g.loc[g["time:timestamp"] == terminal_time, "activity_canonical"].iloc[0]
        default_time = g.loc[g["activity_canonical"] == DEFAULT_ACTIVITY, "time:timestamp"].min()
        label_available_at = g["label_available_at"].dropna().min()
        default_90d = g["default_90d"].dropna().iloc[-1] if g["default_90d"].notna().any() else np.nan
        # Prediction time: first of K events, start+T, or just before activation/terminal.
        # Use 1 ns before boundary so boundary event is not in the prefix.
        candidate_times = [start_time + pd.Timedelta(days=PRED_MAX_DAYS)]
        if len(g) >= PRED_EVENT_COUNT:
            candidate_times.append(g["time:timestamp"].iloc[PRED_EVENT_COUNT - 1])
        for boundary in [activation, terminal_time]:
            if pd.notna(boundary):
                candidate_times.append(boundary - pd.Timedelta(nanoseconds=1))
        t_pred = min([t for t in candidate_times if pd.notna(t)])
        # If no boundary and fewer than K events, use start+T or last event? start+T may be after extraction? cap extraction.
        t_pred = min(t_pred, EXTRACTION_TIME)
        # Label maturity: use actual feed timestamp if present; otherwise conservative horizon + max lag.
        if pd.notna(activation):
            conservative_mature = activation + pd.Timedelta(days=HORIZON_DAYS + FEED_LAG_MAX_DAYS) <= EXTRACTION_TIME
        else:
            conservative_mature = False
        label_matured = bool(pd.notna(label_available_at) and label_available_at <= EXTRACTION_TIME and conservative_mature)
        # Category for cohort flow. Transfer linkage would be added here if available.
        if terminal_activity == "Rejected":
            category = "rejected"
        elif terminal_activity == "Withdrawn":
            category = "withdrawn"
        elif terminal_activity == "Transferred":
            category = "transferred"
        elif pd.notna(activation):
            if label_matured and default_90d == 1:
                category = "labeled_default"
            elif label_matured and default_90d == 0:
                category = "labeled_nondefault"
            else:
                category = "activated_not_matured"
        else:
            category = "open"
        rows.append({...})
    ct = pd.DataFrame(rows)
    ct["start_time"] = pd.to_datetime(ct["start_time"])
    ...

def load_events(path="applications_events.csv"):
    df = pd.read_csv(path)

def build_case_table(events):
    rows = []
    for cid, g in events.groupby("case:concept:name", sort=False):
        start_time = g["time:timestamp"].min()
        activation = g.loc[g["concept:name"] == ACTIVATION_ACTIVITY, "time:timestamp"].min()
        terminal_time = g.loc[g["concept:name"].isin(TERMINAL_ACTIVITIES), "time:timestamp"].min()
        terminal_activity = None
        if pd.notna(terminal_time):
            terminal_activity = g.loc[g["time:timestamp"] == terminal_time, "concept:name"].iloc[0]
        default_time = g.loc[g["concept:name"] == DEFAULT_ACTIVITY, "time:timestamp"].min()
        label_available_at = g["label_available_at"].dropna().min()

        candidates = [start_time + pd.Timedelta(days=PRED_MAX_DAYS)]

        t_pred = min([t for t in candidates if pd.notna(t)], default=EXTRACTION_TIME)
        t_pred = min(t_pred, EXTRACTION_TIME)
        conservative_mature = pd.notna(activation) and (
            activation + pd.Timedelta(days=HORIZON_DAYS + FEED_LAG_MAX_DAYS) <= EXTRACTION_TIME
        )
        label_matured = bool(
            conservative_mature and pd.notna(label_available_at) and label_available_at <= EXTRACTION_TIME
        )
        if terminal_activity == "Rejected":
            category = "rejected"
        elif terminal_activity == "Withdrawn":
            category = "withdrawn"
        elif terminal_activity == "Transferred":

                category = "labeled_nondefault"
            else:
                category = "activated_not_matured"
        else:
            category = "open"
        rows.append({

            "activation_time": activation, "terminal_time": terminal_time,

## 1. Target definition

**Primary target formulation:** a **multi-state / competing-outcome process target** for all applications, with the primary event of interest being **90-day post-activation default**.

- **Unit of analysis:** one credit application case, identified by `case:concept:name`.
- **Population:** all 48,000 applications started between 1 January 2024 and 31 December 2025, observed as of the data-extraction date **15 February 2026**.
- **Prediction time:** an early, as-of checkpoint `t_pred` for each case, defined as the earliest of:
  1. the time of the 3rd observed process event;
  2. `start_time + 14 days`;
  3. one instant before the first of `Account Activated`, `Rejected`, `Withdrawn`, or `Transferred`;
  4. the extraction date, 15 February 2026.  
  Only events with `time:timestamp <= t_pred` are allowed to inform the prediction. The boundary event itself is excluded.
- **Horizon:** the default event is defined as a default occurring **within 90 days after `Account Activated`**. For activated non-default cases, the 90-day post-activation point is treated as a censoring / observed-non-event horizon, not as a causal “safe” label.
- **Primary estimand:** for a case with observed prefix `X(t_pred)`, estimate the conditional distribution over process outcomes:

\[
P(Y_i \in \{\text{Default}, \text{NonDefault90}, \text{Rejected}, \text{Withdrawn}, \text{Transferred}, \text{Censored/Open}\}
\mid X_i(t_{pred}), t_{pred}, \text{cohort})
\]

where:

- **Default** = activated and default recorded within 90 days after activation;
- **NonDefault90** = activated and no default by 90 days after activation, with label matured;
- **Rejected / Withdrawn / Transferred** = competing terminal process outcomes that preclude local activation and local default observation;
- **Censored/Open** = still open at extraction, activated but label not matured, or transfer with missing follow-up.

The operationally reported score can be the **probability of Default** under the observed process, accompanied by probabilities of rejection, withdrawal, transfer, and unresolved/censored status.

This primary formulation is defensible because:

1. It does **not** invent a default outcome for rejected, withdrawn, transferred, or open applications.
2. It preserves the fact that default is only observable after activation.
3. It uses the full 48,000-case process population rather than only the 21,600 matured activated accounts.
4. It separates ordinary censoring from outcomes that are undefined under the observed activation policy.
5. It gives the lender a decision-relevant answer: “What is the probability that this application results in a 90-day default under the current process?” rather than the less defensible “What is the default probability for every application as if all had been activated?”

**Credible alternative target:** **conditional 90-day default among accounts that are eventually activated**:

\[
P(\text{default within 90 days after activation} \mid \text{eventually activated}, X(t_{pred}))
\]

This answers a different question: the default risk **among the accepted/activated portfolio**. It is useful for post-activation monitoring or portfolio risk, but it is not a valid estimate of default risk for all applications unless one adds strong selection or counterfactual assumptions. If applied to rejected or withdrawn prefixes, it becomes a counterfactual estimate of what would have happened had those cases been activated, which is not directly identifiable from the data.

A further alternative is a **time-to-default survival target after activation**, with right censoring at 90 days, extraction, or delayed label maturity. That is statistically clean but still conditional on activation and therefore does not by itself answer the lender’s “every application” question.

## 2. Cohort construction

### Inclusion rules

Include an application in the study population if it satisfies all of the following:

1. It has an `Application Started` event, or at least one event with a valid `case:concept:name` and `time:timestamp`.
2. Its start time lies between **1 January 2024** and **31 December 2025**.
3. It is observed as of **15 February 2026**.
4. Duplicate events created by the 15 October 2025 migration are removed only when they are exact duplicates on case, canonical activity, timestamp, and resource. Non-exact duplicates are retained for audit and process discovery.

### Exclusion rules

Exclude only cases that are structurally unusable, for example:

- missing case identifier;
- missing or unparseable start timestamp;
- cases that cannot be placed in the process timeline.

Do **not** exclude rejected, withdrawn, transferred, open, or immature cases from the process population. They are essential for correct process-level inference.

### As-of rules

For every case:

- Features may use only events with `time:timestamp <= t_pred`.
- Labels may be used only if:
  - the case is activated;
  - `label_available_at <= 15 February 2026`;
  - the 90-day post-activation horizon plus the maximum expected feed lag is matured by the extraction date, or the actual `label_available_at` confirms maturity.
- Outcomes after the extraction date are not used.
- `default_90d` is treated as a label field, not as a feature.
- `final_limit`, `decision_reason`, `collection_status`, and post-activation fields are not used in the primary prefix feature table.

### Cohort-flow table

|---|---:|---:|---|---|

## 3. Censoring and missing-label strategy

### 1. Ordinary right censoring

Ordinary right censoring occurs when the process is still under observation but the target event has not yet occurred by the administrative cutoff.

Examples:

- Activated accounts with no default by 90 days after activation.
- Applications still open at 15 February 2026.
- Cases whose observation window ends before the 90-day post-activation horizon is reached.

In a survival or competing-risk formulation, activated non-default cases are censored at `activation_time + 90 days`. They are not treated as a separate “safe” event unless the analysis is explicitly a multi-state classification where `NonDefault90` is an observed matured state.

### 2. Delayed labels

The external outcome feed arrives 7–35 days after the 90-day horizon. Therefore, a case can be past the 90-day horizon but still not have a usable label.

Handling:

- Use `label_available_at` as the label-maturity timestamp.
- A label is usable only if `label_available_at <= 15 February 2026`.
- If `label_available_at` is missing, use a conservative maturity rule:

\[
activation\_time + 90\text{ days} + 35\text{ days} \le 15\text{ February 2026}
\]

- Immature cases are censored, not imputed as default or non-default.

### 3. Selection through activation

Default is observed only for activated accounts. Activation is not random: risk score, requested amount, region, document behavior, credit-review duration, and policy period can all affect both activation and default.

Consequences:

- A model trained only on activated cases estimates risk in a selected population.
- Rejected and withdrawn cases cannot be labeled non-default for convenience.
- The primary multi-state target keeps activation as part of the process outcome.
- A secondary conditional model can estimate `P(default | activated, prefix)`, but it must be clearly labeled as selection-conditioned.

### 4. Transfer as competing event or missing follow-up

Transfer is ambiguous:

- If the case leaves the local lender and never returns, transfer is a **competing terminal outcome**.
- If the case reappears on a partner platform and later activates, transfer is an intermediate state and follow-up should continue.
- If linkage is missing, transfer is a **missing follow-up mechanism**.

Primary strategy:

- Treat unlinked transfer as a terminal competing outcome for local default observation.
- If a linked transfer has a subsequent partner activation and matured outcome, follow it in a sensitivity analysis.
- If linked but outcome missing, treat it as censored or missing-outcome, not non-default.

### 5. Permanently missing outcome feeds

Some partner-originated activated accounts may never receive an outcome feed.

Handling:

- Do not assume missing feed means non-default.
- Treat as censored at last known time or as missing outcome under an explicit missingness assumption.
- In sensitivity analyses, use:
  - inverse probability weighting for feed availability;
  - pattern-mixture models;
  - multiple imputation with conservative scenarios;
  - exclusion of permanently missing feeds with bias bounds.

### Primary analysis strategy

Use a **multi-state / competing-risk framework**:

- Time origin: `t_pred`.
- Events:
  - Default within 90 days after activation;
  - Rejected;
  - Withdrawn;
  - Transferred, if treated as terminal.
- Censoring:
  - open at extraction;
  - activated non-default at 90 days;
  - immature label;
  - missing transfer follow-up.

A practical implementation can use:

- cause-specific hazard models, e.g. Cox-type or random survival forest models;
- or a multi-class classifier with censoring-aware evaluation;
- plus a separate activation model `P(activation | prefix)` for diagnostics.

The operational default score should be derived from the cumulative incidence of Default, not from a naive binary classifier trained only on matured activated cases.

### Sensitivity analyses

At least the following should be run:

1. **Activated-only conditional model:**  
   Estimate `P(default | activated, prefix)` using only matured activated cases.

2. **Transfer-as-missing analysis:**  
   Treat unlinked transfers as missing follow-up rather than terminal competing events.

3. **Linked-transfer follow-up:**  
   Follow the approximately 60% linked transfers to partner activation and outcome where available.

4. **Missing-feed sensitivity:**  
   Exclude or reweight partner-originated activated accounts with permanently missing feeds.

5. **Policy-stratified analysis:**  
   Fit and validate separately before and after 1 July 2025.

6. **Migration-stratified analysis:**  
   Fit and validate separately before and after 15 October 2025, after canonicalizing activity names.

### Testability of assumptions

| Assumption | Testability |
|---|---|
| Event ordering and duplicate-event handling | Testable from timestamps, case IDs, and activity names |
| Activity renaming after migration | Testable by comparing canonical and original activity distributions before/after 15 October 2025 |
| Label maturity by extraction date | Testable using `label_available_at` and extraction date |
| Distribution shift across policy and migration periods | Testable using feature, variant, and outcome-rate comparisons |
| Missing feed related to observed covariates | Partially testable using partner origin, region, channel, amount, and risk score |
| Transfer missing at random | Partially testable by comparing linked vs unlinked pre-transfer features |
| Activation selection independent of unobserved default risk | Fundamentally untestable without stronger assumptions |
| Counterfactual default risk for rejected/withdrawn cases | Fundamentally untestable from these data alone |

## 4. Prefix extraction

Prefixes must be extracted from the perspective of the prediction time `t_pred`. No event after `t_pred` may enter the prefix.

### General rules

1. Sort events by `case:concept:name` and `time:timestamp`.
2. Canonicalize activity names before process mining or feature extraction.
3. Remove exact migration duplicates only when case, canonical activity, timestamp, and resource are identical.
4. For each case, include only events satisfying:

\[
time:timestamp \le t_pred
\]

5. Exclude the boundary event if `t_pred` is defined as one instant before activation or a terminal event.
6. If multiple events share the same timestamp, use a deterministic tie-breaker, such as event order metadata, resource, or channel, and document it.

### Fixed event-count prefixes

Primary fixed prefix:

- first **3 events** after `Application Started`.

Sensitivity:

- first 5 events;
- first event after the first `Credit Review`;
- first event after first document request, if present.

If a case terminates before the fixed event count is reached, the prefix ends just before the terminal event. For example, a case that is rejected after two events has a two-event prefix and a `short_case` indicator.

### Elapsed-time prefixes

Primary elapsed-time prefix:

- all events up to `start_time + 14 days`.

If the case has fewer than 14 days of observed history by the extraction date, use all observed events up to `t_pred`.

Sensitivity:

- 7-day prefix;
- 21-day prefix;
- prefix ending at first `Credit Review`;
- prefix ending at first document request.

### Short cases

Short cases are not discarded. They are handled with explicit indicators:

- `event_count`;
- `time_to_terminal`;
- `terminal_before_prediction_checkpoint`;
- `short_case_flag`;
- `current_activity`;
- `last_event_age_days`.

A rejected case with only two events is informative: it may indicate fast rejection, which is a competing outcome, not a non-default credit outcome.

### Repeated document loops

Do not collapse repeated document loops into a single activity. Preserve the sequence and derive features such as:

- number of `DocumentRequest -> CreditReview` cycles;
- maximum consecutive document-related events;
- total time spent in document loops;
- time between document requests;
- whether the loop occurred before or after the first credit review.

### Activity renaming and migration

The 15 October 2025 migration renamed activities and created duplicates.

Handling:

- Maintain a canonical activity mapping.
- Keep the original activity name in an audit column, but do not use it as a primary feature.
- Run process discovery separately on:
  - pre-migration data;
  - post-migration data;
  - canonicalized combined data.
- Monitor unknown or newly appearing canonical activities.
- Treat unexpected variants as data-quality flags rather than silently merging them.

## 5. Process-aware features

Features must be available at `t_pred`. They should be grouped into static, prefix-derived, and historical-reference features.

### Static case attributes

These are available at or near application start:

- `case:requested_amount`;
- `case:region`;
- `case:channel`;
- `source_system`;
- start date;
- day of week;
- month;
- policy period: before/after 1 July 2025;
- migration period: before/after 15 October 2025;
- partner origin, if known at start and not derived from later transfer linkage.

### Prefix-derived features

These are computed only from events up to `t_pred`:

#### Control-flow features

- current activity;
- last 3 or last 5 activity sequence;
- prefix variant;
- whether the case has entered `CreditReview`;
- whether a document request has occurred;
- number of document-request loops;
- number of transitions between `CreditReview` and `DocumentRequest`;
- DFG-based transition counts within the prefix;
- whether the prefix matches a known high-risk or high-rejection variant.

#### Timing features

- elapsed time since start;
- time since last event;
- time spent in `CreditReview`;
- time spent in document loops;
- mean, median, and maximum inter-event time;
- time to first `CreditReview`;
- time to first document request;
- time to first decision-like event, if it occurs before `t_pred`.

#### Resource and queue features

- number of distinct resources involved;
- resource sequence;
- whether the same resource handled multiple events;
- resource workload as of the event time, if available from historical reference data;
- queue waiting time as of the event time, if available.

Do **not** use extraction-time resource experience.

#### Risk and application-state features

- first observed `risk_score` at or before `t_pred`;
- change in `risk_score` within the prefix;
- number of `risk_score` updates within the prefix;
- whether `risk_score` changed after a document request;
- requested amount relative to region or channel benchmarks, if benchmarks are computed from historical data only.

### Features requiring historical reference data

Some features require external or historical tables:

- resource experience as of the event time;
- historical queue wait times by resource and date;
- historical duration distributions by activity, region, and policy period;
- historical default or rejection rates by prefix variant, computed with temporal validation to avoid leakage;
- document upload volume by region and date, if available from event-level data.

These features must be computed using only information available before `t_pred` for the target case.

## 6. Leakage audit

| Feature or data element | Leakage or bias risk | Valid availability rule | Keep/repair/drop | Rationale |
|---|---|---|---|---|
| `final_limit` | Populated only after final credit decision; can reveal acceptance or rejection and is downstream of the decision process | Available only after the final decision event; not available at the primary pre-decision prediction checkpoint | Drop from primary prefix features | It is a post-decision field and would leak the decision path |
| `decision_reason` | Populated at or after Accept/Reject; directly related to terminal process outcome | Available only after decision event | Drop from primary prefix features | It reveals or is tightly coupled to rejection/acceptance |
| `collection_status` | Updated after activation and may directly reveal default or collection state | Available only after activation and post-default processes | Drop from primary prefix features | It is a post-activation outcome-related field |
| Future events, including `Contract Signed`, `Account Activated`, `Collections Contact`, `Default Recorded` | Direct leakage of activation, default, or terminal outcome | Only events with `time:timestamp <= t_pred` may be used; boundary event excluded | Drop post-prefix events | The model must predict from the early prefix, not from later process realization |
| `documents_total` | Nightly snapshot may include documents uploaded later on the same calendar day than the nominal prediction time | Use only event-level document uploads with timestamp `<= t_pred`; if only snapshot is available, use previous-day snapshot or lag by one day | Repair or drop | Same-day snapshot can leak future document behavior |
| Updated `risk_score` | Risk score may be updated after decision, after documents, or after other post-prefix events | Use only the value attached to an event with timestamp `<= t_pred`; if multiple updates occur at the same timestamp, use the earliest pre-decision value or a documented tie-break | Keep with repair | The as-of risk score is useful, but post-prefix updates must be excluded |
| Resource experience | Stored at extraction time, not at event time; can leak later career progression and introduce time bias | Reconstruct experience as of the event time from a historical resource table; if impossible, do not use extraction-time experience | Repair or drop | Extraction-time experience is not available at prediction time |
| Label availability, `label_available_at`, repeated `default_90d` | Using label maturity or label value as a feature leaks outcome availability or outcome itself | Use only to determine censoring, label maturity, and evaluation eligibility; never as a model feature | Drop from features | Label fields define the target, not the predictor |
| `data_extract_time` | Administrative extraction metadata, not a case-level predictor at prediction time | Use only to define the as-of cutoff | Drop from features | It is not a valid predictive feature and can introduce batch bias |
| Transfer linkage status | Linkage may be established after the prediction time and may depend on later partner data | Use only if linkage timestamp is `<= t_pred`; otherwise treat as missingness adjustment, not a feature | Repair or drop | Post-hoc linkage can leak future process information |

## 7. Validation design

### Why a random case split is misleading

A random split would be misleading because it ignores the temporal and structural changes in the data:

1. It mixes cases from before and after the 1 July 2025 policy change.
2. It mixes cases from before and after the 15 October 2025 migration.
3. It allows test cases to have the same calendar, policy, and migration context as training cases, inflating apparent performance.
4. It ignores label maturity: late cases may not have matured labels by the extraction date.
5. It violates the as-of perspective: a model deployed at time `t` should only have learned from cases available before `t`.

Therefore, the primary validation must be **temporal**.

### Primary temporal split

Use case start time:

| Split | Case start window | Purpose |
|---|---|---|
| Train | 1 January 2024 through 30 June 2025 | Pre-policy-change training |
| Validation | 1 July 2025 through 14 October 2025 | Post-policy, pre-migration validation |
| Test | 15 October 2025 through 31 December 2025 | Post-migration test |

For binary default metrics, evaluate only cases with `label_matured = True`. For survival or competing-risk metrics, evaluate all test cases with appropriate censoring, but report the number and proportion of censored cases.

Because many late-2025 cases may not have matured 90-day post-activation labels by 15 February 2026, the test set should report:

- number of matured activated cases;
- number of defaults among matured activated cases;
- number of censored cases;
- number of competing outcomes.

### Rolling backtest

Use at least two rolling origins:

1. **Policy-shift backtest**
   - Train: 1 January 2024 through 30 June 2025
   - Test: 1 July 2025 through 30 September 2025

2. **Migration-shift backtest**
   - Train: 1 January 2024 through 14 October 2025
   - Test: 15 October 2025 through 31 December 2025, with maturity-filtered binary evaluation and censoring-aware survival evaluation.

This separates the effects of the policy change and the system migration.

### Calibration

Calibration is essential because the business question is risk prediction, not only ranking.

Use:

- reliability diagrams;
- Brier score;
- calibration slope and intercept;
- expected log-loss;
- calibration of cumulative incidence for Default;
- decision-curve analysis for default probability thresholds.

For competing-risk models, calibrate the cumulative incidence of Default separately from the cumulative incidence of competing outcomes.

### Subgroup performance

Report performance by:

- policy period: pre/post 1 July 2025;
- migration period: pre/post 15 October 2025;
- region;
- channel;
- requested amount bucket;
- risk-score bucket;
- partner origin;
- transfer status;
- label-feed availability;
- activation status.

A model that performs well overall but poorly in post-policy or partner-originated subgroups is not deployable without correction.

### Distribution shift monitoring

Monitor:

- feature distribution shift using PSI or KL divergence;
- activation rate;
- rejection, withdrawal, and transfer rates;
- default rate among matured activated cases;
- variant distribution;
- DFG transition distribution;
- event timing distributions;
- missingness rates for `risk_score`, `label_available_at`, and transfer linkage.

### Evaluation metrics

For the binary default target among matured activated cases:

- PR-AUC;
- ROC-AUC;
- precision at fixed recall;
- lift at fixed recall;
- Brier score;
- calibration metrics.

For the competing-risk / multi-state target:

- time-dependent AUC for Default;
- integrated Brier score;
- cause-specific calibration;
- cumulative incidence calibration;
- multi-class log-loss;
- per-class precision, recall, and F1;
- decision-curve analysis.

Report event counts and censoring fractions alongside metrics. A high AUC with very few matured default events in the test period is not sufficient evidence of deployability.

## 8. Limitations and decision-use boundary

### What the study can claim

The study can support claims such as:

- “Among applications observed as of 15 February 2026, the model predicts the probability of a 90-day post-activation default under the current process.”
- “The model distinguishes early prefixes that are associated with higher default, rejection, withdrawal, or transfer probabilities.”
- “Performance is estimated on temporally held-out cases with explicit label-maturity and censoring handling.”
- “The model does not treat rejected, withdrawn, transferred, or open cases as non-default.”

### What the study cannot claim

The study cannot claim:

- “Rejected applications would not have defaulted if they had been activated.”
- “Withdrawn applications are safe.”
- “Open applications are non-default.”
- “The model estimates counterfactual default risk under a different activation policy.”
- “The model is valid indefinitely after policy, migration, or deployment changes.”
- “The model identifies causal drivers of default.”

### Predictive association vs selection effects vs counterfactual risk

- **Predictive association:** the model learns that certain early prefixes are associated with higher default, rejection, or withdrawal probabilities in the historical process.
- **Selection effect:** activated accounts are a selected subset. Their default risk is not the default risk of all applications.
- **Counterfactual default risk:** the risk that a rejected or withdrawn application would have experienced default had it been activated. This requires untestable assumptions and is not the primary estimand.

### Ways deployment could change the data-generating process

At least three deployment feedback mechanisms are likely:

1. **Selection feedback:**  
   If the model is used to reject or flag high-risk applications, the activated portfolio becomes compositionally different. Future activated cases may have lower observed default rates, causing calibration drift and changing the relationship between prefixes and outcomes.

2. **Behavioral and process feedback:**  
   If the model triggers earlier document requests, manual review, collections contact, or applicant outreach, it can change withdrawal rates, default timing, and process paths. The historical DFG and timing distributions may no longer apply.

3. **Policy and system drift:**  
   Future credit-policy changes, additional migrations, new partner transfers, or changes in the outcome feed can alter activity names, decision thresholds, transfer behavior, and label availability. The model would need revalidation and possibly re-estimation.

## 9. PM4Py-oriented Python implementation skeleton

```python
import pandas as pd
import numpy as np
import pm4py

EXTRACTION_TIME = pd.Timestamp("2026-02-15")
PRED_EVENT_COUNT = 3
PRED_MAX_DAYS = 14
HORIZON_DAYS = 90
FEED_LAG_MAX_DAYS = 35

CANONICAL_ACTIVITY = {
    "Credit Review (new)": "CreditReview",
    "Credit Review": "CreditReview",
    "Documents Requested": "DocumentRequest",
    "Request Documents": "DocumentRequest",
    "Account activated": "AccountActivated",
}
TERMINAL_ACTIVITIES = {"Rejected", "Withdrawn", "Transferred"}
ACTIVATION_ACTIVITY = "AccountActivated"
DEFAULT_ACTIVITY = "DefaultRecorded"
FORBIDDEN_FEATURE_COLUMNS = {
    "final_limit", "decision_reason", "collection_status", "default_90d",
    "label_available_at", "data_extract_time", "documents_total",
    "resource_experience_at_extraction", "activity_original",
}

def load_events(path="applications_events.csv"):
    df = pd.read_csv(path)
    for col in ["time:timestamp", "label_available_at", "data_extract_time"]:
        if col in df.columns:
            df[col] = pd.to_datetime(df[col], errors="coerce")
    df["activity_original"] = df["concept:name"]
    df["concept:name"] = df["concept:name"].map(CANONICAL_ACTIVITY).fillna(df["concept:name"])
    before = len(df)
    df = df.drop_duplicates(subset=["case:concept:name", "concept:name", "time:timestamp", "org:resource"])
    df.attrs["duplicates_removed"] = before - len(df)
    return df.sort_values(["case:concept:name", "time:timestamp"]).reset_index(drop=True)

def build_case_table(events):
    rows = []
    for cid, g in events.groupby("case:concept:name", sort=False):
        start_time = g["time:timestamp"].min()
        activation = g.loc[g["concept:name"] == ACTIVATION_ACTIVITY, "time:timestamp"].min()
        terminal_time = g.loc[g["concept:name"].isin(TERMINAL_ACTIVITIES), "time:timestamp"].min()
        terminal_activity = None
        if pd.notna(terminal_time):
            terminal_activity = g.loc[g["time:timestamp"] == terminal_time, "concept:name"].iloc[0]
        default_time = g.loc[g["concept:name"] == DEFAULT_ACTIVITY, "time:timestamp"].min()
        label_available_at = g["label_available_at"].dropna().min()
        default_90d = g["default_90d"].dropna().iloc[-1] if g["default_90d"].notna().any() else np.nan
        candidates = [start_time + pd.Timedelta(days=PRED_MAX_DAYS)]
        if len(g) >= PRED_EVENT_COUNT:
            candidates.append(g["time:timestamp"].iloc[PRED_EVENT_COUNT - 1])
        for boundary in [activation, terminal_time]:
            if pd.notna(boundary):
                candidates.append(boundary - pd.Timedelta(nanoseconds=1))
        t_pred = min([t for t in candidates if pd.notna(t)], default=EXTRACTION_TIME)
        t_pred = min(t_pred, EXTRACTION_TIME)
        conservative_mature = pd.notna(activation) and (
            activation + pd.Timedelta(days=HORIZON_DAYS + FEED_LAG_MAX_DAYS) <= EXTRACTION_TIME
        )
        label_matured = bool(
            conservative_mature and pd.notna(label_available_at) and label_available_at <= EXTRACTION_TIME
        )
        if terminal_activity == "Rejected":
            category = "rejected"
        elif terminal_activity == "Withdrawn":
            category = "withdrawn"
        elif terminal_activity == "Transferred":
            category = "transferred"
        elif pd.notna(activation):
            if label_matured and default_90d == 1:
                category = "labeled_default"
            elif label_matured and default_90d == 0:
                category = "labeled_nondefault"
            else:
                category = "activated_not_matured"
        else:
            category = "open"
        rows.append({
            "case:concept:name": cid, "start_time": start_time, "t_pred": t_pred,
            "activation_time": activation, "terminal_time": terminal_time,
            "terminal_activity": terminal_activity, "default_time": default_time,
            "label_available_at": label_available_at, "default_90d": default_90d,
            "label_matured": label_matured, "category": category,
        })
    ct = pd.DataFrame(rows)
    for col in ["start_time", "t_pred", "activation_time", "terminal_time", "default_time", "label_available_at"]:
        ct[col] = pd.to_datetime(ct[col], errors="coerce")
    return ct

def temporal_split(ct):
    ct = ct.copy()
    # Random pm4py.split_train_test is only a convenience baseline and is not primary here.
    ct["policy_period"] = np.where(ct["start_time"] < pd.Timestamp("2025-07-01"), "pre_policy", "post_policy")
    ct["migration_period"] = np.where(ct["start_time"] < pd.Timestamp("2025-10-15"), "pre_migration", "post_migration")
    ct["split"] = np.where(ct["start_time"] < pd.Timestamp("2025-07-01"), "train",
                           np.where(ct["start_time"] < pd.Timestamp("2025-10-15"), "validation", "test"))
    ct["binary_evaluable"] = ct["category"].isin(["labeled_default", "labeled_nondefault"])
    return ct

def build_prefix_features(events, ct, prefix_length=PRED_EVENT_COUNT):
    ev = events.merge(ct[["case:concept:name", "t_pred"]], on="case:concept:name", how="left")
    pre = ev[ev["time:timestamp"] <= ev["t_pred"]].copy()
    pre = pre.drop(columns=[c for c in FORBIDDEN_FEATURE_COLUMNS if c in pre.columns])
    log = pm4py.format_dataframe(pre, case_id="case:concept:name", activity_key="concept:name", timestamp_key="time:timestamp")
    prefix_logs = pm4py.get_prefixes_from_log(log, prefix_length=prefix_length, case_id_key="case:concept:name", activity_key="concept:name", timestamp_key="time:timestamp")
    dfg = pm4py.discover_dfg(log, case_id_key="case:concept:name", activity_key="concept:name", timestamp_key="time:timestamp")
    variant_counts = pm4py.get_variants(log, case_id_key="case:concept:name", activity_key="concept:name")
    feature_rows = []
    for cid, g in pre.groupby("case:concept:name", sort=False):
        acts = g["concept:name"].tolist()
        t0 = g["time:timestamp"].min()
        t1 = g["time:timestamp"].max()
        rs = g["risk_score"].dropna() if "risk_score" in g.columns else pd.Series(dtype=float)
        feature_rows.append({
            "case:concept:name": cid,
            "event_count": len(g),
            "elapsed_days": (t1 - t0).total_seconds() / 86400.0 if len(g) else 0.0,
            "current_activity": acts[-1] if acts else None,
            "has_document_request": int("DocumentRequest" in acts),
            "document_loop_count": int(sum(1 for a, b in zip(acts, acts[1:]) if a == "DocumentRequest" and b == "CreditReview")),
            "distinct_resources": g["org:resource"].nunique() if "org:resource" in g.columns else 0,
            "risk_score_first": rs.iloc[0] if len(rs) else np.nan,
            "risk_score_change": (rs.iloc[-1] - rs.iloc[0]) if len(rs) >= 2 else 0.0,
        })
    feat = pd.DataFrame(feature_rows)
    static_cols = [c for c in ["case:requested_amount", "case:region", "case:channel", "source_system"] if c in pre.columns]
    static = pre.drop_duplicates("case:concept:name").set_index("case:concept:name")[static_cols]
    feat = feat.set_index("case:concept:name").join(static).reset_index()
    feat = feat.merge(ct[["case:concept:name", "split", "category", "start_time", "t_pred", "label_matured", "binary_evaluable"]], on="case:concept:name", how="left")
    bad_cols = sorted(set(feat.columns) & FORBIDDEN_FEATURE_COLUMNS)
    max_pre_ts = pre.groupby("case:concept:name")["time:timestamp"].max()
    limit_ts = ct.set_index("case:concept:name")["t_pred"]
    post_prefix_leak = int((max_pre_ts > limit_ts.reindex(max_pre_ts.index)).sum())
    feat.attrs["leakage_bad_columns"] = bad_cols
    feat.attrs["leakage_post_prefix_events"] = post_prefix_leak
    return feat, dfg, variant_counts, prefix_logs

def main():
    events = load_events()
    ct = build_case_table(events)
    ct = temporal_split(ct)
    cohort = ct["category"].value_counts().reindex(
        ["labeled_default", "labeled_nondefault", "activated_not_matured",
         "rejected", "withdrawn", "transferred", "open"], fill_value=0)
    feat, dfg, variants, prefix_logs = build_prefix_features(events, ct)
    print("Cohort flow:\n", cohort)
    print("Temporal split:\n", ct["split"].value_counts())
    print("Leakage audit:", feat.attrs)
    # Connect survival/competing-risk model here: events = default/reject/withdraw/transfer,
    # censoring = open, immature label, 90-day non-default horizon.

if __name__ == "__main__":
    main()
```
