def build_scenario(name):
    demand = [list(row) for row in DEMAND]
    svc = {k: list(v) for k,v in SERVICE.items()}
    absent = set(); outage = []
    if name in ("U1","U5"):
        for row in demand:
            if row[0] in (8,10):
                row[1] = row[1]*1.7
                row[2] = row[2]+8  # P1 pct + 8pp (cap)
                # renormalize p2,p3 so sums 100
                rest = 100-row[2]; cur=row[3]+row[4]; row[3]=row[3]*rest/cur; row[4]=row[4]*rest/cur
    if name == "U2":
        # E and ES case counts double in 12-16, proportions renormalized
        for row in demand:
            if row[0] in (12,14):
                # convert pct to counts, double E and ES, recompute pcts
                n=row[1]; counts=[n*p/100 for p in row[5:10]]
                counts[1]*=2; counts[4]*=2
                tot=sum(counts); row[1]=tot
                row[5:10]=[100*c/tot for c in counts]
    if name in ("U3","U5"): absent.add("Chen")
    if name == "U4":
        for k in svc: svc[k][4]=min(1.0, svc[k][4]*2)
    if name == "U5": outage=[(9.5,10.0)]
    return demand, svc, absent, outage
