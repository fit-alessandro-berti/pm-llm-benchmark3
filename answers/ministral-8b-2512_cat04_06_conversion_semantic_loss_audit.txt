import pm4py
from pm4py.objects.petri_net.obj import PetriNet, Marking
from pm4py.objects.bpmn.obj import BPMNModel
from pm4py.objects.process_tree.obj import ProcessTree
from pm4py.objects.log.obj import Trace
from pm4py.algo.discovery.footprints import algorithm as footprints_discovery
from pm4py.algo.conversion.petri_net import algorithm as conversion_petri_net
from pm4py.algo.conversion.bpmn import algorithm as conversion_bpmn
from pm4py.algo.conversion.process_tree import algorithm as conversion_process_tree
from pm4py.algo.soundness import algorithm as soundness_check
from pm4py.visualization.petri_net import visualizer as pn_visualizer
from pm4py.visualization.bpmn import visualizer as bpmn_visualizer
from pm4py.visualization.process_tree import visualizer as pt_visualizer

def convert_and_validate(source_powl):
    # --- POWL  Petri Net ---
    net_from_powl, im_from_powl, fm_from_powl = conversion_petri_net.apply(source_powl)
    pm4py.save_vis_petri_net(net_from_powl, im_from_powl, fm_from_powl, "net_from_powl.png")

    # --- Petri Net  BPMN (Primary Pipeline) ---
    bpmn_from_net = conversion_bpmn.apply(net_from_powl, im_from_powl, fm_from_powl)
    pm4py.save_vis_bpmn(bpmn_from_net, "bpmn_from_net.png")

    # --- Petri Net  Process Tree  BPMN (Secondary Pipeline) ---
    try:
        tree_from_net = conversion_process_tree.apply(net_from_powl, im_from_powl, fm_from_powl)
        pm4py.save_vis_process_tree(tree_from_net, "tree_from_net.png")
        bpmn_from_tree = conversion_bpmn.apply(tree_from_net)
        pm4py.save_vis_bpmn(bpmn_from_tree, "bpmn_from_tree.png")
    except Exception as e:
        print(f"Process tree conversion failed: {e}")
        bpmn_from_tree = None

    # --- Round-trip validation ---
    round_trip_nets = []
    if bpmn_from_net:
        net_from_bpmn, im_from_bpmn, fm_from_bpmn = conversion_petri_net.apply(bpmn_from_net)
        round_trip_nets.append((net_from_bpmn, im_from_bpmn, fm_from_bpmn))
        pm4py.save_vis_petri_net(net_from_bpmn, im_from_bpmn, fm_from_bpmn, "round_trip_net.png")

    if bpmn_from_tree:
        net_from_bpmn_tree, im_from_bpmn_tree, fm_from_bpmn_tree = conversion_petri_net.apply(bpmn_from_tree)
        round_trip_nets.append((net_from_bpmn_tree, im_from_bpmn_tree, fm_from_bpmn_tree))
        pm4py.save_vis_petri_net(net_from_bpmn_tree, im_from_bpmn_tree, fm_from_bpmn_tree, "round_trip_net_from_tree.png")

    # --- Diagnostics ---
    for i, (net, im, fm) in enumerate([(net_from_powl, im_from_powl, fm_from_powl)] + round_trip_nets):
        print(f"\n--- Diagnostics for Net {i} ---")
        is_wfn = pm4py.check_is_workflow_net(net, im, fm)
        print(f"Workflow Net: {is_wfn}")
        if is_wfn:
            sound = soundness_check.apply(net, im, fm)
            print(f"Soundness: {sound}")
            if sound:
                footprints = footprints_discovery.apply(net, im, fm)
                print(f"Footprints: {footprints}")

    return {
        "net_from_powl": (net_from_powl, im_from_powl, fm_from_powl),
        "bpmn_from_net": bpmn_from_net,
        "bpmn_from_tree": bpmn_from_tree,
        "round_trip_nets": round_trip_nets
    }

# Run conversion
results = convert_and_validate(source_powl)
```

```python
def validate_round_trip(original_net, original_im, original_fm, round_trip_net, round_trip_im, round_trip_fm, bound=3):
    def explore_traces(net, im, fm, bound):
        traces = []
        current_marking = im
        for _ in range(bound):
            enabled = pm4py.get_enabled_transitions(net, current_marking)
            if not enabled:
                break
            for t in enabled:
                if t.label is None:  # Skip silent transitions
                    continue
                new_marking = pm4py.get_successors(net, current_marking, t)[0]
                traces.append(t.label)
                current_marking = new_marking
                break
        return traces

    # Check workflow net and soundness
    is_wfn = pm4py.check_is_workflow_net(round_trip_net, round_trip_im, round_trip_fm)
    sound = soundness_check.apply(round_trip_net, round_trip_im, round_trip_fm)

    # Check footprints
    original_footprints = footprints_discovery.apply(original_net, original_im, original_fm)
    round_trip_footprints = footprints_discovery.apply(round_trip_net, round_trip_im, round_trip_fm)

    # Check traces (bounded exploration)
    original_traces = explore_traces(original_net, original_im, original_fm, bound)
    round_trip_traces = explore_traces(round_trip_net, round_trip_im, round_trip_fm, bound)

    # Compare reachability graphs (simplified)
    original_rg = conversion_petri_net.apply_to_reachability_graph(original_net, original_im, original_fm)
    round_trip_rg = conversion_petri_net.apply_to_reachability_graph(round_trip_net, round_trip_im, round_trip_fm)

    # Custom validation: Check if round-trip net can simulate original behavior
    def can_simulate(original_net, original_im, round_trip_net, round_trip_im):
        # Simplified: Check if round-trip net can produce at least one trace from original
        original_traces = set()
        current_marking = original_im
        while True:
            enabled = pm4py.get_enabled_transitions(original_net, current_marking)
            if not enabled:
                break
            for t in enabled:
                if t.label is None:
                    continue
                new_marking = pm4py.get_successors(original_net, current_marking, t)[0]
                original_traces.add(t.label)
                current_marking = new_marking
                break
        # Check if round-trip net can produce any of these traces
        round_trip_traces = set()
        current_marking = round_trip_im
        while True:
            enabled = pm4py.get_enabled_transitions(round_trip_net, current_marking)
            if not enabled:
                break
            for t in enabled:
                if t.label is None:
                    continue
                new_marking = pm4py.get_successors(round_trip_net, current_marking, t)[0]
                round_trip_traces.add(t.label)
                current_marking = new_marking
                break
        return any(trace in round_trip_traces for trace in original_traces)

    print(f"Workflow Net: {is_wfn}")
    print(f"Soundness: {sound}")
    print(f"Footprints match: {original_footprints == round_trip_footprints}")
    print(f"Can simulate original traces: {can_simulate(original_net, original_im, round_trip_net, round_trip_im)}")
    print(f"Original traces (bounded): {original_traces}")
    print(f"Round-trip traces (bounded): {round_trip_traces}")

    return {
        "is_wfn": is_wfn,
        "sound": sound,
        "footprints_match": original_footprints == round_trip_footprints,
        "can_simulate": can_simulate(original_net, original_im, round_trip_net, round_trip_im),
        "original_traces": original_traces,
        "round_trip_traces": round_trip_traces
    }

# Run validation
for i, (net, im, fm) in enumerate(results["round_trip_nets"]):

# 6. Visual-Semantics Review

The BPMN rendering may mislead stakeholders in several ways:
1. **Parallel Gateways**: BPMN parallel gateways may not perfectly represent POWL concurrency. For example, `Assess Risk || Prepare Evidence` might be visualized as a parallel gateway, but the exact interleaving semantics may not be preserved.
2. **Exclusive Gateways**: The XOR for `optional_legal` may be visualized as a gateway, but the silent skip transition (`skip_legal`) is invisible in BPMN. This could lead stakeholders to believe both branches are always taken.
3. **Loop Visualization**: The verification loop (`verification_loop`) may be visualized as a BPMN loop, but the separator (`correct_defect`) and minimum iteration count may not be clear. Stakeholders might assume zero iterations are allowed.
4. **Silent Routing Nodes**: Silent transitions (e.g., `skip_legal`) are not representable in BPMN, so their absence could lead to incorrect assumptions about behavior.
5. **Graph Layout**: The apparent ordering of activities in the BPMN diagram may suggest dependencies that don't exist in the original POWL model (e.g., `optional_legal` and `verification_loop` are unordered).

**Proposed Annotations**:
- Add a legend explaining silent transitions and their absence in BPMN.
- Document that parallel gateways in BPMN do not guarantee true concurrency (only interleaving).
- Clarify that loops in BPMN may not enforce minimum iteration counts or separators.
- Use color-coding or labels to distinguish optional vs. mandatory branches.

# 7. Rejection Criteria

A conversion result should be rejected or quarantined if:
1. The resulting Petri net is not a workflow net (no initial/final marking or unreachable transitions).
2. The net is not sound (e.g., deadlocks, livelocks, or unreachable final marking).
3. The round-trip net cannot simulate at least one trace from the original POWL model.
4. The BPMN model admits release before verification (violating synchronization invariant O9).
5. The BPMN model allows zero iterations of the verification loop (violating O8).
6. The BPMN model collapses optional branches (e.g., forces both `Legal Review` and `skip_legal` to be taken).
7. The BPMN model misrepresents silent transitions (e.g., omits `skip_legal` entirely).
8. The round-trip net's reachability graph does not include all traces from the original POWL model (within bounded exploration).
9. The BPMN model's footprints do not match the original POWN model's footprints.
10. The process tree conversion fails, and no alternative BPMN model is provided.

# 8. Canonical Representation Recommendation

**Canonical Model for Storage**:
- **Authoritative**: The original POWL model (`source_powl`) should be the canonical representation. It is the most precise and semantically rich.
- **Derivative**: Converted Petri nets, BPMN models, and process trees are derivative artifacts. They should be generated on demand and validated against the POWL model.

**Validation Evidence**:
- For Petri nets: Workflow net check, soundness check, and reachability graph comparison.
- For BPMN models: Footprint comparison, trace simulation, and explicit documentation of silent transitions.
- For process trees: Structural equivalence to the original POWL (if possible) and behavioral equivalence via round-trip validation.

**Process Tree Intermediate**:
- **Allowed**: Only if the process tree conversion succeeds and the resulting tree is structurally equivalent to the POWL model (e.g., preserves XOR, loop, and concurrency semantics). Otherwise, avoid it to prevent semantic loss.

**Future Edits**:
- Edits should be made directly to the POWL model. Derivative models (Petri net, BPMN) should be regenerated from the POWL model to avoid drift.
- A validation pipeline should be run after every edit to ensure behavioral equivalence.

# 9. Residual Uncertainty

Even if all tests pass, the following claims cannot be definitively established:
1. **True Concurrency vs. Interleaving**: The BPMN model may not perfectly distinguish between true concurrency (POWL) and arbitrary interleaving (BPMN). Stakeholders might assume dependencies where none exist.
2. **Silent Transition Behavior**: The absence of silent transitions in BPMN means their effects (e.g., skipping `Legal Review`) cannot be directly observed or validated in the BPMN model.
3. **Minimum Iteration Counts**: While the Petri net may enforce at least one iteration of the verification loop, the BPMN model's loop semantics may not clearly communicate this requirement to stakeholders.
4. **Exact Final Marking**: The round-trip validation uses a bounded exploration, so it cannot guarantee that all possible traces (including those with silent transitions) are covered.
5. **Behavioral Equivalence Under All Markings**: The validation assumes that the initial and final markings are correctly preserved, but silent transitions or hidden states may introduce subtle differences not caught by the tests.
6. **Stakeholder Misinterpretation**: Even if the models are behaviorally equivalent, stakeholders may misinterpret the BPMN visualization due to layout, gateway semantics, or lack of annotations.
