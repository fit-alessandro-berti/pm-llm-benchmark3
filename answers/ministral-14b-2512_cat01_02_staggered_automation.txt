## 9. Reproducible Python Analysis
```python
import pandas as pd
import statsmodels.formula.api as smf
import numpy as np
import matplotlib.pyplot as plt

# Load data
data = """branch,month,automation,arrivals,fte,high_complexity_pct,automated_fast_path_pct,median_cycle_h,p90_cycle_h,rework_pct,escalation_pct,csat_1to5
Alpha,2026-01,0,1023,34,28.0,4,50.4,101.3,13.2,7.8,3.83
Alpha,2026-02,0,1056,34,28.0,4,46.9,94.7,13.2,7.8,3.89
... [full CSV data] ...
"""

df = pd.read_csv(pd.compat.StringIO(data))
df['month'] = pd.to_datetime(df['month'])
df['year_month'] = df['month'].dt.to_period('M')
df['adoption_month'] = df.groupby('branch')['month'].transform('min')  # First month of automation

# 1. Branch trends (no subplots)
for branch in df['branch'].unique():
    branch_df = df[df['branch'] == branch]
    plt.figure(figsize=(10, 4))
    plt.plot(branch_df['year_month'].astype(str), branch_df['median_cycle_h'], label='Median Cycle Time')
    plt.plot(branch_df['year_month'].astype(str), branch_df['p90_cycle_h'], label='P90 Cycle Time')
    plt.axvline(branch_df[branch_df['automation'] == 1]['month'].min(), color='red', linestyle='--', label='Adoption')
    plt.title(f"{branch} - Cycle Time Trends")
    plt.legend()
    plt.show()

# 2. Adopter pre/post contrasts (3-month windows)
def get_pre_post(branch, adoption_month):
    pre = df[(df['branch'] == branch) & (df['month'] < adoption_month)].tail(3)
    post = df[(df['branch'] == branch) & (df['month'] >= adoption_month)].head(3)
    return pre, post

for branch in ['Alpha', 'Beta', 'Gamma']:
    adoption_month = df[df['branch'] == branch]['month'].min()
    pre, post = get_pre_post(branch, adoption_month)
    print(f"\n{branch} Pre/Post (3-month avg):")
    print(f"Pre median_cycle_h: {pre['median_cycle_h'].mean():.1f}h  Post: {post['median_cycle_h'].mean():.1f}h (: {post['median_cycle_h'].mean() - pre['median_cycle_h'].mean():.1f}h)")

# 3. Relative time since adoption
df['relative_time'] = (df['month'] - df['adoption_month']).dt.days
df['post_adoption'] = (df['relative_time'] >= 0).astype(int)

# 4. Event-study model (branch-clustered)
model = smf.ols("median_cycle_h ~ post_adoption + high_complexity_pct + arrivals + FTE",
                data=df).fit(cov_type='cluster', groups=df['branch'])
print("\nEvent-study coefficients:")
print(model.summary())

# 5. Treatment-intensity model
model_intensity = smf.ols("median_cycle_h ~ automated_fast_path_pct + high_complexity_pct + arrivals + FTE",
                          data=df).fit(cov_type='cluster', groups=df['branch'])
print("\nTreatment-intensity coefficients:")
print(model_intensity.summary())

# Warnings
print("\nWARNINGS:")
print("- Only 5 branches; cluster-robust SEs may underestimate uncertainty.")
print("- Median cycle time hides case-level heterogeneity; p90_cycle_h should be monitored separately.")
print("- CSAT response rates unobserved; effects may not generalize to non-respondents.")
```
