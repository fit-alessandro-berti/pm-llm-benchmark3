        # Filter for specific resource
        sub_log = pm4py.filter_event_attribute_values(log,

                                                      parameter='resource')

        complexity_stats[res] = {
            'avg_complexity': avg_complexity,
            'substitution_episodes': substitution_count,
            'similarity_score': sim_matrix.get(res, {})
        }
        
    # 6. Avoid Over-Interpreting Frequency Clusters

        if total_events > 500 and high_value_count < 20:
            high_freq_low_auth.append(res)
            
    return roles, sim_matrix, handover_net, complexity_stats, high_freq_low_auth
