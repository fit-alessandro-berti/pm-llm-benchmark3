```python
def bounded_trace_exploration(net, im, fm, max_loops=3):
    """Explores traces up to `max_loops` iterations, hiding silent transitions."""
    traces = []
    current_marking = im.copy()
    for _ in range(max_loops):
        enabled = pm4py.get_enabled_transitions(net, current_marking)
        if not enabled:
            break
        for transition in enabled:
            if transition.label is None:  # Skip silent
                continue
            new_marking = pm4py.get_next_marking(net, current_marking, transition)
            if new_marking == fm:  # Final state
                traces.append([t.label for t in enabled])
            current_marking = new_marking
    return traces

# Compare original POWL  Petri net with round-tripped nets
original_traces = bounded_trace_exploration(net_from_powl, im_from_powl, fm_from_powl)
bpmn_traces = bounded_trace_exploration(bpmn_net, im_from_powl, fm_from_powl)

print(f"Original traces: {original_traces}")
print(f"BPMN traces: {bpmn_traces}")
assert original_traces == bpmn_traces, "Traces differ after round-trip"
```
