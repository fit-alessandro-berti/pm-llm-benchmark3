# 4. Network and conformance analyses

The following PM4Py workflow isolates true control exceptions from known data-quality and emergency effects before measuring social-network concentration and timing anomalies:

```python
import pandas as pd
import pm4py

# 1. Format dataframe for PM4Py
df = pd.read_csv("p2p_log.csv")
df = pm4py.format_dataframe(
    df,
    case_id="case:concept:name",
    activity_key="concept:name",
    timestamp_key="time:timestamp",
)
log = pm4py.convert_to_event_log(df)

# 2. Isolate source-system data-quality effects (legacy_erp & quarter_close_upload & manual_upload)
log_clean_sys = pm4py.filter_event_attribute_values(
    log, "source_system", ["legacy_erp", "quarter_close_upload"], level="case", retain=False
)
log_manual = pm4py.filter_event_attribute_values(
    log, "source_system", ["manual_upload"], level="case", retain=True
)
log_legacy = pm4py.filter_event_attribute_values(
    log, "source_system", ["legacy_erp"], level="case", retain=True
)

# 3. Split emergency vs. non-emergency cases
log_emerg = pm4py.filter_event_attribute_values(
    log_clean_sys, "emergency_flag", [True, "True"], level="case", retain=True
)
log_non_emerg = pm4py.filter_event_attribute_values(
    log_clean_sys, "emergency_flag", [True, "True"], level="case", retain=False
)

# 4. Four-eyes principle checks (run separately on clean non-emergency, emergency, and full logs)
# Check A: Create Purchase Order vs. Approve Purchase Order
soe_po_violations = pm4py.filter_four_eyes_principle(
    log_non_emerg, "Create Purchase Order", "Approve Purchase Order"
)
# Check B: Change Vendor Bank Details vs. Payment Approval
soe_bank_violations = pm4py.filter_four_eyes_principle(
    log, "Change Vendor Bank Details", "Payment Approval"
)
# Check C: Emergency Override vs. Retrospective Justification
soe_emerg_violations = pm4py.filter_four_eyes_principle(
    log_clean_sys, "Emergency Override", "Retrospective Justification"
)

# 5. Organizational network discovery (compare clean non-emergency vs. emergency vs. manual_upload)
handover_net = pm4py.discover_handover_of_work_network(log_clean_sys)
working_together_net = pm4py.discover_working_together_network(log_clean_sys)
subcontracting_net = pm4py.discover_subcontracting_network(log_clean_sys) # n=2 loops

# Multi-dimensional network analysis (e.g., resource-to-resource conditioned on vendor_id or source_system)
net_analysis = pm4py.discover_network_analysis(
    log_manual,
    out_column="org:resource",
    in_column="org:resource",
    node_column_source="org:resource",
    node_column_target="org:resource",
    edge_column="vendor_id",
)

# 6. Path performance and service/waiting-time analysis
# Isolate fast transitions between Change Vendor Bank Details -> Payment Approval (e.g., <= 30 min = 1800s)
fast_bank_to_pay = pm4py.filter_paths_performance(
    log, ("Change Vendor Bank Details", "Payment Approval"), 0, 1800
)
# Isolate fast PO approvals (<= 15 min = 900s) outside emergency cases
fast_non_emerg_po = pm4py.filter_paths_performance(
    log_non_emerg, ("Create Purchase Order", "Approve Purchase Order"), 0, 900
)
# Inspect sub-traces between key activities and compute waiting/service times
between_bank_pay = pm4py.filter_between(log, "Change Vendor Bank Details", "Payment Approval")
variants_between = pm4py.get_variants(between_bank_pay)
service_times = pm4py.get_service_time(log_emerg)
df_with_times = pm4py.insert_case_service_waiting_time(df)

# 7. Declarative / Log-Skeleton check for Emergency Override -> Retrospective Justification
# Build a normative Log Skeleton or DECLARE model on compliant emergency cases (within 3 working days)
compliant_emerg_cases = pm4py.filter_paths_performance(
    log_emerg, ("Emergency Override", "Retrospective Justification"), 0, 3 * 86400
)
declare_model = pm4py.discover_declare(compliant_emerg_cases)
declare_conf = pm4py.conformance_declare(log_emerg, declare_model)

# Alternatively, discover a log skeleton from compliant emergency cases to check precedence/response
skeleton_model = pm4py.discover_log_skeleton(compliant_emerg_cases)

```
