```python
class ProcessMiningAdapter:
    """Encapsulates every PM4Py interaction. The simulation engine below never
    imports pm4py: it consumes plain parameter dictionaries produced here."""

    def __init__(self, df: pd.DataFrame):
        self.df = pm4py.format_dataframe(
            df,
            case_id="case:concept:name",
            activity_key="concept:name",
            timestamp_key="time:timestamp",
        )
        self.log = pm4py.convert_to_event_log(self.df)

    # -- process model ------------------------------------------------------
    def parse_reference_model(self):
        """Parse the simplified process tree.

        Limitations (documented, not hidden): the tree has no data conditions
        (requires_* flags), no appeal loop, treats Legal check as a free XOR,
        serialises Public notice before inspection although the log shows
        partial overlap, and knows nothing about resources, calendars, or
        batching. It is used for structural reference and conformance checks
        of simulated logs; runtime routing is data-driven (RoutingEngine).
        """
        tree = pm4py.parse_process_tree(PROCESS_TREE_STR)
        net, im, fm = pm4py.convert_to_petri_net(tree)
        return tree, net, im, fm

    # -- structure / behaviour statistics ------------------------------------
    def variants(self) -> pd.DataFrame:
        seq = (
            self.df.sort_values(["case:concept:name", "time:timestamp"])
            .groupby("case:concept:name")["concept:name"]
            .agg(" > ".join)
        )
        return seq.value_counts().rename("cases").to_frame()

    def dfg(self):
        """Directly-follows graph + start/end activities."""
        dfg, start_acts, end_acts = pm4py.discover_dfg(self.log)
        return dfg, start_acts, end_acts

    def performance_dfg(self) -> Dict[Tuple[str, str], Dict[str, float]]:
        """Edge (a,b) -> timing statistics. NOTE: edges measure
        completion-to-completion time = waiting(b) + service(b): the split is
        NOT observable from completion-only timestamps (see design doc §2.3)."""
        return pm4py.discover_performance_dfg(self.log)

    # -- arrivals -------------------------------------------------------------
    def arrival_statistics(self) -> Dict[str, float]:
        starts = self.df.groupby("case:concept:name")["time:timestamp"].min().sort_values()
        iat_days = starts.diff().dropna().dt.total_seconds() / 86400.0
        return {
            "n_cases": int(len(starts)),
            "mean_interarrival_days": float(iat_days.mean()) if len(iat_days) else float("nan"),
            "implied_cases_per_week": float(len(starts) / max(1e-9, (starts.max() - starts.min()).days / 7)),
            "arrivals_by_weekday": starts.dt.day_name().value_counts().to_dict(),
            "arrivals_by_month": starts.dt.month.value_counts().sort_index().to_dict(),
        }

    # -- timing columns --------------------------------------------------------
    def with_timing_columns(self) -> pd.DataFrame:
        """Sojourn-based timing columns (completion-to-completion).

        Because the log records completions only, 'sojourn_since_prev' is
        wait + service. A genuine split requires start timestamps or the
        calibration procedure in §2.3; `has_start_ts` marks where a split
        would be directly computable.
        """
        df = self.df.sort_values(["case:concept:name", "time:timestamp"]).copy()
        g = df.groupby("case:concept:name")
        df["prev_activity"] = g["concept:name"].shift(1)
        df["prev_timestamp"] = g["time:timestamp"].shift(1)
        df["sojourn_since_prev_h"] = (
            df["time:timestamp"] - df["prev_timestamp"]
        ).dt.total_seconds() / 3600.0
        df["case_elapsed_days"] = (
            df["time:timestamp"] - g["time:timestamp"].transform("min")
        ).dt.total_seconds() / 86400.0
        df["has_start_ts"] = False  # would be True for lifecycle:start events
        return df

    def activity_timing_summary(self) -> pd.DataFrame:
        df = self.with_timing_columns()
        return (
            df.groupby("concept:name")["sojourn_since_prev_h"]
            .agg(["count", "median", lambda s: s.quantile(0.9)])
            .rename(columns={"<lambda_0>": "p90"})
        )
```

## 1. Model scope and state variables

**Boundary.** The twin covers one case end-to-end: portal submission  final `Issue permit` / `Reject application`, including rework loops (planning, completeness, inspection, legal) and the applicant-appeal re-entry. It excludes pre-submission applicant design work, post-decision construction oversight, and the emergency fast track (which was deleted from the log — treated only as a demand-surge risk in stress scenarios). The twin runs in two modes:

- **Historical replay mode** (validation): trace-driven — the empirical arrival timestamps and case attributes from the log are replayed through the resource model.
- **Stochastic forward mode** (experiments): fitted arrival process, intervention switches on, used for counterfactual comparisons.

**Level of detail.** Activity-level discrete-event simulation. Two clocks are kept explicitly: *calendar time* (SLA clocks, the 14-day public-notice minimum, applicant waits) and *work time* (service consumes resource-calendar time only). Documents are not modeled individually (the missing-documents loop is one abstraction); geography is at district granularity; resources are modeled individually with pool calendars and skill restrictions.

**State variables.**

| Object | State variables | Notes |
|---|---|---|
| Case | Static: case type, district, architect id, application value, `requires_fire/env/notice`, inferred `requires_legal`, prior-violations flag (partially missing) | Dynamic: current stage; set of open parallel branches {planning, fire, env, notice, inspection}; missing-doc loop counter; rework counters per stage; pending applicant response + deadline; inspection-ready flag; bundle id; age/priority; SLA deadline; latent objection flag |
| Resource pool / individual | Per resource: busy/idle, current task, remaining service; per pool: priority queue (urgent / aging-above-threshold / normal), utilization accumulators | Pools: clerks (5), planners (6 general + 2 simple-only), fire reviewers (2 FTE-equiv, Tue–Thu), environmental (3 FTE × 60%), inspectors (4, Tue–Fri, batchable), legal (2, council-week blocks), schedulers, system (unlimited) |
| Inspection bundle | District buffer: accumulated ready cases, oldest-wait timer, architect grouping | D3 mechanism: closes on size N or max wait T; travel = base + per-stop increment with D3/D5 multipliers |
| Objection event (latent) | District, active window, extension distribution | Never observed; affects all notices open in the district during the window (cross-case correlation) |
| Global | Clock; exception calendar (holidays, council weeks); seasonal arrival state (t); active architect bursts; WIP/backlog history; intervention configuration; random-number streams (for common random numbers) | |

**Cross-case dependency mechanisms (why this is not an independent-case simulation):** (1) shared-resource queues couple all cases in a pool; (2) architect burst arrivals (top-20 architects = 31% of volume submit related applications in the same week); (3) district/architect inspection bundling shares travel time but delays the first case in a bundle; (4) district-scoped objection windows lengthen all overlapping public notices; (5) the Monday portal burst collides with reduced clerk capacity (one clerk on portal support Mondays).

**Explicit non-goals:** document content, inspector learning curves, applicant strategic behavior changes (except as scenario parameters), and legal-precedent effects between cases.

## 2. Parameter-estimation plan

**2.0 PM4Py pipeline.** The full 18-month CSV is loaded with `pandas.read_csv` and passed through `pm4py.format_dataframe(df, case_id="case:concept:name", activity_key="concept:name", timestamp_key="time:timestamp")`, then `pm4py.convert_to_event_log(df)`. On the event log we use: `pm4py.discover_dfg` (structure + **concurrency detection**: activity pairs with edges in both directions, e.g. Planning  Fire, are modeled as parallel branches — this corrects the tree's serialization); `pm4py.discover_performance_dfg` (completion-to-completion edge timings, used as sojourn statistics and as calibration targets); `pm4py.parse_process_tree` + `pm4py.convert_to_petri_net` for the reference model used in conformance checks of *simulated* logs (token replay / alignments). A log-discovered model via `pm4py.discover_petri_net_inductive` (with rare-variant filtering) is compared against the supplied tree as a cross-check. Attribute-conditional statistics (by case type, district, architect) are computed on the flattened DataFrame with pandas *before* conversion, because they need case attributes. PM4Py's Petri-net playout is used only as a structural sanity check — it cannot represent resources, calendars, batching, or non-stationary arrivals, so it is **not** the twin's engine.

**2.1 Arrivals.** Per case, take the first event as submission. Estimate a multiplicative rate model (month, weekday) on daily counts via a Poisson/negative-binomial GLM (log link: `log E[count] =  + month + weekday`); test the month×weekday interaction — if significant, use the full weekday×month rate table instead of the multiplicative form. Check overdispersion and weekly-count autocorrelation. Overlay the **architect-burst component**: from `architect_id` + submission weeks, estimate the empirical distribution of submissions per architect-week for the top-20 architects and the intra-burst time offsets; simulate large-architect volume as a cluster process, the remainder as the NHPP (thinning algorithm). Hour-of-day submission distribution is empirical. The arrival model is validated against the weekly count distribution (variance/mean ratio, ACF), not just the mean.

**2.2 Routing and branching.** Conditional probabilities estimated directly from the log per (case type, district, flags): P(fire | flags, type, district), P(environmental | flags, type, district), P(public notice | type), P(legal check | type), P(missing docs | type), P(reject | type), P(appeal | type). Where the decision rule references data that is incomplete (building-area threshold for fire, environmental sensitivity of districts), the residual gap is covered by an empirical conditional probability and flagged as uncertain. The concurrency structure (which review branches run in parallel, whether the notice window overlaps reviews) is taken from the DFG bidirectional-edge test, not from the tree. Variant-group shares from the log are the acceptance target for the routing model.

**2.3 Service vs. waiting — the critical estimation problem.** Many human activities record **only completion timestamps**, so every performance-DFG edge and every sojourn is `waiting + service` and the provided split is "partly inferred". Worse, a throughput-feasibility check shows the provided service medians are **incompatible with the calendars**:

| Pool | Nominal capacity | Demand implied by summary medians (400 cases/wk) | Implied  |
|---|---|---|---|
| Clerks (5 FTE) | 200 h/wk | 480 checks × 25 min  200 h (Mondays lose 1 clerk) | 1.0 — edge of feasibility |
| Planners (8 FTE) | 320 h/wk | planning 400×3.5 h + drafting 400×2 h  2,200 h | 7 — infeasible |
| Fire (2 FTE, Tue–Thu) | 48 h/wk | 180 reviews × 2.5 h  450 h | 9 — infeasible |
| Environmental (3 × 60%) | 72 h/wk | 70 × 4 h  280 h | 4 — infeasible |
| Inspectors (4, Tue–Fri) | 128 h/wk | 360 × (1 h + travel)  540 h | 4 — infeasible |
| Legal (2, council weeks) | 70 h/wk | 90 × 3 h  270 h | 4 — infeasible |

Since 31,200 cases were actually completed, at least one of {service split, FTE counts, task ownership} is wrong for most pools. The twin therefore treats the **split as a latent parameter** with hard feasibility bounds (implied utilization   0.85–0.9, because queues exist but did not explode), and resolves it by **simulation-based calibration**: choose per-activity service distributions (lognormal , ) such that the *simulated* waiting medians/p90 reproduce the *empirical* waiting summaries, while throughput stays feasible. Candidate resolutions to verify with HR/SMEs: waiting contaminating the "service" estimates (most likely — true touch times for reviews are plausibly 15–60 min), unlisted roles performing tasks, or automation. The resolution adopted is documented and carried as uncertainty in the parameter sets (§2.8). Waiting is **never** an input except for applicant/third-party steps; it is an emergent, validated output.

**2.4 Distributions and tails.** Lognormal MLE per activity × case type where cell sizes allow (31k cases  thousands per cell), pooled across types otherwise. Heavy-tail diagnostics (QQ plots, tail-index comparison against log-logistic/Weibull). Nonparametric fallback (empirical resampling) for the heaviest: applicant missing-documents response (median 5 d, heavy tail), public-notice extensions, and district travel times. Public notice = 14-day calendar minimum + max(0, extension), where the extension is a **mixture** (point mass at 0 = no objection; heavy tail = objection) fit by EM on observed notice durations — explicitly a proxy, never treated as observed objection data.

**2.5 Rework.** Loop-back probabilities per stage (completeness, planning, inspection, legal, appeal) per case type, estimated from repeated-activity patterns per case. Rework-pass service times are estimated separately where sample size allows (they may differ from first-pass). Rework reasons are unstructured text and not reliably coded, so rework propensity is modeled with covariates where identifiable and otherwise as an uncertain rate with sensitivity bands.

**2.6 Calendars.** HR calendars encoded as weekly patterns + dated exceptions (holidays, council meeting weeks). Each calendar is **empirically falsified against the log**: e.g., if Legal-check completion timestamps appear during council weeks, the "no work during council weeks" claim is wrong and must be reconciled. The Monday portal-support clerk is modeled as reduced clerk capacity on Mondays; the fire pool's Tue–Thu pattern and inspectors' Tue–Fri field days are validated against timestamp weekday distributions.

**2.7 Cross-case dependencies.** Bursts: §2.1. Inspection bundling: co-occurrence of same-district inspections in the same inspector-day in the log calibrates current batching tendency; D3's travel-saving factor is an **assumed scenario parameter** (unobservable directly). Objection clusters: district-level event rate and extension distribution from the §2.4 mixture; because the log is flattened by case, dependencies are reconstructed from `architect_id`/`district` joins rather than assumed absent.

**2.8 Missing data  parameter sets and the calibration loop.** Uncertain quantities (service/waiting splits, objection incidence and extension, burst intensity, residual routing gaps, D-intervention effectiveness) are drawn into K  10–16 **parameter sets** (bounded imputations around best estimates). A closed-loop calibration runs per set: simulate baseline  compare activity waiting medians/p90, per-type case durations, variant-group shares  adjust latent splits  iterate (Nelder–Mead / grid over ~10–20 latent parameters) until tolerances in §6 are met. All experiments are replicated across all K sets, so reported effects include epistemic uncertainty, not just seed noise.

## 3. Assumptions

- **Arrivals.** Base arrivals are an NHPP with multiplicative month × weekday factors; bursts are a cluster overlay for large architects (empirical sizes/offsets). Case-type mix is stationary over the horizon. *Risk:* interventions could change applicant behavior (e.g., D1 deterring marginal applicants) — handled only as explicit sensitivity scenarios, not endogenous behavior.
- **Service times.** Lognormal per activity × type; small true touch times for reviews per the §2.3 feasibility argument; travel add-ons per district (D3/D5 higher); batching reduces marginal travel per stop. *Risk:* the split is latent; carried in K parameter sets.
- **Routing.** Data-conditional probabilities estimated empirically; concurrency from DFG, overriding the tree's serialization; Decision drafting waits for all applicable results (AND-join); Legal check is a probabilistic XOR by case type.
- **Resources/calendars.** HR-reconstructed calendars are correct *after* log-based falsification (§2.6); skills restricted as stated (2 planners simple-only); no multitasking; no preemption; system tasks unlimited.
- **Rework.** Loop-back probabilities stationary unless an intervention explicitly changes them (e.g., D1 multiplies missing-doc probability by 0.3–0.7; D2's misclassified skips raise downstream rework/appeal). Rework consumes resources normally.
- **Applicant/third-party response.** Exogenous, heavy-tailed, independent of municipal load; modeled separately from municipal service times (never resource-constrained); not affected by interventions except via explicit scenario multipliers.
- **Cross-case dependencies.** Only the four mechanisms of §1 (queues, bursts, bundles, objection windows). No other interference.
- **Non-stationary demand.** The 18-month seasonal pattern and Monday effect persist into the horizon; stress scenarios S1/S4 perturb them. Long-run demand drift is not modeled (flagged as scenario, not forecast).
- **Missing data.** Objections are latent (proxy via notice-duration mixture; wide sensitivity — never treated as observed). Start-timestamp gaps  latent split (§2.3). Removed emergency cases  duration tails slightly conservative. Flattened log  dependency parameters from attribute joins. Incomplete routing inputs (building area, environmental sensitivity, violation history)  empirical residual probabilities and eligibility-share uncertainty for D2.

## 5. Warm-up, replications, and uncertainty treatment

- **Warm-up.** Arrivals are seasonal, so there is no steady state; each replication is a **terminating run**. In stochastic mode the system is *primed* by simulating 8–12 weeks before the measurement start (or, alternatively, initializing with the empirical in-flight backlog at a comparable calendar date — e.g., a February snapshot — with stage distribution taken from the log); the priming period is discarded. Welch moving-average plots of WIP and flow time across replications confirm transient length; sensitivity of conclusions to priming length (4/8/12 weeks) and to empty-vs-primed initialization is reported. Historical replay mode starts from the actual backlog, so warm-up bias is minimal there.
- **Run length.** 24 months per replication (two full seasonal cycles, including two peak seasons and repeated council/holiday calendars), after priming; measurement from month 0 (post-priming) to month 24. Validation replay uses the 18-month log window.
- **Replications and seeds.** Seeds via `numpy.random.SeedSequence(base).spawn(...)`; **common random numbers** across baseline/intervention pairs (same seed per replication index) enable paired differences; note CRN decorrelates partially when interventions change routing, but pairing still reduces comparison variance. Sequential stopping rule: begin with 30 replications per parameter set; add replications until the 95% CI half-width is 10% of the mean for overall median duration, 15% for p90 by case type; cap 150. Budget: K(12) parameter sets × scenarios × replications — run embarrassingly parallel.
- **Uncertainty layers, reported separately and pooled.** (1) *Aleatory*: replication spread  percentile bootstrap CIs (BCa) on KPIs. (2) *Epistemic*: spread across the K parameter sets  reported as ranges / mixture intervals; conclusions must be stable across sets to be actionable. (3) *Structural*: alternative concurrency models (fully serial vs. parallel reviews) and batching policies as model-variant sensitivity runs. The headline comparative statistic is **P(intervention beats baseline)** over all (seed, parameter set) pairs, plus paired median deltas with intervals.
- **Heavy tails.** Quantile-based KPIs (median, p90, p95) rather than means; per-replication quantiles then bootstrap; empirical-resampling or log-logistic fits for the heaviest distributions (applicant response, notice extensions) so fitted lognormals do not truncate the tail; no winsorization of simulated data for decisions (visualization only). Tail KPIs get more replications because their estimator variance is higher — this is checked in the stopping rule.
- **Sensitivity analysis.** One-at-a-time tornado over the top ~10 uncertain parameters (service split fractions, objection incidence/extension, burst intensity, D-effectiveness levels); global sensitivity (Latin hypercube / Sobol on a metamodel if runtime is limiting) to identify which parameters drive the *ranking* of interventions, not just the KPI levels.

## 6. Validation strategy

Validation gates must pass **before** any experiment in §4 is believed; results are conditional on these gates.

1. **Face validation.** Workshops with permit officers, planners, inspectors, legal: review 20+ animated simulated traces (per case type and district), the bottleneck map, and calendar representations; explicitly sign off on the Monday-portal-support rule, council-week blocking, and the §2.3 capacity reconciliation.
2. **Event-log replay validation.** Historical mode: replay the empirical arrivals and attributes of the 18-month window; compare per-case simulated vs. actual durations (correlation  0.6; MAE  20% of the median) and per-activity waiting distributions. Additionally, run the *simulated* log through the same PM4Py pipeline (same `ProcessMiningAdapter`) and check conformance of simulated traces against the reference Petri net (token replay; alignments on a sample) — simulated behavior must be at least as fitting as the real log, and variant-group shares must match (±3 pp).
3. **Distributional validation.** QQ plots and quantile tolerances (not bare KS tests — with 31k cases they reject trivially): median case duration by type within ±10%, p90 within ±15–20%; variant-group shares within ±3 pp; DFG edge-frequency rank correlation (Spearman  0.9); performance-DFG edge timings within tolerance; waiting medians/p90 per activity (the calibration targets) within ±10%/±20%.
4. **Resource-utilization validation.** Compare simulated pool utilization against log-inferred busy proxies (inferred service minutes per resource-week vs. available minutes). Enforce the **capacity-consistency check** of §2.3: implied utilization per pool must be <0.9 and match the log period's throughput; if calibration cannot satisfy this, the staffing/split reconciliation is reopened with HR rather than silently absorbed. Also verify simulated timestamp weekday patterns reproduce calendar structure (no legal completions in council weeks, fire reviews Tue–Thu, etc.).
5. **Bottleneck validation.** Rank activities by average waiting and mean queue length; the simulated top-3 bottlenecks must match the empirically known pain points (legal queue, inspection waiting, planning queue after Monday spikes) in 2 of 3 positions; queue-length time series should reproduce the post-Monday completeness spike and seasonal backlog waves.
6. **Out-of-sample validation.** Temporal split: calibrate on months 1–12, predict months 13–18 (chosen so the holdout contains a peak season) using only actual arrivals — KPI predictions must land within the §6.3 tolerances. Leave-one-district-out tests for the travel/queue models. Parameter sets are frozen before prediction. Where a partial natural experiment exists in the log (e.g., a past period with an extra inspector), backtest against it.

Deviations are documented and become explicit model limitations attached to any recommendation; the twin is recalibrated quarterly as new log data arrives.

| Resource utilization | Mean and p95-weekly utilization per pool; implied stability | Mean within 50–85%; p95-weekly 90% (above this, queues grow super-linearly); no sustained backlog growth |
| Rework | Missing-doc loop share; inspection rework; appeal rate | No stage worsens >10% unless net duration benefit is large (e.g., D2/D8 risk) |
| Backlog | Peak WIP, end-of-horizon open cases, post-peak recovery time | Peak WIP not above baseline; recovery 6 weeks after peak |
| Robustness | P(beats baseline) over (seed × parameter set); performance under S1–S3; worst-case regression | P 0.8; improvement holds in 2 of 3 stress scenarios; worst-case regression 5% |
| Cost / qualitative gates | Cost per case-day saved (D4 temp staff); applicant effort for D1; legal risk sign-off for D2 | Reported; qualitative veto possible |

**Decision procedure.** (1) Guardrails first: any intervention violating utilization, fairness, or worst-case-regression thresholds is dropped regardless of mean gains. (2) Rank surviving candidates on primary KPIs with interval estimates; inspect the Pareto front (duration vs. utilization vs. fairness). (3) Require robustness across parameter sets and stress scenarios. (4) Recommend a **phased pilot**: staggered rollout in 1–2 districts as a live A/B comparison, with the twin recalibrated on pilot data before city-wide adoption. Explicitly: the twin produces **rankings under uncertainty conditional on the validation gates**, not optimal solutions — no optimality claim is made from an uncalibrated or unvalidated model.

PROCESS_TREE_STR = (
    "->( 'Submit application',"
    " *( ->('Completeness check', X(tau, ->('Request missing documents','Receive missing documents'))),"
    "    'Completeness rework' ),"
    " +('Planning review', X(tau,'Fire-safety review'), X(tau,'Environmental review')),"
    " X(tau,'Public-notice period'),"
    " *( ->('Inspection scheduling','Site inspection'), 'Inspection rework' ),"
    " 'Decision drafting',"
    " X(tau,'Legal check'),"
    " X('Issue permit','Reject application') )"
)


def _p90(s: pd.Series) -> float:
    return s.quantile(0.90)


class ProcessMiningAdapter:
    """Encapsulates every pm4py interaction. The simulation engine below never
    imports pm4py; it consumes parameter dictionaries produced/calibrated here."""

    def __init__(self, df: pd.DataFrame):
        self.df = pm4py.format_dataframe(
            df, case_id="case:concept:name", activity_key="concept:name",
            timestamp_key="time:timestamp")
        self.log = pm4py.convert_to_event_log(self.df)

    # -- reference process model ------------------------------------------
    def parse_reference_model(self):
        """Parse the simplified process tree and convert it to a Petri net.

        LIMITATIONS of the tree (by design, documented not hidden): it has no
        data conditions (requires_* flags), no probabilities on the XORs, no
        appeal loop, treats 'Legal check' as a free choice although it is
        type-dependent, serialises Public notice before Inspection although
        the log shows partial overlap, and knows nothing about resources,
        calendars, or batching. It is used for structural reference and for
        conformance-checking simulated logs; runtime routing is the
        data-driven RoutingEngine. Actual concurrency is taken from the DFG
        (activity pairs with edges in BOTH directions are parallel).
        Optionally: pm4py.discover_petri_net_inductive(self.log) to compare a
        log-discovered model against this hand-supplied tree.
        """
        tree = pm4py.parse_process_tree(PROCESS_TREE_STR)
        net, im, fm = pm4py.convert_to_petri_net(tree)
        return tree, net, im, fm

    # -- structure / behaviour --------------------------------------------
    def variants(self) -> pd.DataFrame:
        seq = (self.df.sort_values(["case:concept:name", "time:timestamp"])
               .groupby("case:concept:name")["concept:name"].agg(" > ".join))
        return seq.value_counts().rename("cases").to_frame()

    def dfg(self):
        """Directly-follows graph + start/end activities. Bidirectional edges
        (e.g. Planning <-> Fire) reveal concurrency to correct the tree."""
        dfg, start_acts, end_acts = pm4py.discover_dfg(self.log)
        return dfg, start_acts, end_acts

    def performance_dfg(self) -> dict:
        """Edge (a,b) -> timing statistics. NOTE: with completion-only
        timestamps each edge measures waiting(b) + service(b); the split is
        NOT observable here (design doc §2.3 — calibrated, not assumed)."""
        return pm4py.discover_performance_dfg(self.log)

    # -- arrivals -----------------------------------------------------------
    def arrival_statistics(self) -> Dict[str, object]:
        starts = (self.df.groupby("case:concept:name")["time:timestamp"]
                  .min().sort_values())
        iat_days = starts.diff().dropna().dt.total_seconds() / 86400.0
        return {
            "n_cases": int(len(starts)),
            "mean_interarrival_days": float(iat_days.mean()) if len(iat_days) else float("nan"),
            "implied_cases_per_week": float(len(starts) / max(1e-9, (starts.max() - starts.min()).days / 7)),
            "arrivals_by_weekday": starts.dt.day_name().value_counts().to_dict(),
            "arrivals_by_month": starts.dt.month.value_counts().sort_index().to_dict(),
        }

    # -- timing columns -------------------------------------------------------
    def with_timing_columns(self) -> pd.DataFrame:
        """Sojourn columns (completion-to-completion). Because the log records
        completions only, sojourn = wait + service; a genuine split needs
        start timestamps (lifecycle:start) or the §2.3 calibration."""
        df = self.df.sort_values(["case:concept:name", "time:timestamp"]).copy()
        g = df.groupby("case:concept:name")
        df["prev_activity"] = g["concept:name"].shift(1)
        df["prev_timestamp"] = g["time:timestamp"].shift(1)
        df["sojourn_since_prev_h"] = (df["time:timestamp"] - df["prev_timestamp"]).dt.total_seconds() / 3600.0
        df["case_elapsed_days"] = (df["time:timestamp"] - g["time:timestamp"].transform("min")).dt.total_seconds() / 86400.0
        df["has_start_ts"] = False  # True where lifecycle starts exist
        return df

    def activity_timing_summary(self) -> pd.DataFrame:
        return (self.with_timing_columns()
                .groupby("concept:name")["sojourn_since_prev_h"]
                .agg(["count", "median", _p90])
                .rename(columns={"median": "median_h", "_p90": "p90_h"}))

    @classmethod
    def from_simulated_events(cls, events: List[dict]) -> "ProcessMiningAdapter":
        """Validation symmetry: feed a simulated log (same schema) through the
        exact same analytics pipeline used for the real log."""
        return cls(pd.DataFrame(events))

**Notes on the skeleton.** (1) The excerpt contains only completion timestamps, so `sojourn_since_prev_h` is wait + service; the split is a calibrated latent parameter (§2.3), and `has_start_ts` marks where a direct split would exist. (2) `pm4py.read_csv` is deliberately not used — ingestion is pandas, then `pm4py.format_dataframe` with the three standard keys. (3) The tree is parsed for structural reference and conformance, but runtime routing, resources, calendars, batching, and cross-case effects live in the engine, because a process tree cannot express data conditions, probabilities, resources, or time. (4) The same `ProcessMiningAdapter` runs on simulated logs, giving a symmetric validation path: variants, DFG, performance DFG, and arrival statistics are compared between real and simulated logs exactly as in §6.
