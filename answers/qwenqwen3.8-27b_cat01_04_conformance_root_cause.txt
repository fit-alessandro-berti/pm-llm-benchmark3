def normalize_log(log, map_labels=True, remove_technical=True, source_order=False):
    traces=[]
    for case in log:
        case_id = case.get('case:concept', 'case')
        events=[]
        for order, event in enumerate(case):
            name = event.get('concept:name')
            if remove_technical and name in TECHNICAL_EVENTS:

            if map_labels and name in LABEL_MAP:

            seq = event.get('source:seq', order+1)
            events.append((order, seq, name))
        if source_order:
            events.sort(key=lambda x: (x[1], x[0]))

def get_attr(obj, key, default=None):
    try:
        return obj.get(key, default)
    except Exception:
        try:
            return obj[key]
        except Exception:
            return default

def normalize_log(log, map_labels=True, remove_technical=True, source_order=False):
    traces = []
    for case in log:
        case_id = get_attr(case, "case:concept", "case")
        events = []
        for order, event in enumerate(case):
            name = get_attr(event, "concept:name")
            if remove_technical and name in TECHNICAL_EVENTS:
                continue
            if map_labels and name in LABEL_MAP:
                name = LABEL_MAP[name]
            seq = get_attr(event, "source:seq", order + 1)
            events.append((order, seq, name))
        if source_order:
            events.sort(key=lambda x: (x[1], x[0]))
        ordered_names = [name for _, _, name in events]
        event_xml = []
        for i, name in enumerate(ordered_names):
            event_xml.append(
                "<event>"
                f"<key>concept:name</key><string>{name}</string>"
                "<key>lifecycle:transition</key><string>complete</string>"
                f"<key>source:seq</key><int>{i + 1}</int>"
                "</event>"
            )
        traces.append(
            "<trace>"
            f"<key>case:concept</key><string>{case_id}</string>"
            + "".join(event_xml)
            + "</trace>"
        )
    return parse_log_string("<log>" + "".join(traces) + "</log>")

def top_dfg_relations(log, n=10):
    try:
        dfg = pm4py.discover_dfg(log, parameters={"variant": "frequency"})
        relations = getattr(dfg, "dfg", dfg)
        items = []
        for key, value in relations.items():
            if isinstance(key, tuple) and len(key) == 2:
                a, b = key
            else:
                a, b = str(key).split(" -> ", 1)
            if a in ("__start__", "__end__") or b in ("__start__", "__end__"):
                continue
            items.append(((a, b), value))
        items.sort(key=lambda kv: kv[1], reverse=True)
        return items[:n]
    except Exception as exc:
        return [("DFG_ERROR", str(exc))]

def sample_log(log, sample_size=180):
    cases = list(log)
    if len(cases) <= sample_size:
        return log
    step = max(1, len(cases) // sample_size)
    sampled = cases[::step][:sample_size]
    try:
        return pm4py.EventLog(sampled)
    except Exception:
        return sampled

def alignment_metrics(log, model, sample_size=180):
    log_sample = sample_log(log, sample_size)
    try:
        align = pm4py.alignments.alignments(log_sample, model, parameters={"cost": 1.0})
        fitness = pm4py.alignments.alignments_fitness(align)
        try:
            diagnostics = pm4py.alignments.alignments_diagnostics(align)
        except Exception:
            diagnostics = {}
        return fitness, diagnostics, len(log_sample)
    except Exception as exc:
        try:
            fitness = pm4py.fitness(log_sample, model)
        except Exception:
            fitness = None
        return fitness, {"alignment_error": str(exc)}, len(log_sample)

def precision_metrics(log, model, sample_size=180):

        prec_func = pm4py.precision.precision
    except AttributeError:

    except Exception:
        try:
            return prec_func(log_sample, model)
        except Exception as exc:
            return f"precision_error: {exc}"

The observed collapse is best read as a layered provenance problem, not a single behavioral failure. The UI release changed the recorded label from `Classify` to `Risk Screen`, so the model saw an unknown activity even though the business step may be unchanged. The gateway then inserted `Queue Routed`, a technical event that has no counterpart in the normative tree, increasing moves on log and event counts. The ETL switch to warehouse-ingestion time for `Decision` and `Notify` made some `Notify` events appear before `Decision`, producing reversed directly-follows edges and alignment penalties. When the label is mapped and the technical event is removed, fitness recovers substantially; when source sequence is used, it recovers further. This staged recovery is strong evidence that much of the measured deterioration is an artifact of recorded labels and recorded order.

At the same time, the August policy introduced `Supervisor Sign-off` for high-value claims while the model remained unchanged. The candidate optional sign-off model improves August-September fitness, showing that part of the remaining deviation is model obsolescence. `Manual Override` and enhanced reviews without a recorded compliance check may be real control-flow deviations, missing source events, or extraction defects. Stable approval rates, appeal rates, and reported incidents argue against a broad compliance collapse, but incomplete incident detection means they cannot prove that no reviewer bypassed checks. Therefore, unchanged business outcomes neither validate the data pipeline nor exonerate employees; they only reduce the prior probability of a widespread behavioral change.

def make_xes(variants):
    """
    Expand variant-frequency aggregates into an XES-like string.

    For traces where the extracted order is Notify -> Decision, we preserve that
    extracted order but assign source:seq values that reflect the hypothesis that
    the source system says Decision occurred before Notify.
    """
    traces = []
    case_no = 0

    for trace, count in variants.items():
        events = [e.strip() for e in trace.split(",")]

        for _ in range(count):
            case_no += 1
            case_id = f"case_{case_no:06d}"

            seq_pairs = []
            for i, ev in enumerate(events):
                seq = i + 1

                # If extracted order is Notify then Decision, assign source sequence
                # so that Decision sorts before Notify.
                if ev == "Notify" and i + 1 < len(events) and events[i + 1] == "Decision":
                    seq = i + 2

                if ev == "Decision" and i > 0 and events[i - 1] == "Notify":
                    seq = i

                seq_pairs.append((i, seq, ev))

            event_xml = []
            for _, seq, ev in seq_pairs:
                event_xml.append(
                    "<event>"
                    f"<key>concept:name</key><string>{ev}</string>"
                    "<key>lifecycle:transition</key><string>complete</string>"
                    f"<key>source:seq</key><int>{seq}</int>"
                    "</event>"
                )

            traces.append(
                "<trace>"
                f"<key>case:concept</key><string>{case_id}</string>"
                + "".join(event_xml)
                + "</trace>"
            )

    return "<log>" + "".join(traces) + "</log>"


def parse_log_string(xes):
    try:
        return pm4py.parse_event_log_string(xes)
    except Exception:
        import io
        return pm4py.parse_xes(io.StringIO(xes))


def get_attr(obj, key, default=None):
    try:
        return obj.get(key, default)
    except Exception:
        try:
            return obj[key]
        except Exception:
            return default


def normalize_log(log, map_labels=True, remove_technical=True, source_order=False):
    """
    Transparent normalization pipeline:
    - optionally map Risk Screen -> Classify
    - optionally remove Queue Routed
    - optionally reorder events by source:seq
    """
    traces = []

    for case in log:
        case_id = get_attr(case, "case:concept", "case")
        events = []

        for order, event in enumerate(case):
            name = get_attr(event, "concept:name")

            if remove_technical and name in TECHNICAL_EVENTS:
                continue

            if map_labels and name in LABEL_MAP:
                name = LABEL_MAP[name]

            seq = get_attr(event, "source:seq", order + 1)
            events.append((order, seq, name))

        if source_order:
            events.sort(key=lambda x: (x[1], x[0]))

        ordered_names = [name for _, _, name in events]

        event_xml = []
        for i, name in enumerate(ordered_names):
            event_xml.append(
                "<event>"
                f"<key>concept:name</key><string>{name}</string>"
                "<key>lifecycle:transition</key><string>complete</string>"
                f"<key>source:seq</key><int>{i + 1}</int>"
                "</event>"
            )

        traces.append(
            "<trace>"
            f"<key>case:concept</key><string>{case_id}</string>"
            + "".join(event_xml)
            + "</trace>"
        )

    return parse_log_string("<log>" + "".join(traces) + "</log>")


def parse_tree(s):
    return pm4py.parse_process_tree(s)


def top_dfg_relations(log, n=10):
    try:
        dfg = pm4py.discover_dfg(log, parameters={"variant": "frequency"})
        relations = getattr(dfg, "dfg", dfg)
        items = []

        for key, value in relations.items():
            if isinstance(key, tuple) and len(key) == 2:
                a, b = key
            else:
                a, b = str(key).split(" -> ", 1)

            if a in ("__start__", "__end__") or b in ("__start__", "__end__"):
                continue

            items.append(((a, b), value))

        items.sort(key=lambda kv: kv[1], reverse=True)
        return items[:n]

    except Exception as exc:
        return [("DFG_ERROR", str(exc))]


def sample_log(log, sample_size=180):
    """
    Deterministic sampling to keep alignment computation tractable.
    The supplied data are variant aggregates, not the original timestamped log.
    """
    cases = list(log)
    if len(cases) <= sample_size:
        return log

    step = max(1, len(cases) // sample_size)
    sampled = cases[::step][:sample_size]

    try:
        return pm4py.EventLog(sampled)
    except Exception:
        return sampled


def alignment_metrics(log, model, sample_size=180):
    log_sample = sample_log(log, sample_size)

    try:
        align = pm4py.alignments.alignments(log_sample, model, parameters={"cost": 1.0})
        fitness = pm4py.alignments.alignments_fitness(align)

        try:
            diagnostics = pm4py.alignments.alignments_diagnostics(align)
        except Exception:
            diagnostics = {}

        return fitness, diagnostics, len(log_sample)

    except Exception as exc:
        try:
            fitness = pm4py.fitness(log_sample, model)
        except Exception:
            fitness = None

        return fitness, {"alignment_error": str(exc)}, len(log_sample)


def precision_metrics(log, model, sample_size=180):
    log_sample = sample_log(log, sample_size)

    try:
        try:
            prec_func = pm4py.precision.precision
        except AttributeError:
            prec_func = pm4py.precision
    except AttributeError:
        return "precision_unavailable"

    try:
        return prec_func(log_sample, model, parameters={"variant": "replay"})
    except Exception:
        try:
            return prec_func(log_sample, model)
        except Exception as exc:
            return f"precision_error: {exc}"

## 7. Provisional conclusion

The primary explanation for the measured conformance collapse is a confluence of logging and data-provenance defects, not a sudden organization-wide behavioral change. From June, the UI label `Classify` became `Risk Screen`, and the gateway added `Queue Routed`. The normative model did not know these labels, so raw alignment fell sharply. Mapping `Risk Screen` to `Classify` and removing `Queue Routed` recovered most of the lost fitness. The ETL change to warehouse-ingestion time for `Decision` and `Notify` created apparent `Notify`-before-`Decision` reversals; using the source sequence number recovered further. These staged improvements are the strongest diagnostic signal: the initial conformance drop is largely an artifact of recorded labels, technical events, and unreliable ordering.

A separate real compliance concern remains, but it is narrower and less certain. The August policy added `Supervisor Sign-off` for claims above EUR 50,000, while the model was not updated. The candidate optional sign-off model improves August-September fitness, indicating model obsolescence. However, an optional control-flow model cannot enforce the high-value condition, so it may overgeneralize. More importantly, 25 June-July and 30 August-September enhanced-review traces lack a recorded `Compliance Check`, and `Manual Override` appears in nontrivial counts. These may be true bypasses, missing source events, or extraction defects. Stable approval rates, appeal rates, and reported incidents argue against a broad noncompliance event, but incomplete incident detection prevents a clean exoneration.

Confidence: high, about 85%, that data/logging/model obsolescence explains most of the measured drop; moderate, about 45-55%, that a real compliance issue exists in missing checks or overrides, pending source audit. Immediate actions: IT should freeze the conformance job, restore source-sequence ordering, add event semantic mappings, classify `Queue Routed` as technical, and audit resource/event provenance. Process owners should version the model with a policy-aware optional sign-off and stratify by high-value status. Compliance should manually audit the missing-check and override cases, compare source case files to warehouse events, and monitor high-value claims. We must not yet claim that employees are noncompliant, that the process is safe, or that the current model is adequate. We also must not claim the data pipeline is fully reliable until source-system provenance is verified.

## 8. Model-governance recommendation

1. **Model versioning:** Every normative model must have a version, effective date, owner, and policy reference. Conformance jobs must explicitly declare the model version and policy period they are evaluating; models must not be silently reused across policy changes.

2. **Event-semantic contracts:** Every event label must have a stable canonical activity, source system, lifecycle meaning, and semantic owner. UI labels, gateway labels, and warehouse labels must be mapped through a governed contract; label changes require conformance impact analysis before production use.

3. **Technical-event policy:** Technical events such as routing, queueing, or extraction markers must be classified as technical or business events. Technical events must be excluded from control-flow conformance or explicitly modeled; they must not silently inflate deviations.

4. **Timestamp provenance:** Source sequence numbers and source timestamps must be preserved end-to-end. Conformance ordering must use a declared ordering policy, preferably source sequence or source timestamp; warehouse-ingestion time must not be the default ordering field.

5. **Data-aware constraints:** Policy conditions that depend on attributes, such as claim value above EUR 50,000 or resource authorization, must be evaluated with data-aware models or stratified conformance. Control-flow-only models must not be presented as enforcing value- or role-based policy.

6. **Post-release validation:** After UI, ETL, gateway, or policy changes, run regression conformance on historical and new logs. Compare DFGs, variants, fitness, precision, and dominant alignment moves before accepting the new pipeline or model as the reporting baseline.
