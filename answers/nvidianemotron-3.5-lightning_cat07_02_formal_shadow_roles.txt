#

df_formatted = pm4py.format_dataframe(

    case_id="case:concept:name",

)

# 2. Discover organisational roles
# ----------------------------------------------------------------------
# pm4py.discover_organizational_roles infers role assignments based on
# activity participation frequencies, caseattribute coupling, and resource
# overlap.  We feed the formatted log and indicate which case attributes
# should influence role shaping.
role_discovery = pm4py.discover_organizational_roles(
    df_formatted,
    case_id="case:concept:name",
    activity_key="concept:name",
    resource_key="org:resource",
    case_attributes=["case_complexity", "temporary_substitution"]
)
# role_discovery typically returns a roles_df with columns such as:
#   resource, proposed_role, core_activity_fraction, etc.

# ----------------------------------------------------------------------
# 3. Discover activitybased resource similarity
# ----------------------------------------------------------------------
# Computes pairwise similarity (cosine over normalized activity counts)
# between all resources in the log.
resource_similarity = pm4py.discover_activity_based_resource_similarity(
    df_formatted,
    activity_key="concept:name",
    resource_key="org:resource"
)
# resource_similarity is a square dataframe / dict with similarity scores;
# specific pairs can be extracted for inspection (e.g., the R03_R04 pair
# with the documented 0.94 caveat).

handover_network = pm4py.discover_handover_of_work_network(

Any staffing change motivated by substitution data should be piloted with controlled metrics (resolution time, rework rate, customer feedback) before organizationwide rollout.
