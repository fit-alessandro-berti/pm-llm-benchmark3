def tree_metrics(root):
    from pm4py.objects.process_tree.obj import Operator
    ops = leaves = visible = silent = 0
    dup = {}
    max_depth = 0
    def rec(n, d):
        nonlocal ops, leaves, visible, silent, max_depth
        max_depth = max(max_depth, d)
        if n.children:
            ops += 1
            for c in n.children:
                rec(c, d + 1)
        else:
            leaves += 1
            if n.label is None:
                silent += 1
            else:
                visible += 1
                dup[n.label] = dup.get(n.label, 0) + 1
    rec(root, 0)
    return {
        'operator_nodes': ops,
        'leaves': leaves,
        'visible_leaves': visible,
        'silent_leaves (tau)': silent,
        'total_nodes': ops + leaves,
        'max_depth_edges': max_depth,
        'duplicate_labels_in_tree': {k: v for k, v in dup.items() if v > 1},
    }

def tree_metrics(root):
    stats = {"operator_nodes": 0, "leaves": 0, "visible_leaves": 0,
             "silent_leaves": 0, "max_depth_edges": 0}
    labels = {}

    def rec(node, depth):
        stats["max_depth_edges"] = max(stats["max_depth_edges"], depth)
        if node.children:
            stats["operator_nodes"] += 1
            for c in node.children:
                rec(c, depth + 1)
        else:
            stats["leaves"] += 1
            if node.label is None:
                stats["silent_leaves"] += 1
            else:
                stats["visible_leaves"] += 1
                labels[node.label] = labels.get(node.label, 0) + 1

    rec(root, 0)
    stats["total_nodes"] = stats["operator_nodes"] + stats["leaves"]
    stats["duplicate_labels_in_tree"] = {k: v for k, v in labels.items() if v > 1}
    # labels repeated at RUNTIME come from the loop, not from duplicate leaves
    return stats

def net_metrics(tree):
    net, im, fm = pm4py.convert_to_petri_net(tree)
    m = {"places": len(net.places),
         "transitions": len(net.transitions),
         "silent_transitions": sum(1 for t in net.transitions if t.label is None),
         "visible_transitions": sum(1 for t in net.transitions if t.label is not None),
         "arcs": len(net.arcs)}
    try:
        m["simplicity"] = pm4py.simplicity_petri_net(net, im, fm)
    except Exception as e:
        m["simplicity"] = f"unavailable ({e})"
    return m

for name, tree in (("A", model_a()), ("B", model_b())):
    print(name, tree_metrics(tree), net_metrics(tree))
