def manual_dfg(sub_df):
    dfg = {}
    for trace in sub_df["trace"]:
        acts = [a.strip() for a in trace.split(">")]
        for pair in zip(acts, acts[1:]):
            dfg[pair] = dfg.get(pair, 0) + 1
    return dfg

def print_dfg(name, dfg):
    print(f"\n{name} DFG frequency:")
    for (a, b), v in sorted(dfg.items(), key=lambda kv: (-kv[1], kv[0])):
        print(f"  {a} -> {b}: {v}")

def compare_dfgs(dfg_a, dfg_b, name_a, name_b):
    set_a = set(dfg_a.keys())
    set_b = set(dfg_b.keys())
    print(f"\nRelations only in {name_a}:")
    for rel in sorted(set_a - set_b):
        print(f"  {rel[0]} -> {rel[1]}")
    print(f"\nRelations only in {name_b}:")
    for rel in sorted(set_b - set_a):
        print(f"  {rel[0]} -> {rel[1]}")
    print("\nRelations in both groups:")
    for rel in sorted(set_a & set_b):
        print(f"  {rel[0]} -> {rel[1]}: {name_a}={dfg_a[rel]}, {name_b}={dfg_b[rel]}")

def build_xes(sub_df):
    lines = [
        "<?xml version='1.0' encoding='UTF-8'?>",
        "<log xes.version='1.0'>",
        "<extension name='Concept' prefix='concept' uri='http://www.xes-standard.org/concept.xesext'/>",
    ]
    for _, row in sub_df.iterrows():
        lines.append("<trace>")
        lines.append(f"<string key='concept:name' value='{row['case_id']}'/>")
        lines.append(f"<string key='group' value='{row['group']}'/>")
        lines.append(f"<string key='complexity' value='{row['complexity']}'/>")
        for act in row["trace"].split(">"):
            lines.append("<event>")
            lines.append(f"<string key='concept:name' value='{act.strip()}'/>")
            lines.append("</event>")
        lines.append("</trace>")
    lines.append("</log>")
    return "\n".join(lines)

## 8. Residual uncertainty

1. Unobserved case severity within `Low`, `Medium`, and `High` may still confound group comparisons. Most valuable addition: a pre-routing granular risk score or initial claim-reserve/severity field.

2. The actual routing rule and eligible alternative groups are not observed. Most valuable addition: routing-decision logs containing rule version, features used, and counterfactual eligible groups.

3. `wait_before_review_h` mixes queueing, compliance waiting, and possibly other non-review delays. Most valuable addition: step-level timestamps with queue-entry/queue-exit and compliance start/end events.

4. Only completed cases are observed, so duration and quality estimates may be affected by censoring or abandonment. Most valuable addition: open, abandoned, and cancelled cases with status timestamps.

5. Group label may proxy reviewer skill, certification, or local work practice. Most valuable addition: reviewer IDs plus certification/tenure data, or a randomized routing pilot that separates group label from reviewer assignment.
