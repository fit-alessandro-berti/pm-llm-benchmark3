# 5. PM4Py role-discovery and similarity code

```python
import pandas as pd
import pm4py

# Load and format the dataframe
df = pd.read_csv("event_log.csv")  # Assume columns: case:concept:name, concept:name, time:timestamp, org:resource, etc.
df = pm4py.format_dataframe(df, case_id='case:concept:name', activity_key='concept:name', timestamp_key='time:timestamp')

# Discover organizational roles using activity-based clustering
# Note: pm4py.discover_organizational_roles is not a direct function; we approximate with similarity and clustering
from pm4py.discovery.organizational_mining import algorithm as org_mining

# Activity-based resource similarity (cosine similarity)
similarity_df = pm4py.discover_activity_based_resource_similarity(df, parameters={org_mining.Parameters.ACTIVITY_KEY: 'concept:name'})

# Handover of work network (who passes work to whom)
handover_df = pm4py.discover_handover_of_work_network(df)

# Working together network (who collaborates on the same case)
working_together_df = pm4py.discover_working_together_network(df)

# Subcontracting network (who delegates to whom)
subcontracting_df = pm4py.discover_subcontracting_network(df)

# Cluster resources based on activity profiles (using similarity_df)
from sklearn.cluster import AgglomerativeClustering
import numpy as np

# Extract similarity matrix (assuming similarity_df is a square matrix)
similarity_matrix = similarity_df.values
clustering = AgglomerativeClustering(n_clusters=None, affinity='precomputed', linkage='average', distance_threshold=0.3)
clusters = clustering.fit_predict(1 - similarity_matrix)  # Convert similarity to distance

# Join with case complexity and substitution data
# Assume df has columns: case_complexity, temporary_substitution
role_df = df.groupby('org:resource').agg({
    'case_complexity': ['mean', 'median', 'count'],
    'temporary_substitution': 'max',  # Flag if resource ever had substitutions
    'concept:name': lambda x: x.value_counts().to_dict()  # Activity counts per resource
}).reset_index()

# Add cluster labels
role_df['cluster'] = clusters
role_df['role'] = role_df['cluster'].map({
    0: 'Front-Office Agent',
    1: 'Technical Analyst',
    2: 'Commercial Specialist',
    3: 'Supervisor',
    # ... map other clusters
})

# Avoid over-interpreting frequency clusters:
# - Check case complexity per cluster (e.g., Erik's cluster should have high complexity).
# - Validate substitutions (e.g., Giorgio's cluster should show temporary_substitution=True for Fatima's absence).
# - Inspect handover networks to confirm authority (e.g., Hana should receive handoffs for A9).

# Example: Filter for high-complexity cases in a cluster
high_complexity_resources = role_df[role_df[('case_complexity', 'median')] > 0.7]['org:resource']
print("High-complexity resources:", high_complexity_resources.tolist())
```
