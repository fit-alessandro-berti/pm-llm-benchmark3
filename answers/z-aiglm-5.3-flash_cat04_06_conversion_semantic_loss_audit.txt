Implementation of epsilon closure:

```python
def epsilon_closure(net, marking, max_silent_steps=10000):
    # BFS over silent (label None) transitions; visited set prevents infinite cycling
    from copy import deepcopy
    closure = {marking_key(marking): marking}
    frontier = [marking]
    silent_steps = 0
    while frontier:
        m = frontier.pop()
        for t in pm4py.get_enabled_transitions(net, m):
            if t.label is None:
                silent_steps += 1
                if silent_steps > max_silent_steps:
                    raise RuntimeError("Silent-step budget exceeded — possible tau livelock")
                m2 = fire(net, m, t)
                k = marking_key(m2)
                if k not in closure:
                    closure[k] = m2
                    frontier.append(m2)
    return list(closure.values())
```

```python
def fire(net, marking, transition):
    new_marking = Marking()
    for p, c in marking.items():
        new_marking[p] = c
    for arc in transition.in_arcs:
        new_marking[arc.source] = new_marking.get(arc.source, 0) - arc.weight
        if new_marking[arc.source] <= 0:
            del new_marking[arc.source]
    for arc in transition.out_arcs:
        new_marking[arc.target] = new_marking.get(arc.target, 0) + arc.weight
    return new_marking
```

Now, code for `accepts(net, im, fm, trace)` with backtracking:

```python
def accepts(net, im, fm, trace):
    key = make_marking_key(net)
    fm_norm = norm(fm)

    def norm(m):
        return frozenset((p.name, c) for p, c in m.items() if c > 0)

    def closure(marking):
        seen = {key(marking): marking}
        stack = [marking]
        budget = 100000
        while stack:
            m = stack.pop()
            for t in pm4py.get_enabled_transitions(net, m):
                if t.label is None:
                    budget -= 1
                    if budget < 0:
                        raise RuntimeError("tau budget exceeded (possible silent livelock)")
                    m2 = fire(m, t)
                    k = key(m2)
                    if k not in seen:
                        seen[k] = m2
                        stack.append(m2)
        return list(seen.values())

    def rec(marking, idx):
        clos = closure(marking)
        if idx == len(trace):
            return any(norm(m) == fm_norm for m in clos)
        label = trace[idx]
        for m in clos:
            for t in pm4py.get_enabled_transitions(net, m):
                if t.label == label:
                    if rec(fire(m, t), idx + 1):
                        return True
        return False

    return rec(im, 0)
```
