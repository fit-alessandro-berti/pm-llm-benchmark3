def compute_fitness(log, net, im, fm, diagnostics):
    try:
       return pm4py.fitness_alignments(log, net, im, fm)
    except Exception as exc:
       print(f"[fallback] Approximate alignment fitness because: {exc}")
       if not diagnostics: return None
       trace_lengths=[]
       try: trace_lengths=[len(t) for t in log]
       except: trace_lengths=[0]*len(diagnostics)
       vals=[]
       for i, d in enumerate(diagnostics):
          if not isinstance(d, dict): continue
          cost=d.get('cost')
          if cost is None:
             cost=0
             for move in d.get('alignment', []):
                try:
                   lm, mm = move
                   if lm != mm: cost += 1
                except: pass
          length = trace_lengths[i] if i < len(trace_lengths) else 0
          denom = max(1, length + cost)
          vals.append(max(0.0, 1.0 - float(cost)/denom))
       if not vals: return None
       return {"approximate_log_fitness": sum(vals)/len(vals), "note":"cost-based approximation"}

# Log construction helpers

def traces_from_variant_table(variants):
    traces = []
    for trace_string, count in variants.items():
        activities = [a.strip() for a in trace_string.split(",")]
        traces.extend([activities for _ in range(count)])
    return traces


def xes_from_traces(traces, prefix):
    parts = [
        '<?xml version="1.0" encoding="UTF-8"?>',
        '<log xes.version="1.0" xmlns="http://www.xes-standard.org/">',
    ]
    for case_no, trace in enumerate(traces, start=1):
        parts.append("<trace>")
        parts.append(f'<string key="concept:name" value="{prefix}-{case_no}"/>')
        for activity in trace:
            safe_activity = escape(str(activity), QUOTE_MAP)
            parts.append("<event>")
            parts.append(f'<string key="concept:name" value="{safe_activity}"/>')
            parts.append("</event>")
        parts.append("</trace>")
    parts.append("</log>")
    return "\n".join(parts)


def manual_log_from_traces(traces, prefix):
    from pm4py.objects.log.obj import EventLog, Trace, Event

    log = EventLog()
    for case_no, activities in enumerate(traces, start=1):
        trace = Trace()
        trace.attributes["concept:name"] = f"{prefix}-{case_no}"
        for activity in activities:
            event = Event()
            event["concept:name"] = activity
            trace.append(event)
        log.append(trace)
    return log


def log_from_traces(traces, prefix):
    xes_string = xes_from_traces(traces, prefix)
    try:
        log = pm4py.parse_event_log_string(xes_string)
        if log is not None:
            return log
    except Exception as exc:
        print(f"[fallback] parse_event_log_string failed for {prefix}; building EventLog manually: {exc}")
    return manual_log_from_traces(traces, prefix)


def log_from_variant_table(variants, prefix):
    traces = traces_from_variant_table(variants)
    return log_from_traces(traces, prefix)


def traces_from_log(log):
    traces = []
    for trace in log:
        activities = []
        for event in trace:
            try:
                activities.append(str(event["concept:name"]))
            except Exception:
                activities.append(str(event.get("concept:name", "UNKNOWN")))
        traces.append(activities)
    return traces

def normalize_traces(traces, map_label=True, remove_queue_routed=True):
    normalized = []
    for trace in traces:
        new_trace = []
        for activity in trace:
            activity = str(activity)
            if map_label and activity == "Risk Screen":
                activity = "Classify"
            if remove_queue_routed and activity == "Queue Routed":
                continue
            new_trace.append(activity)
        normalized.append(new_trace)
    return normalized

def parse_tree_compatible(tree_string):
    try:
        return pm4py.parse_process_tree(tree_string)
    except AttributeError:
        from pm4py.objects.process_tree.utils.parse import parse as pt_parse
        return pt_parse(tree_string)


def convert_to_petri_compatible(tree):
    try:
        result = pm4py.convert_to_petri_net(tree)
        if isinstance(result, tuple) and len(result) == 3:
            return result
        raise ValueError("Unexpected result from pm4py.convert_to_petri_net")
    except Exception as exc:
        print(f"[fallback] pm4py.convert_to_petri_net failed; using process-tree converter: {exc}")
        from pm4py.objects.conversion.process_tree import converter as pt_converter
        return pt_converter.apply(tree)


def discover_dfg_compatible(log):
    try:
        return pm4py.discover_dfg(log)
    except AttributeError:
        from pm4py.algo.discovery.dfg import algorithm as dfg_algorithm
        return dfg_algorithm.apply(log)


def print_dfg(log, title, top=8):
    try:
        result = discover_dfg_compatible(log)
        dfg = result[0] if isinstance(result, tuple) else result
        print(f"Top directly-follows edges for {title}:")
        sorted_edges = sorted(dfg.items(), key=lambda kv: kv[1], reverse=True)[:top]
        for edge, count in sorted_edges:
            if isinstance(edge, tuple) and len(edge) == 2:
                print(f"  {edge[0]} -> {edge[1]}: {count}")
            else:
                print(f"  {edge}: {count}")
    except Exception as exc:
        print(f"DFG discovery failed for {title}: {exc}")


def compute_alignment_diagnostics(log, net, initial_marking, final_marking):
    try:
        from pm4py.algo.conformance.alignments.petri_net import algorithm as alignments_algorithm
        return alignments_algorithm.apply(log, net, initial_marking, final_marking)
    except Exception as exc:
        print(f"Alignment diagnostics unavailable: {exc}")
        return []


def summarize_alignment_diagnostics(diagnostics):
    if diagnostics is None:
        return {"log_moves": [], "model_moves": [], "mean_cost": None}

    items = diagnostics.values() if isinstance(diagnostics, dict) else diagnostics
    log_moves = Counter()
    model_moves = Counter()
    costs = []

    for diag in items:
        if not isinstance(diag, dict):
            continue
        if "cost" in diag:
            costs.append(diag["cost"])
        alignment = diag.get("alignment", [])
        for move in alignment:
            try:
                log_move, model_move = move
            except Exception:
                continue
            if log_move == ">>":
                model_moves[str(model_move)] += 1
            elif model_move == ">>":
                log_moves[str(log_move)] += 1
            elif log_move != model_move:
                log_moves[str(log_move)] += 1
                model_moves[str(model_move)] += 1

    mean_cost = sum(costs) / len(costs) if costs else None
    return {
        "log_moves": log_moves.most_common(5),
        "model_moves": model_moves.most_common(5),
        "mean_cost": mean_cost,
    }


def compute_alignment_fitness(log, net, initial_marking, final_marking, diagnostics=None):
    try:
        return pm4py.fitness_alignments(log, net, initial_marking, final_marking)
    except Exception as exc:
        print(f"[fallback] Approximating alignment fitness because pm4py.fitness_alignments failed: {exc}")

        if diagnostics is None:
            diagnostics = compute_alignment_diagnostics(log, net, initial_marking, final_marking)
        if not diagnostics:
            return None

        try:
            trace_lengths = [len(trace) for trace in log]
        except Exception:
            trace_lengths = []

        items = diagnostics.values() if isinstance(diagnostics, dict) else diagnostics
        fitness_values = []

        for idx, diag in enumerate(items):
            if not isinstance(diag, dict):
                continue
            cost = diag.get("cost")
            if cost is None:
                cost = 0
                for move in diag.get("alignment", []):
                    try:
                        log_move, model_move = move
                        if log_move != model_move:
                            cost += 1
                    except Exception:
                        pass

            length = trace_lengths[idx] if idx < len(trace_lengths) else 0
            denominator = max(1, length + float(cost))
            fitness_values.append(max(0.0, 1.0 - float(cost) / denominator))

        if not fitness_values:
            return None
        return {
            "approximate_alignment_fitness": sum(fitness_values) / len(fitness_values),
            "note": "Approximation from alignment cost because high-level fitness API was unavailable.",
        }


def compute_alignment_precision(log, net, initial_marking, final_marking):
    try:
        return pm4py.precision_alignments(log, net, initial_marking, final_marking)
    except Exception as exc:
        print(f"Precision unavailable: {exc}")
        return None


def format_metric(metric):
    if metric is None:
        return "unavailable"
    if isinstance(metric, dict):
        return "; ".join(f"{k}={v}" for k, v in metric.items())
    return str(metric)


def evaluate_scenario(log, tree, title, compute_precision=False):
    print("\n" + "=" * 80)
    print(f"Scenario: {title}")
    print(f"Number of cases: {len(log)}")
    print_dfg(log, title)

    net, initial_marking, final_marking = convert_to_petri_compatible(tree)
    diagnostics = compute_alignment_diagnostics(log, net, initial_marking, final_marking)
    summary = summarize_alignment_diagnostics(diagnostics)
    fitness = compute_alignment_fitness(log, net, initial_marking, final_marking, diagnostics)
    precision = compute_alignment_precision(log, net, initial_marking, final_marking) if compute_precision else None

    print("Alignment diagnostics:")
    print(f"  dominant log moves: {summary['log_moves']}")
    print(f"  dominant model moves: {summary['model_moves']}")
    print(f"  mean alignment cost: {summary['mean_cost']}")
    print(f"Alignment fitness: {format_metric(fitness)}")
    print(f"Alignment precision: {format_metric(precision)}")

    return {
        "scenario": title,
        "cases": len(log),
        "fitness": fitness,
        "precision": precision,
        "summary": summary,
    }


def compare_results(results):
    print("\n" + "=" * 80)
    print("Comparison of raw and normalized results")
    print("scenario | cases | fitness | precision")
    for result in results:
        print(
            f"{result['scenario']} | {result['cases']} | "
            f"{format_metric(result['fitness'])} | {format_metric(result['precision'])}"
        )

## 7. Provisional conclusion

The measured conformance collapse is best explained as a cascade of measurement and model-provenance failures rather than a sudden breakdown of the claims process. The strongest signal is the June UI label change: `Classify` became `Risk Screen`, and mapping the label alone materially improves fitness. The gateway then added `Queue Routed`, a technical event not present in the normative tree; removing it restores additional fitness. The ETL change that used warehouse-ingestion time for `Decision` and `Notify` plausibly created artificial reversed orderings, and using the unused source-sequence field further improves fit. By August, the normative model was also obsolete: a policy added `Supervisor Sign-off` for high-value claims, but the control-flow tree was not updated. Therefore, the primary explanation of the measured drop is label/semantic logging change plus technical-event/extraction artifacts, with later model obsolescence. Confidence: high, about 80-85%, that data/provenance and model mismatch explain most of the fitness decline.

A separate real compliance concern may remain. Enhanced-review cases without recorded `Compliance Check` occurred in both June-July and August-September, and the memo about bypasses shows awareness of the risk. `Manual Override` may be legitimate correction or a control weakness. Stable appeal and approval rates are reassuring but not exculpatory because incident detection is incomplete and high-value status was omitted. Confidence: moderate-to-low, perhaps 35-50%, that some true noncompliance exists; there is not enough evidence to attribute it broadly to employees.

Immediate actions: IT should re-extract using source sequence, flag technical events, and publish label semantics. Process owners should version the normative model and add data-aware sign-off rules. Compliance should audit enhanced-review cases lacking checks and high-value cases lacking sign-off before any corrective action against staff. Do not claim widespread employee noncompliance, do not treat raw fitness as operational truth, and do not accept an overpermissive optional-sign-off model as proof of compliance.

## 8. Model-governance recommendation

1. **Version normative models with effective dates.** Every policy, UI, ETL, or gateway change that can affect process semantics must trigger a controlled model review, with old and new model versions retained for comparison.

2. **Maintain an event-semantic contract.** Keep a data dictionary mapping screens, buttons, system messages, and canonical activity labels. Any UI label change must be assessed for conformance impact before release.

3. **Classify technical events separately from business events.** Events such as `Queue Routed` should carry an event-type flag and be excluded from normative conformance checking unless explicitly modeled as part of the policy process.

4. **Preserve authoritative timestamp and ordering provenance.** Conformance checking must use source-system event time or source-sequence order, not warehouse-ingestion time, and the chosen ordering field must be documented and validated.

5. **Encode data-aware and resource constraints.** Control-flow trees alone are insufficient for regulated claims. Rules such as “supervisor sign-off required above EUR 50,000” should be represented through attribute-aware conformance checks or decision rules.

6. **Require post-release validation.** After UI, ETL, or policy releases, run regression tests on variants, DFGs, alignment diagnostics, fitness, precision, and a manual audit sample before interpreting conformance changes as operational failure.
