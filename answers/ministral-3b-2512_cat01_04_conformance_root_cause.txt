import pm4py
from pm4py.objects.log.util import dataframe_util

def generate_logs(variant_counts):
    logs = []
    for trace, count in variant_counts.items():
        for _ in range(count):
            logs.append(trace.split(','))
    return logs

# April-May 2026
april_may_logs = generate_logs({
    "Receive,Classify,Standard Review,Decision,Notify": 380,
    "Receive,Classify,Enhanced Review,Compliance Check,Decision,Notify": 180,
    "Receive,Classify,Standard Review,Request Info,Standard Review,Decision,Notify": 40

alignment_april_may = pm4py.align_log(

alignment_june_jul = pm4py.align_log(
    pm4py.format_dataframe_dataframe(june_jul_logs),

def normalize_logs(logs):
    normalized = []
    for trace in logs:
        normalized_trace = [x.replace("Risk Screen", "Classify") for x in trace]
        normalized_trace = [x for x in normalized_trace if x not in ["Queue Routed", "Manual Override"]]
        normalized.append(normalized_trace)
    return normalized_logs

normalized_june_jul = normalize_logs(june_jul_logs)
normalized_alignment = pm4py.align_log(
