DIAGNOSIS_ALIASES = {
    "Diagnostic Assessment Completed": "Initial Diagnosis Completed",
}

def normalize_events(df: pd.DataFrame) -> pd.DataFrame:
    df = df.copy()
    df["timestamp"] = pd.to_datetime(df["timestamp"])
    df = df.sort_values(["case_id", "timestamp"]).reset_index(drop=True)
    df["activity"] = df["activity"].replace(DIAGNOSIS_ALIASES)
    df["activity"] = df["activity"].fillna("UNKNOWN_ACTIVITY").astype(str)
    # exact duplicate removal
    df = df.drop_duplicates(subset=["case_id", "activity", "timestamp"], keep="first")
    # collapse rapid repeated diagnosis events (example: <=15 min) to earliest
    diag = df[df["activity"] == "Initial Diagnosis Completed"]
    if not diag.empty:
        keep_idx = []
        for _, g in diag.groupby("case_id"):
            g = g.sort_values("timestamp")
            keep_idx.append(g.index[0])
            last = g["timestamp"].iloc[0]
            for idx, ts in zip(g.index[1:], g["timestamp"][1:]):
                if ts - last > pd.Timedelta(minutes=15):
                    keep_idx.append(idx)
                    last = ts
        df = df.drop(index=[i for i in diag.index if i not in keep_idx])
    df["lifecycle:transition"] = df["timestamp"]
    return df

def edge_median_hours(df: pd.DataFrame, start: str, end: str) -> float:
    a = df[df["activity"] == start][["case_id", "timestamp"]].rename(columns={"timestamp": "t_start"})
    b = df[df["activity"] == end][["case_id", "timestamp"]].rename(columns={"timestamp": "t_end"})
    m = a.merge(b, on="case_id")
    m = m[m["t_end"] >= m["t_start"]]
    if m.empty:
        return float("nan")
    return (m["t_end"] - m["t_start"]).dt.total_seconds().div(3600).median()

                "variant_counts": pd.Series(variants).value_counts().to_dict(),
                "dfg_edge_counts": {f"{a}->{b}": int(c) for (a, b), c in dfg.items()},

import pandas as pd
import pm4py

DIAGNOSIS_ALIASES = {
    "Diagnostic Assessment Completed": "Initial Diagnosis Completed",
}

def normalize_events(df: pd.DataFrame) -> pd.DataFrame:
    df = df.copy()
    df["timestamp"] = pd.to_datetime(df["timestamp"])
    df = df.sort_values(["case_id", "timestamp"]).reset_index(drop=True)
    df["activity"] = df["activity"].replace(DIAGNOSIS_ALIASES)
    df["activity"] = df["activity"].fillna("UNKNOWN_ACTIVITY").astype(str)

    # Duplicate strategy: remove exact duplicates; collapse rapid repeated diagnosis events.
    df = df.drop_duplicates(subset=["case_id", "activity", "timestamp"], keep="first")
    diag = df[df["activity"] == "Initial Diagnosis Completed"]
    if not diag.empty:
        keep_idx = []
        for _, g in diag.groupby("case_id"):
            g = g.sort_values("timestamp")
            keep_idx.append(g.index[0])
            last = g["timestamp"].iloc[0]
            for idx, ts in zip(g.index[1:], g["timestamp"][1:]):
                if ts - last > pd.Timedelta(minutes=15):
                    keep_idx.append(idx)
                    last = ts
        df = df.drop(index=[i for i in diag.index if i not in keep_idx])

    df["lifecycle:transition"] = df["timestamp"]
    return df

def edge_median_hours(df: pd.DataFrame, start: str, end: str) -> float:
    a = df[df["activity"] == start][["case_id", "timestamp"]].rename(columns={"timestamp": "t_start"})
    b = df[df["activity"] == end][["case_id", "timestamp"]].rename(columns={"timestamp": "t_end"})
    m = a.merge(b, on="case_id")
    m = m[m["t_end"] >= m["t_start"]]
    if m.empty:
        return float("nan")
    return (m["t_end"] - m["t_start"]).dt.total_seconds().div(3600).median()

def monthly_process_evidence(csv_path: str = "field_service_events.csv", periods=None) -> pd.DataFrame:
    df = normalize_events(pd.read_csv(csv_path))
    log = pm4py.format_dataframe(
        df,
        case_id="case_id",
        activity="activity",
        timestamp="lifecycle:transition",
        resource="resource",
    )

    starts = df[df["activity"] == "Case Created"].copy()
    if starts.empty:
        starts = df.groupby("case_id")["timestamp"].min().reset_index()
        starts["activity"] = "Case Created"
    starts = starts.drop_duplicates("case_id", keep="first")
    starts["start_period"] = starts["timestamp"].dt.to_period("M")

    if periods is None:
        periods = sorted(starts["start_period"].unique())

    rows = []
    for p in periods:
        t0 = p.to_timestamp()
        t1 = (p + 1).to_timestamp()

        # Event-calendar cohort: events whose timestamp is in month p.
        event_log = pm4py.filter_time_range(log, time_range=(t0, t1), parameter="lifecycle:transition")
        try:
            event_df = pm4py.format_log_to_dataframe(
                event_log,
                case_id="case_id",
                activity="activity",
                timestamp="lifecycle:transition",
                resource="resource",
            )
        except Exception:
            event_df = df[(df["timestamp"] >= t0) & (df["timestamp"] < t1)]

        # Case-start cohort: all events for cases whose Case Created timestamp is in month p.
        start_case_ids = set(starts.loc[starts["start_period"] == p, "case_id"])
        case_df = df[df["case_id"].isin(start_case_ids)]

        for cohort_name, cohort_df in [("event_calendar", event_df), ("case_start", case_df)]:
            if cohort_df.empty:
                rows.append({"period": str(p), "cohort": cohort_name, "cases": 0})
                continue

            cohort_log = pm4py.format_dataframe(
                cohort_df,
                case_id="case_id",
                activity="activity",
                timestamp="lifecycle:transition",
                resource="resource",
            )

            variants = pm4py.get_variants(cohort_log)
            dfg = pm4py.discover_dfg(cohort_log)
            perf_dfg = pm4py.discover_performance_dfg(cohort_log, parameter="lifecycle:transition")

            try:
                temporal = pm4py.extract_temporal_features_dataframe(
                    cohort_df,
                    case_id="case_id",
                    activity="activity",
                    timestamp="timestamp",
                    resource="resource",
                )
            except Exception:
                temporal = pd.DataFrame()

            rows.append(
                {
                    "period": str(p),
                    "cohort": cohort_name,
                    "events": len(cohort_df),
                    "cases": cohort_df["case_id"].nunique(),
                    "unknown_share": float(cohort_df["activity"].eq("UNKNOWN_ACTIVITY").mean()),
                    "variant_counts": pd.Series(variants).value_counts().to_dict(),
                    "dfg_edge_counts": {f"{a}->{b}": int(c) for (a, b), c in dfg.items()},
                    "perf_dfg": {f"{a}->{b}": v for (a, b), v in perf_dfg.items()},
                    "median_order_part_hours": edge_median_hours(cohort_df, "Order Part", "Part Received"),
                    "median_diag_remote_hours": edge_median_hours(cohort_df, "Initial Diagnosis Completed", "Remote Fix"),
                    "temporal_feature_rows": int(len(temporal)),
                }
            )
    return pd.DataFrame(rows)

   - V3 is high-risk and process-sensitive.
   - V1 is relatively stable.

2. **Variant-specific or hierarchical modeling** if V3 remains materially different after feature repair.
   - V3 has enough volume in later months to justify separate treatment.
   - A hierarchical model can share common signal across variants while allowing V3-specific part-lead-time effects.

3. **Partner-specific treatment only if Partner Z has enough matured cases**.
   - If Partner Z is a persistent population with different rural/device characteristics, a partner feature or partner-specific calibration is justified.
   - Do not create a permanent separate model solely from immature September/October evidence.

4. **Avoid over-fragmentation**.
   - Too many small segments will reduce stability.
   - Use segment features and regularization before creating many independent models.

## 5. Monitoring architecture

### Monitor categories

#### 1. Predictive performance monitor
Track:
- AUROC.
- Brier score.
- ECE.
- Precision/recall at operational thresholds.
- Predicted risk distribution.
- Predicted-vs-observed breach by risk decile.

Stratify by:
- Month.
- Variant: V1, V2, V3.
- Partner: Partner Z vs non-Partner-Z.
- Maturity bucket.
- Data-quality pass/fail.

Alert only when:
- Label maturity is sufficient.
- Data-quality gates pass.
- Degradation persists for a defined window, e.g., 2–4 weeks.
- Degradation is not explained by a known process or population event.

---

#### 2. Calibration monitor
Track:
- ECE by variant and partner.
- Reliability diagrams.
- Mean predicted risk vs observed breach rate.
- Calibration slope and intercept.
- Segment-specific calibration drift.

Alert when:
- Matured ECE exceeds threshold, e.g., 0.05.
- Mean predicted risk diverges from observed breach rate in a matured, data-quality-pass cohort.
- Variant-specific calibration worsens while overall calibration appears stable.

---

#### 3. Label-maturity monitor
Track:
- Label maturity by start month.
- Median label delay.
- Share of unresolved cases.
- Outcome API backlog.
- Maturity by variant and partner.
- Maturity by path length.

Alert when:
- Maturity for a recent cohort falls below threshold, e.g., 90%.
- Median label delay increases materially.
- Long-path cases, especially V3/Partner Z, have disproportionately low maturity.

**Alert dependency:**  
A label-delay alarm should **not** automatically trigger retraining. It should trigger outcome-ops investigation and mark predictive metrics as provisional.

---

#### 4. Data-quality monitor
Track:
- Unknown activity share.
- Missing queue feature.
- Diagnosis label mismatch.
- Duplicate event rate.
- Unmapped variant share.
- Feature null rates.
- Feature version and lineage.
- Backfill completion.

Alert when:
- Unknown activity exceeds baseline, e.g., >2%.
- Missing queue feature exceeds baseline, e.g., >5%.
- Diagnosis label mismatch appears.
- Duplicate rate spikes.
- Derived feature version is stale after a schema change.

**Alert dependency:**  
Data-quality failure should gate predictive and calibration alerts. If features are unreliable, do not trigger model retraining; trigger pipeline repair.

---

#### 5. Process-variant monitor
Track:
- V1, V2, V3, unmapped/other shares.
- Variant transitions.
- Routing policy changes.
- Partner Z routing share.
- Unmapped variant share.

Alert when:
- V2 or V3 share shifts beyond a threshold, e.g., ±5 percentage points.
- Unmapped/other share exceeds 2%.
- Partner Z V3 routing share rises materially.

---

#### 6. Performance-DFG monitor
Track:
- Median, mean, and p90 elapsed time on key DFG edges:
  - `Order Part -> Part Received`.
  - `Initial Diagnosis Completed -> Remote Fix`.
  - `Dispatch Technician -> Repair Completed`.
  - `Case Created -> Triage`.
  - `Triage -> Initial Diagnosis Completed`.
- Edge frequencies.
- Missing or abnormal transitions.
- Event-calendar vs case-start cohort differences.

Alert when:
- `Order Part -> Part Received` p90 or median exceeds threshold, e.g., >120 h.
- Remote-fix time increases materially.
- DFG structure changes unexpectedly.
- Case-start cohort process times differ materially from event-calendar activity times.

---

#### 7. Population monitor
Track:
- Partner share.
- Rural travel distance distribution.
- Device age distribution.
- Customer segment distribution.
- Feature PSI/KS for population-sensitive features.
- Partner Z V3 share.

Alert when:
- Partner share changes materially.
- Rural distance or device age distributions shift.
- PSI/KS for population features exceeds threshold.
- Partner Z cases have different maturity or variant mix.

---

#### 8. Intervention/deployment monitor
Track:
- Model version.
- Feature pipeline version.
- Calibration version.
- Backfill status.
- A/B or shadow deployment status.
- Known business interventions:
  - Remote-routing policy.
  - Supplier issue.
  - Partner onboarding.
  - Outcome API backlog.

Alert when:
- A new model or feature version is deployed without validation.
- Calibration changes are applied to immature cohorts.
- A known intervention is active but not annotated in monitoring.

---

### Alert dependency logic

Use a gate-based alerting architecture:

1. **Data-quality gate**
   - If data-quality gate fails:
     - Mark predictive metrics as unreliable.
     - Suppress retraining triggers.
     - Route to data engineering.

2. **Label-maturity gate**
   - If label maturity is below threshold:
     - Mark predictive metrics as provisional.
     - Do not trigger retraining.
     - Route to outcome/data-ops.

3. **Process/ population gate**
   - If process or population alarm is active:
     - Annotate predictive alerts with the known business cause.
     - Route to process owner and model owner.
     - Prefer recalibration or feature addition over immediate full retraining.

4. **Model-performance trigger**
   - Retraining is triggered only if:
     - Data-quality gate passes.
     - Label-maturity gate passes.
     - Predictive degradation persists.
     - Degradation is not fully explained by a known process/population event.
     - Stratified analysis confirms model-level issue.

This prevents a label-delay alarm from causing unnecessary retraining on immature labels.

### Python sketch: monthly process-evidence extraction

```python
import pandas as pd
import pm4py

DIAGNOSIS_ALIASES = {
    "Diagnostic Assessment Completed": "Initial Diagnosis Completed",
}


def _standardize_columns(df: pd.DataFrame) -> pd.DataFrame:
    renames = {
        "case:concept:name": "case_id",
        "activity:concept:name": "activity",
        "time:timestamp": "timestamp",
        "lifecycle:transition": "timestamp",
        "org:resource": "resource",
    }
    return df.rename(
        columns={k: v for k, v in renames.items() if k in df.columns and v not in df.columns}
    )


def normalize_events(df: pd.DataFrame) -> pd.DataFrame:
    """
    Normalize diagnosis labels and apply a conservative duplicate strategy.

    Duplicate strategy:
    1. Remove exact duplicates: same case, activity, timestamp.
    2. For diagnosis events, keep the earliest event per case.
       Additional diagnosis events are kept only if they occur more than 15 minutes later,
       which may represent a legitimate re-diagnosis rather than a duplicate.
    """
    df = df.copy()
    df["timestamp"] = pd.to_datetime(df["timestamp"])
    df = df.sort_values(["case_id", "timestamp"]).reset_index(drop=True)

    df["activity"] = df["activity"].replace(DIAGNOSIS_ALIASES)
    df["activity"] = df["activity"].fillna("UNKNOWN_ACTIVITY").astype(str)

    # Exact duplicate removal.
    df = df.drop_duplicates(subset=["case_id", "activity", "timestamp"], keep="first")

    # Collapse rapid repeated diagnosis events.
    diag = df[df["activity"] == "Initial Diagnosis Completed"]
    if not diag.empty:
        keep_idx = []
        for _, g in diag.groupby("case_id"):
            g = g.sort_values("timestamp")
            keep_idx.append(g.index[0])
            last = g["timestamp"].iloc[0]
            for idx, ts in zip(g.index[1:], g["timestamp"][1:]):
                if ts - last > pd.Timedelta(minutes=15):
                    keep_idx.append(idx)
                    last = ts
        df = df.drop(index=[i for i in diag.index if i not in keep_idx])

    # PM4Py-friendly timestamp attribute.
    df["lifecycle:transition"] = df["timestamp"]
    return df


def edge_median_hours(df: pd.DataFrame, start: str, end: str) -> float:
    """Median elapsed time from start activity to subsequent end activity."""
    a = df[df["activity"] == start][["case_id", "timestamp"]].rename(
        columns={"timestamp": "t_start"}
    )
    b = df[df["activity"] == end][["case_id", "timestamp"]].rename(
        columns={"timestamp": "t_end"}
    )
    m = a.merge(b, on="case_id")
    m = m[m["t_end"] >= m["t_start"]]
    m = m.sort_values(["case_id", "t_start", "t_end"])
    m = m.drop_duplicates(subset=["case_id", "t_start"], keep="first")
    if m.empty:
        return float("nan")
    return (m["t_end"] - m["t_start"]).dt.total_seconds().div(3600).median()


def monthly_process_evidence(csv_path: str = "field_service_events.csv", periods=None) -> pd.DataFrame:
    """
    Extract auditable monthly process evidence.

    This does not fit a model and does not call a PM4Py drift detector.
    It produces monthly summaries that can be used for external thresholds and investigation.
    """
    df = normalize_events(pd.read_csv(csv_path))

    log = pm4py.format_dataframe(
        df,
        case_id="case_id",
        activity="activity",
        timestamp="lifecycle:transition",
        resource="resource",
    )

    # Case-start reference.
    starts = df[df["activity"] == "Case Created"].copy()
    if starts.empty:
        starts = df.groupby("case_id")["timestamp"].min().reset_index()
        starts["activity"] = "Case Created"
    starts = starts.drop_duplicates("case_id", keep="first")
    starts["start_period"] = starts["timestamp"].dt.to_period("M")

    if periods is None:
        periods = sorted(starts["start_period"].unique())

    rows = []

    for p in periods:
        t0 = p.to_timestamp()
        t1 = (p + 1).to_timestamp()

        # Cohort 1: event-calendar cohort.
        # Interpretation: activity volume occurring in calendar month p.
        # This may include cases started earlier and completed later.
        try:
            event_log = pm4py.filter_time_range(
                log,
                time_range=(t0, t1),
                parameter="lifecycle:transition",
            )
            event_df = pm4py.format_log_to_dataframe(
                event_log,
                case_id="case_id",
                activity="activity",
                timestamp="lifecycle:transition",
                resource="resource",
            )
        except Exception:
            event_df = df[(df["timestamp"] >= t0) & (df["timestamp"] < t1)]

        event_df = _standardize_columns(event_df).copy()

        # Cohort 2: case-start cohort.
        # Interpretation: all events for cases whose Case Created timestamp falls in month p.
        # This is the cohort relevant to monthly model evaluation and outcome maturity.
        start_case_ids = set(starts.loc[starts["start_period"] == p, "case_id"])
        case_df = df[df["case_id"].isin(start_case_ids)]

        for cohort_name, cohort_df in [
            ("event_calendar", event_df),
            ("case_start", case_df),
        ]:
            cohort_df = _standardize_columns(cohort_df).copy().reset_index(drop=True)
            if cohort_df.empty:
                rows.append({"period": str(p), "cohort": cohort_name, "cases": 0})
                continue

            cohort_df["timestamp"] = pd.to_datetime(cohort_df["timestamp"])
            cohort_df["lifecycle:transition"] = cohort_df["timestamp"]

            cohort_log = pm4py.format_dataframe(
                cohort_df,
                case_id="case_id",
                activity="activity",
                timestamp="lifecycle:transition",
                resource="resource",
            )

            variants = pm4py.get_variants(cohort_log) or []
            variant_counts = (
                pd.Series([tuple(v) for v in variants]).value_counts().to_dict()
            )

            dfg = pm4py.discover_dfg(cohort_log) or {}
            perf_dfg = pm4py.discover_performance_dfg(
                cohort_log,
                parameter="lifecycle:transition",
            ) or {}

            # Optional temporal feature extraction for auditability.
            try:
                temporal = pm4py.extract_temporal_features_dataframe(
                    cohort_df,
                    case_id="case_id",
                    activity="activity",
                    timestamp="timestamp",
                    resource="resource",
                )
            except Exception:
                temporal = pd.DataFrame()

            rows.append(
                {
                    "period": str(p),
                    "cohort": cohort_name,
                    "events": len(cohort_df),
                    "cases": cohort_df["case_id"].nunique(),
                    "unknown_share": float(cohort_df["activity"].eq("UNKNOWN_ACTIVITY").mean()),
                    "variant_counts": variant_counts,
                    "dfg_edge_counts": {f"{a}->{b}": int(c) for (a, b), c in dfg.items()},
                    "perf_dfg": {f"{a}->{b}": v for (a, b), v in perf_dfg.items()},
                    "median_order_part_hours": edge_median_hours(
                        cohort_df,
                        "Order Part",
                        "Part Received",
                    ),
                    "median_diag_remote_hours": edge_median_hours(
                        cohort_df,
                        "Initial Diagnosis Completed",
                        "Remote Fix",
                    ),
                    "temporal_feature_rows": int(len(temporal)),
                }
            )

    return pd.DataFrame(rows)
```

## 6. Risks of acting on the wrong diagnosis

### 1. Retraining on immature October labels

**Risk:**  
October has only 55% label maturity and a 52-day median label delay. The outcome API backlog disproportionately affects longer resolution paths, especially V3 and Partner Z cases. Retraining on October labels could teach the model a biased, artificially low breach rate or noisy label distribution. The model could become worse for exactly the long, high-risk cases it needs to catch.

**Mitigation:**
- Exclude October from training until labels mature.
- Use fixed-vintage backtesting.
- Use censoring-aware labels or exclude immature cases.
- Gate retraining on label maturity and outcome-API health.
- Report October metrics as provisional, not comparable to matured months.

---

### 2. Calibrating around an April pipeline bug

**Risk:**  
April’s high mean predicted risk, poor AUROC, and high ECE are driven by renamed diagnosis events, duplicates, unknown activities, wrong prefixes, and stale cached features. If the team only recalibrates or adjusts thresholds without repairing the pipeline, the model will be tuned to corrupted features. Once the pipeline is fixed, the recalibrated model may become miscalibrated again.

**Mitigation:**
- Quarantine April until features are recomputed.
- Normalize diagnosis labels.
- Remove duplicates.
- Rebuild prediction prefixes.
- Recompute cached features.
- Validate April against clean event lineage before using it for calibration or training.
- Do not apply permanent threshold changes based on April raw metrics.

---

### 3. Ignoring genuine supplier-driven concept drift in V3

**Risk:**  
August and September show a real process and concept change in V3: part lead times increase sharply, and matured V3 breach rates rise to 0.64 and 0.61. If this is dismissed as mix shift, label delay, or population shift, the model will continue to underpredict parts-dependent breach risk. The business will miss high-risk cases that are delayed by supplier performance.

**Mitigation:**
- Stratify performance by V3 and part lead time.
- Monitor `Order Part -> Part Received` as a first-class process KPI.
- Add part lead time and supplier-status features.
- Consider V3-specific calibration or a hierarchical model.
- Escalate supplier performance separately from model performance.
- Validate that matured V3 cases with long lead times have elevated breach rates.

---

### 4. Attributing Partner Z’s population shift to unfair model behavior without stratified evidence

**Risk:**  
Partner Z introduces a new population with higher rural travel distance, older devices, and higher V3 routing. The original model lacks rural distance and device age features. September’s observed Partner Z breach rate is 0.45 versus 0.27 for other sources, but Partner Z labels mature more slowly. Jumping to a conclusion of model unfairness or bias without maturity-adjusted, stratified evidence could lead to unnecessary retraining, incorrect fairness claims, or misallocation of remediation effort.

**Mitigation:**
- Stratify by Partner Z versus non-Partner-Z.
- Wait for matured Partner Z labels before making fairness or performance claims.
- Run transport/reweighting checks to see how much of the gap is population shift.
- Add partner, rural distance, and device age features.
- Compare matured Partner Z cases with matured non-Partner-Z cases after adjusting for V3 and part lead time.

---

### 5. Treating June’s mix shift as concept drift and over-retraining

**Risk:**  
June’s main change is a routing policy that increases V2 usage. V2 has a lower matured breach rate, so the overall observed breach rate falls. The model’s mean predicted risk remains higher than the observed rate, causing calibration error. If this is misdiagnosed as concept drift, the team may perform an unnecessary full retrain, introduce instability, and delay a simpler recalibration.

**Mitigation:**
- Stratify by variant.
- Check whether V1, V2, and V3 discrimination and calibration remain stable within variant.
- Use recalibration or variant-specific calibration first.
- Retrain only if within-variant performance degrades or new features are needed.

---

### 6. Treating October’s low observed breach rate as evidence of improvement

**Risk:**  
October’s observed breach rate is 0.16, lower than September’s 0.31, but only 55% of labels are mature. The outcome API backlog disproportionately affects longer paths. Interpreting October as an improvement could suppress necessary supplier escalation, delay model repair, and create false confidence.

**Mitigation:**
- Do not compare October’s raw observed breach rate to matured months.
- Use maturity-adjusted or fixed-vintage breach rates.
- Monitor outcome API backlog and label delay.
- Treat October predictive metrics as provisional.
- Re-evaluate October after labels mature.
