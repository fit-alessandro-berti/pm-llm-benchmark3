def duration_intervals(start_intervals, end_intervals, constraint=None):
    res = []
    for s in start_intervals:
        for e in end_intervals:
            if constraint and not constraint(s,e):
                continue
            lo = (e[0] - s[1]).total_seconds()
            hi = (e[1] - s[0]).total_seconds()
            res.append((lo,hi))

        else:
            merged[-1][1] = max(merged[-1][1], hi)

def classify_sla(intervals, max_allowed):
    if not intervals: return "INDETERMINATE"
    if all(hi <= max_allowed for lo,hi in intervals): return "CERTAIN_PASS"
    if all(lo > max_allowed for lo,hi in intervals): return "CERTAIN_VIOLATION"

def classify_profile(intervals, mean, sd, zeta):
    if not intervals: return "INDETERMINATE"
    lo_band = mean - zeta*sd
    hi_band = mean + zeta*sd
    if all(hi < lo_band or lo > hi_band for lo,hi in intervals):
        return "CERTAIN_VIOLATION"
    if all(lo >= lo_band and hi <= hi_band for lo,hi in intervals):
        return "CERTAIN_PASS"

from itertools import product
import pandas as pd
from datetime import datetime, timedelta, timezone
from zoneinfo import ZoneInfo

UTC = timezone.utc

def to_utc(dt):
    return dt.astimezone(UTC)

def localize(naive_dt, tz, fold=None):
    z = ZoneInfo(tz)
    if fold is None:
        return naive_dt.replace(tzinfo=z).astimezone(UTC)
    return naive_dt.replace(tzinfo=z, fold=fold).astimezone(UTC)

def hub_interval(s):
    r = datetime.strptime(s, "%Y-%m-%d %H:%M:%S").replace(tzinfo=UTC)
    return [(r - timedelta(seconds=1.5), r + timedelta(seconds=1.5))]

def ops_ber_interval(s, offset_low=-90, offset_high=120):
    r = datetime.strptime(s, "%Y-%m-%d %H:%M")
    # floor-to-minute: recorded clock in [r, r+60)
    # true = recorded - offset; offset in [offset_low, offset_high]
    local_start = r - timedelta(seconds=offset_high)
    local_end = r + timedelta(seconds=60) - timedelta(seconds=offset_low)
    return [(localize(local_start, "Europe/Berlin"), localize(local_end, "Europe/Berlin"))]

def sensor_interval(s):
    r = datetime.strptime(s, "%Y-%m-%d %H:%M:%S").replace(tzinfo=UTC)
    # round to nearest 10s +/-5s, offset +/-5s
    return [(r - timedelta(seconds=10), r + timedelta(seconds=10))]

def vendor_ny_interval(s):
    r = datetime.strptime(s, "%Y-%m-%d %H:%M")
    # floor-to-minute: [r, r+60); offset [-180,180]
    local_start = r - timedelta(seconds=180)
    local_end = r + timedelta(seconds=60) - timedelta(seconds=-180)
    return [(localize(local_start, "America/New_York"), localize(local_end, "America/New_York"))]

def legacy_ber_interval(s):
    r = datetime.strptime(s, "%Y-%m-%d %H:%M")
    # floor-to-minute: [r, r+60); offset [-60,60]
    local_start = r - timedelta(seconds=60)
    local_end = r + timedelta(seconds=60) - timedelta(seconds=-60)
    return [
        (localize(local_start, "Europe/Berlin", fold=0), localize(local_end, "Europe/Berlin", fold=0)),
        (localize(local_start, "Europe/Berlin", fold=1), localize(local_end, "Europe/Berlin", fold=1)),
    ]

def mobile_lon_interval(s, entry_shift_occurrence=True):
    r = datetime.strptime(s, "%Y-%m-%d %H:%M")
    # round-to-minute +/-30s; offset +/-300s => true clock [r-330, r+330]
    local_start = r - timedelta(seconds=330)
    local_end = r + timedelta(seconds=330)
    if entry_shift_occurrence:
        # entry delay [0,300] after occurrence => occurrence may be earlier
        local_start -= timedelta(seconds=300)
    return [(localize(local_start, "Europe/London"), localize(local_end, "Europe/London"))]

    sysname = row["system"]

    if sysname == "SENSOR":
        return sensor_interval(row["recorded"])
    if sysname == "VENDOR-NY":
        return vendor_ny_interval(row["recorded"])
    if sysname == "LEGACY-BER":
        return legacy_ber_interval(row["recorded"])
    if sysname == "MOBILE-LON":
        return mobile_lon_interval(row["recorded"], entry_shift_occurrence=True)
    raise ValueError(sysname)

def duration_intervals(start_intervals, end_intervals, order_constraint=False):
    out = []
    for s in start_intervals:
        for e in end_intervals:
            if order_constraint:
                # require start before end; if intervals overlap, min duration can be 0
                if s[0] > e[1]:
                    continue
                lo = max(0.0, (e[0] - s[1]).total_seconds())
                hi = (e[1] - s[0]).total_seconds()
            else:
                lo = (e[0] - s[1]).total_seconds()
                hi = (e[1] - s[0]).total_seconds()
            out.append((lo, hi))
    out.sort()
    merged = []
    for lo, hi in out:
        if not merged or lo > merged[-1][1]:
            merged.append([lo, hi])
        else:
            merged[-1][1] = max(merged[-1][1], hi)
    return [tuple(x) for x in merged]

def get_intervals(case, activity):
    row = events[(events["case"] == case) & (events["activity"] == activity)]
    if row.empty:
        return []
    return row["intervals"].iloc[0]

def classify_sla(intervals, max_allowed):
    if not intervals:
        return "INDETERMINATE"
    if all(hi <= max_allowed for lo, hi in intervals):
        return "CERTAIN_PASS"
    if all(lo > max_allowed for lo, hi in intervals):
        return "CERTAIN_VIOLATION"
    return "POSSIBLE_VIOLATION"

def classify_profile(intervals, mean, sd, zeta):
    if not intervals:
        return "INDETERMINATE"
    lo_band = mean - zeta * sd
    hi_band = mean + zeta * sd
    if all(hi < lo_band or lo > hi_band for lo, hi in intervals):
        return "CERTAIN_VIOLATION"
    if all(lo >= lo_band and hi <= hi_band for lo, hi in intervals):
        return "CERTAIN_PASS"
    return "POSSIBLE_VIOLATION"

## 6. Evidential Sufficiency

For every base-case `POSSIBLE_VIOLATION`, the minimum additional evidence is generally one of the following:

- **Synchronization / resolution evidence:** tighter NTP/offset telemetry or raw sub-second timestamps that shrink the relevant interval enough to put the whole interval on one side of the SLA/band.
  - T-401 S1, T-401 AI: need OPS-BER offset/resolution narrowed so max 900s or min >900s.
  - T-402 S2, T-402 IC: need OPS-BER/HUB intervals narrowed around 2700s.
  - T-403 S1, T-403 AI, T-403 AC: need VENDOR-NY stronger sync or event-level exact time.
  - T-409 S1, T-409 AI, T-409 IC: need normal NTP diagnostics instead of weak `[-12,+12]` min estimate.

- **Missing activity evidence:**
  - T-404 S1/S2, T-404 AI, T-404 IC: need actual `Investigation_Start` timestamp or a tighter signed work-record duration than `[18,25]` min.
  - T-407 S1, T-407 AI: need receipt acknowledgment or a tighter delivery latency bound; to make certain pass, need upper receipt time high enough that max duration 900s, or to make certain violation need receipt early enough that min duration >900s.

- **DST fold evidence:**
  - T-405 S1, T-405 AI: need logged DST fold or raw UTC timestamps. Ingestion order alone leaves a 12–18 min uncertainty for AlertInvestigation.

- **Semantic evidence:**
  - T-406 S1/S2, T-406 AI, T-406 IC: need to clarify whether the 5-minute mobile entry delay shifts actual occurrence or only record creation. If it only delays record creation, T-406 S2 and T-406 IC become certain pass under the interval model.

## 7. Governance Recommendation

- **UTC storage:** store all canonical event times in UTC with nanosecond or at least millisecond precision. Keep local display times as derived fields only.
- **Raw timestamp retention:** retain original raw timestamp strings, timezone identifiers, DST fold indicators, and display/rounding semantics before normalization.
- **Source offset telemetry:** continuously log per-source NTP/clock offset bounds, not just daily summaries. Treat missing NTP windows as reduced evidence.
- **Resolution metadata:** record whether timestamps are exact, floored, rounded, user-entered, or sensor-completion events, including rounding width.
- **Start/complete semantics:** require distinct lifecycle fields for `Investigation_Start`, `Investigation_Complete`, `Containment_Start`, and `Containment_Complete`. Do not infer one from the other without retaining uncertainty.
- **DST fold:** require fold-aware ingestion (`fold=0/1`) or UTC-only event capture for legal/operational timelines. Never resolve ambiguous local times silently.
- **Causal identifiers:** store alert IDs, receipt IDs, investigation IDs, and containment IDs so that ordering can be enforced by identifiers rather than by possibly skewed clocks.
- **Enforcement policy:** for legal/SLA enforcement, require a predeclared policy for interval overlap. A conservative policy treats any feasible violation as a possible violation; a lenient policy may require the entire feasible interval to violate. State this policy before adjudication.
- **Mobile/user-entered timestamps:** record whether the stamp is entry time, event time, or back-entered time. Capture separate entry-delay bounds.
- **Provenance:** preserve source system, ingestion path, clock-offset bounds, resolution, timezone, DST fold, and inference method for every normalized timestamp.
