```python
import pm4py
from pm4py.objects.process_tree.obj import Operator, ProcessTree

def act(label: str) -> ProcessTree:
    return ProcessTree(label=label)

def tau() -> ProcessTree:
    return ProcessTree(label=None)

def seq(*children):
    return ProcessTree(operator=Operator.SEQUENCE, children=list(children))

def xor(*children):
    return ProcessTree(operator=Operator.XOR, children=list(children))

def par(*children):
    return ProcessTree(operator=Operator.PARALLEL, children=list(children))

def loop(do, redo, exit_):
    return ProcessTree(operator=Operator.LOOP, children=[do, redo, exit_])

def _link_parents(pt: ProcessTree) -> ProcessTree:
    for child in pt.children:
        child.parent = pt
        _link_parents(child)
    return pt
```

```python
import pandas as pd
import pm4py
from pm4py.util import constants as pm4py_constants

CASE_ID = "case:concept:name"
ACTIVITY = "concept:name"
TIMESTAMP = "time:timestamp"
RESOURCE = "org:resource"

FINAL_OUTCOMES = {"Reject Vendor", "Activate Vendor", "Close Onboarding"}

def _events_of(trace_df, activity):
    return trace_df[trace_df[ACTIVITY] == activity]

def _res(trace_df, activity):
    evs = _events_of(trace_df, activity)
    if evs.empty:
        return None
    return evs.iloc[0][RESOURCE]

def _ts(trace_df, activity, occurrence=0):
    evs = _events_of(trace_df, activity)
    if len(evs) <= occurrence:
        return None
    return evs.iloc[occurrence][TIMESTAMP]

def check_resource_segregation(df):
    """R9: returns list of findings per case."""
    findings = []
    for case_id, trace in df.groupby(CASE_ID, sort=False):
        # R9a: Verify Bank Account vs applicable approval(s)
        vba_res = _res(trace, "Verify Bank Account")
        for approval in ("Manager Approval", "Compliance Approval"):
            appr_res = _res(trace, approval)
            if appr_res is None:
                continue  # vacuous: approval not on this route
            if vba_res is None or appr_res is None:
                findings.append((case_id, "R9", f"{approval}: resource attribute missing", "not evaluated"))
                continue
            if vba_res == appr_res:
                findings.append((case_id, "R9", f"four-eyes violated: 'Verify Bank Account' and '{approval}' both by {vba_res}", "fail"))
        # R9b: Capture Master Data vs Activate Vendor
        cmd_res = _res(trace, "Capture Master Data")
        av_res = _res(trace, "Activate Vendor")
        if av_res is not None:
            if cmd_res is None or av_res is None:
                findings.append((case_id, "R9", "Activate Vendor/Capture Master Data: resource attribute missing", "not evaluated"))
            elif cmd_res == av_res:
                findings.append((case_id, "R9", f"four-eyes violated: 'Capture Master Data' and 'Activate Vendor' both by {cmd_res}", "fail"))
    return findings

def check_emergency_policy(df, resolution="gated-exception"):
    """R5/R6. resolution in {'legal-strict', 'gated-exception'}"""
    findings = []
    for case_id, trace in df.groupby(CASE_ID, sort=False):
        trace = trace.sort_values(TIMESTAMP).reset_index(drop=True)
        se = _events_of(trace, "Sanctions Escalation")
        av = _events_of(trace, "Activate Vendor")
        if se.empty or av.empty:
            continue
        olo = _events_of(trace, "Obtain Legal Opinion")
        av_ts = av.iloc[0][TIMESTAMP]
        if olo.empty:
            findings.append((case_id, "R5/R6", "Sanctions Escalation without any Obtain Legal Opinion (R4/R5 breach)", "fail"))
            continue
        olo_ts = olo.iloc[0][TIMESTAMP]
        if olo_ts < av_ts:
            continue  # R5 satisfied: opinion precedes activation
        # OLO after AV: emergency activation
        gap_h = (olo_ts - av_ts).total_seconds() / 3600.0
        if resolution == "legal-strict":
            findings.append((case_id, "R5", f"Activate Vendor precedes Obtain Legal Opinion (R5 breach; remediation required)", "fail"))
        else:  # gated-exception
            authorized = trace.attrs.get("emergency_authorized")  # not in the log: always None
            if gap_h > 24.0:
                findings.append((case_id, "R6", f"emergency opinion arrived {gap_h:.1f}h after activation (> 24h)", "fail"))
            elif authorized is None:
                findings.append((case_id, "R6", f"emergency activation: opinion within 24h ({gap_h:.1f}h) but joint authorization is not recorded in the log", "policy-dependent"))
    return findings

def check_post_activation_bank_change(df):
    """R11."""
    findings = []
    for case_id, trace in df.groupby(CASE_ID, sort=False):
        trace = trace.sort_values(TIMESTAMP).reset_index(drop=True)
        av_ts = _ts(trace, "Activate Vendor")
        if av_ts is None:
            continue
        cbd = _events_of(trace, "Change Bank Details")
        cbd = cbd[cbd[TIMESTAMP] > av_ts]
        if cbd.empty:
            continue
        rfp_ts = _ts(trace, "Release First Payment")
        for _, ev in cbd.iterrows():
            cbd_ts = ev[TIMESTAMP]
            sv = _events_of(trace, "Suspend Vendor")
            sv = sv[sv[TIMESTAMP] > cbd_ts]
            if sv.empty:
                findings.append((case_id, "R11", "post-activation Change Bank Details without subsequent Suspend Vendor", "fail"))
            if rfp_ts is not None and rfp_ts > cbd_ts:
                vba = _events_of(trace, "Verify Bank Account")
                recheck = vba[(vba[TIMESTAMP] > cbd_ts) & (vba[TIMESTAMP] < rfp_ts)]
                if recheck.empty:
                    findings.append((case_id, "R11", "no re-verification (Verify Bank Account) between Change Bank Details and Release First Payment", "fail"))
    return findings

def check_clarification_deadline(df):
    """R3 residual: clarifications must be answered before the first final outcome."""
    findings = []
    for case_id, trace in df.groupby(CASE_ID, sort=False):
        trace = trace.sort_values(TIMESTAMP).reset_index(drop=True)
        outcome_ts = [trace.iloc[i][TIMESTAMP] for i in range(len(trace))
                      if trace.iloc[i][ACTIVITY] in FINAL_OUTCOMES]
        outcome_ts = min(outcome_ts) if outcome_ts else None
        open_rc = False
        for _, ev in trace.iterrows():
            if ev[ACTIVITY] == "Request Clarification":
                if outcome_ts is not None and ev[TIMESTAMP] > outcome_ts:
                    findings.append((case_id, "R3", "Request Clarification raised after a final outcome", "fail"))
                open_rc = True
            elif ev[ACTIVITY] == "Receive Clarification":
                open_rc = False
            elif outcome_ts is not None and ev[TIMESTAMP] >= outcome_ts and open_rc:
                findings.append((case_id, "R3", "final outcome reached with an unanswered Request Clarification", "fail"))
                open_rc = False
    return findings
```

```python
from __future__ import annotations

import pandas as pd
import pm4py

# ---------------------------------------------------------------------------
# Schema contract
# ---------------------------------------------------------------------------
CASE_ID   = "case:concept:name"
ACTIVITY  = "concept:name"
TIMESTAMP = "time:timestamp"
RESOURCE  = "org:resource"          # fall back to "Resource" if needed

FINAL_OUTCOMES = {"Reject Vendor", "Activate Vendor", "Close Onboarding"}
HOURS = 3600.0

def _ordered(df: pd.DataFrame) -> pd.DataFrame:
    """One case, sorted by timestamp (stable: log order breaks ties)."""
    return df.sort_values(TIMESTAMP, kind="stable").reset_index(drop=True)

def _events(case_df: pd.DataFrame, activity: str) -> pd.DataFrame:
    return case_df[case_df[ACTIVITY] == activity]

def _first(case_df: pd.DataFrame, activity: str):
    ev = _events(case_df, activity)
    return None if ev.empty else ev.iloc[0]

def _res(case_df: pd.DataFrame, activity: str):
    ev = _first(case_df, activity)
    if ev is None:
        return None
    r = ev.get(RESOURCE)
    return None if pd.isna(r) else r
```

```python
def check_segregation_of_duties(df: pd.DataFrame) -> list[dict]:
    """R9 — activity-only models cannot express this: needs org:resource.

    Missing-resource policy: if either event of a pair lacks org:resource the
    check is reported as 'not evaluated' and the case is routed to manual
    review (fail-closed). It never silently passes.
    """
    findings = []
    for case, g in df.groupby(CASE_ID, sort=False):
        trace = _ordered(g)
        vba = _first(trace, "Verify Bank Account")
        for approval in ("Manager Approval", "Compliance Approval"):
            appr = _first(trace, approval)
            if appr is None:
                continue                       # vacuous: route not taken
            if vba is None:
                continue                       # structural layer already flags this
            r_vba, r_appr = _res(trace, "Verify Bank Account"), _res(trace, approval)
            if r_vba is None or r_appr is None:
                findings.append(_finding(case, "R9", "resource",
                    "not evaluated",
                    f"org:resource missing on Verify Bank Account or {approval} "
                    "(fail-closed: manual review required)"))
            elif r_vba == r_appr:
                findings.append(_finding(case, "R9", "resource", "fail",
                    f"four-eyes violated: Verify Bank Account and {approval} "
                    f"both performed by '{r_vba}'",
                    f"Verify Bank Account@{vba[TIMESTAMP]} / {approval}@{appr[TIMESTAMP]}"))
        cmd, av = _first(trace, "Capture Master Data"), _first(trace, "Activate Vendor")
        if av is not None and cmd is not None:
            r_cmd, r_av = _res(trace, "Capture Master Data"), _res(trace, "Activate Vendor")
            if r_cmd is None or r_av is None:
                findings.append(_finding(case, "R9", "resource", "not evaluated",

                findings.append(_finding(case, "R9", "resource", "fail",
                    f"four-eyes violated: Capture Master Data and Activate Vendor "
                    f"both performed by '{r_cmd}'"))
```

```python
    """R5 + R6 — the conflict resolved in §1.
```

Section 5 — dict + orientation explanation + overlap/disagreement.

Orientation conventions:
- `equivalence`: symmetric; both activities occur equally often in every case.
- `always_after (A,B)`: forward — every occurrence of A is eventually followed by 1 B.
- `always_before (A,B)`: backward — B only occurs if A occurred earlier (A is always before B).
- `never_together`: symmetric exclusion.
- `directly_follows (A,B)`: A may be immediately followed by B.
- `activ_freq`: allowed occurrence-count sets.

# 1. Policy Interpretation and Conflict Resolution

**Material assumptions**

- **A1 — Case completeness.** Every trace is a closed case containing `Close Onboarding`; prefix conformance for running cases is out of scope.
- **A2 — Alphabet partition.** The 14 *backbone* activities (request, three checks, clarification cycle, classification, three routes, payment, close) are modeled procedurally. `Sanctions Escalation`, `Obtain Legal Opinion`, `Change Bank Details`, `Suspend Vendor` are *policy overlays*: the procedural and Log-Skeleton layers are evaluated on the projected trace, while DECLARE, the temporal profile, and the policy layer see the full trace. This prevents false structural failures caused by discretionary escalation placement, at the price of the tree being silent about overlays (defense-in-depth via DECLARE `precedence`/`response`).
- **A3 — Risk class is not an attribute.** The route taken (`Manager Approval` vs. `Enhanced Due Diligence`+`Compliance Approval`) is the only observable proxy. "Applicable approval" in R9 = the approval on the route actually taken; vacuous in rejected cases.
- **A4 — R3 "final outcome"** = the first of `Reject Vendor`, `Activate Vendor`, `Close Onboarding`; every clarification must be closed before it, and no new request may follow it.
- **A5 — R6's 24-hour window is anchored at the emergency `Activate Vendor`** ("the opinion must then arrive"). An escalation-anchored reading is stricter and noted as a variant.
- **A6 — R6's joint authorization and emergency flag are not in the log** and can therefore never be *confirmed*, only flagged. Any `Activate Vendor`-before-`Obtain Legal Opinion` trace is at best *policy-dependent*.
- **A7 — Resources and time.** Each event carries exactly one `org:resource`; timestamps are trustworthy, case-relative hours; same-hour ties are broken by log order.
- **A8 — At most one bank-change cycle per case**, hence `Verify Bank Account` occurs 1–2 times in the Log Skeleton (initial verification plus the R11 re-verification).
- **A9 — R12 deviations are flags, never hard rejects** (explicit Compliance instruction); `zeta = 2.0`.
- **A10 — "Several clarification cycles" = an unbounded loop**; therefore no finite `activ_freq` set is defined for the clarification pair (see §5, §9).
- **A11 — Route-exclusivity is taken literally:** `Manager Approval` only on the low-risk route, `Enhanced Due Diligence`+`Compliance Approval` only on the high-risk route; `Release First Payment` is mandatory before `Close Onboarding` on accepted routes (backbone step 5 read literally).

**The R5/R6 conflict.** R5 ("under all circumstances") and R6 ("may occur before, when jointly authorized") are mutually inconsistent as stated: R5 forbids exactly what R6 permits. They must not be silently merged.

- **Resolution A — Legal-strict (R5 wins absolutely).** `Obtain Legal Opinion` must strictly precede `Activate Vendor` whenever `Sanctions Escalation` occurred. Emergency activations are violations; the late opinion is remediation, not conformance. *Risk:* operationally rigid — genuine emergencies are either blocked or pushed into unrecorded off-process workarounds, which is worse for auditability than a controlled exception.
- **Resolution B — Gated exception (implemented).** R5 is the default; R6 carves a narrow exception: `Activate Vendor` before `Obtain Legal Opinion` is tolerated **only** when (i) joint authorization exists outside the log *and* (ii) the opinion arrives within 24 hours of the emergency activation. Condition (ii) is enforced as a hard timestamp gate; condition (i) is unverifiable, so every activation of the exception is flagged `policy-dependent` and routed to a compliance officer — nothing passes silently.
- *(Resolution C, considered and rejected: split lifecycles — a standard model with hard R5 and a separate deviation process for emergencies. It fragments case history and pushes emergencies out of the audited model entirely.)*

**Choice: Resolution B**, because the log's silence about authorization is a data gap that should surface as a review task, not be silently resolved in either direction. **Compliance risk of the choice:** detection-after-the-fact rather than prevention; authorization can be asserted post hoc; if the off-log evidence is weak, a regulator may re-characterize every such case as an R5 breach. Mitigations: the 24-hour deadline is a hard fail condition, the exception is never auto-passed, and every activation is auditable.

    """Visible activity leaf."""

    """Silent activity (tau)."""
    return ProcessTree(label=None)


def _seq(*children: ProcessTree) -> ProcessTree:
    return ProcessTree(operator=Operator.SEQUENCE, children=list(children))


def _xor(*children: ProcessTree) -> ProcessTree:
    return ProcessTree(operator=Operator.XOR, children=list(children))

    return ProcessTree(operator=Operator.PARALLEL, children=list(children))


def _loop(do: ProcessTree, redo: ProcessTree, exit_: ProcessTree) -> ProcessTree:
    return ProcessTree(operator=Operator.LOOP, children=[do, redo, exit_])


def _link_parents(pt: ProcessTree) -> ProcessTree:
    for child in pt.children:
        child.parent = pt
        _link_parents(child)
    return pt
