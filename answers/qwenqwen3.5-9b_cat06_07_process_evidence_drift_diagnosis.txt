```python
import pandas as pd
import pm4py

def audit_monthly_pipeline(month, events_df):
    # 1. Normalization: Rename diagnosis labels to standardize features
    events_df['activity'] = events_df['activity'].replace(
        {'Diagnostic Assessment Completed': 'Initial Diagnosis Completed'}
    )
    
    # 2. Duplicate Handling: Remove exact duplicates based on case_id + timestamp
    # Assumes 'case_id' and 'event_timestamp' columns exist
    events_df = events_df.drop_duplicates(subset=['case_id', 'event_timestamp'])
    
    # 3. Cohort Creation
    # Event-Calendar: All events in the month regardless of case start
    event_cohort = pm4py.filter_time_range(events_df, from_date=month, to_date=month)
    
    # Case-Start: Only cases where 'Case Created' happened in the month
    case_cohort = events_df[events_df['activity'] == 'Case Created'][
        events_df['event_timestamp'].dt.to_period('M') == month
    ]
    
    # 4. Process Mining
    # Filter to only include valid cases for DFG discovery
    filtered_dfg = pm4py.discover_dfg(case_cohort)
    
    # Performance DFG to check edge times
    perf_dfg = pm4py.discover_performance_dfg(case_cohort)
    
    # 5. Variant Extraction
    variants = pm4py.get_variants(case_cohort)
    
    # 6. Temporal Features
    # Extract features for each variant to check for missingness
    features_df = pm4py.extract_temporal_features_dataframe(case_cohort)
    
    # 7. Summary Audit
    summary = {
        'month': month,
        'case_count': len(case_cohort),
        'event_count': len(event_cohort),
        'variants': list(variants.keys()),
        'dfg_depth': len(filtered_dfg),
        'feature_missingness': features_df.isnull().sum().sum() / features_df.size,
        'edge_stats': perf_dfg.describe()
    }
    
    return summary
```
