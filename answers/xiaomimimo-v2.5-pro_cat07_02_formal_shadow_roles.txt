# 5. PM4Py role-discovery and similarity code

```python
import pm4py
import pandas as pd

# 1. Load and format the dataframe
# Assuming the raw data is in a pandas DataFrame called 'df'
df = pm4py.format_dataframe(
    df, 
    case_id='case:concept:name', 
    activity_key='concept:name', 
    timestamp_key='time:timestamp',
    resource_key='org:resource'
)

# To avoid over-interpreting activity-frequency clusters as formal authority, 
# we can filter out temporary substitution events for the initial baseline role discovery.
df_permanent = df[df['temporary_substitution'] == False]

# 2. Run role discovery (uses resource profiles / similarity under the hood)
# This will cluster resources based on what activities they perform.
roles = pm4py.discover_organizational_roles(df_permanent)
print("Discovered Roles:", roles)

# 3. Run activity-based resource similarity to find out how similar resources are
# even if they are in different formal clusters (e.g., finding R07_Giorgio's dual nature).
similarity_matrix = pm4py.discover_activity_based_resource_similarity(df_permanent)
print("Activity Similarity Matrix:\n", similarity_matrix)

# 4. Compute handover-of-work and working-together networks
# Handover network shows who passes work to whom (e.g., R13_Mateo passing to R05_Erik)
handover_network = pm4py.discover_handover_of_work_network(df)

# Working-together network shows who works sequentially on the same cases
working_together_network = pm4py.discover_working_together_network(df)

# Subcontracting network can also reveal if senior agents (R07) are pulling work 
# from junior agents (R01/R02) during complex paths.
subcontracting_network = pm4py.discover_subcontracting_network(df)

# 5. Join role-discovery output with complexity and substitution indicators
# Extract median complexity per resource from the event log
resource_complexity = df.groupby('org:resource')['case_complexity'].median().reset_index()
resource_complexity.columns = ['resource', 'median_complexity']

# Map the discovered roles back to the dataframe to analyze risk exposure per role
# (Illustrative logic to bind role clusters with attributes)
for role_name, role_resources in roles.items():
    role_df = df[df['org:resource'].isin(role_resources)]
    avg_complexity = role_df['case_complexity'].mean()
    substitution_count = role_df[role_df['temporary_substitution'] == True].shape[0]
    print(f"Role: {role_name}")
    print(f" - Members: {role_resources}")
    print(f" - Average Complexity Handled: {avg_complexity:.2f}")
    print(f" - Events Handled during Substitution: {substitution_count}")

# 6. Avoiding over-interpretation: 
# We do not use roles = pm4py.discover_organizational_roles(df) as the sole source of truth.
# Instead, we manually inspect 'similarity_matrix' for pairs with 0.8 < sim < 1.0 
# (like R03_Chen and R04_Dina) because these indicate shared core activities but 
# divergent peripheral activities or complexity scopes.
```
