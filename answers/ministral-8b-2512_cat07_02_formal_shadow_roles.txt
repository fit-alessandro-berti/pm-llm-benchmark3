# 5. PM4Py role-discovery and similarity code

```python
import pandas as pd
import pm4py

# --- Step 1: Load and format the event log ---
def load_and_format_event_log(event_log_df):
    """
    Convert a pandas DataFrame into a PM4Py event log.
    Assumes columns: case:concept:name, concept:name, time:timestamp, org:resource,
    case_complexity, customer_segment, product_line, temporary_substitution, escalation_level.
    """
    # Filter out automation events if needed (e.g., R14_Nora)
    human_events = event_log_df[event_log_df['org:resource'] != 'R14_Nora']

    # Format as PM4Py event log
    event_log = pm4py.format_dataframe(
        human_events,
        case_id='case:concept:name',
        activity='concept:name',
        timestamp='time:timestamp',
        resource='org:resource'
    )
    return event_log

# --- Step 2: Discover organizational roles ---
def discover_roles(event_log, min_support=0.05, max_depth=3):
    """
    Discover roles using activity-based similarity and case complexity.
    """
    # Discover roles based on activity similarity
    roles = pm4py.discover_organizational_roles(
        event_log,
        variant_filtering=True,
        min_support=min_support,
        max_depth=max_depth
    )

    # Enrich roles with case complexity and substitutions
    enriched_roles = []
    for role in roles:
        role_df = event_log[event_log['resource'] == role['resource']]
        role_stats = {
            'role': role['resource'],
            'case_complexity': role_df['case_complexity'].median(),
            'substitution_episodes': len(role_df[role_df['temporary_substitution'] == 1]),
            'critical_cases': len(role_df[role_df['escalation_level'] == 'critical'])
        }
        enriched_roles.append(role_stats)

    return pd.DataFrame(enriched_roles)

# --- Step 3: Compute activity-based resource similarity ---
def compute_resource_similarity(event_log):
    """
    Compute cosine similarity between resources based on activity profiles.
    """
    similarity_matrix = pm4py.discover_activity_based_resource_similarity(
        event_log,
        similarity_threshold=0.7,
        variant_filtering=True
    )
    return similarity_matrix

# --- Step 4: Discover handover-of-work and working-together networks ---
def discover_networks(event_log):
    """
    Discover handover-of-work and working-together networks.
    """
    # Handover-of-work network (who hands off to whom)
    handover_network = pm4py.discover_handover_of_work_network(
        event_log,
        variant_filtering=True
    )

    # Working-together network (who collaborates on the same case)
    working_network = pm4py.discover_working_together_network(
        event_log,
        variant_filtering=True
    )

    return handover_network, working_network

# --- Step 5: Join role discovery with case complexity ---
def join_with_case_complexity(event_log, case_complexity_df):
    """
    Join discovered roles with case complexity and substitution data.
    """
    # Merge case complexity data
    enriched_log = event_log.merge(
        case_complexity_df,
        left_on='resource',
        right_on='Resource',
        how='left'
    )

    # Filter for substitutions
    enriched_log['substitution'] = enriched_log.apply(
        lambda x: 1 if x['temporary_substitution'] == 1 else 0,
        axis=1
    )

    return enriched_log

# --- Step 6: Avoid over-interpreting frequency clusters ---
def analyze_role_authority(event_log, authority_df):
    """
    Analyze role authority by joining with approval/decision data.
    """
    # Example: Join with approval authority (e.g., A9 Supervisor Approval)
    authority_log = event_log.merge(
        authority_df,
        left_on='resource',
        right_on='Resource',
        how='left'
    )

    # Filter for approval events
    approval_events = authority_log[authority_log['concept:name'] == 'A9 Supervisor Approval']

    # Analyze authority distribution
    authority_distribution = approval_events['resource'].value_counts()
    return authority_distribution

# --- Example Usage ---
if __name__ == "__main__":
    # Load data (example)
    event_log_df = pd.read_csv("event_log.csv")  # Assume this is your formatted DataFrame
    case_complexity_df = pd.read_csv("case_complexity.csv")
    authority_df = pd.read_csv("approval_authority.csv")

    # Step 1: Format event log
    event_log = load_and_format_event_log(event_log_df)

    # Step 2: Discover roles
    roles_df = discover_roles(event_log)

    # Step 3: Compute similarity
    similarity_matrix = compute_resource_similarity(event_log)
    print("Resource Similarity Matrix:")
    print(similarity_matrix)

    # Step 4: Discover networks
    handover_network, working_network = discover_networks(event_log)
    print("Handover Network:")
    print(handover_network)

    # Step 5: Join with case complexity
    enriched_log = join_with_case_complexity(event_log, case_complexity_df)

    # Step 6: Analyze authority
    authority_distribution = analyze_role_authority(event_log, authority_df)
    print("Approval Authority Distribution:")
    print(authority_distribution)
```

### **Key PM4Py Functions Used:**
1. `pm4py.format_dataframe`: Converts pandas DataFrame to PM4Py event log.
2. `pm4py.discover_organizational_roles`: Clusters resources by activity similarity.
3. `pm4py.discover_activity_based_resource_similarity`: Computes cosine similarity between resources.
4. `pm4py.discover_handover_of_work_network`: Identifies handoff patterns between resources.
5. `pm4py.discover_working_together_network`: Identifies collaboration patterns.
6. **Custom Joins**: Merge with case complexity, substitutions, and authority data to avoid misclassification.

### **Avoiding Misclassification:**
- **Case Complexity**: Joined with `case_complexity_df` to ensure roles are not misclassified by frequency alone.
- **Substitution Data**: Used to identify shadow roles (e.g., Giorgio during Fatima’s absence).
- **Authority Data**: Separate from activity frequency to avoid treating "doers" as "authorities."
