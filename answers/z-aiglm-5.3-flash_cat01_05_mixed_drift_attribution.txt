def validate_data(monthly, dfg):
    issues = []
    # months parseable, unique, ordered
    m = pd.PeriodIndex(monthly["month"], freq="M")
    if not (m.is_monotonic_increasing and m.is_unique):
        issues.append("monthly index not strictly increasing")
    # bounded proportion-like columns
    bounds = {"cases": (0, np.inf), "high_complexity_pct": (0,100),
              "supplier_z_pct": (0,100), "fast_variant_pct": (0,100),
              "rework_variant_pct": (0,100),
              "raw_model_fitness": (0,1), "mapped_model_fitness": (0,1),
              "missing_resource_pct": (0,100), "duplicate_open_case_pct": (0,100)}
    for col,(lo,hi) in bounds.items():
        outside = ~monthly[col].between(lo,hi)
        if outside.any():
            issues.append(f"{col}: out-of-range rows {outside.sum()}")
    # structural expectations
    pre = monthly[monthly["month"].isin(["2026-01","2026-02"])]
    if (pre["duplicate_open_case_pct"] > 0).any():
        issues.append("pre-migration duplicates present (expected 0)")
    if (monthly["mapped_model_fitness"] < monthly["raw_model_fitness"] - 1e-9).any():
        issues.append("mapped fitness below raw fitness: mapping definitions suspect")
    # DFG fragment checks
    if not dfg[EDGE_COLS].apply(lambda s: s.between(0,100)).all().all():
        issues.append("DFG rates outside [0,100]")
    if (dfg["reg_elig"] < 99).any():   # every case should leave Register
        issues.append("reg_elig materially below 100 - fragment or dropped cases")
    print("VALIDATION:", "OK" if not issues else issues)
    return issues

def dfg_distances(dfg, ref_month="2026-01"):
    V = dfg.set_index("month")[EDGE_COLS].astype(float)
    l1_ref = V.sub(V.loc[ref_month]).abs().sum(axis=1)
    l1_adj = V.diff().abs().sum(axis=1)     # NaN at first row
    cos = {}
    for mo in V.index:
        a,b = V.loc[mo],V.loc[ref_month]
        num=float(np.dot(a,b)); den=float(np.linalg.norm(a)*np.linalg.norm(b))
        cos[mo]= round(num/den,4) if den else np.nan
    out=pd.DataFrame({"L1_vs_ref":l1_ref.round(1),
                      "L1_adjacent":l1_adj.round(1),
                      "cosine_vs_ref":pd.Series(cos)})
    return out

def format_event_table(raw_df, case_col="case_id", act_col="activity",
                       ts_col="timestamp", res_col="resource"):
    """Format the raw event table with pm4py WITHOUT mutating the input."""
    df = raw_df.copy()
    df[ts_col] = pd.to_datetime(df[ts_col])
    formatted = pm4py.format_dataframe(
        df, case_id=case_col, activity_key=act_col, timestamp_key=ts_col)
    if res_col in formatted.columns:
        formatted = formatted.rename(columns={res_col: "org:resource"})
    return formatted

def normalized_view(event_df, label_map=LABEL_MAP,
                    technical_activity=TECHNICAL_ACTIVITY):
    """Return a NORMALIZED COPY (labels mapped, technical duplicates removed).
    The original dataframe is never overwritten."""
    df = event_df.copy()
    if "concept:name" not in df.columns:
        raise ValueError("pass a pm4py-formatted dataframe")
    df["concept:name"] = df["concept:name"].replace(label_map)
    dup_mask = ((df["concept:name"] == technical_activity)
                & df.duplicated(
                    subset=["case:concept:name", "concept:name",
                            "time:timestamp"], keep="first"))
    return df.loc[~dup_mask].reset_index(drop=True)

def _as_log(obj):
    """Accept a DataFrame or an EventLog; return an object pm4py can mine."""
    return obj if not isinstance(obj, pd.DataFrame) \
        else pm4py.convert_to_event_log(obj)

def slice_window(log_or_df, start, end, mode="traces_contained"):
    """Time-window filter. start/end are 'YYYY-MM-DD HH:MM:SS'.
    mode: 'events' | 'traces' | 'traces_contained'."""
    return pm4py.filter_time_range(_as_log(log_or_df), str(start), str(end),
                                   mode=mode, retain=True)

def _month_bounds(year, month):
    start = pd.Timestamp(year=year, month=month, day=1)
    end = start + pd.offsets.MonthBegin(1)
    return start.strftime("%Y-%m-%d %H:%M:%S"), end.strftime("%Y-%m-%d %H:%M:%S")

def monthly_dfg(year, month, event_df, normalized=True):
    """Discover the DFG for one calendar month on the chosen view."""
    s,e = _month_bounds(year, month)
    view = normalized_view(event_df) if normalized else event_df
    log = slice_window(view, s, e)
    dfg, start_acts, end_acts = pm4py.discover_dfg(log)
    return dfg, start_acts, end_acts

def monthly_variants(year, month, event_df, normalized=True):
    s,e = _month_bounds(year, month)
    view = normalized_view(event_df) if normalized else event_df
    variants = pm4py.get_variants(slice_window(view, s, e))
    counts = {}
    for key, val in variants.items():
        counts[key] = val if isinstance(val, int) else len(val)
    return counts

def temporal_profile_reference(event_df, normalized=True):
    view = normalized_view(event_df) if normalized else event_df
    return pm4py.discover_temporal_profile(_as_log(view))

def temporal_profile_anomalies(ref_profile, new_profile, z_threshold=3.0):
    """Compare [mean, stdev] per activity pair against the reference."""
    out=[]
    for pair,stats in new_profile.items():
        if pair in ref_profile:
            mu,sd = ref_profile[pair][0], ref_profile[pair][1]
            if sd and sd>0:
                zz=(stats[0]-mu)/sd
                if abs(zz)>z_threshold:
                    out.append({"pair":pair,"z":round(zz,2),
                                "ref_mean":mu,"new_mean":stats[0]})
    return sorted(out,key=lambda d:-abs(d["z"]))

def discover_log_skeleton_view(event_df, year=None, month=None,
                               noise_threshold=0.05, normalized=True):
    view = normalized_view(event_df) if normalized else event_df
    if year and month:
        s,e=_month_bounds(year,month)
        view = slice_window(view,s,e)
    return pm4py.discover_log_skeleton(_as_log(view),
                                       noise_threshold=noise_threshold)

def skeleton_deviations(event_df, skeleton):
    diag = pm4py.conformance_log_skeleton(_as_log(event_df), skeleton)
    n_bad = sum(1 for d in diag if d.get("is_fit") is False)
    return {"cases_checked":len(diag),"non_fitting":n_bad,
            "deviating_fraction": round(n_bad/max(len(diag),1),3)}

def discover_declare_view(event_df, noise_threshold=0.05, normalized=True):
    """DECLARE discovery (available in recent pm4py releases)."""
    view = normalized_view(event_df) if normalized else event_df
    try:
        return pm4py.discover_declare(_as_log(view),
                                      noise_threshold=noise_threshold)
    except AttributeError:
        print("pm4py.discover_declare unavailable in this version "
              "- use log skeleton instead.")
        return None

def raw_vs_mapped_conformance(formatted_raw_df, petri_net, im, fm):
    """Token-based fitness BEFORE and AFTER normalisation, never overwriting."""
    fit_raw = pm4py.fitness_token_based_replay(_as_log(formatted_raw_df),
                                               petri_net, im, fm)
    fit_map = pm4py.fitness_token_based_replay(
        _as_log(normalized_view(formatted_raw_df)), petri_net, im, fm)
    return {"raw":fit_raw,"mapped":fit_map,
            "recovery":fit_map["log_fitness"]-fit_raw["log_fitness"]}

### 3.4 Candidate change points per artifact

- **Logging drift:** (a) **2026-03-01** — definite (labels 911, missingness 1.054.8%, dup 04%, raw fitness 0.37 in one month); (b) **~2026-10-01** — suspected second semantic event: SrA rate triples (+450%) while seasonality alone predicts roughly a doubling; a recording-practice shift is plausible (verify via documentation audit).
- **Behavioral (control-flow) drift:** (a) **early 2026-05** — rework loop step (1224/100) preceding any internal staffing change, supplier-caused but behavioral in artifact space; (b) **2026-07-01** — priority scheduling (ordering-level behavior; predicted signature invisible in aggregates; dispersion hints only).
- **Performance drift:** (a) **2026-05** — first measurable cycle inflation (+17% median, supply-mediated); (b) **2026-06-15  07** — acceleration into the worst regime (median 40.844.245.0; p90 95104110) aligned with the assessor exit; (c) **2026-12-05** — partial de-escalation.

### 3.5 One global change point or many?

The data **reject a single global change point**: (i) artifact divergence — label/metadata metrics step on Mar-1 while flow metrics move from May; (ii) divergent reversal dynamics — rework and supplier share recover after September, case mix stays high through November, missingness decays slowly; (iii) distinct slopes — the fitness plateau versus the accelerating cycle curve. The coherent reading is **four to six mechanism-specific change points** (Mar-1; May-5; Jun-15; Jul-1; Oct-1; Dec-5). Formal confirmation (CUSUM/PELT per metric family on event-level daily series) is still outstanding and should be run before these dates are treated as statistical estimates rather than event-anchored hypotheses.

## 6. Segmentation plan

| segment level | variables | purpose | minimum support rule | comparisons enabled | risk of conditioning bias |
|---|---|---|---|---|---|
| L0 epoch | Calendar window mapped to regimes: Jan–Feb baseline; Mar-1–May-4 post-migration; May-5–Jun-14 supplier shock; Jun-15–Sep-30 capacity+priority; Oct-1–Nov-30 season; Dec+ intervention | Isolate regimes; anchor change points | Full census of each epoch | Cross-regime artifact deltas | Epoch boundaries blur — announced dates precede full effect (transition lag) |
| L1 supplier channel | supplier_z membership | Decouple supply-side from organization-side drift | 150 completed cases per epoch×channel; pool adjacent epochs below that | Channel-stratified DFG/variant/conformance; DiD around May-5 | Supplier serves non-random customers/regions  selection confounding |
| L2 priority × complexity | Recorded queue priority; audited complexity band | Expose delay redistribution and SrA logic | 100 cases per cell-month, else two-month pooling | Waiting quantiles, variant mixes, SrA propensity by cell | Historic complexity tags unreliable; analyzing only tagged rows imports informative-missingness bias |
| L3 delivery unit | Team / role / region (where resource present) | Localize capacity effects | 30 cases and 3 active assessors per team-month, else quarterly aggregation | Throughput, handovers, wait by team | MNAR resources post-migration  complete-case bias; cross-region case-mix differences |
| L4 record-quality cohort | Legacy-label residue vs migrated-label cases | Quarantine logging contamination from behavioral reads | Census of affected cases | Stability of artifacts before/after cleanup | Cohort membership coincides with migration timing — conflates era with channel |

**Primary monitoring population:** the mapped, de-duplicated log restricted to **non-Supplier-Z cases, with priority × complexity strata preserved as separate panels**. Rationale: it carries the bulk of sustained volume (~55–75% of monthly cases, comfortably clearing support rules), it is uncontaminated by the largest exogenous shock (supplier), it maximizes signal-to-noise for staffing-, priority-, and seasonality-monitors, and it avoids the selection bias introduced by filtering on resource completeness. The supplier arm must still run as a **parallel dedicated panel** (not discarded) — otherwise recurring supplier incidents become invisible in the sentinel series.

No universal threshold is used: count-type metrics get Poisson/Wilson logic, duration metrics get quantile/shift tests, distributional metrics get permutation nulls.

**Seasonality handling:** reference distributions are built per calendar month (YoY once 24 months exist), STL-adjusted series feed break detectors, and known-contaminated windows (Mar-1–Apr-30; supplier shock core) are quarantined from reference estimation.

**Multiple testing:** monitors are tiered into critical and watch families; within each family, monthly p-values pass Benjamini–Hochberg FDR (q=0.05), page-level alerts additionally require Holm correction; an advisory alert must replicate in the next window before paging, unless classified critical.

def validate_data(monthly, dfg):
    issues = []
    m = pd.PeriodIndex(monthly["month"], freq="M")
    if not (m.is_monotonic_increasing and m.is_unique):
        issues.append("monthly index not strictly increasing/unique")
    bounds = {"cases": (0, np.inf), "high_complexity_pct": (0, 100),
              "supplier_z_pct": (0, 100), "fast_variant_pct": (0, 100),
              "rework_variant_pct": (0, 100), "raw_model_fitness": (0, 1),
              "mapped_model_fitness": (0, 1), "missing_resource_pct": (0, 100),
              "duplicate_open_case_pct": (0, 100)}
    for col, (lo, hi) in bounds.items():
        n_out = (~monthly[col].between(lo, hi)).sum()
        if n_out:
            issues.append(f"{col}: {n_out} out-of-range rows")
    pre = monthly[monthly["month"].isin(["2026-01", "2026-02"])]
    if (pre["duplicate_open_case_pct"] > 0).any():
        issues.append("pre-migration duplicates present (expected 0)")
    if (monthly["mapped_model_fitness"] <
            monthly["raw_model_fitness"] - 1e-9).any():
        issues.append("mapped fitness below raw: mapping definitions suspect")
    if not dfg[EDGE_COLS].apply(lambda s: s.between(0, 100)).all().all():
        issues.append("DFG rates outside [0,100]")
    if (dfg["reg_elig"] < 99).any():
        issues.append("reg_elig below 100: fragment omits cases or dropped starts")
    print("VALIDATION:", "OK" if not issues else issues)
    return issues

def baseline_changes(monthly, baseline_months=("2026-01", "2026-02")):
    s = monthly.set_index("month")
    base = s.loc[list(baseline_months)].mean(numeric_only=True)
    rows = []
    cols = ["cases", "high_complexity_pct", "supplier_z_pct", "fast_variant_pct",
            "rework_variant_pct", "median_cycle_h", "p90_cycle_h",
            "raw_model_fitness", "mapped_model_fitness", "missing_resource_pct"]
    for mo, row in s.iterrows():
        for col in cols:
            rel = (row[col] - base[col]) / base[col] * 100 if base[col] else np.nan
            rows.append({"month": mo, "metric": col, "value": row[col],
                         "baseline": round(base[col], 3),
                         "delta_abs": round(row[col] - base[col], 3),
                         "delta_rel_pct": round(rel, 1)})
    return pd.DataFrame(rows)

def dfg_distances(dfg, ref_month="2026-01"):
    V = dfg.set_index("month")[EDGE_COLS].astype(float)
    l1_ref = V.sub(V.loc[ref_month]).abs().sum(axis=1).round(1)
    l1_adj = V.diff().abs().sum(axis=1).round(1)
    cos = {}
    for mo in V.index:
        a, b = V.loc[mo].values, V.loc[ref_month].values
        den = float(np.linalg.norm(a) * np.linalg.norm(b))
        cos[mo] = round(float(np.dot(a, b)) / den, 4) if den else np.nan
    return pd.DataFrame({"L1_vs_ref": l1_ref, "L1_adjacent": l1_adj,
                         "cosine_vs_ref": pd.Series(cos)})

                                 - monthly["raw_model_fitness"]))

    "median_cycle_h": 0.10, "p90_cycle_h": 0.12,
    "rework_variant_pct": 0.30, "high_complexity_pct": 0.30,
    "supplier_z_pct": 0.50, "missing_resource_pct": 1.00,

def flag_change_points(monthly):
    s = monthly.set_index("month")
    base = s.loc[["2026-01", "2026-02"]].mean(numeric_only=True)
    flags, active = {}, set()
    for mo, row in s.iterrows():
        fired = []
        for col, thr in FLAG_THRESHOLDS.items():
            rel = abs((row[col] - base[col]) / base[col]) if base[col] else 0.0
            if rel >= thr and col not in active:
                fired.append(col); active.add(col)
            elif rel < thr * 0.8 and col in active:       # hysteresis reset
                active.discard(col)
        if fired:
            flags[mo] = fired
    return flags

import pandas as pd

LABEL_MAP = {"Eligibility Screen": "Validate Eligibility"}
TECHNICAL_ACTIVITY = "Open Case"

def format_event_table(raw_df, case_col="case_id", act_col="activity",
                       ts_col="timestamp", res_col="resource"):
    """Format a raw event table with pm4py WITHOUT mutating the input."""
    df = raw_df.copy()
    df[ts_col] = pd.to_datetime(df[ts_col])
    formatted = pm4py.format_dataframe(
        df, case_id=case_col, activity_key=act_col, timestamp_key=ts_col)
    if res_col in formatted.columns:
        formatted = formatted.rename(columns={res_col: "org:resource"})
    return formatted

def normalized_view(event_df, label_map=LABEL_MAP,
                    technical_activity=TECHNICAL_ACTIVITY):
    """Return a NORMALIZED COPY: labels mapped, technical duplicates removed.
    The original dataframe is never overwritten."""
    df = event_df.copy()
    if "concept:name" not in df.columns:
        raise ValueError("pass a pm4py-formatted dataframe")
    df["concept:name"] = df["concept:name"].replace(label_map)
    dup_mask = ((df["concept:name"] == technical_activity)
                & df.duplicated(subset=["case:concept:name", "concept:name",
                                        "time:timestamp"], keep="first"))
    return df.loc[~dup_mask].reset_index(drop=True)

def _as_log(obj):
    """DataFrame -> EventLog; pass logs through unchanged."""
    return obj if not isinstance(obj, pd.DataFrame) \
        else pm4py.convert_to_event_log(obj)

def slice_window(log_or_df, start, end, mode="traces_contained"):
    """Filter a time window. start/end: 'YYYY-MM-DD HH:MM:SS'.
    mode: 'events' | 'traces' | 'traces_contained' (fully-contained traces)."""
    return pm4py.filter_time_range(_as_log(log_or_df), str(start), str(end),
                                   mode=mode, retain=True)

def _month_bounds(year, month):
    start = pd.Timestamp(year=year, month=month, day=1)
    end = start + pd.offsets.MonthBegin(1)
    return (start.strftime("%Y-%m-%d %H:%M:%S"),
            end.strftime("%Y-%m-%d %H:%M:%S"))

def monthly_dfg(year, month, event_df, normalized=True):
    """Discover (dfg, start_activities, end_activities) for one month."""
    s, e = _month_bounds(year, month)
    view = normalized_view(event_df) if normalized else event_df
    return pm4py.discover_dfg(slice_window(view, s, e))

def monthly_variants(year, month, event_df, normalized=True):
    """Variant -> count dictionary for one month (handles API variants)."""
    s, e = _month_bounds(year, month)
    view = normalized_view(event_df) if normalized else event_df
    variants = pm4py.get_variants(slice_window(view, s, e))
    counts = {}
    for key, val in variants.items():
        counts[key] = val if isinstance(val, int) else len(val)
    return counts

def temporal_profile_reference(event_df, normalized=True):
    """Reference temporal profile: {(src,tgt): [mean, stdev]}."""
    view = normalized_view(event_df) if normalized else event_df
    return pm4py.discover_temporal_profile(_as_log(view))

def temporal_profile_anomalies(ref_profile, new_profile, z_threshold=3.0):
    """Flag activity pairs whose mean moved > z reference sigmas."""
    out = []
    for pair, stats in new_profile.items():
        if pair in ref_profile:
            mu, sd = ref_profile[pair][0], ref_profile[pair][1]
            if sd and sd > 0:
                z = (stats[0] - mu) / sd
                if abs(z) > z_threshold:
                    out.append({"pair": pair, "z": round(z, 2),
                                "ref_mean": mu, "new_mean": stats[0]})
    return sorted(out, key=lambda d: -abs(d["z"]))

def discover_log_skeleton_view(event_df, year=None, month=None,
                               noise_threshold=0.05, normalized=True):
    """Log skeleton over the whole table or one month of it."""
    view = normalized_view(event_df) if normalized else event_df
    if year and month:
        s, e = _month_bounds(year, month)
        view = slice_window(view, s, e)
    return pm4py.discover_log_skeleton(_as_log(view),
                                       noise_threshold=noise_threshold)

def skeleton_deviations(event_df, skeleton):
    """Fraction of cases deviating from a discovered log skeleton."""
    diag = pm4py.conformance_log_skeleton(_as_log(event_df), skeleton)
    def _bad(d):
        if "is_fit" in d:
            return not d["is_fit"]
        return any(isinstance(v, list) and v for v in d.values())
    n_bad = sum(1 for d in diag if _bad(d))
    return {"cases_checked": len(diag), "non_fitting": n_bad,
            "deviating_fraction": round(n_bad / max(len(diag), 1), 3)}

def discover_declare_view(event_df, noise_threshold=0.05, normalized=True):
    """DECLARE discovery (guarded: depends on pm4py release)."""
    view = normalized_view(event_df) if normalized else event_df
    try:
        return pm4py.discover_declare(_as_log(view),
                                      noise_threshold=noise_threshold)
    except AttributeError:
        print("pm4py.discover_declare unavailable in this version - "
              "fall back to discover_log_skeleton_view().")
        return None

def raw_vs_mapped_conformance(formatted_raw_df, petri_net, im, fm):
    """Token-based fitness BEFORE vs AFTER normalization (copies only)."""
    fit_raw = pm4py.fitness_token_based_replay(
        _as_log(formatted_raw_df), petri_net, im, fm)
    fit_map = pm4py.fitness_token_based_replay(
        _as_log(normalized_view(formatted_raw_df)), petri_net, im, fm)
    return {"raw": fit_raw, "mapped": fit_map,
            "recovery": fit_map["log_fitness"] - fit_raw["log_fitness"]}

## 9. Provisional attribution

**Strongest conclusions.** First, the March collapse is logging/semantic drift, held with high confidence: the label inventory expanded, duplicated technical events and resource missingness appeared the same week, and raw fitness fell ~0.37 points while mapped fitness stayed near baseline — a constellation no behavioral mechanism reproduces. Second, genuine control-flow/rework drift began in early May and is mechanistically tied to the Supplier Z disruption: the loop rate roughly doubled in lock-step with supplier share (r0.8), the direction and timing match the documented API/staff failure, and the partial reversal after the 2026-09-10 recovery is what a supply-side cause predicts. Third, the dominant cycle-time channel is queueing, not touch work: Assess service moved only ~0.6 h while waiting ballooned, the p90/median ratio widened 2.182.47, and the acceleration tracks the June staffing loss.

**Only plausible, not established.** The priority-rule redistribution (July 1) fits the widening tails and timing, but monthly aggregates are structurally blind to opposing class-level shifts — it needs segmented waits before I would promote it. The October Senior Assessment surge is ambiguous between seasonal case mix and a recording-propensity change; seasonality predicts roughly a doubling, the data show ~5×. Persistent behavioral adaptation (residual rework of 20–21% vs 11.5% baseline after supply recovery) is suggestive but competes with October backlog and seasonal hardness. The December improvement cannot be credited to overtime or the quality campaign while seasonal roll-off runs concurrently.

**Where mediation blocks clean attribution.** Supplier disruption  rework loops  added assessor workload  queueing: mid-year cycle time therefore mixes H2 and H3 irreducibly at this granularity. Case mix mediates both rework and SrA propensity, so population and behavior are entangled in October. Non-random resource missingness contaminates every resource-conditioned analysis used to test organizational hypotheses.

**Revision triggers.** Class-stratified waiting data showing opposite-signed shifts would upgrade the priority mechanism to high confidence. A prior-year seasonal baseline reproducing October volume/complexity but *not* the rework step would sharpen mix-versus-behavior separation. Supplier-stratified rework staying elevated in Q4 would cement behavioral persistence. A documented SrA recording-policy change dated before October would move that surge from "population" toward "logging."

## 10. Monitoring decision rules

1. IF raw fitness drops sharply while mapped fitness holds steady and the label inventory expands, THEN freeze performance dashboards, publish the label mapping, and re-run affected analyses on normalized logs, BECAUSE the change is provenance/semantic and reporting raw values manufactures phantom drift.

2. IF the AssessRequestInfo rate rises 1.5× baseline specifically within Supplier-Z cases, THEN trigger the supplier bundle-quality incident workflow (API and intake audit) rather than assessor coaching, BECAUSE incomplete upstream documents mechanically generate rework loops.

3. IF queue waiting before Assess rises while Assess service time stays inside control limits, THEN investigate capacity and scheduling (vacancies, rosters, batching) before changing assessor practice, BECAUSE flat service plus rising wait localizes deterioration to queueing.

4. IF high-complexity waiting falls while standard-case waiting rises after a queue-priority deployment, THEN record priority class explicitly and report dual-class service levels instead of a single median target, BECAUSE priority redistributes delay between classes and aggregates conceal the transfer.

5. IF a metric breaches limits only during the same calendar windows as prior years with matching direction, THEN classify it as recurring/seasonal, widen that month's reference limits, and pre-plan capacity, BECAUSE in-season attribution hunts for novel root causes waste effort on expected variation.

6. IF missing-resource or duplicate-event rates rise only in migrated activities, THEN route the ticket to data engineering and suspend resource-sensitive monitors until completeness is restored, BECAUSE measurement faults corrupt organizational analyses without proving process change.

7. IF rework remains above baseline after supplier recovery even when controlling for case mix, THEN launch a behavioral-adaptation review (checklist drift, defensive info requests) with front-line input, BECAUSE persistent residual loop rates indicate learned practice rather than a transient supply fault.
