def positive_overlap_rate(g):
    intervals = [(row.start, row.complete) for row in g.itertuples() if pd.notna(row.start) and pd.notna(row.complete)]
    if len(intervals) < 2: return np.nan
    total = 0; overlapping = 0
    for i in range(len(intervals)):
        for j in range(i+1, len(intervals)):
            total += 1
            s1,e1 = intervals[i]; s2,e2 = intervals[j]
            if max(s1,s2) < min(e1,e2):
                overlapping += 1
    return overlapping/total if total else np.nan

def max_equal_start(g): return g["start"].value_counts().max()
def max_equal_end(g): return g["complete"].value_counts().max()
def count_adjacencies(g):
    g2 = g.sort_values("start")
    adj = 0
    for i in range(len(g2)-1):
        if g2["start"].iloc[i+1] == g2["complete"].iloc[i]:
            adj += 1
    return adj

def queue_backlog_at_arrivals(g):
    g2 = g.sort_values("queue_arrival").reset_index(drop=True)
    max_backlog = 0
    for i, row in g2.iterrows():
        arrived = g2[g2["queue_arrival"] <= row["queue_arrival"]]
        started_before = g2[(g2["start"] <= row["queue_arrival"])]
        backlog = len(arrived) - len(started_before)
        if backlog > max_backlog:
            max_backlog = backlog
    return max_backlog

def positive_overlap_rate(g):
    intervals = [(row.start, row.complete) for row in g.itertuples() if pd.notna(row.start) and pd.notna(row.complete)]
    if len(intervals) < 2:
        return np.nan
    total = 0
    overlapping = 0
    for i in range(len(intervals)):
        for j in range(i + 1, len(intervals)):
            total += 1
            s1, e1 = intervals[i]
            s2, e2 = intervals[j]
            if max(s1, s2) < min(e1, e2):
                overlapping += 1
    return overlapping / total if total else np.nan

def max_equal_start(g):
    return int(g["start"].value_counts().max())

def max_equal_end(g):
    return int(g["complete"].value_counts().max())

def count_adjacencies(g):
    g2 = g.sort_values("start")
    return int(sum(g2["start"].iloc[i + 1] == g2["complete"].iloc[i] for i in range(len(g2) - 1)))

def max_queue_backlog_at_arrival(g):
    g2 = g.sort_values("queue_arrival").reset_index(drop=True)
    backlogs = []
    for _, row in g2.iterrows():
        arrived = g2[g2["queue_arrival"] <= row["queue_arrival"]]
        started_before = g2[g2["start"] <= row["queue_arrival"]]
        backlogs.append(len(arrived) - len(started_before))
    return max(backlogs) if backlogs else 0

def precision_seconds(p):
    p = str(p).lower()
    if "sec" in p:
        return int(p.split()[0])
    if "min" in p:
        return int(p.split()[0]) * 60
    if "hour" in p:
        return int(p.split()[0]) * 3600
    return np.nan

def positive_overlap_rate(g):
    intervals = [
        (row.start, row.complete)
        for row in g.itertuples()
        if pd.notna(row.start) and pd.notna(row.complete)
    ]
    if len(intervals) < 2:
        return np.nan
    total = 0
    overlapping = 0
    for i in range(len(intervals)):
        for j in range(i + 1, len(intervals)):
            total += 1
            s1, e1 = intervals[i]
            s2, e2 = intervals[j]
            if max(s1, s2) < min(e1, e2):
                overlapping += 1
    return overlapping / total if total else np.nan

def max_equal_start(g):

def max_equal_end(g):

def count_adjacencies(g):
    g2 = g.sort_values("start")

        g2["start"].iloc[i + 1] == g2["complete"].iloc[i]

def max_queue_backlog_at_arrival(g):
    g2 = g.sort_values("queue_arrival").reset_index(drop=True)

    for _, row in g2.iterrows():

        started_before = g2[g2["start"] <= row["queue_arrival"]]
        backlogs.append(len(arrived) - len(started_before))
    return max(backlogs) if backlogs else 0

# 5. Robustness and alternative explanations

**Resource IDs trusted versus shared accounts**

- If RBT-7, T-D, and T-H are trusted specific resources, then G1 and G4 have reliable physical interpretations: a four-item automated batch and a single-server sequential queue.
- If `analyst_pool` were trusted as one worker, G5 would imply heavy multitasking or implausible utilization. Treating it as a shared account/pool changes the conclusion from “one worker doing overlapping cases” to “up to three analysts working concurrently under one login.”
- The other labels such as T-B, T-C, T-F, and T-G are likely workstations or screens. If these are shared accounts, then apparent batching at start or end could partly be an identity artifact. The prompt only explicitly supports shared-account status for `analyst_pool`, so this remains a caveat rather than a finding.

**Timestamps exact versus interval-censored by stated precision**

- G1, G4, and G8 have 1-second scanner/PLC precision and appear sufficiently exact for adjacency and overlap analysis, except that G8’s safety system may use a different clock.
- G2, G3, G5, and G6 have 1-minute precision. Point estimates such as 11:30 or 17:00 should be interpreted as minute buckets, not exact seconds. For G6, even interval-censoring to one minute leaves a near-zero service-time problem.
- G7 has 15-minute precision. Treating 14:00 and 14:15 as exact points would invent a 15-minute simultaneous batch; the correct treatment is interval censoring or exclusion until unrounded timestamps exist.
- PM4Py batch labels are especially fragile under coarse precision. With 15-minute data, PM4Py may detect “batches” that are purely rounding artifacts.

**At least three cases where the same temporal pattern supports multiple physical explanations**

1. **G1 equal start/end** could be a genuine four-item automated batch, or PLC batch-completion logging in which all lifecycle timestamps are written at the end.  
2. **G2 equal start, distinct ends** could be one inspector starting a group and then inspecting sequentially, or a shared workstation with several people after a common setup.  
3. **G3 distinct starts, equal end** could be bulk form submission after sequential reviews, or true simultaneous completion at folder close, or 1-minute rounding hiding slightly different completions.  
4. **G5 overlapping intervals** could be multiple analysts on a shared login, or one analyst multitasking with preemption.  
5. **G7 equal start/end** could be quarter-hour rounding, or a real 15-minute batch.  
6. **G6 start equals complete** could be bulk closure, a legitimate zero-duration batch action, or 1-minute coarsening.

# 6. Data-collection redesign

| Priority | Data element | Recommendation |
|---|---|---|
| Critical | Execution timestamps | Record actual start, completion, pause, and resume from the system performing the work. Do not overwrite execution timestamps with form submission times. |
| Critical | Recording timestamps | Store both `occurred_at` and `recorded_at`; monitor recording lag; retain source insertion order. |
| Critical | Group/batch identifiers | Add `batch_id`, `fixture_id`, `batch_size`, `position_in_batch`, and group start/end for true physical batches such as G1. |
| Critical | Resource/badge identity | Replace shared logins with badge-level identity; if a pool is unavoidable, record `pool_id`, `badge_id`, `session_id`, and workstation. |
| Critical | Form/bulk submission event | Add an explicit `submitted_at` distinct from `completed_at`. For G3 and G6, the form submission should be an extra event, not the lifecycle completion. |
| High | Source clocks | Synchronize all systems to a reference clock; record clock source and offset. This is essential for G8 because repair queue and safety system clocks differ. |
| High | Workstation capacity | Record machine capacity, slots/fixtures, and whether concurrent processing is possible. |
| High | Pause/resume and downtime | Record capacity-loss events such as safety checks, breaks, faults, and downtime with start/end and same clock source. |
| Medium | Timestamp precision metadata | Store precision, accuracy, rounding rule, source system version, and raw unrounded timestamps. For G7, retain the raw timestamp before rounding. |
| Medium | Validation rules | Alert on `start == complete`, impossible overlaps for a trusted single resource, shared-account overlap, large recording lag, and cross-system clock mismatches. |

# 7. Final adjudication

High-confidence temporal interpretations:

- **G1**: high confidence for a genuine simultaneous batch on RBT-7, assuming the PLC writes true cycle start/complete and the fixture capacity is actually used.
- **G4**: high confidence for sequential processing with reliable 1-second scanner timestamps.
- **G6**: high confidence that the observed pattern is end-of-shift bulk submission; actual close/service times are not captured.
- **G7**: high confidence that the observed pattern is timestamp coarsening from 15-minute legacy MES precision.
- **G2 and G3**: high confidence in the temporal-pattern category—group-start logging for G2 and bulk completion/submission for G3—but not in physical start/completion times.

Clusters to exclude or interval-censor in queue analysis:

- **G6**: exclude start-to-complete service durations; if used, treat 17:00 only as a closure/submission record.
- **G7**: interval-censor start and complete to quarter-hour buckets; do not compute point estimates of waiting or service.
- **G2**: use queue arrival to 10:00 as group wait only; do not calculate per-case service as start-to-complete without correction.
- **G3**: use start only if operational validation confirms it is a real review start; treat 11:30 as submission/completion aggregation.
- **G5**: do not compute individual-resource utilization from the shared login; either exclude or model as a pool.
- **G8**: include only after cross-system clock validation; otherwise treat the 15:00–15:35 safety check as an interval with uncertain alignment.

Clusters requiring operational validation before any performance conclusion:

- **G3**: need actual review completion times separate from form submission.
- **G5**: need badge-level identity or number of active analysts.
- **G6**: need actual start/completion times for ticket closure work.
- **G7**: need unrounded legacy MES timestamps or documented rounding rules.
- **G8**: need clock synchronization between repair queue and safety system, plus confirmed downtime start/end.
- **G2**: need confirmation of the begin-group button semantics and whether one inspector performed all inspections sequentially.
