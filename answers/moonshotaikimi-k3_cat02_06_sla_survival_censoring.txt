```python
def km_curve(times, events):
    # events: 1=event,0=censored
    df_ = pd.DataFrame({"t": times, "e": events}).sort_values("t")
    out = []
    n = len(df_); s = 1.0
    for t, g in df_.groupby("t"):
        d = int(g["e"].sum()); c = len(g) - d
        if d > 0:
            s *= (1 - d / n)
            out.append((t, s, n, d))
        n -= (d + c)
    return pd.DataFrame(out, columns=["time","surv","n_risk","n_event"])
```

AJ CIF:

```python
def aj_cif(times, codes, cause):
    # codes: 0 censored, else cause labels
    df_ = pd.DataFrame({"t": times, "code": codes}).sort_values("t")
    n = len(df_); s = 1.0; f = 0.0; rows=[]
    for t, g in df_.groupby("t"):
        d_all = int((g["code"]>0).sum())
        d_cause = int((g["code"]==cause).sum())
        if d_cause>0:
            f += s * d_cause / n
        if d_all>0:
            s *= (1 - d_all/n)
        rows.append((t, s, f, n))
        n -= len(g)
    return pd.DataFrame(rows, columns=["time","surv_any","cif_cause","n_risk"])
```

```python
CODE = {"resolved": 1, "withdrawn": 2, "rejected": 3, "transferred": 4, "open": 0}

def encode(dfx, transfer="competing"):
    """Return arrays (time, code) under a transfer scenario.
    transfer='competing': transfer is an absorbing competing event at transfer time.
    transfer='censor':    transfer is (possibly informative) right-censoring.
    transfer='breach':    transfer counts as SLA failure at min(t_transfer, SLA).
    """
    t = dfx["duration_or_age_h"].astype(float).copy()
    code = dfx["observed_status"].map(CODE).copy()
    if transfer == "censor":
        code = code.replace(4, 0)
    elif transfer == "breach":
        m = code == 4
        t = t.mask(m, np.minimum(t, SLA_H))
        code = code.mask(m, 5)  # 5 = SLA-breach terminal (no internal resolution)
    return t.to_numpy(), code.to_numpy()
```

```python
def km_surv(times, event_flag):
    """Kaplan-Meier. Returns step table; open cases retained in risk sets."""
    d = pd.DataFrame({"t": np.asarray(times, float), "e": np.asarray(event_flag, int)})
    d = d.sort_values("t", kind="mergesort")
    n, s, rows = len(d), 1.0, []
    for t, g in d.groupby("t", sort=True):
        d_i = int(g["e"].sum())
        if d_i:
            s *= 1.0 - d_i / n
            rows.append({"time": t, "surv": s, "n_risk": n, "d": d_i})
        n -= len(g)
    return pd.DataFrame(rows)

def aj_cif(times, codes, cause=1):
    """Aalen-Johansen CIF for `cause` with all other codes as competing events."""
    d = pd.DataFrame({"t": np.asarray(times, float), "c": np.asarray(codes, int)})
    d = d.sort_values("t", kind="mergesort")
    n, s, f, rows = len(d), 1.0, 0.0, []
    for t, g in d.groupby("t", sort=True):
        d_all = int((g["c"] > 0).sum())
        d_c = int((g["c"] == cause).sum())
        if d_c:
            f += s * d_c / n
        if d_all:
            s *= 1.0 - d_all / n
        rows.append({"time": t, "surv_any": s, "cif": f, "n_risk": n})
        n -= len(g)
    return pd.DataFrame(rows)

def value_at(step, col, horizon):
    step = step[step["time"] <= horizon]
    return step[col].iloc[-1] if len(step) else (1.0 if col=="surv" else 0.0)
```

```python
# ------------------------------------------------------------------
# (B) Event/censoring encoding: 2+ estimands x 3 transfer scenarios
# ------------------------------------------------------------------
CODE = {"resolved": 1, "withdrawn": 2, "rejected": 3, "transferred": 4, "open": 0}

def encode(dfx, transfer="competing"):
    """Return (time, code). code: 0=censored, 1=Resolve, 2=Withdraw, 3=Reject,
    4=Transfer (competing), 5=SLA failure without internal resolution."""
    t = dfx["duration_or_age_h"].to_numpy(dtype=float).copy()
    c = dfx["observed_status"].map(CODE).to_numpy(dtype=int).copy()
    if transfer == "censor":                       # (a) loss to follow-up (informative?)
        c = np.where(c == 4, 0, c)
    elif transfer == "breach":                     # (c) conservative compliance view
        m = c == 4
        t = np.where(m, np.minimum(t, SLA_H), t)
        c = np.where(m, 5, c)
    return t, c                                    # "competing": (b) as recorded

def km_surv(times, event):
    """Kaplan-Meier step table; open cases stay in risk sets until their age."""
    d = pd.DataFrame({"t": np.asarray(times, float), "e": np.asarray(event, int)}).sort_values("t")
    n, s, rows = len(d), 1.0, []
    for tt, g in d.groupby("t", sort=True):
        d_i = int(g["e"].sum())
        if d_i:
            s *= 1.0 - d_i / n
            rows.append({"time": tt, "surv": s, "n_risk": n, "d": d_i})
        n -= len(g)
    return pd.DataFrame(rows)

def aj_cif(times, codes, cause=1):
    """Aalen-Johansen CIF for `cause`; every other code competes."""
    d = pd.DataFrame({"t": np.asarray(times, float), "c": np.asarray(codes, int)}).sort_values("t")
    n, s, f, rows = len(d), 1.0, 0.0, []
    for tt, g in d.groupby("t", sort=True):
        d_all = int((g["c"] > 0).sum())
        d_c = int((g["c"] == cause).sum())
        if d_c:
            f += s * d_c / n
        if d_all:
            s *= 1.0 - d_all / n
        rows.append({"time": tt, "surv_any": s, "cif": f, "n_risk": n})
        n -= len(g)
    return pd.DataFrame(rows)

def value_at(step, col, horizon, default):
    s = step[step["time"] <= horizon]
    return float(s[col].iloc[-1]) if len(s) else default

def estimands(dfx, transfer="competing"):
    t, c = encode(dfx, transfer)
    cif_r = aj_cif(t, c, cause=1)
    F72 = value_at(cif_r, "cif", SLA_H, 0.0)
    Fall = float(cif_r["cif"].iloc[-1])
    disp_codes = {"competing": [1, 2, 3, 4], "breach": [1, 2, 3, 5], "censor": [1, 2, 3]}[transfer]
    km = km_surv(t, np.isin(c, disp_codes).astype(int))
    S72 = value_at(km, "surv", SLA_H, 1.0)
    return {"E1_not_resolved_by_72": 1.0 - F72,
            "E2_late_resolution": Fall - F72,
            "E3_no_disposition_at_72": S72}
```

# 1. SLA estimands and time definitions

**Time origin and clock.** Time origin `t=0` is the `Submit` event; the time scale is elapsed hours since `Submit`; the SLA horizon is  = 72 h. For each case define latent event times `T_R` (Resolve), `T_W` (Withdraw), `T_J` (Reject), `T_X` (Transfer), and censoring time `C` (age at the observation cutoff for `open` cases). We observe `Y = min(first event, C)` = `duration_or_age_h` and an event-type indicator. Times are treated as continuous; the only engineered tie is at exactly 72 h under the "transfer = breach" scenario below.

**Four estimands** (estimand first, estimator later):

| ID | Estimand | Formal definition | Operational decision it serves |
|---|---|---|---|
| E1 | Probability of **not being resolved by 72 h** | `1  F_R(72)`, where `F_R` is the *cumulative incidence* of Resolve with Withdraw/Reject/Transfer as competing events | Customer-facing SLA compliance; breach alerting; the headline "tail risk" number |
| E2 | **Cumulative incidence of resolution after 72 h** | `F_R()  F_R(72)` | "How much of the tail eventually completes?" — improvement targeting, late-stage staffing |
| E3 | Probability of **no internal terminal disposition by 72 h** | `S_any(72)`, survival of the composite {Resolve, Withdraw, Reject(, Transfer)} | Backlog/WIP and capacity planning |
| E4 | **Cause-specific CIFs at 72 h** | `F_W(72)`, `F_J(72)`, `F_X(72)` | Triage quality (early rejects), partner/network management (transfers) |

E1 is **not** `1  KM` where competing events are censored: that KM estimand is the resolution probability in a *hypothetical world where withdrawal/rejection/transfer are abolished*, which no stakeholder is promised.

**Three distinct "non-event" situations must not be conflated:**
- *Right censoring*: `open` cases with age < 72 (S10, S15, S24, S29, S31) — even their 72-h breach status is unknown.
- *Competing events*: `withdrawn`/`rejected` — fully observed absorbing outcomes that make future resolution impossible, not missing data.
- *Already-observed breaches*: cases that crossed 72 h unresolved — S03, S13, S20, S28 (resolved late), S09, S11, S25, S26 (open, age > 72), S32 (transferred at 96 h). For these, "not resolved by 72" is a **fact**, not a probability. Only the five sub-72 open cases (plus transfers under the censoring scenario) contribute genuine probabilistic mass to E1.

**How the four anchor cases contribute under each estimand:**

| Case | E1 (not resolved by 72) | E2 (late resolution) | E3 (no disposition at 72) | E4 |
|---|---|---|---|---|
| S03, resolved at 76 | Certain contributor (a post-72 resolution is *not* in `F_R(72)`) | Event at 76, in the numerator | At risk 0–76  certain contributor | Resolve cause, event at 76 |
| S09, open at 92 (gap 31) | Certain contributor (open at 92  unresolved at 72) | Right-censored at 92 — may still resolve | Certain contributor (still no disposition) | Censored for all causes at 92; censoring plausibly informative (31 h silence) |
| S06, withdrawn at 27 | Certain contributor (can never be resolved) | Non-event | Event at 27 — leaves the no-disposition risk set | Withdraw cause, event at 27. **Not censoring.** |
| S08, transferred at 68 | Scenario-dependent: (a) censored at 68  72-h status unknown; (b) competing event  certain contributor; (c) breach at 72 | (a) censored; (b) non-event; (c) non-event | (a) censored at 68; (b) disposition at 68; (c) still without disposition  contributor | Transfer cause under (b); S08 is escalated P1, so scenario (a) is informative censoring |

# 3. Why naive completed-case analysis is biased

A naive analysis keeps only `resolved` rows and compares durations: cohort A median 47.5 h (mean 54.7) vs cohort B median 39 h (mean 45.6) — conclusion: "B is faster, SLA tail is fine." Every step of that is wrong:

- **Survivor / length bias.** Conditioning on `resolved` discards the tail itself: S09 (open at 92 h), S26 (open at 104 h), S11 (open at 81 h). At any fixed cutoff, slow resolutions are under-represented because they have not happened yet; the resolved-only sample is length-biased toward fast cases. The proper E1 contrast is 73.3% (A) vs 70.6% (B) — a ~2.7 pp gap with widely overlapping intervals, not the clear win the medians suggest.
- **Informative censoring.** All three transfers (S08, S21, S32) are escalated cases; two are P1. Censoring them "at random" removes the hardest cases from the risk set and inflates estimated resolution probabilities (quantified in §6: E2 nearly doubles). Likewise, open cases with long silence (S26: 72 h since last event; S11: 55 h; S25: 48 h) are plausibly stalled, not representative survivors.
- **Competing risks misread as censoring.** S05 and S23 are rejected within 11 h; S12 withdrawn at 15 h. Treating these as censoring in a resolution-only KM estimates a hypothetical world without rejection/withdrawal and overstates resolution-by-72 (30.0% vs the correct 28.5% here — small numerically, large conceptually, and growing with early-exit volume).
- **Cohort imbalance.** Cohort B carries more P1 (5/17 = 29% vs A's 3/15 = 20%) and more transfers (2 vs 1). P1 has the worst E1 (75%). Raw cohort contrasts are confounded by priority mix; the policy change was not randomized and AutoCheck coverage in B is only 13/17.
- **Open-below-72 vs open-above-72 are different objects.** S31 (age 23) is a genuine unknown — it belongs in the risk set and contributes through KM/AJ reweighting. S26 (age 104) is a **known breach**; excluding it as "censored" throws away observed failures, while treating it as still-at-risk-of-breach pretends a fact is a probability. Active cases below 72 (S10, S15, S24, S29) are the only ones for which E1 is truly a prediction.

# 4. Statistical analysis plan

| Method | What it estimates | Key assumption | Fit to *these* data |
|---|---|---|---|
| Kaplan–Meier | Single-event survival `S(t)` | Non-informative censoring | Right tool for E3 (composite disposition); **wrong** tool for "probability of resolution" under competing events |
| Aalen–Johansen CIF | Cause-specific cumulative incidence `F_k(t)` | Non-informative censoring of open cases | **Primary**: nonparametric, handles 4 causes, no covariates to overfit at n = 32 |
| Cause-specific hazards (Cox) | Hazard of each cause among cases still active | Proportional hazards; correct covariate timing | Etiologic interpretation; n = 32 supports at most 1–2 covariates — use descriptively/stratified, not as a fitted multivariable model |
| Subdistribution hazards (Fine–Gray) | Direct regression on `F_k(t)` | Competing-event cases retained in risk set (odd interpretation) | Prediction-oriented; too data-hungry here |
| Restricted mean time | `² S_any(t) dt` = expected case-hours active in the first 72 h | Minimal | Robust workload metric; computed below  **50.4 of 72 h** per case; does not separate causes |
| Landmarking | `P(breach  alive at landmark s, prefix-so-far)` | Correct truncation of features at `s` | The operationally right design for 24/48-h alerts; currently blocked for mid-case landmarks by missing activity timestamps |
| Multi-state model | Transition hazards between process stages; state-occupation probabilities | Markov or clock-reset assumptions | The ideal dynamic representation (§5); needs per-activity timestamps we do not have |

**Recommendation.** *Primary:* Aalen–Johansen CIFs for all four causes, pooled and stratified by cohort and priority, run under all three transfer scenarios, with bootstrap percentile intervals. Rationale: with only **9 of 32 cases under observation at 72 h** (S03, S09, S11, S13, S20, S25, S26, S28, S32 — four of them open), any parametric or regression approach would be false precision; scenario-based transfer handling is more honest than a modeling assumption. *Secondary:* KM for E3 plus RMST-in-system for workload; cause-specific hazard contrasts reported descriptively; landmarking and the multi-state model specified now and activated once event timestamps arrive. Estimates beyond ~110 h (the largest observed time) should be treated as flat-tail extrapolation and not reported.

| 4 | Cohort/policy contrast (T2) | A 73.3% vs B 70.6% (  2.7 pp, intervals overlap widely) | — | A 40.0% vs B 47.1% |
| 5 | Within-B as-treated: AutoCheck (n = 13) vs no-AutoCheck (n = 4: S20, S23, S26, S30) | 61.5% vs 100% | — | — |

Composition at 72 h under T2 (sums to ~100%): resolved 28.5%, withdrawn 12.8%, rejected 9.5%, transferred 6.4%, still open 42.9%.

**What could reverse:**
- **E1 is robust** to transfer handling (70.9–71.5%) and to stalled-case assumptions, because it is identified by pre-72 information. The conclusion "roughly 7 in 10 cases are not resolved within SLA" is stable.
- **E2 is assumption-driven**: it nearly doubles (13.9%  27.1%) if transfers are censored rather than treated as competing, because censoring hard, escalated cases up-weights later resolutions. Any statement like "the tail eventually resolves" is fragile: even under the base case, only ~19% of the not-resolved-by-72 mass is observed to resolve later.
- **E3 swings 13 pp** (34.8–47.8%) purely on the transfer convention — backlog claims must state the convention.
- **"Cohort B improved tail risk" is reversible**: the raw  is ~2.7 pp with overlapping intervals; B has more P1 (priority adjustment would likely favor B) but more transfers (T3 accounting penalizes B via S32 at 96 h); the within-B AutoCheck contrast (61.5% vs 100%) is as-treated, tiny-n, and confounded — it is a hypothesis, not an effect.

# 7. Operational interpretation

- **Alerting (landmark design, associational scores).** At 24 h and 48 h landmarks, compute `P(not resolved by 72  still active at landmark, current stage, priority, reviews-so-far, inactivity-so-far)` using only prefix-truncated features. Trigger example: at 48 h, a case in Review or Escalated with > 24 h of inactivity routes to a senior queue. Independently of any model, a hard rule "no event for  48 h  audit" would already catch S11, S25, S26. Scores are associations, not effects of escalation or any intervention.
- **Workload.** Expected 72 h+ WIP  E3 × inflow; RMST says an average case spends  50.4 of its first 72 h active in the internal system; E2 quantifies late-stage completion work. Under T2 vs T3 these backlog numbers differ materially — fix the convention in writing.
- **SLA governance.** Report E1 with intervals, stratified by cohort and priority, weekly. Frame the A-vs-B gap as a monitored hypothesis (non-randomized rollout, 13/17 AutoCheck compliance), never as "the policy reduced breaches by x%".
- **Calibration.** Bin predicted breach risks vs observed breach frequency, overall and within each priority × cohort cell; track Brier score; recalibrate per priority if systematic over/under-prediction appears. With n = 32, calibration is currently only demonstrable as a procedure.
- **Fairness / visibility.** Outcome observability itself is unequal: 3 of 8 P1 cases were transferred (their outcomes unknown) vs 0 of 10 P3 cases — P1 SLA performance is the least verifiable, and any dashboard that silently censors transfers will look best exactly where visibility is worst. Check alert rates, transfer decisions, and escalation latency by priority; do not let historical transfer decisions enter training data as "ground truth" non-breaches.

```python
# ---- (B) Encoding: 2+ estimands x 3 transfer scenarios ----
CODE = {"resolved": 1, "withdrawn": 2, "rejected": 3, "transferred": 4, "open": 0}

def encode(dfx, transfer="competing"):
    """Return (time, code). code: 0=censored, 1=Resolve, 2=Withdraw, 3=Reject,
    4=Transfer (competing), 5=SLA failure without internal resolution."""
    t = dfx["duration_or_age_h"].to_numpy(dtype=float).copy()
    c = dfx["observed_status"].map(CODE).to_numpy(dtype=int).copy()
    if transfer == "censor":                     # T1: loss to follow-up (informative?)
        c = np.where(c == 4, 0, c)
    elif transfer == "breach":                   # T3: conservative compliance view
        m = c == 4
        t = np.where(m, np.minimum(t, SLA_H), t)
        c = np.where(m, 5, c)
    return t, c                                  # T2 "competing": as recorded

def km_surv(times, event):
    """Kaplan-Meier step table; open cases remain in risk sets until their age."""
    d = pd.DataFrame({"t": np.asarray(times, float),
                      "e": np.asarray(event, int)}).sort_values("t")
    n, s, rows = len(d), 1.0, []
    for tt, g in d.groupby("t", sort=True):
        d_i = int(g["e"].sum())
        if d_i:
            s *= 1.0 - d_i / n
            rows.append({"time": tt, "surv": s, "n_risk": n, "d": d_i})
        n -= len(g)
    return pd.DataFrame(rows)

def aj_cif(times, codes, cause=1):
    """Aalen-Johansen CIF for `cause`; every other event code competes."""
    d = pd.DataFrame({"t": np.asarray(times, float),
                      "c": np.asarray(codes, int)}).sort_values("t")
    n, s, f, rows = len(d), 1.0, 0.0, []
    for tt, g in d.groupby("t", sort=True):
        d_all = int((g["c"] > 0).sum())
        d_c = int((g["c"] == cause).sum())
        if d_c:
            f += s * d_c / n
        if d_all:
            s *= 1.0 - d_all / n
        rows.append({"time": tt, "surv_any": s, "cif": f, "n_risk": n})
        n -= len(g)
    return pd.DataFrame(rows)

def value_at(step, col, horizon, default):
    s = step[step["time"] <= horizon]
    return float(s[col].iloc[-1]) if len(s) else default

def estimands(dfx, transfer="competing"):
    t, c = encode(dfx, transfer)
    cif_r = aj_cif(t, c, cause=1)
    F72 = value_at(cif_r, "cif", SLA_H, 0.0)
    Fall = float(cif_r["cif"].iloc[-1])
    disp = {"competing": [1, 2, 3, 4], "breach": [1, 2, 3, 5],
            "censor": [1, 2, 3]}[transfer]
    S72 = value_at(km_surv(t, np.isin(c, disp).astype(int)), "surv", SLA_H, 1.0)
    return {"E1_not_resolved_by_72": 1.0 - F72,
            "E2_late_resolution": Fall - F72,
            "E3_no_disposition_at_72": S72}

def rmst_in_system(dfx, horizon=SLA_H):
    t, c = encode(dfx, "competing")
    km = km_surv(t, np.isin(c, [1, 2, 3, 4]).astype(int))
    area, pt, ps = 0.0, 0.0, 1.0
    for tt, ss in zip(km["time"], km["surv"]):
        if tt > horizon:
            break
        area += (tt - pt) * ps; pt, ps = tt, ss
    return area + (horizon - pt) * ps
```

# 9. Limitations and additional data

Prioritized missing data:

1. **Per-activity event timestamps** (highest value). Everything time-varying — escalation timing, review counts-so-far, inactivity at landmark times, multi-state transition hazards, proper landmarking — is blocked without them. Today only the last activity's time is recoverable, and only for open cases (age  gap).
2. **External outcomes after transfer.** The single largest sensitivity found (E2: 13.9%  27.1%) comes from not knowing what happens to S08, S21, S32 after handoff. Partner-side resolution status/time would collapse assumptions T1–T3 into measurement.
3. **Submit timestamps and the cutoff date.** Needed to align cases on calendar time, detect drift, and separate cohort (policy) effects from period effects; Assumption A2 (single common cutoff) is stated but case start dates are not.
4. **Time-varying workload / staffing / concurrency.** A strong confounder of durations and of escalation decisions.
5. **Assignment and escalation mechanism.** Who escalates, under what rule, and how priority is assigned — required before any causal reading of escalation or priority contrasts; the `escalated` flag alone cannot support one.
6. **Per-case rollout compliance.** AutoCheck presence is only a proxy for actual policy exposure (13/17 in B; S17 shows ordering varies); eligibility and compliance logs would separate intention-to-treat from as-treated contrasts.
7. **Sample size and follow-up length.** n = 32 with 9 cases at risk at 72 h; tail statements beyond ~110 h are extrapolation. A consecutive, longer-frozen extract; priority-change history; and business cost weights (breach vs withdrawal vs transfer) would sharpen both estimation and governance.

**Identifiability ceiling of the current extract:** no analysis of this table alone can separate "policy effect" from "case-mix + period effect," and no model can make transfer censoring non-informative — all cohort and escalation contrasts reported here are associational by construction.
