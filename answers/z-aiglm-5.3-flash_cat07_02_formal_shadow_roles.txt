def similarities_to_frame(sim):
    """Normalize whatever container PM4Py returns into long-form pairs."""
    recs = []
    if isinstance(sim, pd.DataFrame):
        mat = sim.copy()
        for a in mat.index:
            for b in mat.columns:
                if str(a) < str(b):
                    recs.append({"r1": a, "r2": b, "similarity": float(mat.loc[a, b])})
    else:  # dict-of-dicts or similar
        items = list(sim.items()) if hasattr(sim, "items") else []
        for a, inner in items:
            for b, v in (inner.items() if hasattr(inner, "items") else []):
                if str(a) < str(b):
                    recs.append({"r1": str(a), "r2": str(b), "similarity": float(v)})
    out = pd.DataFrame(recs)
    return out.drop_duplicates(["r1", "r2"]).sort_values("similarity", ascending=False)

def net_to_long(net, name):
    recs = []
    if isinstance(net, dict):
        for src, targets in net.items():
            inner = targets.items() if isinstance(targets, dict) else []
            for tgt, w in inner:
                recs.append({"src": str(src), "tgt": str(tgt), "w": float(w), "net": name})
    elif isinstance(net, pd.DataFrame):
        d = net.copy()
        if set(d.columns) >= {"src", "tgt"}:
            d["net"] = name
            return d
    return pd.DataFrame(recs)

    (pm4py.discover_handover_of_work_network, "handover", {}),
    (pm4py.discover_working_together_network, "working_together", {}),
    (pm4py.discover_subcontracting_network,   "subcontracting", {"n": 2}),
]:
    try:
        nets.append(net_to_long(fn(df, **kw), nm))
    except Exception as exc:
        print(f"[skip] {nm}: {exc}")

def median_step_duration(frame, activity, mask=None):
    f = frame.sort_values(["case:concept:name", "time:timestamp"]).copy()
    f["dur_min"] = (
        f.groupby("case:concept:name")["time:timestamp"].shift(-1)
         - f["time:timestamp"]

    sel = f[(f["concept:name"] == activity) & f["dur_min"].notna()]
    if mask is not None:
        sel = sel[mask(sel)]
    return float(sel["dur_min"].median())

Key design choices for avoiding over-interpretation:
- The **separate authority audit** (final executor per gating activity) is computed *in addition to* the frequency-based role list; any divergence between "who does a lot" and "who signs last" is reported explicitly.
- Complexity and substitution overlays are joined at **case level before** being aggregated to resources, preventing survivorship bias toward busy-but-shallow actors.
- System and temporary actors are kept in dedicated flags (`SYSTEM_ACTORS`, `TEMP_AGENTS`) so similarity and brokerage statistics can be recomputed with/without them.
- `cluster_log` is executed only inside a labelled, guarded block as a navigational tool.

**Cautions before changing staffing based on mined roles:**

1. **Behavioral clusters are not authority mandates.** Promoting Mateo's observed approvals or Giorgio's shadow pricing into permanent rights via log evidence alone would bypass finance/H delegation controls and could embed idiosyncratic workarounds as policy. Conversion of shadow into formal structure requires case-by-case ratification of actual past outputs plus a policy amendment — not a dashboard.
2. **Similarity equals substitutability only at your peril.** The two pairs that naive clustering most wants to merge (KaiLina at 0.98, ChenDina at 0.94) are precisely where intervention went wrong (reopened cases; hidden complexity tiers) or where risk concentrates quietly (senior diagnostics). Always weight swaps by *risk-adjusted* case mix, and remember that even a certified, well-intentioned substitute (Giorgio) halved effective pricing throughput — the substitution cost was measurable, and it was still the best available move.
3. **Logs under-report, especially for seasonality.** Two ten-week windows contain one commercial absence, one auditor workshop, and one security surge; any headcount change extrapolated from such episodes inherits their biases, and offline decisions (phone negotiations, informal coaching) never enter the record. Re-mine quarterly and diff role structures over time before acting on them.
