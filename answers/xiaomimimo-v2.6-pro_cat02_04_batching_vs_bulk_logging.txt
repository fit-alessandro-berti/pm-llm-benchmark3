```python
def largest_equal_group(t, tol=0.0):
    """size of the biggest set of timestamps equal within tol seconds"""
    t = t.dropna().sort_values().reset_index(drop=True)
    if t.empty:
        return 0
    if tol <= 0:
        return int(t.value_counts().max())
    best, run = 1, 1
    for i in range(1, len(t)):
        if (t[i] - t[i - 1]).total_seconds() <= tol:
            run += 1
            best = max(best, run)
        else:
            run = 1
    return int(best)
```

```python
def chain_count(g, tol=0.0):
    r = g.sort_values("sequence_no")
    e = r["complete_ts"].tolist(); s = r["start_ts"].tolist()
    n = 0; tot = 0
    for i in range(len(r) - 1):
        if pd.isna(e[i]) or pd.isna(s[i+1]):
            continue
        tot += 1
        if abs((e[i] - s[i+1]).total_seconds()) <= tol:
            n += 1
    return n, tot
```

```python
def max_equal_group(ts, tol_s=0.0):
    t = ts.dropna().sort_values().reset_index(drop=True)
    if t.empty: return 0
    if tol_s <= 0: return int(t.dt.strftime("%H:%M:%S").value_counts().max())
    best = run = 1
    for i in range(1, len(t)):
        run = run + 1 if (t[i] - t[i-1]).total_seconds() <= tol_s else 1
        best = max(best, run)
    return int(best)

def chained_handoffs(g, tol_s=0.0):
    """complete_i == start_{i+1} within tol (source insertion order)"""
    r = g.sort_values("sequence_no")
    hits = pairs = 0
    e, s = r.complete_ts.tolist(), r.start_ts.tolist()
    for i in range(len(r) - 1):
        if pd.isna(e[i]) or pd.isna(s[i+1]): continue
        pairs += 1
        if abs((e[i] - s[i+1]).total_seconds()) <= tol_s: hits += 1
    return hits, pairs

def pair_overlap_rate(g):
    """fraction of case pairs whose [start, complete] intervals intersect"""
    rows = [(a, b) for a, b in zip(g.start_ts, g.complete_ts) if pd.notna(a) and pd.notna(b)]
    ovl = valid = 0
    for (s1, e1), (s2, e2) in combinations(rows, 2):
        valid += 1
        if min(e1, e2) > max(s1, s2): ovl += 1
    return (ovl / valid) if valid else np.nan, valid
```

- **Conformance** "no repair work during a safety check" cannot be evaluated while the two clocks are unaligned.

```python
def to_ts(x):
    """wall clock -> naive timestamp; blanks survive as NaT (never imputed)"""
    if x is None:
        return pd.NaT
    s = str(x).strip()
    return pd.NaT if s in ("", "nan", "NaT", "None", "-") else pd.to_datetime(f"{DAY} {s}")


def prec_seconds(txt):
    m = re.match(r"\s*(\d+)\s*(sec|min)", str(txt))
    return np.nan if not m else int(m.group(1)) * (1 if m.group(2) == "sec" else 60)

def max_equal_group(ts, tol_s=0.0):
    """size of the largest set of stamps equal (exact) or within tol seconds (chained)"""
    t = ts.dropna().sort_values().reset_index(drop=True)
    if t.empty:
        return 0
    if tol_s <= 0:
        return int(t.dt.strftime("%H:%M:%S").value_counts().max())
    best = run = 1
    for i in range(1, len(t)):
        run = run + 1 if (t[i] - t[i - 1]).total_seconds() <= tol_s else 1
        best = max(best, run)
    return int(best)


def chained_handoffs(g, tol_s=0.0):
    """complete_i == start_(i+1) within tol, in source insertion order"""
    r = g.sort_values("sequence_no")
    e, s, d = r["complete_ts"].tolist(), r["start_ts"].tolist(), r["service_s"].tolist()
    hits = pairs = degenerate = 0
    for i in range(len(r) - 1):
        if pd.isna(e[i]) or pd.isna(s[i + 1]):
            continue
        if d[i] == 0 or d[i + 1] == 0:          # bulk-submitted rows: adjacency is meaningless
            degenerate += 1
            continue
        pairs += 1
        if abs((e[i] - s[i + 1]).total_seconds()) <= tol_s:
            hits += 1
    return hits, pairs, degenerate


def pair_overlap_rate(g):
    """fraction of case pairs whose [start, complete] intervals intersect"""
    rows = [(a, b) for a, b in zip(g["start_ts"], g["complete_ts"]) if pd.notna(a) and pd.notna(b)]
    ovl = valid = 0
    for (s1, e1), (s2, e2) in combinations(rows, 2):
        valid += 1
        if min(e1, e2) > max(s1, s2):           # zero-length intervals never intersect (G6)
            ovl += 1
    return (ovl / valid if valid else np.nan), valid


def summarize(g):
    life = pd.concat([g["start_ts"], g["complete_ts"]]).dropna()
    life_grid_s = 60.0 if bool(((life - life.dt.normalize()).dt.total_seconds() % 60 == 0).all()) else 1.0
    rec = g["recorded_at_ts"].dropna()
    rec_sub_minute = bool(not ((rec - rec.dt.normalize()).dt.total_seconds() % 60 == 0).all())
    wall = (g["complete_ts"].max() - g["start_ts"].min()).total_seconds()
    svc = g["service_s"].dropna()
    ovl, npairs = pair_overlap_rate(g)
    hits, pairs, degen = chained_handoffs(g, tol_s=0.0)
    ia = g["queue_arrival_ts"].sort_values().diff().dropna().dt.total_seconds()
    ia_cv = ia.std(ddof=0) / ia.mean() if len(ia) > 1 and ia.mean() else np.nan
    return dict(
        cluster=g["cluster"].iloc[0], resource=g["resource"].iloc[0], n=len(g),
        src=g["source"].iloc[0], prec_s=g["precision_s"].iloc[0],
        life_grid_s=life_grid_s, grid_conflict=bool(life_grid_s > g["precision_s"].iloc[0]),
        rec_sub_min=rec_sub_minute,
        start_eq=f'{max_equal_group(g["start_ts"])}/{len(g)}',
        end_eq=f'{max_equal_group(g["complete_ts"])}/{len(g)}',
        handoffs=f"{hits}/{pairs}" + (f"+{degen}deg" if degen else ""),
        ovl_rate=round(ovl, 2), pairs=npairs,
        parallelism=round(svc.sum() / wall, 2) if wall > 0 else np.nan,   # n/a when 0/0 (G6)
        svc_tot_min=round(svc.sum() / 60, 1),
        wait_mean_min=round(g["wait_s"].mean() / 60, 1),
        neg_wait=int(g["wait_s"].lt(0).sum()),
        lag_med_s=round(g["lag_s"].median(), 1),
        rec_mono_in_seq=bool(g.sort_values("sequence_no")["recorded_at_ts"].is_monotonic_increasing),
        ia_cv=round(ia_cv, 2) if pd.notna(ia_cv) else np.nan,
        block_gap_min=round((g["start_ts"].min() - g["queue_arrival_ts"].max()).total_seconds() / 60, 1),
        censor_half_min=round(g["precision_s"].iloc[0] / 120, 1),
        suspect_rows=int(g["f_suspect"].sum()),
    )


def own_pattern(g, tol_s=0.0):
    """temporal shape only - deliberately NOT a physical semantics claim"""
    n = len(g)
    ns, ne = max_equal_group(g["start_ts"], tol_s), max_equal_group(g["complete_ts"], tol_s)
    hits, pairs, _ = chained_handoffs(g, tol_s)
    if ns == n and ne == n:
        return "Simultaneous (start&end)"
    if ns == n:
        return "Batch at start"
    if ne == n:
        return "Batch at end"
    if pairs and hits == pairs:
        return "Serial chain (no batch)"
    return "Spread / mixed"
```

# 5. Robustness and alternative explanations

**5.1 Resource IDs trusted (one ID = one worker) vs shared accounts.**
The decisive quantity is `parallelism factor` against *known physical capacity*.

| cluster | parallelism | if ID = one unit | if shared/pooled is allowed | verdict |
|---|---:|---|---|---|
| G1 | 4.00 | impossible for a single sequential machine  artefact or multihead batch | matches documented fixture capacity **exactly (4)** | the capacity crosscheck **favors a real 4up cycle** |
| G2 | 2.59 | impossible for one tablet/screen  start must be batched | still impossible (one screen) | start batching stands under both readings |
| G3 | 2.39 | impossible for one reviewer  completion must be batched | still implausible (one desktop form) | end batching stands under both readings |
| G4, G8 | 1.00 | perfectly consistent | consistent | serial flow stands under both readings |
| G5 | 2.33 | **impossible** for one human  inflated durations or multitasking | 2.33/3 analysts  0.78 utilization each — entirely plausible | conclusion **flips** on the identity assumption alone |
| G7 | 5.00 | impossible | impossible unless the bench has 5 people | coarsening stands under both readings |
| G6 | n/a (0/0) | no information | no information | unaffected |

[Obs] So: trusting IDs turns G2/G3/G5/G7 into *provable* logging artefacts (overunit parallelism); allowing shared accounts dissolves only the G5 contradiction (and G7 only if a 5person bench exists). Waiting time is robust to identity (arrivalstart needs no resource attribution); **utilization, handover, perperson productivity and separationofduty conformance are not**.

**5.2 Timestamps exact vs intervalcensored by stated precision.** Censoring width is `max(documented precision, observed grid)` per A3 (so 60 s for G1/G4/G8, 1 min for G2/G3/G5/G6, 15 min for G7).

| cluster | under "timestamps are exact" | under interval censoring | conclusion stability |
|---|---|---|---|
| G1 | 4 items truly simultaneous, service 3.00 min | one shared cycle of 2–4 min; perlabel simultaneity unresolved (±60 s) | **cyclelevel conclusion stable**; perlabel simultaneity not stable |
| G2 | 4 items started together and ended 8/14/21/27 min later | starts all within ±30 s of 10:00; end spacing 6–7 min = 6–7× the width | **stable** (the end spacing is the robust signal) |
| G3 | all four finished at 11:30 | finishes are only bounded by 11:30 ± 30 s; true ends  [start, 11:30] | "submission batch" **stable**; true durations lost |
| G4 | perfect handoffs, service 4.00 min | handoff gap  (60 s, +60 s); service  (3, 5) min | seriality **stable**; "zerogap handoff" not stable |
| G5 | 3way overlap with 70 min of work in 30 min | ±30 s changes nothing | identity conclusion is the fragile one, not the timing |
| G6 | 7 simultaneous closures at 17:00 | ±30 s changes nothing; the true ends are anywhere in [arrival, 17:00] | "logging batch" **stable**; everything else unknown |
| G7 | 5 × 15.00 min service, waits 11…+2 min | service  (0, 30 min), wait  (15, +30) min | **nothing** survives except "a 15min grid exists" |
| G8 | 7.00 min service, zerogap handoffs | service  (6, 8) min; handoff gap ±60 s | seriality **stable**; exact chaining and the safetywindow overlap not stable |

**[Obs]/[Hyp] Extra deduction for G7.** Negative waits are impossible for real starts, so they discriminate among rounding rules. If `start` were **floored** to the quarter hour, true start  [14:00, 14:15) and every arrival (14:01–14:11) can precede it — selfconsistent. If `start` were **ceiled**, true start  14:00 and M05's 14:11 arrival still yields a negative wait — inconsistent, unless the upstream clock is skewed. The floor rule is therefore the parsimonious reading, and it implies service  (0, 30 min), not 15 min. The naive 15.00 min is *exactly one grid step*, which is what quantization produces regardless of the truth.

**5.3 The same temporal pattern, several physical explanations (each unresolved in the data).**
1. **G1 — four coextensive intervals:** one 4up fixture cycle with parallel heads (physical batch) | one load/unload cycle with serial labelling inside (start batching/common setup) | one cyclelevel PLC record or minute truncation of a serial cycle (logging batch). Discriminator: perhead PLC subsignals; the capacity4 × parallelism4.00 match is corroborating but not decisive.
2. **G2 — one common start:** the "begin group" button (UI logging artefact) | a real common setup/lot presentation at 10:00 | floorrounding of four starts into the 10:00 cell. Discriminator: screen audit log of button presses (all 4 arrivals were complete by 09:54, so a real common start is not excluded).
3. **G3 — one common end:** folderlevel bulk submit (deferred logging) | all four reviews genuinely finished at 11:30 | a form defaulting the completion field to the folder close time. Discriminator: distinct 7–8 min start spacing + "reviews performed throughout the halfhour" + all `recorded_at` = 11:31.
4. **G7 — both boundaries equal on a grid:** 15min coarsening | a real group measurement finishing inside one grid cell | crosssystem rounding plus clock skew. Discriminator: the negative waits (only coarsening with a floor rule, or clock skew, explains them).
5. **G4/G8 — exact chaining `complete_i = start_(i+1)`:** genuine onepiece flow | a single scan action writing both events | minute truncation hiding up to 59 s of slack. Discriminator: scanlevel audit or raw seconds.
6. **G5 — heavy overlap on one resource:** 3 analysts on a shared login | one analyst multitasking | durations inflated by open/close form conventions. Discriminator: badge data (absent).

# 6. Data-collection redesign

Priorities are ordered by "how many of the ambiguities above they remove per unit of effort".

**P0 — semantics of the lifecycle (removes G2, G3, G6, G7 ambiguity).**
1. **Execution timestamps:** emit separate, named fields `started_at`, `work_completed_at`, `submitted_at`, `recorded_at`. Never let a form default `start = complete`; a missing start must be null, never the submit time. Reject zerolength intervals at entry or mark them `mode=bulk_logging`. *[Rec]*
2. **Group/batch identifiers:** add `group_id` (lot/fixture/folder/batch), `group_role` (`member_of_batch` / `batch_head`), `items_in_group`, and one row per item with its own lifecycle. A UI "begin group" button must write one *group* event, not four item starts. *[Rec]*
3. **Source clocks:** NTP/PTPsync all sources and log `source_clock_id` + measured offset/skew per record (or per session). This is the only way to test G8's downtime hypothesis and to convert G7's negative waits into an attributable defect. *[Rec]*

**P1 — identity and capacity (removes G5, enables real utilization).**
4. **Resource/badge identity:** store `acting_account` and `acting_person` (badge) separately, plus `session_id`; forbid shared logins for execution events or require badge reauthentication at `start`. *[Rec]*
5. **Workstation capacity:** store `station_id`, `capacity` (e.g. fixture = 4), `heads_or_operators`, and `batch_cycle_id`, so that "4 intervals of 3 min" can be read as one cycle of 3 min instead of 12 min of demand. *[Rec]*
6. **Pause/resume and downtime events:** firstclass `blocked_start/blocked_end` with `reason_code` (safety check, changeover, break, no demand), logged in the *same* clock as execution; plus a downtime register the queue system can read. This separates G8's "blocked" from "queueing". *[Rec]*

**P2 — metrology of the timestamps (removes the precision contradictions).**
7. **Precision metadata:** store `precision`, `accuracy_estimate`, `rounding_rule` (`floor`/`ceil`/`nearest`) and `resolution` per source and per field, and keep the raw sensor timestamp at native resolution alongside the rounded display value. *[Rec]*
8. **Recording timestamps:** `recorded_at` at millisecond resolution from one ingestion sequencer, guaranteed monotone with `sequence_no` (G5 currently violates this); if rows can be resequenced, keep `original_sequence_no`. *[Rec]*
9. **Queue arrival:** carry `queue_arrival` with its own `source_clock_id` and resolution, so crosssystem wait can be bounded instead of silently negative. *[Rec]*

**P3 — validation and calibration.**
10. **Continuous dataquality monitors:** negative waits, zero/negative durations, `recorded_at` vs `sequence_no` monotonicity, "documented precision vs observed grid" (the 1 s vs 60 s conflict in G1/G4/G8), and ` service / wall clock > known capacity` (the 2.3–5.0 anomalies). *[Rec]*
11. **Small observational time study** (2–3 shifts) at G1, G2, G5 and G8 to physically verify fixture simultaneity, the begingroup semantics, analyst staffing behind `analyst_pool`, and the scanhandoff protocol. Cheap, and it is the only evidence that can settle the §5.3 cases. *[Rec]*

# 7. Final adjudication

**A. Interpretable now, with high confidence (no operational validation required).**
- **G1** — the unit of work is **one machine cycle over 4 items (09:00–09:03)**, not four independent 3min jobs; the 4.00 parallelism factor matches the documented fixture capacity of exactly 4. Use batchserver semantics (capacity 4, cycle 3 min). *Confidence 65 for the cycle claim; the intracycle simultaneity of the labelling itself remains unproven and must be reported as such.*
- **G4** — **strictly serial onepiece flow** (parallelism 1.00, 4/4 handoffs); peritem service 4.00 min ± 1 min and waiting 719 min are usable. The only caveat is the ±60 s handoff gap.
- **G6** — confidently a **logging batch** (bulk form at 17:00, service = 0, single write). What is *not* recoverable is where the work happened.
- **G3** — confidently an **end/submission batch**: treat every `complete` as a rightcensored upper bound at 11:30 and use `start` values as the usable signal.

**B. Exclude from, or intervalcensor in, queue/performance analysis.**
- **G7 — exclude entirely** from waiting, service, utilization and bottleneck statistics. Every lifecycle value must be intervalcensored to its 15min cell (service  (0, 30 min), wait  (15, +30 min)); the zero service variance and the four negative waits make point estimates indefensible. Only "cycle time to the grid cell" is reportable.
- **G6 — exclude from wait/service decomposition**; report only cycle time (arrival  17:00, bias  2 min from entry lag + rounding) and recover true service from the ticket audit trail. Do **not** use it for utilization or bottleneck ranking (it hides TF completely).
- **G2 — censor the start side**: `start` is a group action; rederive peritem starts as `complete_(i1)` or keep start as leftcensored at 10:00. Recorded waiting (6–18 min) understates true waiting (18–27 min).
- **G3 — censor the end side** as above (service 6–28 min is inflated up to 4×).
- **G5 — exclude all resourcelevel measures** (utilization, peranalyst productivity, handover) until identity is resolved; percase waiting is still usable.

**C. Require operational validation before any performance conclusion.**
- **G8** — the queue is explainable either by a **capacity block (15:00–15:35 safety check)** or by chronic undercapacity; the two source clocks have an unknown offset, so the overlap is a hypothesis. Validate the clock offset and the scanhandoff protocol before recommending capacity.
- **G5** — needs badgelevel identity (or an explicit staffing record for `analyst_pool`) to choose between "3 analysts at 78%" and "impossible 233%".
- **G1** — needs perhead/peritem PLC signals before any claim about peritem service time or operatorequivalent utilization.
- **G2 / G3** — needs UI/form instrumentation (begingroup audit, "finished vs submitted" fields) before the recorded service times are used in any SLA or capacity model.

**Crosscutting rule adopted throughout [Rec]:** a PM4Py (or any) batch label is admitted as *operational* batching only when it is corroborated by at least one nontemporal fact — known capacity (G1's 4up fixture), UI/protocol semantics (G2's begingroup button, G3's folder submit), an explicit group identifier, or a physical observation. Otherwise the label is reported as "temporal coincidence within tolerance", and — as G4/G8 at `merge_distance = 300 s` and G6's "Simultaneous" batch of zerolength intervals demonstrate — it may describe the logger rather than the shop floor.
