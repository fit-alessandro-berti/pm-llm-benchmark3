"""

Scope: descriptive / structural diagnostics ONLY.

    print(f"  !! WARNING: {msg}")


def head(title: str) -> None:
    print("\n" + "=" * 78)
    print(title)
    print("=" * 78)

assert set(df["group"]) == set(GROUPS)

        median_cycle=("cycle_time_h", "median"),
        mean_wait=("wait_before_review_h", "mean"),
        mean_service=("review_service_h", "mean"),
        rework_case_pct=("rework_count", lambda s: 100.0 * (s > 0).mean()),

    )
    .loc[GROUPS]

    df.groupby(["complexity", "group"])
    .agg(
        n=("case_id", "count"),
        mean_cycle=("cycle_time_h", "mean"),
        mean_wait=("wait_before_review_h", "mean"),
        mean_service=("review_service_h", "mean"),
        rework_case_pct=("rework_count", lambda s: 100.0 * (s > 0).mean()),

)

        warn(f"complexity={cx}, group={g}: n={int(row['n'])} (<{MIN_CELL}); stratum mean unstable")

weights = df["complexity"].value_counts(normalize=True).sort_index()
print("Common weights:", {k: round(v, 4) for k, v in weights.items()})

for metric in ["cycle_time_h", "wait_before_review_h", "review_service_h"]:
    sm = df.groupby(["complexity", "group"])[metric].mean()
    est = {}
    for g in GROUPS:
        num = den = 0.0
        for cx in weights.index:
            if (cx, g) in sm.index:
                num += weights[cx] * sm.loc[(cx, g)]
                den += weights[cx]
            else:
                warn(f"{metric}: no {g} cases at complexity={cx}; renormalising weights")
        est[g] = num / den if den else float("nan")
    print(
        f"{metric:>22}: Orion={est['Orion']:6.2f}  Nova={est['Nova']:6.2f}  "
        f"contrast={est['Orion'] - est['Nova']:+.2f} h"

    .size()

    .reindex(columns=GROUPS, fill_value=0)
)

for (cx, sh), row in overlap.iterrows():
    o, n = int(row.get("Orion", 0)), int(row.get("Nova", 0))
    if o == 0 or n == 0:
        warn(
            f"NO OVERLAP at complexity={cx}, shift={sh} "
            f"(Orion={o}, Nova={n}): group contrast unidentified in this cell"
        )
    elif o < MIN_CELL or n < MIN_CELL:
        warn(f"thin support at complexity={cx}, shift={sh} (Orion={o}, Nova={n})")

# ------------------------------------- 5a. trace variants per group
head("5a) TRACE VARIANTS PER GROUP (frequency, then mean cycle)")
variants = (
    df.groupby(["group", "trace"])
    .agg(n=("case_id", "count"), mean_cycle=("cycle_time_h", "mean"))
    .round(2)
)
for g in GROUPS:
    print(f"\n-- {g} --")

head("1) CRUDE GROUP COMPARISON (completed cases only)")

    .agg(

    if int(row["n"]) < MIN_CELL:
        warn(f"complexity={cx}, group={g}: n={int(row['n'])} (<{MIN_CELL}); stratum mean unstable")

head(
    "3) COMPLEXITY-STANDARDIZED CONTRAST\n"
    "   Common weighting = pooled complexity mix of all 30 completed cases:\n"
    "   Low 12/30=0.400, Medium 10/30=0.333, High 8/30=0.267"
)
weights = df["complexity"].value_counts(normalize=True).sort_index()

for metric in ["cycle_time_h", "wait_before_review_h", "review_service_h"]:
    sm = df.groupby(["complexity", "group"])[metric].mean()
    est = {}
    for g in GROUPS:
        num = den = 0.0
        for cx in weights.index:
            if (cx, g) in sm.index:
                num += weights[cx] * sm.loc[(cx, g)]
                den += weights[cx]
            else:
                warn(f"{metric}: no {g} cases at complexity={cx}; renormalising weights")
        est[g] = num / den if den else float("nan")

        f"{metric:>22}: Orion={est['Orion']:6.2f}  Nova={est['Nova']:6.2f}  "

        "Request Info) are entered - a routing/case-mix difference, not a\n"

    if int(row["count"]) < MIN_CELL:

## 8. Residual uncertainty

1. **Residual case severity within strata** — the router may use signals (amounts, fraud flags, history) beyond the 3-level `complexity` field, leaving within-stratum confounding intact. *Most valuable addition: capture the routing engine's decision inputs/score at Classify for every case.*
2. **Queue dynamics are unobserved** — `arrival_load` is a coarse binary, `wait_before_review_h` conflates queueing with compliance waiting, and there are no timestamps to decompose them. *Most valuable addition: timestamped activity log plus a queue-length/staffing snapshot at the moment of routing.*
3. **Selection into the completed sample** — abandoned, withdrawn, or still-open cases are absent, so conditioning on `case_completed` may distort every comparison if drop-out depends on delay and group/shift. *Most valuable addition: a full intake cohort including open, abandoned, and withdrawn cases with closure reasons.*
4. **Group–shift collinearity and positivity holes** — Orion has no Medium/High day cases, so no design on current data can separate group from shift there. *Most valuable addition: a design change — randomized or alternating assignment within shift×complexity cells (the pilot) to manufacture overlap.*
5. **Quality is coarsely measured** — a single 30-day appeal flag, potentially contaminated by post-process factors, plus the anomaly that rework cases show no extra service time. *Most valuable addition: longer-horizon quality fields (reopens, final appeal outcomes, adjuster error flags) linked to case and reviewer.*
