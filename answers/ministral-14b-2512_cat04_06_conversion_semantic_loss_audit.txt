# --- Bounded Trace Exploration ---
print("\n=== Bounded Trace Exploration ===")
def explore_traces(net, im, fm, max_visible_length=5, max_loop_iterations=2):
    from pm4py.objects.petri_net.utils import get_enabled_transitions
    from collections import deque

    traces = []
    visited = set()
    queue = deque()
    queue.append((im, [], 0, 0))  # (marking, trace, visible_length, loop_iterations)

    while queue:
        marking, trace, visible_len, loop_iter = queue.popleft()
        if marking == fm and visible_len > 0:
            traces.append(trace)
            continue

        enabled = get_enabled_transitions(net, marking)
        for t in enabled:
            new_marking = marking.copy()
            new_marking[t] += 1
            new_marking[t.input_places[0]] -= 1
            new_trace = trace + [t.label if t.label else "tau"]

            # Skip silent transitions for trace length
            if t.label:
                new_visible_len = visible_len + 1
            else:
                new_visible_len = visible_len

            # Skip if bounds exceeded
            if new_visible_len > max_visible_length or loop_iter > max_loop_iterations:
                continue

            # Check for loops (e.g., Correct Defect)
            if t.label == "Correct Defect":
                new_loop_iter = loop_iter + 1
            else:
                new_loop_iter = loop_iter

            if (new_marking, new_loop_iter) not in visited:
                visited.add((new_marking, new_loop_iter))
                queue.append((new_marking, new_trace, new_visible_len, new_loop_iter))

    return traces

# Explore original Petri net
original_traces = explore_traces(net_from_powl, im_from_powl, fm_from_powl)

**Key Observations:**
- **True concurrency vs. interleaving**: Petri net preserves true concurrency, but BPMN may approximate it as parallel gateways (which can be misinterpreted as sequential).
- **Optional Legal Review**: The silent transition in POWL is lost in BPMN, which may force explicit routing (e.g., exclusive gateway) that is not semantically equivalent.
- **Binary-loop semantics**: The loop in POWL (with `Correct Defect` separating iterations) is preserved in Petri net and BPMN, but the minimum iteration count may not be visually obvious in BPMN.
- **Join before Release**: The synchronization before `Release Change` is preserved in Petri net and BPMN, but the lack of an explicit edge in POWL between `optional_legal` and `verification_loop` may lead to ambiguity in BPMN (e.g., whether the join is implicit or explicit).
- **Activity-label uniqueness**: All labels in the source are unique, but BPMN may introduce duplicate labels for routing nodes (e.g., gateways).
- **Initial/final markings**: Petri net preserves exact initial and final markings; BPMN uses start/end events, which are semantically equivalent but may not be visually distinct.

# 5. Round-Trip Validation Code

```python
def validate_round_trip(original_net, original_im, original_fm, roundtrip_net, roundtrip_im, roundtrip_fm):
    print("\n=== Round-Trip Validation ===")

    # 1. Soundness check
    original_sound = pm4py.check_soundness(original_net, original_im, original_fm)
    roundtrip_sound = pm4py.check_soundness(roundtrip_net, roundtrip_im, roundtrip_fm)
    print(f"Original soundness: {original_sound}")
    print(f"Round-trip soundness: {roundtrip_sound}")

    # 2. Footprint comparison
    original_footprints = pm4py.discover_footprints(original_net, original_im, original_fm)
    roundtrip_footprints = pm4py.discover_footprints(roundtrip_net, roundtrip_im, roundtrip_fm)
    footprint_match = set(original_footprints) == set(roundtrip_footprints)
    print(f"Footprint match: {footprint_match}")

    # 3. Reachability graph comparison (bounded)
    original_rg = pm4py.convert_to_reachability_graph(original_net, original_im, original_fm)
    roundtrip_rg = pm4py.convert_to_reachability_graph(roundtrip_net, roundtrip_im, roundtrip_fm)
    print(f"Original RG states: {len(original_rg.states)}")
    print(f"Round-trip RG states: {len(roundtrip_rg.states)}")

    # 4. Bounded trace comparison
    original_traces = explore_traces(original_net, original_im, original_fm)
    roundtrip_traces = explore_traces(roundtrip_net, roundtrip_im, roundtrip_fm)
    trace_match = set(tuple(t) for t in original_traces) == set(tuple(t) for t in roundtrip_traces)
    print(f"Trace match (bounded): {trace_match}")

    # 5. Final marking equality
    final_marking_match = original_fm == roundtrip_fm
    print(f"Final marking match: {final_marking_match}")

    # 6. Positive/negative witnesses
    original_witnesses = pm4py.get_enabled_transitions(original_net, original_im)
    roundtrip_witnesses = pm4py.get_enabled_transitions(roundtrip_net, roundtrip_im)
    print(f"Original initial transitions: {len(original_witnesses)}")
    print(f"Round-trip initial transitions: {len(roundtrip_witnesses)}")

    return {
        "soundness": original_sound == roundtrip_sound,
        "footprints": footprint_match,
        "traces": trace_match,
        "final_marking": final_marking_match,
        "exhaustive": False  # Bounded exploration is not exhaustive
    }

# Run validation if round-trip Petri net exists
if 'net_from_bpmn' in locals():
    validation = validate_round_trip(
        net_from_powl, im_from_powl, fm_from_powl,
        net_from_bpmn, im_from_bpmn, fm_from_bpmn
    )
    print("\nValidation summary:")
    for k, v in validation.items():
        print(f"{k}: {v}")

# 6. Visual-Semantics Review

**Potential Misreads in BPMN:**
1. **Parallel vs. Exclusive Gateways**: The unordered concurrency between `Assess Risk` and `Prepare Evidence` may be rendered as a parallel gateway in BPMN, but the lack of an explicit edge between `optional_legal` and `verification_loop` could lead to confusion about whether they are truly concurrent or just unordered.
2. **Silent Optional Behavior**: The silent transition (`skip_legal`) is invisible in BPMN, which may force the use of an exclusive gateway with no incoming edge, making the optional nature unclear.
3. **Loop Visualization**: The loop in BPMN may not explicitly show the minimum iteration count (at least one verification iteration), which is critical for the source semantics.
4. **Silent Routing Nodes**: If silent transitions are not handled explicitly, BPMN may introduce implicit routing nodes (e.g., invisible gateways) that obscure the original semantics.
5. **Graph Layout Ambiguity**: The lack of an edge between `optional_legal` and `verification_loop` in POWL may lead to a BPMN layout where these branches appear sequentially rather than concurrently, even if the underlying Petri net preserves concurrency.

**Proposed Annotations:**
- Add a comment in BPMN to clarify that `Assess Risk` and `Prepare Evidence` are truly concurrent (not just unordered).
- Use a labeled exclusive gateway for `optional_legal` with a note that one branch is silent.
- Explicitly annotate the loop to indicate a minimum iteration count.
- Include a legend or documentation explaining that silent transitions are hidden in the BPMN visualization.

# 7. Rejection Criteria

A conversion result should be rejected or quarantined if:
1. **Unsoundness**: The converted Petri net or BPMN is not sound (e.g., dead transitions, livelocks).
2. **Loss of Required Interleaving**: The BPMN or process tree does not preserve the unordered concurrency between `Assess Risk` and `Prepare Evidence`.
3. **Admission of Release Before Verification**: The converted model allows `Release Change` to occur without completing the verification loop.
4. **Collapse of Optional Behavior**: The silent transition (`skip_legal`) is not represented in the BPMN, leading to a forced path through `Legal Review`.
5. **Failed Exact-Final-Marking Test**: The final marking of the round-trip Petri net does not match the original.
6. **Non-Block-Structured Process Tree**: The process tree conversion fails due to non-block-structured POWL (e.g., arbitrary concurrency or silent transitions).
7. **Missing Minimum Iteration**: The BPMN or Petri net does not enforce at least one verification iteration.
8. **Ambiguous Gateway Interpretation**: The BPMN uses an exclusive gateway where a parallel gateway is semantically correct (or vice versa).
9. **Silent Transition Handling**: Silent transitions are not explicitly represented in the BPMN, leading to implicit routing that obscures the original semantics.
10. **Failed Trace Validation**: The bounded trace exploration reveals that the round-trip model admits or rejects traces that the original does not.

# 8. Canonical Representation Recommendation

**Authoritative Representation:**
- **Canonical Model**: The original POWL model should be the authoritative representation. It is the most precise and unambiguous representation of the source semantics.
- **Derivative Artifacts**:
  - **Petri Net**: A faithful conversion of POWL to Petri net should be stored as a derivative artifact, with validation evidence (soundness, footprints, traces) attached.
  - **BPMN**: The BPMN model should be treated as a visualization artifact, not authoritative. It should include annotations to clarify ambiguities (e.g., concurrency, optional behavior).
  - **Process Tree**: Only allowed if the POWL is block-structured (which it is not in this case). If conversion succeeds, it should be treated as a derivative artifact with validation evidence.

**Validation Evidence:**
- For the Petri net: Soundness check, footprint discovery, bounded trace exploration.
- For the BPMN: Visual annotations, round-trip validation to Petri net, and explicit documentation of semantic approximations.
- For the process tree (if applicable): Process tree validation, round-trip to Petri net, and comparison with original POWL.

**Future Edits:**
- Edits should be made to the POWL model, with derivative artifacts regenerated and validated.
- Round-trip drift should be avoided by treating POWL as the single source of truth and regenerating all derivative artifacts from it.
- Any manual edits to BPMN or Petri net should be flagged as non-authoritative and require regeneration from POWL.

# 9. Residual Uncertainty

Even if all tests pass, the following claims cannot be established:
1. **Exhaustive Trace Equivalence**: The bounded trace exploration is not exhaustive and may miss rare or complex traces (e.g., those involving many loop iterations or silent transitions).
2. **Visual Interpretation of Concurrency**: While the Petri net may preserve true concurrency, the BPMN visualization may still be misinterpreted as sequential due to layout or gateway ambiguity.
3. **Silent Transition Semantics**: The silent transition (`skip_legal`) is invisible in BPMN, and its exact semantics (e.g., whether it represents a true choice or just a no-op) may not be recoverable from the visualization.
4. **Minimum Iteration Enforcement**: While the Petri net enforces at least one verification iteration, the BPMN may not visually or semantically enforce this constraint in all cases.
5. **Gateway Type Ambiguity**: The BPMN may use a parallel gateway where an exclusive gateway is semantically correct (or vice versa), leading to ambiguity in the routing logic.
6. **Final Marking Interpretation**: Even if the final marking matches, the exact meaning of the marking (e.g., which activities are truly complete) may not be clear from the BPMN alone.
