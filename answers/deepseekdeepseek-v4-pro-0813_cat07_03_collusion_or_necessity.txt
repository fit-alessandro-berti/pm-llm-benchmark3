def retro_rule_violations(df):
    viol=[]
    for c,g in df.groupby("case:concept:name"):
        ev = g[g["concept:name"]=="Emergency Override"]
        rj = g[g["concept:name"]=="Retrospective Justification"]
        for _,e in ev.iterrows():
            follows = rj[rj["time:timestamp"] > e["time:timestamp"]]
            within3 = follows[follows["time:timestamp"] <= e["time:timestamp"] + pd.offsets.BDay(3)]
            if len(within3)==0 or all(within3["org:resource"]==e["org:resource"]):
                viol.append(c)
    return viol
