# 5. Round-Trip Validation Code

```python
from collections import deque
import pm4py

# ------------------------------------------------------------------
# Witness traces (visible labels only). tau / None labels are hidden.
# ------------------------------------------------------------------
POSITIVE_WITNESSES = [
    ["Register Change", "Assess Risk", "Prepare Evidence",
     "Legal Review", "Configure Trial", "Inspect Trial",
     "Release Change", "Archive Change"],
    ["Register Change", "Prepare Evidence", "Assess Risk",
     "Configure Trial", "Inspect Trial",
     "Release Change", "Archive Change"],            # legal skipped
    ["Register Change", "Assess Risk", "Prepare Evidence",
     "Configure Trial", "Inspect Trial", "Correct Defect",
     "Configure Trial", "Inspect Trial",
     "Release Change", "Archive Change"],            # 2 iterations
    ["Register Change", "Assess Risk", "Legal Review", "Prepare Evidence",
     "Configure Trial", "Inspect Trial",
     "Release Change", "Archive Change"],            # legal overlaps with verification pre-start
]

NEGATIVE_WITNESSES = [
    # Release before verification completed
    ["Register Change", "Assess Risk", "Prepare Evidence",
     "Release Change", "Archive Change"],
    # Verification without Prepare Evidence
    ["Register Change", "Assess Risk", "Configure Trial", "Inspect Trial",
     "Release Change", "Archive Change"],
    # Configure after Inspect
    ["Register Change", "Assess Risk", "Prepare Evidence",
     "Inspect Trial", "Configure Trial",
     "Release Change", "Archive Change"],
    # Loop re-entry without Correct Defect
    ["Register Change", "Assess Risk", "Prepare Evidence",
     "Configure Trial", "Inspect Trial",
     "Configure Trial", "Inspect Trial",
     "Release Change", "Archive Change"],
    # Archive before Release
    ["Register Change", "Assess Risk", "Prepare Evidence",
     "Configure Trial", "Inspect Trial",
     "Archive Change", "Release Change"],
    # Legal Review before Assess Risk
    ["Register Change", "Legal Review", "Assess Risk", "Prepare Evidence",
     "Configure Trial", "Inspect Trial", "Release Change", "Archive Change"],
]

VISIBLE_LEN_BOUND = 14      # bound on visible (non-tau) steps
LOOP_ITER_BOUND = 3         # bound on re-entry count per loop construct
SILENT_RUN_BOUND = 8        # bound on consecutive silent steps before giving up

def is_final(net, marking, fm):
    try:
        return marking == fm
    except Exception:
        return False

def visible_label(t):
    try:
        return t.label
    except Exception:
        return None

def explore_accepts(net, im, fm, target_visible):
    """
    Bounded DFS over the reachability graph, hiding transitions whose label is None.
    Requires exact final-marking equality for a completed trace.
    Avoids infinite silent cycling by bounding consecutive silent steps and total visible length.
    Returns True iff the net can execute exactly the visible sequence in target_visible
    and end exactly in fm.
    """
    if net is None or im is None or fm is None:
        return False
    if len(target_visible) > VISIBLE_LEN_BOUND:
        return False

    start = (im, 0, 0)   # (marking, idx_into_visible, consecutive_silent)
    seen = set()
    stack = [start]

    def key(marking, idx, silent_run):
        try:
            return (frozenset(marking.items()) if hasattr(marking, "items")
                    else frozenset(marking), idx, silent_run)
        except Exception:
            return (repr(marking), idx, silent_run)

    while stack:
        marking, idx, silent_run = stack.pop()
        k = key(marking, idx, silent_run)
        if k in seen:
            continue
        seen.add(k)

        if idx == len(target_visible):
            if is_final(net, marking, fm):
                return True
            # Continue silently only, in case tau steps lead to final marking
            if silent_run >= SILENT_RUN_BOUND:
                continue
        else:
            if silent_run >= SILENT_RUN_BOUND:
                # Cannot keep silently drifting; require a visible step now.
                pass

        try:
            enabled = pm4py.get_enabled_transitions(net, marking)
        except Exception:
            enabled = []

        for t in enabled:
            lbl = visible_label(t)
            try:
                from pm4py.objects.petri_net.semantics import fire_transition
                new_marking = fire_transition(t, marking)
            except Exception:
                # Fallback: try pm4py utility if available
                try:
                    new_marking = pm4py.play_out_transition(net, marking, t)  # not guaranteed
                except Exception:
                    continue

            if lbl is None:
                # Silent move: do NOT advance visible index
                if silent_run + 1 <= SILENT_RUN_BOUND:
                    stack.append((new_marking, idx, silent_run + 1))
            else:
                if idx < len(target_visible) and lbl == target_visible[idx]:
                    stack.append((new_marking, idx + 1, 0))
        # end for
    return False

def validate_accepting_net(net, im, fm, label):
    print(f"\n=== Trace validation: {label} ===")
    if net is None or im is None or fm is None:
        print("  null net/markings; skipped")
        return
    # Footprints (defensible semantic fingerprint)
    try:
        fps = pm4py.discover_footprints(net, im, fm)
        print(f"  footprints discovered: {len(fps) if hasattr(fps, '__len__') else 'n/a'} entries")
    except Exception as e:
        print(f"  discover_footprints raised: {e!r}")

    # Reachability graph (defensible for sound, bounded nets)
    try:
        rg = pm4py.convert_to_reachability_graph(net, im, fm)
        n_states = getattr(rg, "states", None)
        print(f"  reachability graph: {len(n_states) if n_states is not None else 'n/a'} states")
    except Exception as e:
        print(f"  convert_to_reachability_graph raised: {e!r}")

    # Positive witnesses
    for w in POSITIVE_WITNESSES:
        ok = explore_accepts(net, im, fm, w)
        print(f"  positive witness {'ACCEPTED' if ok else 'REJECTED'}: {w}")

    # Negative witnesses
    for w in NEGATIVE_WITNESSES:
        ok = explore_accepts(net, im, fm, w)
        status = "REJECTED (good)" if not ok else "ACCEPTED (BAD)"
        print(f"  negative witness {status}: {w}")

# Run on the source net and every round-tripped net
validate_accepting_net(net_from_powl, im_from_powl, fm_from_powl, "net_from_powl")
validate_accepting_net(net_rt_from_bpmn_net, im_rt_from_bpmn_net, fm_rt_from_bpmn_net,
                       "net_rt_from_bpmn_net")
validate_accepting_net(net_rt_from_bpmn_tree, im_rt_from_bpmn_tree, fm_rt_from_bpmn_tree,
                       "net_rt_from_bpmn_tree")
```

**Bounded exploration disclaimer.** The visible-trace exploration is **not exhaustive** over the full language. It bounds visible length to `VISIBLE_LEN_BOUND`, loop re-entry to `LOOP_ITER_BOUND`, and consecutive silent moves to `SILENT_RUN_BOUND`. It is exhaustive **over the chosen bound and over the chosen positive/negative witnesses**: every trace within these bounds that reaches the final marking must do so with exact final-marking equality. It cannot, by construction, prove language equality of the source and round-tripped nets.

# 6. Visual-Semantics Review

Even when the underlying Petri net is sound and the BPMN serialization round-trips to a behaviorally equivalent net, the **rendered** BPMN diagram can be misread:

- **Parallel gateways.** PM4Py may emit parallel gateways for the POWL partial order between `Assess Risk` and `Prepare Evidence`. A reader can mistake a parallel gateway for an AND-split that **forces** both branches, which is semantically correct here, but the same drawing convention may be reused for the unordered pair `Optional Legal Review  Verification Loop`, where one branch contains a silent skip — visually confusing.
- **Exclusive gateways.** The XOR-with-`tau` for optional legal review is likely rendered as an exclusive gateway with one labeled outgoing edge (`Legal Review`) and one empty edge. The empty edge is easily overlooked; stakeholders may believe `Legal Review` is mandatory.
- **Loop visualization.** The binary LOOP is typically flattened into a sequence of gateways in BPMN. There is no native BPMN “redo” marker, so the reader may see `Configure Trial  Inspect Trial  Correct Defect  Configure Trial  …` as a linear sequence rather than a loop. This obscures INV-07 and INV-08.
- **Silent routing nodes.** BPMN diagrams produced from Petri nets often contain anonymous gateways with no label and no semantic hint. These are not no-ops; they encode control flow that was explicit in the POWL operator structure.
- **Apparent ordering of graph layout.** A layered layout engine may draw `Optional Legal Review` to the left of the verification loop, suggesting `Legal Review` precedes `Configure Trial`. There is **no such edge in the source**. Layout is not a semantic statement; only the underlying net’s arcs are.
- **Start/end events.** BPMN start and end events are an approximation of the Petri-net initial and final markings. They are not interchangeable with the marking semantics of the workflow net.

**Recommended annotations / documentation to ship alongside the BPMN file:**

1. A sidecar JSON or Markdown file listing, for each BPMN gateway, its origin (which POWL operator produced it: XOR, LOOP, or partial-order).
2. An explicit note on the diagram (or in documentation) that `Legal Review` is optional via a silent skip branch, and that the empty edge from the XOR gateway is intentional.
3. A note that `Optional Legal Review` and `Verification Loop` are unordered; any left-to-right appearance is a layout artifact.
4. An explicit loop marker on the `Correct Defect  Configure Trial` arc, or a sub-process boundary marking the LOOP scope.
5. A legend distinguishing **silent routing gateways** (no label, introduced by conversion) from **semantic gateways** (derived from a POWL operator).
6. A statement that the BPMN is a **derived visualization**; the authoritative artifact is the POWL and its accepting Petri net.

# 7. Rejection Criteria

A conversion result should be **rejected or quarantined** if any of the following holds:

1. **Unsound net.** `pm4py.check_soundness(...)` on `net_from_powl` or any round-tripped net reports the net as unsound (deadlocks, livelocks, or unreachable markings).
2. **Not a workflow net.** `pm4py.check_is_workflow_net(...)` returns False for `net_from_powl` or for a round-tripped net that is meant to be authoritative.
3. **Null markings.** `convert_to_petri_net(...)` returns a net with a null initial or final marking, or `convert_to_bpmn(...)` returns None.
4. **Failed exact-final-marking test.** A positive witness trace reaches a marking other than `fm` and is nonetheless reported as accepted, or fails to reach `fm` exactly when the visible sequence is completed.
5. **Loss of a required interleaving.** Either of the two orderings of `Assess Risk` / `Prepare Evidence` is rejected, or any valid interleaving of `Optional Legal Review` with the verification loop is rejected.
6. **Admission of release before verification.** A trace reaching `Release Change` while `Configure Trial`/`Inspect Trial` have not executed (or while `Assess Risk` / `Prepare Evidence` are incomplete) is accepted.
7. **Collapse of optional behavior.** `Legal Review` becomes mandatory (skip branch lost), or `Legal Review` becomes unreachable.
8. **Loop semantics broken.** A second verification iteration is accepted without `Correct Defect`, or zero iterations are accepted (loop body skipped entirely).
9. **Finality violation.** `Archive Change` is not the unique final visible activity, or the final marking is reachable without `Archive Change`.
10. **Label collision.** Two semantically distinct control-flow positions share a visible label in a way that is not authorized by the source’s loop re-entry.
11. **Conversion exception swallowed silently.** Any `convert_to_*` call raised an exception that the pipeline did not log, or whose exception was hidden to make the pipeline “succeed”.
12. **Round-trip drift.** A round-tripped net accepts a negative witness or rejects a positive witness that the source net handles correctly.

# 8. Canonical Representation Recommendation

- **Authoritative representation for storage: the source POWL object** (`source_powl`), together with its **accepting Petri net** (`net_from_powl`, `im_from_powl`, `fm_from_powl`) as the canonical executable form. The POWL is the original specification; the Petri net is the formal, markable operationalization produced by PM4Py’s own converter, with full invariant audit.
- **Separate model for stakeholder visualization: the BPMN diagram** (`bpmn_from_net`). BPMN should be treated as a **derived visualization artifact**, not as the system of record. It exists to communicate with non-PM4Py stakeholders.
- **Derivative artifacts and required validation evidence:**
  - `net_from_powl.png` — visual rendering of the authoritative net. Validation: workflow-net check, soundness check, positive/negative witness results.
  - `bpmn_from_net.png` — derived BPMN. Validation: round-trip back to a Petri net; workflow-net and soundness checks on the round-tripped net; trace-equivalence witness tests under the chosen bound; sidecar annotation file describing gateway provenance.
  - `tree_from_net.png`, `bpmn_from_tree.png` — only if the process-tree conversion succeeds. Validation: same as above plus an explicit check that the tree’s language is not a strict subset/superset of the source net’s language on the witness traces.
  - Round-tripped Petri nets (`net_rt_from_bpmn_net`, `net_rt_from_bpmn_tree`) — used only as validation evidence, never as the authoritative model.
- **Should a process-tree intermediate be allowed?** Only as a **non-authoritative, optional** derivative. The source POWL mixes a non-block-structured partial order with an XOR-with-skip and a binary LOOP; a single block-structured process tree cannot, in general, capture this combination. The process-tree intermediate may be used for visualization or compactness **only if** it survives the witness-trace validation; otherwise it must be discarded and the failure logged.
- **Avoiding round-trip drift in future edits:**
  1. Edits are made **only** to the POWL source; the Petri net and BPMN are regenerated from the POWL on every change.
  2. The BPMN file is never edited by hand; if a stakeholder needs a different BPMN layout, a layout-only transform is applied and recorded as a separate, non-authoritative artifact.
  3. Every regeneration runs the witness-trace validation suite; any regression quarantines the new artifact.
  4. A versioned manifest records, for each artifact: its source POWL hash, the PM4Py version, the conversion calls used, and the validation results.
  5. The process-tree intermediate is never persisted as authoritative; if it is used, it is regenerated and re-validated on every edit.

# 9. Residual Uncertainty

Even if every test above passes, the following claims **cannot** be established by the proposed validation:

1. **Full language equivalence between POWL and round-tripped BPMN/Petri net.** The bounded visible-trace exploration is not exhaustive over all trace lengths and loop iteration counts; it cannot prove the absence of any divergence beyond the chosen bounds.
2. **Behavioral equivalence under all markings, not just reachable ones.** Footprints and reachability-graph checks cover reachable states; they do not, by themselves, prove properties of unreachable or pathological markings that a different conversion path might introduce.
3. **Preservation of true concurrency (pomset/partial-order semantics), not just interleaving semantics.** Trace-based validation compares interleaved visible sequences; two models with identical interleaving languages can still differ in their true-concurrency semantics (e.g., one may collapse two independent events into a sequence).
4. **Faithfulness of BPMN **visualization** to the underlying net.** Layout, gateway labels, and silent routing nodes can still mislead a human reader even when the underlying net is behaviorally correct; visual review is qualitative, not a proof.
5. **Stability under future PM4Py versions.** Conversion functions may change; today’s validation evidence does not guarantee that a future PM4Py release will produce the same net or BPMN from the same POWL.
6. **Equivalence of the optional-legal silent skip branch across all hidden behaviors.** Hiding `None`-labeled transitions conflates all silent moves; the validation cannot distinguish a “correct” silent skip from an unrelated silent path that happens to produce the same visible trace.
7. **Industrial soundness** (e.g., absence of resource contentions, timing, data flow). The audit covers control-flow soundness only; it does not establish soundness with respect to data, resources, or business rules associated with the labels.
