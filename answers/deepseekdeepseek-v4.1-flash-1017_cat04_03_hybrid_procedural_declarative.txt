    n = ProcessTree(operator=Operator.SEQUENCE)
    n.children = list(children)
    return n

    n = ProcessTree(operator=Operator.XOR)
    n.children = list(children)
    return n

def parallel(*children):
    n = ProcessTree(operator=Operator.PARALLEL)
    n.children = list(children)
    return n

def loop(do, redo):
    n = ProcessTree(operator=Operator.LOOP)
    n.children = [do, redo]
    return n

import pandas as pd

def check_resource_and_data(df, case_attrs, zeta_emergency_hours=24):
    """
    df columns: case_id, activity, timestamp, resource
    case_attrs: dataframe columns case_id, emergency_authorization (bool),
                joint_authorization_by (str/list), maybe authorization_timestamp
    Returns list of dicts with case_id, check, verdict, detail
    """

    for cid, g in df.sort_values(["case_id", "timestamp"]).groupby("case_id"):
        acts = g.set_index("activity")
        # R9a: Verify Bank Account resource != applicable approval resource
        verify_resources = set(g.loc[g.activity == "Verify Bank Account", "resource"].dropna())
        approval_acts = ["Manager Approval", "Compliance Approval"]
        approval_rows = g[g.activity.isin(approval_acts)]
        for _, arow in approval_rows.iterrows():
            if not verify_resources:
                violations.append((cid, "R9-approval", "missing-resource", "Verify Bank Account resource missing"))
            elif arow["resource"] in verify_resources:
                violations.append((cid, "R9-approval", "fail", f"{arow['activity']} resource {arow['resource']} equals a Verify Bank Account resource"))
        # R9b: Capture Master Data resource != Activate Vendor resource
        cap = g[g.activity == "Capture Master Data"]
        act = g[g.activity == "Activate Vendor"]
        if not cap.empty and not act.empty:
            cap_res = set(cap["resource"].dropna())
            act_res = set(act["resource"].dropna())
            if not cap_res or not act_res:
                violations.append((cid, "R9-activate", "missing-resource", "Capture or Activate resource missing"))
            elif cap_res & act_res:
                violations.append((cid, "R9-activate", "fail", f"shared resource(s) {cap_res & act_res}"))
        # R10: at most once (also in log skeleton)
        for activity in ["Activate Vendor", "Reject Vendor", "Release First Payment"]:
            n = int((g.activity == activity).sum())
            if n > 1:
                violations.append((cid, "R10", "fail", f"{activity} occurs {n} times"))
        # R11: if Change Bank Details occurs after Activate Vendor
        act_time = g.loc[g.activity == "Activate Vendor", "timestamp"].min() if (g.activity == "Activate Vendor").any() else None
        changes = g[(g.activity == "Change Bank Details") & (g.timestamp > act_time)] if act_time is not None else g.iloc[0:0]
        for _, ch in changes.iterrows():
            susp = g[(g.activity == "Suspend Vendor") & (g.timestamp > ch.timestamp)]
            if susp.empty:
                violations.append((cid, "R11-suspend", "fail", "Change Bank Details after activation without later Suspend Vendor"))
            # another Verify after change and before any later Release First Payment after change
            later_releases = g[(g.activity == "Release First Payment") & (g.timestamp > ch.timestamp)]
            if not later_releases.empty:
                next_release_time = later_releases["timestamp"].min()
                reverify = g[(g.activity == "Verify Bank Account") & (g.timestamp > ch.timestamp) & (g.timestamp < next_release_time)]
                if reverify.empty:
                    violations.append((cid, "R11-reverify", "fail", "No extra Verify Bank Account between Change Bank Details and next Release First Payment"))
        # R5/R6 emergency check
        esc = g[g.activity == "Sanctions Escalation"]
        if not esc.empty:
            esc_time = esc["timestamp"].min()
            legal = g[g.activity == "Obtain Legal Opinion"]
            act_rows = g[g.activity == "Activate Vendor"]
            if act_rows.empty:
                # R4 eventually handled by DECLARE; if no legal at all, R4 violation
                if legal.empty:
                    violations.append((cid, "R4", "fail", "Sanctions Escalation without Obtain Legal Opinion"))
            else:
                act_time = act_rows["timestamp"].min()
                legal_after = legal[legal.timestamp > esc_time]
                if legal_after.empty:
                    violations.append((cid, "R4/R5", "fail", "Sanctions Escalation without Obtain Legal Opinion"))
                else:
                    legal_time = legal_after.timestamp.min()
                    if legal_time < act_time:
                        # R5 satisfied
                        pass
                    else:
                        # emergency exception under R6
                        ca = case_attrs[case_attrs.case_id == cid]
                        auth = False if ca.empty else bool(ca.iloc[0].get("emergency_authorization", False))
                        joint = None if ca.empty else ca.iloc[0].get("joint_authorization_by", None)
                        delay_hours = (legal_time - act_time).total_seconds() / 3600.0
                        if not auth or not joint or delay_hours > 24:
                            violations.append((cid, "R5/R6", "fail", f"Legal after activation without valid emergency exception (delay={delay_hours:.2f}h, auth={auth}, joint={joint})"))
    return violations

# 1. Policy Interpretation and Conflict Resolution

Material assumptions:

- A case is a trace keyed by case id. In conforming traces, accepted cases contain `Activate Vendor`; rejected cases contain `Reject Vendor`; `Close Onboarding` terminates the case.
- The procedural model is the canonical hard control-flow skeleton. DECLARE and Log Skeleton are used for cross-cutting and declarative constraints that are awkward or non-local in a ProcessTree.
- Clarification cycles are modeled procedurally only before `Classify Risk`. DECLARE is used to enforce the more general eventual/alternating response obligation for every `Request Clarification`.
- `Sanctions Escalation` is modeled only on the high-risk route, after `Run Sanctions Screening` and before/around `Enhanced Due Diligence`. If escalation is allowed on other routes, the procedural model must be revised; DECLARE still enforces R7.
- Post-activation bank changes are modeled only before the single `Release First Payment`. Post-release changes are handled by the external resource/data check.
- Missing resources or missing emergency-authorization attributes are not silently treated as compliance. Under the chosen policy, R5 is strict by default; R6 is a separate, externally evidenced exception mode.
- Support/confidence values in the DECLARE dictionary are configuration metadata, not empirical proof that a rule is legally correct.

R5/R6 conflict resolutions:

1. **Strict R5 (chosen canonical policy):** If `Sanctions Escalation` occurs, `Obtain Legal Opinion` must precede `Activate Vendor` under all circumstances. Emergency activation before the opinion is nonconforming in the event log. R6 is delegated to an external resource/data layer and is only applicable if a permissive deployment policy is explicitly enabled and external joint authorization is present. Compliance risk: legitimate emergency activations may be flagged as violations; however, the event-log default is legally conservative and avoids silently inventing authorization that is not in the log.
2. **Permissive R6 exception:** If `Sanctions Escalation` occurs and `Activate Vendor` happens before `Obtain Legal Opinion`, the case is compliant only if external case attributes prove joint authorization outside the event log and the opinion arrives within 24 hours. Compliance risk: this depends on attributes not present in the event log; missing, untrusted, or incomplete authorization data could allow unauthorized activation or create a weak audit trail.
3. **Hybrid operational policy:** Use strict R5 for automatic conformance and surface emergency cases as `policy-dependent` until a human verifies external joint authorization. This is not a silent combination: the event-log check remains strict, and R6 is an explicit override requiring external evidence.

Chosen implementation: **Strict R5 is the canonical model.** R6 is supported only as an explicit external validation mode. Under default execution, T6 fails at the procedural/R5 layer; under an authorized permissive R6 mode with valid external attributes, T6 becomes policy-dependent and may pass.

```python
from pm4py.objects.process_tree.obj import ProcessTree, Operator

def leaf(label):
    return ProcessTree(label=label)

def seq(*children):
    node = ProcessTree(operator=Operator.SEQUENCE)
    node.children = list(children)
    return node

def xor(*children):
    node = ProcessTree(operator=Operator.XOR)
    node.children = list(children)
    return node

def parallel(*children):
    node = ProcessTree(operator=Operator.PARALLEL)
    node.children = list(children)
    return node

def loop(do_node, redo_node):
    node = ProcessTree(operator=Operator.LOOP)
    node.children = [do_node, redo_node]
    return node

def tau():
    return ProcessTree()

# Zero or more clarification request/response cycles before Classify Risk.
clarification_cycle = loop(
    seq(
        leaf("Request Clarification"),
        leaf("Receive Clarification")
    ),
    tau()
)
clarifications = xor(tau(), clarification_cycle)

# Optional post-activation bank-change / suspension / re-verification cycle.
bank_change_cycle = loop(
    seq(
        leaf("Change Bank Details"),
        leaf("Suspend Vendor"),
        leaf("Verify Bank Account")
    ),
    tau()
)
post_activation_bank_checks = xor(tau(), bank_change_cycle)

def accepted_suffix():
    return seq(
        leaf("Activate Vendor"),
        post_activation_bank_checks,
        leaf("Release First Payment"),
        leaf("Close Onboarding")
    )

reject_branch = seq(
    leaf("Reject Vendor"),
    leaf("Close Onboarding")
)

low_risk_branch = seq(
    leaf("Manager Approval"),
    accepted_suffix()
)

high_risk_no_escalation = seq(
    leaf("Enhanced Due Diligence"),
    leaf("Compliance Approval"),
    accepted_suffix()
)

high_risk_with_escalation_strict_R5 = seq(
    leaf("Sanctions Escalation"),
    leaf("Enhanced Due Diligence"),
    parallel(
        leaf("Compliance Approval"),
        leaf("Obtain Legal Opinion")
    ),
    accepted_suffix()
)

high_risk_branch = xor(
    high_risk_no_escalation,
    high_risk_with_escalation_strict_R5
)

pre_checks = seq(
    leaf("Receive Onboarding Request"),
    parallel(
        leaf("Capture Master Data"),
        leaf("Run Sanctions Screening"),
        leaf("Verify Bank Account")
    ),
    clarifications,
    leaf("Classify Risk")
)

procedural_model = seq(
    pre_checks,
    xor(
        reject_branch,
        low_risk_branch,
        high_risk_branch
    )
)
```

Explanation:

- Concurrency: after `Receive Onboarding Request`, `Capture Master Data`, `Run Sanctions Screening`, and `Verify Bank Account` are in a PARALLEL block. `Classify Risk` is after the parallel block, so all three checks must complete first.
- Choices: after `Classify Risk`, an XOR selects rejection, low-risk acceptance, or high-risk acceptance. This enforces R1 and the route-specific R2 behavior.
- Clarification loops: optional zero-or-more `Request Clarification -> Receive Clarification` cycles are placed before `Classify Risk`. This matches T2. T5 fails because it has a request but no receive before classification.
- Terminal outcomes: rejection closes directly; accepted routes activate, optionally change bank details/suspend/re-verify, release first payment once, then close.
- Post-activation behavior: a bank-change cycle is allowed only before `Release First Payment`. Post-release bank changes are deliberately excluded from the procedural tree and delegated to the resource/data layer.
- Approximation: the tree does not allow `Sanctions Escalation` on low-risk or rejected routes, and it places escalation before `Enhanced Due Diligence`. R5 is strict. R6 emergency activation before legal opinion is not in the canonical tree; it is an external policy mode.

- `never_together` duplicates DECLARE `noncoexistence`. This is intentional redundancy for R1/R2/R8.

These are temporal-profile deviations and are reported as `warning`, not as automatic control-flow rejection. Timestamps may be affected by time zones, batch delays, weekends, approvals outside the system, and missing events. Hard policy violations are still found by the procedural, DECLARE, Log Skeleton, and resource/data layers.

# 7. Resource- and Data-Aware Checks

Required event attributes: `case:concept:name`, `concept:name`, `time:timestamp`, and `org:resource`. Emergency-policy case attributes: `emergency_authorization` (boolean), `joint_authorization_by` (set/list of at least two distinct authorities), and optionally `authorization_timestamp`. If `org:resource` is missing, resource checks are `not evaluated` or fail-closed depending deployment policy. If emergency attributes are missing, strict R5 applies: activation before legal opinion after escalation is a violation.

```python
import pandas as pd
import pm4py

def check_resource_and_data(df, case_attrs, strict_r5=True):
    """
    df columns: case_id, activity, timestamp, resource
    case_attrs columns: case_id, emergency_authorization,
                        joint_authorization_by
    """
    findings = []

    # Four-eyes checks using PM4Py, when possible.
    # These calls require proper PM4Py event-log columns.
    try:
        v = pm4py.filter_four_eyes_principle(
            df, "Verify Bank Account", "Manager Approval"
        )
        if len(v) > 0:
            findings.append(("R9", "fail", "Verify Bank Account and Manager Approval share a resource"))
    except Exception as e:
        findings.append(("R9", "not evaluated", f"four-eyes Manager check not evaluable: {e}"))

    try:
        v = pm4py.filter_four_eyes_principle(
            df, "Verify Bank Account", "Compliance Approval"
        )
        if len(v) > 0:
            findings.append(("R9", "fail", "Verify Bank Account and Compliance Approval share a resource"))
    except Exception as e:
        findings.append(("R9", "not evaluated", f"four-eyes Compliance check not evaluable: {e}"))

    try:
        v = pm4py.filter_four_eyes_principle(
            df, "Capture Master Data", "Activate Vendor"
        )
        if len(v) > 0:
            findings.append(("R9", "fail", "Capture Master Data and Activate Vendor share a resource"))
    except Exception as e:
        findings.append(("R9", "not evaluated", f"four-eyes Activate check not evaluable: {e}"))

    for cid, g in df.sort_values(["case_id", "timestamp"]).groupby("case_id"):
        g = g.copy()

        # R10 cardinalities, checked again operationally.
        for activity in ["Activate Vendor", "Reject Vendor", "Release First Payment"]:
            if (g["activity"] == activity).sum() > 1:
                findings.append(("R10", "fail", f"{cid}: {activity} occurs more than once"))

        # R11 post-activation bank-change handling.
        act = g[g["activity"] == "Activate Vendor"]
        if not act.empty:
            act_time = act["timestamp"].min()
            changes = g[
                (g["activity"] == "Change Bank Details") &
                (g["timestamp"] > act_time)
            ]
            for _, ch in changes.iterrows():
                change_time = ch["timestamp"]

                later_suspensions = g[
                    (g["activity"] == "Suspend Vendor") &
                    (g["timestamp"] > change_time)
                ]
                if later_suspensions.empty:
                    findings.append((
                        "R11-suspend", "fail",
                        f"{cid}: Change Bank Details after activation without later Suspend Vendor"
                    ))

                later_releases = g[
                    (g["activity"] == "Release First Payment") &
                    (g["timestamp"] > change_time)
                ]
                if not later_releases.empty:
                    next_release_time = later_releases["timestamp"].min()
                    extra_verify = g[
                        (g["activity"] == "Verify Bank Account") &
                        (g["timestamp"] > change_time) &
                        (g["timestamp"] < next_release_time)
                    ]
                    if extra_verify.empty:
                        findings.append((
                            "R11-reverify", "fail",
                            f"{cid}: no extra Verify Bank Account before later Release First Payment"
                        ))

        # R4/R5/R6 sanctions escalation and legal opinion.
        esc = g[g["activity"] == "Sanctions Escalation"]
        if not esc.empty:
            esc_time = esc["timestamp"].min()
            legal = g[g["activity"] == "Obtain Legal Opinion"]
            act_rows = g[g["activity"] == "Activate Vendor"]

            if legal.empty:
                findings.append((
                    "R4", "fail",
                    f"{cid}: Sanctions Escalation without Obtain Legal Opinion"
                ))
            else:
                legal_time = legal["timestamp"].min()

                if not act_rows.empty:
                    activation_time = act_rows["timestamp"].min()

                    if legal_time < activation_time:
                        # Strict R5 satisfied.
                        pass
                    else:
                        # Activation happened before legal opinion; R6 exception required.
                        ca = case_attrs[case_attrs["case_id"] == cid]
                        auth = False
                        joint = None
                        if not ca.empty:
                            auth = bool(ca.iloc[0].get("emergency_authorization", False))
                            joint = ca.iloc[0].get("joint_authorization_by", None)

                        delay_hours = (legal_time - activation_time).total_seconds() / 3600.0
                        valid_joint = isinstance(joint, (list, tuple, set)) and len(set(joint)) >= 2
                        valid_emergency = auth and valid_joint and delay_hours <= 24.0

                        if strict_r5 and not valid_emergency:
                            findings.append((
                                "R5/R6", "fail",
                                f"{cid}: Activate before Legal without valid emergency exception "
                                f"(delay={delay_hours:.2f}h, auth={auth}, joint={joint})"
                            ))
                        elif not strict_r5 and not valid_emergency:
                            findings.append((
                                "R5/R6", "policy-dependent",
                                f"{cid}: emergency exception claimed but external authorization missing/invalid"
                            ))

    return findings
```

If resource attributes are absent, the four-eyes checks are not silently passed. They are reported as `not evaluated` in permissive reporting, or as `fail` in a fail-closed deployment for high-risk vendor onboarding.

# 9. Contradictions, Redundancies, and Blind Spots

- Duplicated enforcement across layers: R1 and R8 are enforced by ProcessTree XOR, DECLARE `noncoexistence`, and Log Skeleton `never_together`. This is useful for defense-in-depth but can produce multiple findings. The user interface should show the earliest decisive hard violation and then list later redundant findings as corroborating evidence.
- R2 is only partially expressible in activity-only DECLARE. `noncoexistence(Manager Approval, Compliance Approval)` prevents both approvals, and `noncoexistence(Reject Vendor, Manager Approval)` and `noncoexistence(Reject Vendor, Compliance Approval)` prevent approvals in rejected cases, but “exactly one approval in an accepted case” still relies on the procedural XOR route or an external outcome-aware check.
- Clarification loops interact with `always_before` and `response`: multiple request/response cycles require `altresponse`, not simply `always_before(Request, Receive)`. Otherwise repeated requests can make orientation checks ambiguous. The procedural tree restricts clarifications to before `Classify Risk`; DECLARE is more general.
- R11 is difficult for activity-only formalisms because “another `Verify Bank Account`” cannot be distinguished from the initial verification. The DECLARE `response(Change Bank Details, Verify Bank Account)` is conservative: it requires some later verify after any change, even if there is no later release. The external timestamp check is more precise.
- Vacuity in declarative rules: `response(Manager Approval, Activate Vendor)` is vacuously satisfied in rejected cases because no `Manager Approval` occurs. `precedence(Classify Risk, Reject Vendor)` is vacuously satisfied in accepted cases because no `Reject Vendor` occurs. Vacuity is acceptable for conformance, but it should not be interpreted as positive evidence that a route was taken.
- Missing case attributes: R6 cannot be evaluated from the event log because the emergency flag and joint authorization are absent. R9 cannot be evaluated if `org:resource` is missing. These should be reported as `not evaluated` or `policy-dependent`, not silently passed.
- Timestamp-only limitations: temporal profiles and the 24-hour emergency check rely on trustworthy timestamps. Clock skew, time zones, manual backdating, or delayed event ingestion can cause false warnings or false exceptions. “Within 24 hours” should use normalized UTC timestamps and an authoritative clock.
- Support/confidence metadata risk: `support=1.0` and `confidence=1.0` in the DECLARE dictionary mean the rule is configured as a required policy rule. They do not mean the rule was empirically learned from data or that every historical case satisfied it. Policy authors, not the numbers, must decide whether a rule is normative.
- Conflict risk: R5 and R6 are contradictory if both are treated as unconditional. The model avoids silent combination by making strict R5 the default and R6 an explicit external override requiring authorization data. If the deployment enables permissive R6, the procedural model or the policy configuration must be changed, and the change must be auditable.

# 10. Deployment Recommendation

Use the ProcessTree `procedural_model` as the canonical hard control-flow representation, or compile it to an equivalent accepting Petri net if the execution engine requires token-based conformance. Treat it as authoritative for route choice, approval exclusivity, activation/rejection exclusivity, and the stable lifecycle.

Use DECLARE and Log Skeleton as declarative guardrails, not as replacements for the procedural model. DECLARE is authoritative for `Request Clarification -> Receive Clarification` alternation, escalation-to-legal eventual response, and mutual exclusions. Log Skeleton is authoritative for cardinality constraints such as at-most-once activation, rejection, and first payment, and as a redundant check for never-together constraints.

Use the resource/data-aware layer as authoritative for R9, R6, and the detailed R11 post-activation re-verification check. The event-log-only layers cannot see resources, joint authorization, or the distinction between the initial and an additional `Verify Bank Account`.

Recommended conformance-check order:

1. Validate event-log schema, case ids, timestamps, activity labels, and required resource/emergency attributes.
2. Run procedural conformance. Any hard control-flow violation is a primary failure with the earliest violating event.
3. Run DECLARE conformance, then Log Skeleton conformance. De-duplicate findings that duplicate procedural or earlier declarative findings, but retain them as corroboration.
4. Run resource/data-aware checks for R9, R6, R10 operational cardinalities, and R11.
5. Run temporal-profile conformance with `zeta=2.0` and report only `warning` severity.
6. Aggregate findings by case and requirement, preserving layer, timestamp, resource, and explanation.

When layers disagree, surface the disagreement explicitly. For example, T6 passes DECLARE and temporal checks but fails strict procedural R5; the UI should say “hard R5 failure under strict policy; permissive R6 requires external joint authorization and legal opinion within 24h.” Do not hide the procedural failure because a declarative rule passed. For missing emergency authorization, strict R5 should fail or mark the case as requiring manual review, not silently pass. For missing resources, report `not evaluated` in advisory mode or fail closed in high-risk production mode.
