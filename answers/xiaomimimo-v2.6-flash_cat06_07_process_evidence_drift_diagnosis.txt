```python
# ---------------------------------------------------------------- load / clean
def load_normalise_dedupe(path: str = EVENTS_CSV) -> pd.DataFrame:
    df = pd.read_csv(path)
    df = df.rename(columns={"case_id": "case:concept:name",
                            "activity": "concept:name",
                            "timestamp": "time:timestamp"})
    df["time:timestamp"] = pd.to_datetime(df["time:timestamp"], utc=True)

    # (a) NORMALISE the 1 Apr 2026 rename BEFORE any prefix/feature extraction.
    name = df["concept:name"].astype(str).str.strip()
    df["concept:name"] = np.where(name.isin(DIAG_ALIASES), CANON_DIAG, name)

    # (b) DUPLICATES: same (case, canonical activity, timestamp).
    #     Applied uniformly to the WHOLE period -- the audit found it had been
    #     applied only to the new source after 20 Apr 2026, which is itself a
    #     defect. UNKNOWN_ACTIVITY is counted, never silently renamed away.
    dup = df.duplicated(subset=["case:concept:name", "concept:name", "time:timestamp"],
                        keep="first")
    df.attrs["dup_rate_raw"] = float(dup.mean())
    df = df[~dup]

    # (c) PM4Py schema
    df = pm4py.format_dataframe(
        df, case_id="case:concept:name",
        activity_key="concept:name", timestamp_key="time:timestamp")
    return df.sort_values(["case:concept:name", "time:timestamp"]).reset_index(drop=True)


# ------------------------------------------------------------------- cohorts
def event_calendar_cohort(df, m0, m1):
    """Events timestamped inside [m0, m1).

    Interpretation: "what the process did this calendar month" -> DFG,
    performance-DFG, edge times. Traces are CUT at the window edges, so start /
    end activities and variant frequencies are window artefacts.
    """
    return pm4py.filter_time_range(df, dt1=m0, dt2=m1, mode="events",
                                   level="events", timestamp_key="time:timestamp")


def case_start_cohort(df, m0, m1):
    """Cases whose FIRST event falls inside [m0, m1).

    Interpretation: the cohort a prediction / label / breach-rate report is
    built on. Whole traces are kept, so variants and case duration are
    faithful, but some of their events occurred in *later* calendar months
    (edge timing therefore mixes calendar regimes).
    pm4py.filter_time_range(mode="cases") would instead select traces fully
    contained in the window -- a third, different cohort -- and it does not
    expose "first event", so case IDs are selected explicitly here.
    """
    starts = df.groupby("case:concept:name")["time:timestamp"].min()
    keep = starts[(starts >= m0) & (starts < m1)].index
    return df[df["case:concept:name"].isin(keep)].copy()
```

def classify_variant(acts) -> str:
    a = set(acts)
    if UNKNOWN in a or CANON_DIAG not in a:
        return "unmapped/other"
    if "Order Part" in a and "Remote Fix" in a:
        return "unmapped/other"          # impossible mix -> mapping defect
    if "Order Part" in a:
        return "V3_parts"
    if "Remote Fix" in a:
        return "V2_remote"
    return "V1_standard"


def _edge_value(perf_dfg, edge):
    raw = perf_dfg.get(edge)
    if raw is None:
        return np.nan
    if isinstance(raw, dict):                       # defensive: nested measures
        raw = raw.get("mean", next(iter(raw.values())))
    return float(raw)


def monthly_summary(df, year, month):
    m0 = pd.Timestamp(year=year, month=month, day=1, tz="UTC")
    m1 = pd.Timestamp(year=year + (month == 12), month=(month % 12) + 1,
                      day=1, tz="UTC")
    ev = event_calendar_cohort(df, m0, m1)
    cs = case_start_cohort(df, m0, m1)
    if ev.empty:
        return {"month": f"{year}-{month:02d}", "note": "no events"}

    dfg, start_act, end_act = pm4py.discover_dfg(ev)
    perf, _, _ = pm4py.discover_performance_dfg(ev, measure="mean")

    variants = pm4py.get_variants(cs)               # case-start: full traces
    vser = pd.Series({v: len(cid) for v, cid in variants.items()})
    labels = vser.groupby(vser.index.map(classify_variant)).sum()
    n_cases = int(labels.sum()) or 1

    tf = pm4py.extract_temporal_features_dataframe(cs)
    dur = tf["duration"] if "duration" in tf.columns else pd.Series(dtype=float)

    n_ev = len(ev)
    return {
        "month": f"{year}-{month:02d}",
        "cohort_cases_case_start": int(cs["case:concept:name"].nunique()),
        "events_event_calendar": n_ev,
        "unknown_activity_share": float((ev["concept:name"] == UNKNOWN).mean()),
        "cases_without_diagnosis": float(

## 2. Competing explanations and timeline

**January–March — healthy reference regime.** All metrics are stable (AUROC 0.81–0.82, Brier ~0.131, ECE 0.022–0.025), label maturity is 100% with a constant ~33d delay, DQ is at floor (unknown 0.2–0.3%, missing 1%), variant mix is stable (V1 44–45%, V2 35–36%, V3 20%), and `Order Part  Part Received` sits at 28–35h. Within-variant breach rates (V1 ~0.16, V2 ~0.10, V3 ~0.31) are flat. This month block is the only fully trusted baseline against which every later month must be judged.

**April — a schema/pipeline incident, not model decay.** Everything that could reflect reality was normal: 100% maturity, 34d label delay, observed breach 0.19, `Order Part  Part Received` 34h. What broke was the *representation*: the 1 Apr rename meant the pipeline could not locate `Initial Diagnosis Completed`, so prefixes were built at the wrong event or not built, `UNKNOWN_ACTIVITY` hit 17%, the queue feature went missing on 14% of cases, and 18% of cases could not be assigned to a variant. The direction of error is diagnostic: mean predicted risk rose to 0.28 while outcomes stayed at 0.19 (over-prediction). **[Hyp]** prefixes built *later* than diagnosis (larger event counts, longer time-in-process, later-stage status) and/or a high-risk default for the missing queue feature both push risk up — testable by comparing prediction timestamps to true diagnosis timestamps. April is therefore a data-pipeline failure with a nominal/schema component; it is emphatically *not* evidence that the learned risk relationship deteriorated.

**May — partial repair with residual contamination.** The 6 May patch restored most of the label mapping and unknown activity fell to 2.0%, missing to 3%; AUROC recovered to 0.78 and ECE to 0.050 but did not return to baseline (0.025). That residual is expected: the cached "events before diagnosis" feature was never recomputed for April, and duplicate removal still only covered the new source after 20 Apr. May is a *pipeline* residual, not a model issue.

**June — a real, deliberate process/mix change that need not imply concept drift.** The 1 Jun remote-routing policy moved V2 from 34% to 52% and V1 from 43% to 30%. Because V2 is intrinsically low-risk, the overall observed breach rate *fell* to 0.15 — an improvement, not deterioration. Within-variant rates were essentially unchanged (V1 0.16, V2 0.09, V3 0.35 vs Mar 0.15/0.11/0.30), so the mapping XY did not move: **this is covariate/process mix drift, not concept drift.** Two calibration residuals need explanation: (i) the mix alone implies 0.16 while mean predicted risk is 0.20, so ~0.03–0.04 of over-prediction is *unexplained by mix* **[Hyp: stale within-V2 risk estimates after the policy changed which cases route remotely (selection effect inside a variant), or residual queue-feature imputation — resolved by test T3]**; (ii) AUROC slipped 0.810.79 partly because 52% of cases now sit in a low-risk, low-variance bucket, which shrinks between-class separation without any per-case performance loss. **June is a beneficial intervention that produced an apparent calibration problem — treating it as decay would trigger an unnecessary and harmful retrain.**

**July — onset of genuine post-prediction concept drift.** The supplier began missing targets on 10 Jul; the parts edge jumped 35h65h and the matured V3 breach rate jumped 0.30–0.350.46 while V1/V2 stayed flat. Because prediction happens at diagnosis, *before* `Order Part`, the model cannot see this coming from its inputs. Maturity was still 98%, so censoring is a minor confounder only.

**August — performance/process drift plus concept drift, at clean data quality.** The parts edge hit 130h, V3 share reached 35% and V3 breach reached 0.64. The arithmetic separates the two effects cleanly: applying training-era variant rates to August's mix gives 0.19 — exactly the model's mean predicted risk — while applying *August's own* variant rates gives 0.31, consistent with the reported 0.28 at 93% maturity. So the model was behaving consistently with its training-era variant knowledge; what changed was the V3 outcome relationship (concept drift) on top of a genuine mix shift (covariate drift). DQ was near baseline (0.7%/2%), so pipeline failure is excluded.

**September — three overlapping shifts plus the first real censoring.** (a) *Population shift*: Partner Z onboarded at 24%, rural and older-device cases outside training support, 58% V3 vs 36% for others. (b) *Covariate drift*: V3 share peaks at 42%. (c) *Concept drift*: the supplier edge is at 148h and V3 breach stays at 0.61. (d) *Censoring begins*: maturity 82%, delay 42d, and Z labels mature more slowly — so the headline "Z 0.45 vs others 0.27" is maturity-confounded in an unknown direction and must not be used for fairness or partner action yet. The AUROC fall to 0.66 plausibly combines unmodelled rural/device covariates, the V3 relationship change, and a changing matured subset.

**October — heavily confounded by label maturity; the metrics are not comparable.** Maturity is 55% and median delay 52d, and the backlog hits long-path (V3, breach-prone) cases hardest. That mechanically depresses the observed breach rate to 0.16 while the *real* signals keep worsening (edge 155h, V3 41%, Partner Z 27%). The tell-tale combination — Brier improving while AUROC worsens, and the sign of (predicted  observed) flipping from +0.04 to 0.07 — indicates informative right-censoring rather than a model that suddenly became over-confident. Pipeline indicators are near baseline, so April-style failure is not the explanation.

**Why these are not one homogeneous failure.** The *directions* differ: April over-predicts at full maturity with a DQ spike; June over-predicts while outcomes *improve* and DQ is clean; August–September *under*-predict while DQ is clean and a documented external performance failure is visible on a specific DFG edge; October over-predicts only because half the labels are missing. The *mechanisms* differ: broken features  stale mix  changed outcome relationship + new population  missing labels. And the *remedies* differ: pipeline repair  variant recalibration  parts-aware features and business escalation  backlog clearance and vintage-based reporting. A single "the model has decayed, retrain it" narrative would be wrong in April (retraining on corrupt prefixes), wrong in June (retraining on a beneficial policy period), incomplete in August–September (blind to the supplier), and actively destructive in October (learning that high predicted risk means non-breach).

## 3. Diagnostic tests

**T1 — Label-maturity curves and fixed-vintage backtesting.**
*Procedure:* For every case-start cohort, recompute breach rate, AUROC, Brier, ECE at vintages 30/45/60/75/90 days after cohort start; plot maturity % and label-delay p50/p95 by cohort month; also compare the predicted-risk distribution of *matured* vs *not-yet-matured* cases within October.
*Expected under H1 (censoring dominates October):* October's observed breach rate climbs monotonically with vintage from 0.16 toward 0.30+, ECE and the signed gap shrink toward Aug–Sep levels, and the matured subset has materially lower mean predicted risk than the immature subset (informative censoring).
*Expected under H2 (real model degradation):* October's metrics stay poor and stable across vintages, and matured vs immature cases have similar predicted-risk distributions.
*Expected under H3 (April-style pipeline failure, excluded):* vintage does not matter because labels were complete — April's metrics are identical at every vintage, which is exactly what we already observe.

**T2 — Recomputation after activity-label normalization and duplicate cleaning.**
*Procedure:* Rebuild from raw events: map both `Initial Diagnosis Completed` and `Diagnostic Assessment Completed` to one canonical label **before** prefix construction; apply duplicate removal uniformly to *all* months (not just the new source after 20 Apr); invalidate and recompute every derived feature (including the cached "events before diagnosis"); then re-score Jan–Oct without refitting.
*Expected under H1 (pipeline failure):* April returns to AUROC 0.80 and ECE 0.03, unknown share  0.2%, unmapped variants  ~0%, missing queue  ~1%; May's residual ECE 0.05 collapses; Jun–Oct move only marginally.
*Expected under H2 (genuine April concept drift):* April metrics remain near AUROC 0.67 after repair.
*Expected under H3 (residual post-June DQ):* Jun–Oct improve modestly (unknown 0.7–1.0%  0.2–0.3%, missing 2–3%  1%), leaving the Aug–Sep under-prediction essentially intact — which would confirm supplier-driven concept drift as a separate, real phenomenon.

**T3 — Calibration and discrimination stratified by variant × source × month.**
*Procedure:* Per (month, variant, source) cell, compute AUROC, Brier, ECE, and the *signed* calibration gap (mean predicted  observed), all on 95% matured vintages with n and CI reported; report a calibration intercept/slope per cell.
*Expected under H1 (concept drift localised to V3):* V1 and V2 cells keep AUROC 0.80 and gaps 0; the V3 cell shows a strongly negative intercept (under-prediction) from Jul onward, growing JulSep.
*Expected under H2 (Partner-Z population/omitted-covariate shift):* within non-Z cases and within each variant, metrics stay near baseline; the residual is concentrated in the Z cell and shrinks after rural distance/device age are added (T6).
*Expected under H3 (pipeline residual):* degradation is concentrated in cells with high missingness (Apr, and mildly May) regardless of variant, and disappears after T2.
*Also answers June:* if the ~0.03 June over-prediction is concentrated in V2  stale within-variant estimates (policy selection effect); if concentrated where the queue feature was imputed  residual DQ.

**T4 — Comparison of process variants and DFG / performance-DFG edges.**
*Procedure:* Monthly DFG + performance-DFG on the *event-calendar* cohort and variant frequency on the *case-start* cohort; track `Order Part  Part Received` p50/p90, `diagnosis  Remote Fix`, `Initial Diagnosis  Dispatch Technician`, `Repair Completed  Case Closed`; record unmapped-variant share.
*Expected under H1 (process/performance drift):* structural breaks land exactly on 1 Jun (V2 share), 10 Jul (parts edge), 1 Sep (V3 share) — sharp, dated, monotone.
*Expected under H2 (pipeline failure):* breaks land on 1 Apr in *variant assignment and unknown-activity share only*, while edge times and outcome rates stay normal (already observed: Apr edge 34h  normal) and normalise after T2.
*Expected under H3 (no real process change, only measurement):* edge times flat everywhere while variant shares swing — falsified by the observed 28h155h progression.

**T5 — Has the outcome relationship conditional on the same path changed?**
*Procedure:* Restrict to cases with an identical canonical variant and matched prediction-time prefix (same number of events to diagnosis, same time-in-case, same queue, same region, same parts-required flag, same days of promise-slack). Compare breach rate Jan–Mar vs Jul–Sep. Complement with a logistic model `breach ~ prefix features + variant + month indicators` fitted on matured cohorts; test the post-10-Jul month coefficients within V3 and within V1/V2.
*Expected under H1 (concept drift):* a large, significant post-July coefficient **within V3 only**; V1/V2 month effects null.
*Expected under H2 (pure covariate/mix drift):* all month effects null once variant and prefix features are controlled — the entire Aug–Sep gap is composition.
*Expected under H3 (omitted covariate, e.g. parts lead time):* month effects shrink to null when a forecasted parts lead-time/ETA feature is added, without any change to the model's structure — distinguishing "relationship changed" from "we stopped measuring a driver".

**T6 — Population reweighting / transport checks for Partner Z.**
*Procedure:* (i) Maturity-match Z and non-Z (compare only cases at the same vintage, e.g. both at 60d). (ii) Estimate a source propensity on rural distance, device age, variant, region; compute overlap/positivity diagnostics (propensity histogram, effective sample size). (iii) Reweight non-Z September cases to Z's covariate distribution and compare expected vs observed breach; (iv) evaluate the model on Z-only and non-Z-only matured slices; (v) add rural distance + device age and repeat.
*Expected under H1 (population shift via missing covariates):* reweighting and/or adding the two covariates removes most of the Z gap; overlap is poor (positivity violation)  the model is extrapolating, and AUROC recovers once the covariates are present.
*Expected under H2 (partner-specific concept drift):* a substantial Z residual persists after maturity matching, variant conditioning, reweighting, and covariate addition  a partner-specific intercept (not a full model) may be warranted.
*Expected under H3 (artifact of slower Z maturation):* the Z gap shrinks or reverses once both sides are compared at identical vintages — which would invalidate the current 0.45 vs 0.27 headline.

**T7 — Feature missingness and lineage audit.**
*Procedure:* Per feature per month: missingness rate, imputation value actually written, cache/freshness timestamp, and the end-to-end lineage (raw event  normalisation  prefix  feature  prediction). Reconcile against the mapping-patch deployment (6 May) and the dedupe deployment (20 Apr).
*Expected under H1 (pipeline failure):* missingness and wrong-lineage rows spike in Apr and decay after 6 May; the cached "events before diagnosis" for April retains pre-patch values; duplicate-adjusted event counts for 1–20 Apr differ materially from post-20-Apr counts; lineage gaps align exactly with the renamed activity.
*Expected under H2 (genuine model decay):* missingness is flat at ~1% throughout and lineage is complete — the April/Aug/Sept failures would then have to come from somewhere else.
*Expected under H3 (residual post-June defect):* small but non-zero lineage gaps persist in Jun–Oct (unknown 0.7–1.0% vs baseline 0.2–0.3%), quantifying how much of the residual ECE is pipeline.

**T8 — Prediction-timestamp alignment (supports T7).**
*Procedure:* Distribution of (prediction timestamp  true `Initial Diagnosis Completed` timestamp) by month; share of predictions where no diagnosis event precedes within a tolerance.
*Expected under H1:* April spikes (predictions landing at dispatch or later, or missing), normalising after T2; this also *explains the direction* of April's over-prediction. *Expected under H2:* alignment flat all year — which would force a different explanation for April's over-prediction (e.g. queue-feature imputation).

**T9 — Backlog / label-arrival audit (supports T1).**
*Procedure:* Weekly backlog depth; label-arrival delay distribution conditional on variant and path length; share of September/October cases whose labels arrived in the last 7 days.
*Expected under H1 (censoring):* delay is strongly increasing in path length and in V3 membership, and the matured subset is enriched for short/non-breaching traces. *Expected under H2:* delay is roughly independent of variant/path length, which would break the mechanism and force a different reading of October.

## 4. Recalibration-versus-retraining decision

### Stage 0 — Immediate containment (0–5 days)
1. **Freeze all automated retraining and automated calibration triggers.**
2. **Fix the pipeline first**: canonicalise both diagnosis labels to one; apply duplicate removal *uniformly to the whole period* (the asymmetry — new source only, after 20 Apr — is itself a defect); invalidate and recompute **every** derived feature from raw events, including the cached "events before diagnosis" for April; re-run T2.
3. **Repair the outcome API backlog** as an ops incident (it is currently corrupting the scoreboard).
4. **Reporting gates**: publish maturity % and label delay p50 next to every metric; mark Apr–May "pipeline-affected", Sep "partially matured", **Oct "not interpretable"**; suppress ECE/AUROC alarms when maturity <95% or delay p50 >45d.
5. **Business containment independent of the model**: the parts edge at 155h and V3 breach at ~0.6 are real regardless of model quality — escalate the supplier, add parts-ETA tracking, and staff V3/rural cases using *process* signals (edge time, V3 flag, promise-slack) rather than the model's October scores.
6. **Regression test**: after repair, Jan–Mar metrics must be reproduced to within noise; April must recover toward baseline.

### Stage 1 — Short-term recalibration (days 5–14, post-repair only)
- Recalibrate **intercepts/slopes per variant** (not a global intercept) using matured, DQ-trusted vintages — Jun–Sep for V1/V2, Jul–Sep for V3.
- Concretely: raise the V3 risk score to reflect the current parts regime (a simple variant-conditional intercept fitted on Aug–Sep matured cases), and add a fast "parts lead time vs promise slack" adjustment.
- **Do not retrain.** Discrimination within V1/V2 appears intact; retraining now would (a) absorb October's censored labels, (b) absorb April's corrupt prefixes if the repair is incomplete, and (c) discard useful pre-June data.
- Require recalibration parameters to be stable across two independent vintages before deployment, and hold a Jan–Mar and Jun regression slice as a guardrail against over-correction.

### Stage 2 — Feature repair (weeks 2–6)
- Lineage-correct features; add **rural travel distance**, **device age**, **parts lead time / supplier SLA status**, **source/partner**, and **promise-slack at diagnosis**.
- Leakage control: only *forecasted* part ETA is admissible — actual `Part Received` occurs after the prediction point and would leak the label mechanism.
- Re-run T3/T5/T6 with the new features to determine how much of Aug–Sep "concept drift" was actually omitted covariate.

### Stage 3 — Full retraining (only after gates are green)
**Training-month safety table:**

| Cohort month | Maturity | Pipeline state | Safe for training? | Condition |
|---|---|---|---|---|
| Jan–Mar | 100% | clean | **Yes** | — |
| Apr | 100% | **corrupt** | **No  Yes only after repair** | T2 shows AUROC recovery toward ~0.80 and Jan–Mar regression passes; otherwise exclude entirely |
| May | 100% | partially repaired | **Conditional** | recompute cached features; residual ECE  ~0.03 |
| Jun | 100% | near-baseline DQ | **Yes** | annotate routing-policy change as a regime feature/flag |
| Jul | 98% | near-baseline | **Yes at vintage 95%** | annotate supplier incident start |
| Aug | 93% | clean | **Not yet** | wait to 98–100% (60–75d after cohort start) |
| Sep | 82% | clean | **Not yet** | wait to 95% (Nov) *and* maturity-match Partner Z |
| Oct | 55% + backlog | clean but label-starved | **No** | wait for backlog clearance + 95% maturity (likely Dec–Jan); use for backtesting only |

- Use **time-ordered splits** (e.g., train  Aug, validate Sep, backtest Oct at fixed vintage). Never shuffle across regimes without encoding the regime (routing policy version, supplier status, partner mix) as features or strata.
- Retraining requires: green DQ gate, green maturity gate, two consecutive trusted mature vintages, documented annotation of all interventions, champion–challenger backtest on matured months, and human sign-off.

### Is a segment- or variant-specific model justified?
- **V3-specific treatment: yes, now — but as recalibration + features first, not a separate model.** V3's conditional risk regime changed by ~2×, its data-generating mechanism (parts lead time) is qualitatively different, and it is now 41–42% of volume. A variant-conditional calibration layer is proportionate and reversible; a separate V3 *model* should wait until T2/T5/T6 show the V3 relationship is stable at its new level.
- **Separate V1 and V2 models: no.** Their within-variant breach rates are flat all year (V1 0.15–0.19, V2 0.09–0.13) and their discrimination appears intact; splitting them only reduces training data and raises variance. The June calibration gap is an intercept problem, not a model-structure problem.
- **Partner-Z-specific model: not yet.** The 0.45 vs 0.27 gap is maturity-confounded, Z is only 24–27% of volume with 58% V3, and two obvious covariates (rural distance, device age) are missing. First: maturity-matched, variant-stratified, reweighted comparison (T6) and add the covariates. If a residual persists, add a **partner intercept**, not a partner model.
- **Precondition for any variant-specific model:** variant-mapping coverage 99%. April's 18% unmapped share shows this prerequisite can silently fail; without the gate, a variant-specific system would route unknown cases into an unmonitored fallback.

```
Jan–Mar + Jun regression slices passing, 99% variant-mapping coverage, human sign-off.
```

In short: **data-quality and label-delay alarms gate the *validity* of performance alarms rather than escalating them**, and no retraining path exists that bypasses both gates.

### Python sketch — monthly process-evidence extraction

```python
# monthly_process_evidence.py
"""Auditable monthly process evidence for the field-service prediction audit.

Reads -> normalises the renamed diagnosis label -> de-duplicates -> PM4Py schema
-> monthly summaries (DFG, performance-DFG, variants, temporal features).
No model fitting. No PM4Py drift detector is called: PM4Py does not provide a
general-purpose drift detector we are willing to rely on here, so drift
statistics (PSI/KS, CUSUM on signed residuals) are computed downstream from
these summaries with pandas/scipy plus human review.
"""
from __future__ import annotations

import numpy as np
import pandas as pd
import pm4py

EVENTS_CSV = "field_service_events.csv"

CANON_DIAG = "Initial Diagnosis Completed"
DIAG_ALIASES = {"Initial Diagnosis Completed", "Diagnostic Assessment Completed"}
UNKNOWN = "UNKNOWN_ACTIVITY"
KEY_EDGE = ("Order Part", "Part Received")


# ------------------------------------------------------------------ load/clean
def load_normalise_dedupe(path: str = EVENTS_CSV) -> pd.DataFrame:
    df = pd.read_csv(path)
    df = df.rename(columns={"case_id": "case:concept:name",
                            "activity": "concept:name",
                            "timestamp": "time:timestamp"})
    df["time:timestamp"] = pd.to_datetime(df["time:timestamp"], utc=True)

    # (a) NORMALISE the 1 Apr 2026 rename BEFORE any prefix/feature extraction.
    name = df["concept:name"].astype(str).str.strip()
    df["concept:name"] = np.where(name.isin(DIAG_ALIASES), CANON_DIAG, name)

    # (b) DUPLICATES: collapse on (case, canonical activity, timestamp) so a
    #     rename-induced duplicate also collapses. Applied UNIFORMLY to the
    #     whole period -- the audit found it had been applied only to the new
    #     source after 20 Apr 2026, which is itself a defect. UNKNOWN_ACTIVITY
    #     is counted and reported, never silently renamed away.
    dup = df.duplicated(subset=["case:concept:name", "concept:name", "time:timestamp"],
                        keep="first")
    df.attrs["dup_rate_raw"] = float(dup.mean())
    df = df[~dup]

    # (c) PM4Py schema
    df = pm4py.format_dataframe(
        df, case_id="case:concept:name",
        activity_key="concept:name", timestamp_key="time:timestamp")
    return df.sort_values(["case:concept:name", "time:timestamp"]).reset_index(drop=True)


# --------------------------------------------------------------------- cohorts
def event_calendar_cohort(df, m0, m1):
    """Events timestamped inside [m0, m1).

    Interpretation: "what the process DID this calendar month" -> correct basis
    for DFG / performance-DFG / edge times. Traces are CUT at the window edges,
    so start/end activity sets and variant frequencies are window artefacts.
    """
    return pm4py.filter_time_range(df, dt1=m0, dt2=m1, mode="events",
                                   level="events", timestamp_key="time:timestamp")


def case_start_cohort(df, m0, m1):
    """Cases whose FIRST event falls inside [m0, m1).

    Interpretation: the cohort a prediction / label / breach-rate report is
    built on. Whole traces are kept, so variants and case duration are faithful
    -- but some of their events happened in LATER calendar months, so edge
    timings computed here mix calendar regimes.
    pm4py.filter_time_range(mode="cases") instead selects traces fully
    contained in the window (a third, different cohort) and does not expose
    "first event", so case IDs are selected explicitly.
    """
    starts = df.groupby("case:concept:name")["time:timestamp"].min()
    keep = starts[(starts >= m0) & (starts < m1)].index
    return df[df["case:concept:name"].isin(keep)].copy()


# ------------------------------------------------------------------- helpers
def classify_variant(acts) -> str:
    a = set(acts)
    if UNKNOWN in a or CANON_DIAG not in a:
        return "unmapped/other"
    if "Order Part" in a and "Remote Fix" in a:      # impossible mix -> mapping defect
        return "unmapped/other"
    if "Order Part" in a:
        return "V3_parts"
    if "Remote Fix" in a:
        return "V2_remote"
    return "V1_standard"


def _edge_hours(perf_dfg, edge):
    raw = perf_dfg.get(edge)
    if raw is None:
        return np.nan
    if isinstance(raw, dict):                        # defensive: nested measures
        raw = raw.get("mean", next(iter(raw.values())))
    return float(raw) / 3600.0


def _month_window(year, month):
    m0 = pd.Timestamp(year=year, month=month, day=1, tz="UTC")
    m1 = pd.Timestamp(year=year + (month == 12), month=(month % 12) + 1,
                      day=1, tz="UTC")
    return m0, m1


# ------------------------------------------------------------- monthly summary
def monthly_summary(df, year, month):
    m0, m1 = _month_window(year, month)
    ev = event_calendar_cohort(df, m0, m1)           # calendar view
    cs = case_start_cohort(df, m0, m1)               # cohort view
    tag = f"{year}-{month:02d}"
    if ev.empty:
        return {"month": tag, "note": "no events"}

    dfg, start_act, end_act = pm4py.discover_dfg(ev)
    perf, _, _ = pm4py.discover_performance_dfg(ev, measure="mean")

    variants = pm4py.get_variants(cs)                # case-faithful: full traces
    vser = pd.Series({v: len(cid) for v, cid in variants.items()})
    labels = vser.groupby(vser.index.map(classify_variant)).sum()
    n_cases = int(labels.sum()) or 1

    has_diag = cs.groupby("case:concept:name")["concept:name"].apply(
        lambda s: bool((s == CANON_DIAG).any()))

    tf = pm4py.extract_temporal_features_dataframe(cs)
    dur = tf["duration"] if "duration" in tf.columns else pd.Series(dtype=float)
    if pd.api.types.is_timedelta64_dtype(dur):       # newer PM4Py returns timedelta
        dur_h = dur.dt.total_seconds() / 3600.0
    else:                                            # older versions: nanoseconds
        dur_h = pd.to_numeric(dur, errors="coerce") / 3.6e9

    return {
        "month": tag,
        "cases_case_start": int(cs["case:concept:name"].nunique()),
        "events_event_calendar": int(len(ev)),
        "unknown_activity_share": float((ev["concept:name"] == UNKNOWN).mean()),
        "share_cases_without_diagnosis": float((~has_diag).mean()),
        "v1_share": float(labels.get("V1_standard", 0) / n_cases),
        "v2_share": float(labels.get("V2_remote", 0) / n_cases),
        "v3_share": float(labels.get("V3_parts", 0) / n_cases),
        "unmapped_share": float(labels.get("unmapped/other", 0) / n_cases),
        "n_distinct_variants": int(vser.size),
        "order_part_to_received_h": round(_edge_hours(perf, KEY_EDGE), 2),
        "diagnosis_to_remote_fix_h": round(
            _edge_hours(perf, (CANON_DIAG, "Remote Fix")), 2),
        "median_case_duration_h": round(float(dur_h.median()), 2) if len(dur_h) else np.nan,
        "top_start_activities": sorted(start_act)[:5],
        "top_end_activities": sorted(end_act)[:5],
        "highest_volume_edge": max(dfg, key=dfg.get) if dfg else None,
    }


def build_monthly_table(df, year=2026, months=range(1, 13)) -> pd.DataFrame:
    """Auditable monthly evidence table; attach maturity / DQ context beside it."""
    out = pd.DataFrame([monthly_summary(df, year, m) for m in months])
    out.attrs["global_duplicate_rate"] = df.attrs.get("dup_rate_raw")
    out.attrs["note"] = ("join externally with label maturity, observed breach rate "
                         "and DQ indicators before drawing any drift conclusion")
    return out
```

## 6. Risks of acting on the wrong diagnosis

**Failure mode 1 — Retraining on immature labels (October).**
Because the backlog disproportionately removes long-path, breach-prone cases, the matured October subset is enriched for fast non-breaches. A model fitted on it learns *"high predicted risk  non-breach"*, systematically destroying calibration — and, cruelly, Brier may initially *look better* (hard positives were dropped), so the error can pass a naive quality gate. The damage surfaces weeks later when the backlog clears and the model under-predicts the real ~0.30 rate.
**Mitigation:** hard maturity gate (95% + delay p50 45d) on any training or calibration input; fixed-vintage backtesting as the only accepted metric view; freeze retraining while backlog depth > threshold; re-evaluate October retrospectively at 100% maturity before accepting it into the training set.

**Failure mode 2 — Calibrating around a pipeline bug (April/May).**
Fitting a global intercept to April's +0.09 over-prediction would encode *corrupt prefixes and imputed queue features* as if they were signal. Once the mapping/dedupe/cache issues are repaired, that calibration becomes a systematic *under*-prediction, and the real defect is never found — it will reappear at the next schema migration.
**Mitigation:** never fit calibration on cohorts flagged DQ-SUSPECT/BROKEN; repair first, recalibrate second (Stage 0  Stage 1 order); regression slices on Jan–Mar plus repaired April; T2 must show recovery before any parameter is deployed; require lineage completeness and prediction-timestamp alignment as prerequisites for calibration.

**Failure mode 3 — Ignoring genuine supplier-driven concept drift (July–September).**
If the Aug–Sep AUROC drop is blamed wholly on data quality or on Partner Z, the V3 under-prediction (~0.10–0.25) persists: promised dates are missed, SLA penalties accumulate, and customers in the parts path are repeatedly given unrealistic expectations. A knee-jerk "exclude V3 / stop scoring parts cases" response would remove the very segment that most needs early warning.
**Mitigation:** performance-DFG edge monitoring (M6) runs *independently of labels*, so the 65h130h155h signal is visible even when metrics are suppressed; variant-stratified calibration (T3) localises the damage to V3; parts lead-time/ETA feature plus supplier SLA escalation treat the cause; explicitly separate "business intervention needed" from "model intervention needed" in the change registry.

**Failure mode 4 — Attributing Partner Z's population shift to unfair model behaviour without stratified evidence.**
Z's observed 0.45 vs 0.27 gap is maturity-confounded (Z labels mature more slowly), Z is 58% V3 (a mix difference, not a model defect), and rural distance/device age — plausible genuine drivers — were never in the model. A knee-jerk "fairness fix" (blanket up-weighting of Z scores) would bake in a number derived from censored labels and would over-correct if the gap is mostly variant mix.
**Mitigation:** maturity-matched, variant-stratified, reweighted comparison (T6) *before* any action; add the missing covariates and re-test; run overlap/positivity diagnostics — poor overlap means the model is extrapolating, which is a *feature-coverage* problem, not unfairness; only a persistent residual after all of that justifies a partner-specific intercept (and even then prefer features over partner dummies).

**Failure mode 5 — Treating June's drop in breach rate and rise in ECE as model decay.**
June's ECE (0.060) rose while the business *improved* (breach 0.15) and DQ was near-clean. Retraining here would discard or down-weight useful pre-policy data, possibly learn an elevated V2 risk estimate that the policy made obsolete, and would not fix the ~0.03 residual over-prediction, which is an intercept issue.
**Mitigation:** variant-conditional recalibration rather than retraining for mix changes; keep a variant-mix monitor with a *mix-only counterfactual* line on the calibration chart so mix-driven ECE moves are visibly separated from conditional-performance moves; register the routing policy as an intervention so the chart is annotated.

**Failure mode 6 — Reading October's 0.16 as improvement.**
If October's observed breach rate is accepted as comparable to mature months, teams may relax staffing, thresholds or supplier escalation exactly when the underlying process is at its worst (edge 155h, V3 41%, Z 27%). The metrics already show the tell-tale censoring signature — Brier improving while AUROC worsens, and the sign of (predicted  observed) flipping.
**Mitigation:** publish maturity and label delay beside every number; suppress cross-month comparisons when maturity differs by >5pp; during a backlog, rely on *leading* indicators (predicted-risk distribution, V3 share, performance-DFG edge times) rather than outcome metrics; state explicitly in reporting that October's rate is an artifact of label availability.

**Failure mode 7 — Over-segmenting into variant-specific models on unstable variant assignment.**
April's 18% unmapped share proves variant assignment can fail silently. Variant-specific models would then route a large slice into an unmonitored fallback, splitting training data and raising variance for variants (V1/V2) whose relationships never changed.
**Mitigation:** enforce a 99% variant-mapping coverage precondition (M5) before enabling any variant-specific path; prefer a variant-conditional *calibration layer* over separate models until stability is demonstrated across two trusted vintages; maintain an explicit fallback with its own monitored metrics.
