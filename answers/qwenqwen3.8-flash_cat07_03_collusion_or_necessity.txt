# Custom rule: override must be followed by justification within 3 working days
# by a different human resource.
def retro_rule_violations(dataframe):
    overrides = dataframe[dataframe["concept:name"] == "Emergency Override"][
        [
            "case:concept:name",
            "time:timestamp",
            "org:resource",
            "entered_by",
            "approved_by",
            "source_system",
        ]
    ].rename(columns={"org:resource": "override_resource"})

    justifications = dataframe[dataframe["concept:name"] == "Retrospective Justification"][
        [
            "case:concept:name",
            "time:timestamp",
            "org:resource",
            "entered_by",
            "approved_by",
            "source_system",
        ]
    ].rename(columns={"org:resource": "just_resource"})

    m = overrides.merge(
        justifications,
        on="case:concept:name",
        how="left",
        suffixes=("_override", "_just"),
    )

    m["has_justification"] = m["just_resource"].notna()
    m["working_days"] = np.nan

    valid = m["has_justification"]
    m.loc[valid, "working_days"] = np.busday_count(
        m.loc[valid, "time:timestamp_override"].dt.date,
        m.loc[valid, "time:timestamp_just"].dt.date,
    )

    m["same_human"] = (
        m["has_justification"] &
        (m["override_resource"] == m["just_resource"])
    )
    m["late"] = (
        m["has_justification"] &
        (m["working_days"] > 3)
    )

    return m[~m["has_justification"] | m["same_human"] | m["late"]]

retro_violations = retro_rule_violations(df)
