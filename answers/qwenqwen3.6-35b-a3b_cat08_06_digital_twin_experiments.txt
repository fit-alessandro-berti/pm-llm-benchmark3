# We parse manually for demonstration, acknowledging limitations.
def parse_simplified_tree(tree_str: str) -> dict:
    """Minimal recursive parser for the provided process tree string."""
    tree_str = tree_str.replace("\n", " ").replace("  ", " ")
    if tree_str.startswith("->("):
        inner = tree_str[3:-1]
        parts = _split_top_level(inner)
        return {"type": "sequence", "children": [parse_simplified_tree(p) for p in parts]}
    elif tree_str.startswith("*(..."):
        # Simplified: treat as loop
        return {"type": "loop", "children": [parse_simplified_tree(tree_str[4:-2])]}
    elif tree_str.startswith("+("):
        return {"type": "parallel", "children": [parse_simplified_tree(tree_str[2:-2])]}
    elif tree_str.startswith("X("):
        return {"type": "xor", "children": [parse_simplified_tree(c) for c in _split_top_level(tree_str[2:-2])]}
    else:
        return {"type": "activity", "name": tree_str.strip("'")}

def _split_top_level(s: str) -> List[str]:
    """Split string by commas at top level only."""
    parts = []
    depth = 0
    current = []
    for char in s:
        if char == "(": depth += 1
        elif char == ")": depth -= 1
        elif char == "," and depth == 0:
            parts.append("".join(current).strip())
            current = []
            continue
        current.append(char)
    parts.append("".join(current).strip())
    return parts
