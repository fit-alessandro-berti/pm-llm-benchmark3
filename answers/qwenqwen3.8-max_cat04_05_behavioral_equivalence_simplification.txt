```python
def visible_label(t):
    lab = getattr(t, "label", None)
    if lab is None or lab == "tau" or lab == "":
        return None
    return lab

def marking_key(m):
    # pm4py markings are dict-like; sort by place name for deterministic key
    return frozenset(m.items())

def fire(marking, trans):
    new_m = copy(marking)
    for arc in trans.in_arcs:
        place = arc.source
        new_m[place] = new_m[place] - arc.weight
        if new_m[place] == 0:
            del new_m[place]
    for arc in trans.out_arcs:
        place = arc.target
        new_m[place] = new_m.get(place, 0) + arc.weight
    return new_m

def tau_closure(net, markings):
    closure = {}
    stack = []
    for m in markings:
        k = marking_key(m)
        if k not in closure:
            closure[k] = m
            stack.append(m)
    while stack:
        m = stack.pop()
        for t in pm4py.get_enabled_transitions(net, m):
            if visible_label(t) is None:
                nm = fire(m, t)
                k = marking_key(nm)
                if k not in closure:
                    closure[k] = nm
                    stack.append(nm)
    return closure

def bounded_visible_language(net, im, fm, max_visible=12, max_analyze=3):
    fm_key = marking_key(fm)
    traces = set()
    initial = tau_closure(net, [im])
    # worklist entries: closure dict, trace tuple, analyze_count
    stack = [(initial, (), 0)]
    seen_entries = set()
    while stack:
        closure, trace, a_count = stack.pop()
        # deduplicate identical search nodes (same closure, same visible prefix)
        entry_key = (frozenset(closure.keys()), trace, a_count)
        if entry_key in seen_entries:
            continue
        seen_entries.add(entry_key)

        if fm_key in closure:
            traces.add(trace)

        if len(trace) >= max_visible:
            continue

        # group all enabled visible transitions by label
        next_by_label = {}
        for m in closure.values():
            for t in pm4py.get_enabled_transitions(net, m):
                lab = visible_label(t)
                if lab is not None:
                    nm = fire(m, t)
                    k = marking_key(nm)
                    group = next_by_label.setdefault(lab, {})
                    group[k] = nm
        for lab, next_markings in next_by_label.items():
            new_a = a_count + (1 if lab == "Analyze Revision" else 0)
            if new_a > max_analyze:
                continue
            new_closure = tau_closure(net, list(next_markings.values()))
            stack.append((new_closure, trace + (lab,), new_a))
    return traces
```

```python
def random_bounded_playout(net, im, fm, runs=2000, max_visible=12, max_analyze=3, max_total_steps=200, seed=1):
    rng = random.Random(seed)
    fm_key = marking_key(fm)
    out = set()
    for _ in range(runs):
        m = im
        trace = []
        a_count = 0
        for _ in range(max_total_steps):
            if marking_key(m) == fm_key:
                out.add(tuple(trace))
                break
            enabled = list(pm4py.get_enabled_transitions(net, m))
            if not enabled: break
            t = rng.choice(enabled)
            lab = visible_label(t)
            if lab is not None:
                if len(trace) >= max_visible: break
                if lab == 'Analyze Revision':
                    if a_count >= max_analyze: break
                    a_count += 1
                trace.append(lab)
            m = fire(m, t)
        # discard incomplete
    return out
```

```python
def process_tree_stats(tree):
    nodes = 0
    max_depth = 0
    visible = 0
    silent = 0
    labels = []
    def visit(node, depth):
        nonlocal nodes, max_depth, visible, silent
        nodes += 1
        max_depth = max(max_depth, depth)
        children = getattr(node, "children", [])
        if children:
            for c in children:
                visit(c, depth+1)
        else:
            lab = getattr(node, "label", None)
            if lab is None or lab == "tau":
                silent += 1
            else:
                visible += 1
                labels.append(lab)
    visit(tree, 0)
    cnt = Counter(labels)
    duplicate_labels = {k: v for k, v in cnt.items() if v > 1}
    duplicate_extra = sum(v - 1 for v in cnt.values() if v > 1)
    return {"node_count": nodes, "max_edge_depth": max_depth, "visible_leaves": visible, "silent_leaves": silent, "duplicate_labels": duplicate_labels, "duplicate_extra_occurrences": duplicate_extra}
```

# 1. Equivalence Notions

Let the visible alphabet be:

```text
 = {
  Open Case,
  Analyze Revision,
  Check Revision,
  Approve Revision,
  Rework Case,
  Close Case
}
```

`tau` is silent and is not part of the visible alphabet.

For brevity, I will also use:

```text
O = Open Case
A = Analyze Revision
C = Check Revision
P = Approve Revision
R = Rework Case
Z = Close Case
```

A complete trace is a firing/execution path from the initial marking/state to the final marking/state. Hiding `tau` means projecting a complete execution onto ``.

## Visible trace equivalence after hiding `tau`

For a model `M`, define:

```text
L_vis(M) = { projection_() |  is a complete execution of M }
```

Two models are visible-trace equivalent iff:

```text
L_vis(M_A) = L_vis(M_B)
```

Duplicate visible labels are distinguished by their occurrence position in the trace sequence. There are no external event identities beyond the label sequence. Thus:

```text
O A C Z
```

and

```text
O C A Z
```

are different traces, and:

```text
O A C R A C Z
```

contains two `A` occurrences and two `C` occurrences, distinguished by position.

## Bounded visible-language equivalence under P3

Under P3, we hide `tau`, retain all visible labels, and compare only complete visible traces satisfying:

```text
length(w) <= 12
count_A(w) <= 3
```

Define:

```text
L_vis^P3(M) = {
  w  L_vis(M)
  | |w| <= 12 and #A(w) <= 3
}
```

Models are P3-bounded equivalent iff:

```text
L_vis^P3(M_A) = L_vis^P3(M_B)
```

Duplicate labels are again distinguished by position in the trace word.

## Business-relevant equivalence under P1

P1 is an operational compliance view. The relevant observation is not merely the multiset of labels, but the association between analysis attempts and checks.

A reasonable P1 projection is:

1. Remove `tau`.
2. Keep `Approve Revision` visible.
3. Split the case between `Open Case` and `Close Case` into attempt windows separated by `Rework Case`.
4. For each window containing an `Analyze Revision`, record whether that same window contains a `Check Revision` before the next `Rework Case` or before `Close Case`.
5. Preserve the number and placement of `Approve Revision` events.

Under this view, duplicate `Analyze Revision` and `Check Revision` occurrences are distinguished by occurrence position and by the attempt window in which they appear. They are not collapsed to a Boolean or a count only.

Model A structurally guarantees one `Check Revision` per loop iteration, hence one check per analysis attempt. Model B has only one global `Check Revision` concurrent with the whole loop, so when there are two or more analysis attempts, at most one attempt can be associated with that check.

## Abstracted dashboard equivalence under P2

P2 is an executive dashboard abstraction. It hides `Approve Revision` and all silent behavior, collapses repeated `Check Revision` occurrences to the Boolean fact “a check occurred”, records the number of `Analyze Revision` occurrences, and records only whether any `Rework Case` occurred.

Define the dashboard abstraction:

```text
dash(w) = (
  #A(w),
  #C(w) > 0,
  #R(w) > 0
)
```

after hiding `P` and `tau`.

The dashboard also requires `Open Case` first and `Close Case` last, but both models satisfy that structurally, so I omit those coordinates from the tuple.

Models are P2-equivalent iff:

```text
dash(L_vis(M_A)) = dash(L_vis(M_B))
```

Under P2:

- `Analyze Revision` duplicates are counted by label occurrence.
- `Check Revision` duplicates are collapsed to Boolean.
- `Rework Case` duplicates are collapsed to Boolean.
- Exact interleaving is ignored.
- `Approve Revision` is hidden.

## Reference visible-language shapes

Using the abbreviations above:

Model A visible language has the form:

```text
O (B1 R B2 R ... R Bn) Z
n >= 1
Bi  { AC, CA, ACP, CAP }
```

Each loop iteration contains one `A`, one `C`, optional `P`, and then either exits or is followed by `R`.

Model B visible language has the form:

```text
O shuffle(C, D1 R D2 R ... R Dm) Z
m >= 1
Di  { A, AP }
```

There is exactly one global `C`, arbitrarily interleaved with the loop consisting of one or more `A` blocks separated by `R`.

# 2. Structural Comparison

| Feature | Model A | Model B | Semantic consequence |
|---|---|---|---|
| Loop scope | The check is inside the loop body: `+(A, C)` is part of each iteration. | The check is outside the loop and concurrent with the entire loop. | A produces one `C` per analysis iteration; B produces exactly one global `C`. |
| Minimum `Check Revision` occurrences | 1 | 1 | Both satisfy “at least one check” in the shortest run. |
| Maximum `Check Revision` occurrences | Unbounded: one per loop iteration | Exactly 1 | A can show repeated checks; B cannot. |
| Possible placement of `Check Revision` | Inside each iteration, concurrent with `A`, before optional `P`, and before the following `R` or `Z`. | Any interleaving with the whole loop before `Close Case`: before all `A`, between `A` and `P`, between iterations, or after the loop but before the join. | B can place the single check in a position that does not correspond to a per-analysis check. |
| Optional approval occurrences | One optional `P` after both `A` and `C` in each iteration. | One optional `P` after each `A`, independent of the global `C`. | Approval count per analysis can be similar, but approval placement relative to `C` differs. |
| Concurrency | Concurrency only between `A` and `C` inside an iteration. | Concurrency between the single `C` and the entire revision loop. | B has more interleavings, including `P` before `C`. |
| Join before `Close Case` | Loop completion is sequentially followed by `Close Case`. | Parallel join of `C` and the loop must complete before `Close Case`. | Both force `Close Case` last, but B’s join waits for the global check. |
| Rework separation | `Rework Case` occurs between loop iterations. | `Rework Case` occurs between loop iterations inside the loop child. | Both separate analysis attempts, but the check scope differs. |

# 3. Verdicts

| View | Verdict | Confidence | Decisive reason | Caveat |
|---|---|---|---|---|
| P1 — operational compliance | not equivalent | High | Model A guarantees one `Check Revision` per `Analyze Revision` attempt before the next `Rework Case` or `Close Case`; Model B has only one global `Check Revision`. | If the process were restricted to exactly one analysis attempt and check placement relative to approval were ignored, the practical difference would shrink, but under stated P1 they are not equivalent. |
| P2 — executive dashboard | equivalent only under stated abstraction | High | After hiding approvals and collapsing checks to a Boolean, both models yield the same dashboard summaries: `#A = n >= 1`, `check = true`, `rework = n >= 2`. | If the dashboard later exposes exact check counts, approval counts, or check/approval ordering, equivalence disappears. |
| P3 — bounded visible testing | not equivalent | High | The bounded visible languages differ within the P3 bound. Counterexamples exist with  12 visible events and  3 `Analyze Revision` occurrences. | Bounded equality would only be evidence, but here bounded inequality gives concrete counterexamples that are also valid unbounded traces. |

# 4. Counterexample Traces

The following traces are complete visible traces. Arrows separate visible events.

| Trace ID | Trace | Model A | Model B | First decisive semantic difference | Relevant view |
|---|---|---|---|---|---|
| T1 | Open Case  Analyze Revision  Check Revision  Rework Case  Analyze Revision  Check Revision  Close Case | Yes | No | Model A has one check per analysis iteration; Model B has only one global check. | P1, P3 |
| T2 | Open Case  Check Revision  Analyze Revision  Approve Revision  Rework Case  Analyze Revision  Check Revision  Close Case | Yes | No | Two analysis attempts require two checks in Model A; Model B cannot repeat `Check Revision`. | P1, P3 |
| T3 | Open Case  Check Revision  Analyze Revision  Rework Case  Analyze Revision  Close Case | No | Yes | Model B’s single check is concurrent with the whole loop; Model A would require a second check for the second analysis attempt. | P1, P3 |
| T4 | Open Case  Analyze Revision  Approve Revision  Check Revision  Close Case | No | Yes | In Model A, the iteration check must complete before `Approve Revision`; in Model B, the global check may occur after approval. | P3, P1 placement |

At least one counterexample contains two analysis attempts: T1, T2, and T3 all do.

Optional `Approve Revision` and hidden `tau` matter as follows:

- `X('Approve Revision', tau)` means each iteration may either emit a visible `Approve Revision` or silently skip it.
- Hiding `tau` does not collapse a trace with `Approve Revision` into a trace without it; it merely removes the silent alternative.
- Under P3, `Approve Revision` remains visible, so traces such as T4 are relevant.
- Under P2, `Approve Revision` is hidden, which removes the T4-style distinction.
- Under P1, approval placement is potentially relevant, so B’s ability to place the global check after approval can be a compliance-relevant difference.

# 5. Bounded Comparison Code

The code below uses an exhaustive bounded visible-language exploration over tau-closed marking sets. It is not a random playout. A separate random playout function is included only to show the difference between exhaustive bounded exploration and heuristic sampling.

```python
import pm4py
import random
from copy import copy
from collections import Counter

# ---------------------------------------------------------------
# 1. Parse both process trees
# ---------------------------------------------------------------

TREE_A = """->(
  'Open Case',
  *(
    ->(
      +('Analyze Revision', 'Check Revision'),
      X('Approve Revision', tau)
    ),
    'Rework Case'
  ),
  'Close Case'
)"""

TREE_B = """->(
  'Open Case',
  +(
    'Check Revision',
    *(
      ->(
        'Analyze Revision',
        X('Approve Revision', tau)
      ),
      'Rework Case'
    )
  ),
  'Close Case'
)"""

tree_a = pm4py.parse_process_tree(TREE_A)
tree_b = pm4py.parse_process_tree(TREE_B)

# ---------------------------------------------------------------
# 2. Convert to accepting Petri nets
# ---------------------------------------------------------------

net_a, im_a, fm_a = pm4py.convert_to_petri_net(tree_a)
net_b, im_b, fm_b = pm4py.convert_to_petri_net(tree_b)

# ---------------------------------------------------------------
# 3. Check soundness
# ---------------------------------------------------------------

print("Model A soundness:", pm4py.check_soundness(net_a, im_a, fm_a, print_diagnostics=False))
print("Model B soundness:", pm4py.check_soundness(net_b, im_b, fm_b, print_diagnostics=False))

# ---------------------------------------------------------------
# PM4Py marking helper functions
# ---------------------------------------------------------------

try:
    from pm4py.objects.petri_net.obj import Marking
except Exception:
    Marking = dict


def visible_label(transition):
    """
    Return the visible label of a transition.
    Return None for silent transitions.
    """
    label = getattr(transition, "label", None)
    if label is None or label == "tau" or label == "":
        return None
    return label


def marking_key(marking):
    """
    Hashable representation of a marking.
    PM4Py markings are dict-like from place to token count.
    """
    return frozenset(marking.items())


def fire(marking, transition):
    """
    Fire a transition and return a new marking.
    """
    new_m = Marking(marking.copy())

    for arc in transition.in_arcs:
        weight = arc.weight if arc.weight is not None else 1
        new_m[arc.source] = new_m[arc.source] - weight
        if new_m[arc.source] == 0:
            del new_m[arc.source]

    for arc in transition.out_arcs:
        weight = arc.weight if arc.weight is not None else 1
        new_m[arc.target] = new_m.get(arc.target, 0) + weight

    return new_m


def tau_closure(net, markings):
    """
    Compute the set of markings reachable by firing only silent transitions.
    Returns a dict: marking_key -> marking.
    """
    closure = {}
    stack = []

    for m in markings:
        k = marking_key(m)
        if k not in closure:
            closure[k] = m
            stack.append(m)

    while stack:
        m = stack.pop()
        enabled = pm4py.get_enabled_transitions(net, m)

        for t in enabled:
            if visible_label(t) is None:
                nm = fire(m, t)
                k = marking_key(nm)

                if k not in closure:
                    closure[k] = nm
                    stack.append(nm)

    return closure


# ---------------------------------------------------------------
# 4/5. Exhaustive bounded visible-language exploration under P3
# ---------------------------------------------------------------

def bounded_visible_language(net, im, fm, max_visible=12, max_analyze=3):
    """
    Exhaustively enumerate distinct complete visible traces under:
      - visible length <= max_visible
      - number of Analyze Revision events <= max_analyze

    This explores tau closures explicitly.
    It returns a set of tuples of visible labels.
    """
    fm_key = marking_key(fm)
    traces = set()

    initial_closure = tau_closure(net, [im])

    # Work item: (tau-closed marking set, visible prefix, analyze count)
    stack = [(initial_closure, (), 0)]

    # Avoid processing the same search node multiple times.
    # The visible prefix is included because different prefixes matter.
    seen = set()

    while stack:
        closure, trace, a_count = stack.pop()

        entry_key = (frozenset(closure.keys()), trace, a_count)
        if entry_key in seen:
            continue
        seen.add(entry_key)

        # Complete trace if final marking is reachable in this tau closure.
        if fm_key in closure:
            traces.add(trace)

        # Visible bound.
        if len(trace) >= max_visible:
            continue

        # Group enabled visible transitions by visible label.
        next_by_label = {}

        for m in closure.values():
            enabled = pm4py.get_enabled_transitions(net, m)

            for t in enabled:
                lab = visible_label(t)
                if lab is not None:
                    nm = fire(m, t)
                    k = marking_key(nm)
                    next_by_label.setdefault(lab, {})[k] = nm

        for lab, next_markings in next_by_label.items():
            new_a_count = a_count + (1 if lab == "Analyze Revision" else 0)

            if new_a_count > max_analyze:
                continue

            new_closure = tau_closure(net, list(next_markings.values()))
            stack.append((new_closure, trace + (lab,), new_a_count))

    return traces


lang_a = bounded_visible_language(net_a, im_a, fm_a, max_visible=12, max_analyze=3)
lang_b = bounded_visible_language(net_b, im_b, fm_b, max_visible=12, max_analyze=3)

only_a = sorted(lang_a - lang_b)
only_b = sorted(lang_b - lang_a)

print()
print("Exhaustive bounded comparison under P3")
print("--------------------------------------")
print(f"Model A bounded visible traces: {len(lang_a)}")
print(f"Model B bounded visible traces: {len(lang_b)}")
print(f"Traces only in Model A: {len(only_a)}")
print(f"Traces only in Model B: {len(only_b)}")

print()
print("Examples unique to Model A:")
for t in only_a[:10]:
    print("  " + " > ".join(t))

print()
print("Examples unique to Model B:")
for t in only_b[:10]:
    print("  " + " > ".join(t))


# ---------------------------------------------------------------
# Optional: random playout, explicitly NOT exhaustive
# ---------------------------------------------------------------

def random_bounded_playout(
    net,
    im,
    fm,
    runs=2000,
    max_visible=12,
    max_analyze=3,
    max_total_steps=200,
    seed=42
):
    """
    Heuristic random playout.

    This is NOT exhaustive. It may miss rare branches, may miss counterexamples,
    and its result depends on the random choices and run count.
    """
    rng = random.Random(seed)
    fm_key = marking_key(fm)
    sampled = set()

    for _ in range(runs):
        m = im
        trace = []
        a_count = 0

        for _ in range(max_total_steps):
            if marking_key(m) == fm_key:
                sampled.add(tuple(trace))
                break

            enabled = list(pm4py.get_enabled_transitions(net, m))
            if not enabled:
                break

            t = rng.choice(enabled)
            lab = visible_label(t)

            if lab is not None:
                if len(trace) >= max_visible:
                    break

                if lab == "Analyze Revision":
                    if a_count >= max_analyze:
                        break
                    a_count += 1

                trace.append(lab)

            m = fire(m, t)

        # Incomplete runs are discarded.

    return sampled


sample_a = random_bounded_playout(net_a, im_a, fm_a)
sample_b = random_bounded_playout(net_b, im_b, fm_b)

print()
print("Random playout sample sizes")
print("---------------------------")
print(f"Sampled Model A traces: {len(sample_a)}")
print(f"Sampled Model B traces: {len(sample_b)}")
print(f"Sampled only in A: {len(sample_a - sample_b)}")
print(f"Sampled only in B: {len(sample_b - sample_a)}")
print()
print("WARNING: random playout is heuristic and not exhaustive.")
```

The exhaustive part is `bounded_visible_language`. It explicitly:

- hides `tau`;
- retains all visible labels;
- respects the P3 bounds;
- computes tau closures;
- handles repeated markings by using tau-closed sets of markings;
- checks final-marking equality by comparing final marking keys;
- returns distinct visible traces, not firing-sequence multiplicities.

The random playout function is included only to contrast heuristic sampling with exhaustive bounded exploration. It is not sufficient for a verdict.

# 6. Stochastic and Footprint Comparison

Footprint and stochastic comparisons can provide evidence, but they are not substitutes for trace-language equivalence.

## Footprints

`pm4py.discover_footprints(...)` computes pairwise behavioral relations, such as sequence, parallelism, and exclusiveness, between activities.

For these models, footprints are likely to differ because:

- In Model A, `Check Revision` is inside each loop iteration and precedes the optional approval in that iteration.
- In Model B, `Check Revision` is concurrent with the whole loop, so it may be footprint-parallel to `Rework Case`, `Approve Revision`, and multiple `Analyze Revision` occurrences.

Example code:

```python
# Footprint comparison
# Depending on PM4Py version, footprints may be discovered from a process tree
# or from an accepting Petri net.

try:
    fp_a = pm4py.discover_footprints(tree_a)
    fp_b = pm4py.discover_footprints(tree_b)
except Exception:
    fp_a = pm4py.discover_footprints((net_a, im_a, fm_a))
    fp_b = pm4py.discover_footprints((net_b, im_b, fm_b))

print("Footprints A:")
print(fp_a)

print()
print("Footprints B:")
print(fp_b)

print()
print("Footprint objects structurally equal?", fp_a == fp_b)
```

Equal footprints would still not prove trace equivalence. Footprints are pairwise local relations and can miss higher-order dependencies, loop cardinalities, and global counting constraints. For example, a footprint may say that `Check Revision` and `Analyze Revision` are related, but it may not distinguish “one check per analysis” from “one global check for all analyses.”

## Stochastic language and Earth Mover’s Distance

`pm4py.get_stochastic_language(...)` and `pm4py.compute_emd(...)` can compare probabilistic trace distributions if a stochastic semantics is available.

However, probabilities are not given in the task. We must not invent transition probabilities without declaring them. If one wanted to use stochastic comparison, one would need to declare, for example:

- probability of choosing `Approve Revision` versus `tau`;
- probability of exiting the loop versus doing `Rework Case`;
- scheduling weights for concurrent interleavings.

Without such declarations, a stochastic distance has no defined business meaning.

Illustrative API shape:

```python
# Illustrative stochastic comparison.
# This requires an explicit stochastic semantics or transition weights.
# Do not interpret the result unless probabilities have been declared.

try:
    lang_a_stoch = pm4py.get_stochastic_language((net_a, im_a, fm_a))
    lang_b_stoch = pm4py.get_stochastic_language((net_b, im_b, fm_b))

    emd = pm4py.compute_emd(lang_a_stoch, lang_b_stoch)

    print("Stochastic language comparison")
    print("Earth Mover's Distance:", emd)

except Exception as exc:
    print("Stochastic language/EMD skipped because explicit stochastic weights are required.")
    print("Details:", exc)
```

A low EMD would not establish trace equivalence because:

- EMD measures distributional distance, not set equality of traces.
- Rare counterexample traces can have very low probability and still make the models trace-inequivalent.
- The distance depends on the chosen probabilities, which are not part of the original process-tree semantics.
- Sampling-based stochastic comparison adds statistical error.

Even EMD equal to zero would only show equality of the assigned stochastic language, not necessarily equality of all possible traces if some traces have zero probability under the chosen stochastic semantics.

# 7. Complexity Comparison

Under the usual n-ary PM4Py process-tree representation, the following syntactic values can be derived directly from the tree expressions.

| Metric | Model A | Model B |
|---|---:|---:|
| Process-tree node count | 12 | 12 |
| Maximum edge depth from root to leaf | 4 | 5 |
| Visible leaves | 6 | 6 |
| Silent leaves | 1 | 1 |
| Duplicate leaf labels in the tree | 0 | 0 |

Explanation:

- Both models have five operator nodes: root sequence, loop, inner sequence, parallel or top-level parallel, and exclusive choice.
- Both models have seven leaves: `Open Case`, `Close Case`, `Analyze Revision`, `Check Revision`, `Approve Revision`, `Rework Case`, and `tau`.
- Model B is deeper because the loop is nested inside the top-level parallel operator.
- Duplicate leaf labels are zero in both trees, but behavioral repetitions are unbounded because of the loop.

The following code computes these structural counts and also reports PM4Py’s Petri-net simplicity metric.

```python
# Structural complexity comparison
# Assumes tree_a, tree_b, net_a, im_a, fm_a, net_b, im_b, fm_b
# and visible_label(...) from the bounded comparison code.

def process_tree_stats(tree):
    nodes = 0
    max_depth = 0
    visible_leaves = 0
    silent_leaves = 0
    labels = []

    def visit(node, depth):
        nonlocal nodes, max_depth, visible_leaves, silent_leaves

        nodes += 1
        max_depth = max(max_depth, depth)

        children = getattr(node, "children", [])

        if children:
            for child in children:
                visit(child, depth + 1)
        else:
            lab = getattr(node, "label", None)

            if lab is None or lab == "tau":
                silent_leaves += 1
            else:
                visible_leaves += 1
                labels.append(lab)

    visit(tree, 0)

    label_counts = Counter(labels)
    duplicate_labels = {
        label: count
        for label, count in label_counts.items()
        if count > 1
    }

    duplicate_extra_occurrences = sum(
        count - 1
        for count in label_counts.values()
        if count > 1
    )

    return {
        "node_count": nodes,
        "max_edge_depth": max_depth,
        "visible_leaves": visible_leaves,
        "silent_leaves": silent_leaves,
        "duplicate_labels": duplicate_labels,
        "duplicate_extra_occurrences": duplicate_extra_occurrences,
    }


def petri_net_stats(net, im, fm):
    return {
        "places": len(net.places),
        "transitions": len(net.transitions),
        "silent_transitions": sum(
            1
            for t in net.transitions
            if visible_label(t) is None
        ),
        "simplicity_petri_net": pm4py.simplicity_petri_net(net, im, fm),
    }


print("Model A tree stats:")
print(process_tree_stats(tree_a))

print()
print("Model B tree stats:")
print(process_tree_stats(tree_b))

print()
print("Model A Petri net stats:")
print(petri_net_stats(net_a, im_a, fm_a))

print()
print("Model B Petri net stats:")
print(petri_net_stats(net_b, im_b, fm_b))
```

I do not assert exact numeric PM4Py simplicity scores here because those depend on the PM4Py conversion and the metric implementation. The code above prints the exact values for the installed PM4Py version.

# 8. Simplified Model

A single simplification cannot preserve P1, P2, and P3 simultaneously:

- P1 requires per-analysis checks.
- P3 retains all visible labels and distinguishes approval/check ordering.
- P2 deliberately hides approvals and collapses repeated checks.

The most appropriate target for a deliberately simplified model is therefore P2, the executive dashboard view. For operational compliance or bounded visible-language testing, the simplified model below should not replace Model A.

## Simplified process tree for P2

```text
->(
  'Open Case',
  'Check Revision',
  *('Analyze Revision', 'Rework Case'),
  'Close Case'
)
```

This model says:

1. Open the case.
2. Perform one global check.
3. Perform at least one analysis.
4. Optionally rework and analyze again.
5. Close the case.

It intentionally hides approvals and collapses the check to a single Boolean-relevant occurrence.

## PM4Py code

```python
simplified_model = pm4py.parse_process_tree(
    "->('Open Case', 'Check Revision', *('Analyze Revision', 'Rework Case'), 'Close Case')"
)

simplified_net, simplified_im, simplified_fm = pm4py.convert_to_petri_net(simplified_model)

print(
    "Simplified model soundness:",
    pm4py.check_soundness(simplified_net, simplified_im, simplified_fm, print_diagnostics=False)
)

print("Simplified model tree stats:")
print(process_tree_stats(simplified_model))
```

| At least one `Check Revision` | Exact under P2 | The simplified model has one check; P2 only needs Boolean existence. |
| Number of `Analyze Revision` occurrences | Exact under P2 | Loop iterations correspond to analysis counts. |
| Whether any `Rework Case` occurred | Exact under P2 | Rework occurs iff there is more than one analysis iteration. |
| Exact number of `Check Revision` occurrences | Abstracted / weakened | Model A may have many checks; simplified model has one. P2 collapses this. |

# 9. Final Recommendation

Model suitability by view:

| View | Suitable original model | Recommendation |
|---|---|---|
| P1 — operational compliance | Model A | Use Model A. Model B is not appropriate because it does not guarantee a check per analysis attempt. |
| P2 — executive dashboard | Both originals are equivalent under the stated abstraction | The simplified P2 model may replace both originals for dashboard purposes only. |
| P3 — bounded visible-language testing | Neither model is equivalent to the other | Use the model that reflects the intended operational semantics. If the intended semantics are per-analysis checks, use Model A. |

The simplified model should not replace Model A for P1 or P3. It is only justified as a dashboard abstraction under P2.

Warning about bounded comparison:

- If bounded comparison finds a difference, as it does here, that difference is a real counterexample within the bound and also a valid unbounded trace.
- If bounded comparison finds no difference, that does not prove unbounded equivalence. The models could still differ beyond the tested bound.
- Therefore, bounded testing is useful for falsification and evidence, but not as a general proof of equivalence.
