from pathlib import Path

OUT = Path("conversion_audit_outputs")

def unpack_petri_result(result, name):
    """Unpack (net, im, fm) and reject null components."""
    if result is None or not isinstance(result, tuple) or len(result) != 3:
        raise RuntimeError(f"{name}: expected (net, im, fm), got {result!r}")
    net, im, fm = result
    if net is None or im is None or fm is None:
        raise RuntimeError(f"{name}: returned a None component")
    return net, im, fm


def require_not_none(obj, name):
    if obj is None:
        raise RuntimeError(f"{name}: conversion returned None")
    return obj


def diagnose(net, im, fm, name):
    """Workflow-net and soundness diagnostics for an accepting Petri net."""
    is_wf = False
    try:
        is_wf = pm4py.check_is_workflow_net(net)
        print(f"[{name}] is_workflow_net={is_wf}")
    except Exception as e:
        print(f"[{name}] check_is_workflow_net raised: {e!r}")

    try:
        sound = pm4py.check_soundness(net, im, fm, print_diagnostics=True)
        print(f"[{name}] soundness returned: {sound!r}")
    except Exception as e:
        print(f"[{name}] check_soundness raised: {e!r}")

    return is_wf


# ----------------------------------------------------------------------
# 1. POWL -> accepting Petri net
# ----------------------------------------------------------------------
net_from_powl, im_from_powl, fm_from_powl = unpack_petri_result(
    pm4py.convert_to_petri_net(source_powl),
    "POWL->Petri"
)

diagnose(net_from_powl, im_from_powl, fm_from_powl, "net_from_powl")

pm4py.save_vis_powl(source_powl, str(OUT / "00_source_powl.png"))

)

    if tree_from_net is None:

    else:
        pm4py.save_vis_process_tree(

            str(OUT / "03_tree_from_net.png")

roundtrip_nets = {}

        rt_net,
        rt_im,

if bpmn_from_tree is not None:
    try:
        tree_rt_net, tree_rt_im, tree_rt_fm = unpack_petri_result(
            pm4py.convert_to_petri_net(bpmn_from_tree),
            "BPMN_from_tree->Petri"
        )
        roundtrip_nets["bpmn_from_tree"] = (
            tree_rt_net,
            tree_rt_im,
            tree_rt_fm
        )
        diagnose(
            tree_rt_net,
            tree_rt_im,
            tree_rt_fm,
            "roundtrip_bpmn_from_tree"
        )
        pm4py.save_vis_petri_net(
            tree_rt_net,
            tree_rt_im,
            tree_rt_fm,
            str(OUT / "06_roundtrip_bpmn_from_tree.png")
        )
    except Exception as e:
        print(f"[roundtrip] bpmn_from_tree round trip failed: {e!r}")

print("tree_from_net:", tree_from_net)
print("tree_error:", tree_error)
print("roundtrip_nets:", list(roundtrip_nets.keys()))

| I7: `Archive Change` final | expected preserved | expected preserved | conversion failed | expected preserved | Every complete trace ends with `Archive Change`; exact final-marking equality is required in exploration. |
| Initial/final markings | expected preserved | ambiguous | conversion failed | expected preserved | Different place objects; compare by accepting traces and exact final-marking reachability, not object identity. |
| Activity-label uniqueness | expected preserved | ambiguous | conversion failed | ambiguous | Source has one visible transition per label; BPMN tools may duplicate or rearrange task nodes, and trace labels cannot identify occurrences. |
| Gateway/routing-node mapping | expected preserved | ambiguous | conversion failed | ambiguous | BPMN gateways are visual approximations of routing nodes; silent skip can be collapsed or read as an exclusive decision. |

The main uncertainties are not in the accepting Petri net but in the BPMN view and in any attempted process-tree intermediate. In particular:

- **True concurrency versus arbitrary interleaving:** trace sets show that both orders are possible, but they cannot prove that two events may happen at the same time. A BPMN parallel gateway visually suggests simultaneity; the source `StrictPartialOrder` only guarantees lack of order dependency.
- **Lack of an edge between optional legal and verification:** this absence is a real semantic fact. A BPMN layout that always places `Legal Review` before the loop adds a dependency that the POWL does not state.
- **Silent optional behavior:** `skip_legal` is a tau transition. In BPMN, an empty XOR branch may be omitted or appear as an implicit default flow, making the optionality harder to inspect.
- **Binary-loop semantics:** `LOOP(trial_body, correct_defect)` means `trial_body`, then `(correct_defect, trial_body)*`. BPMN loop notation can be misread as a precondition loop that allows zero iterations.
- **Join before `Release Change`:** the BPMN join gateway must be a parallel/AND join, not an exclusive merge. If the silent skip branch is invisible, readers may not see that two branches are being synchronized.
- **Label uniqueness:** labels are unique in the source model, but repeated loop iterations reuse the same label, so a trace cannot identify which loop iteration an event came from without additional positional or state information.

from copy import deepcopy

import pm4py

MAX_VISIBLE_EVENTS = 12
MAX_TOTAL_STEPS = 500
MAX_LOOP_ITERATIONS = 3


def marking_key(marking):
    return tuple(sorted((id(p), c) for p, c in marking.items() if c > 0))


def same_marking(m1, m2):
    return dict(m1) == dict(m2)


def fire_transition(net, marking, t):
    """Return the marking after firing t, or None if t is disabled."""
    m = dict(marking)

    # Check enabledness
    for arc in t.in_arcs:
        p = arc.source
        if m.get(p, 0) < arc.weight:
            return None

    # Consume input tokens
    for arc in t.in_arcs:
        p = arc.source
        m[p] = m.get(p, 0) - arc.weight
        if m[p] == 0:
            del m[p]

    # Produce output tokens
    for arc in t.out_arcs:
        p = arc.target
        m[p] = m.get(p, 0) + arc.weight

    return m


def explore_bounded_visible_traces(net, im, fm):
    """Return accepted visible-label traces under a bounded exploration."""
    accepted = set()

    def dfs(marking, trace, steps, seen_silent):
        if same_marking(marking, fm):
            accepted.add(tuple(trace))
            return

        if steps >= MAX_TOTAL_STEPS:
            return

        for t in pm4py.get_enabled_transitions(net, marking):
            new_marking = fire_transition(net, marking, t)
            if new_marking is None:
                continue

            if t.label is None:
                key = marking_key(new_marking)
                if key in seen_silent:
                    continue
                seen_silent.add(key)
                dfs(new_marking, trace, steps + 1, seen_silent)
                seen_silent.remove(key)
            else:
                if len(trace) >= MAX_VISIBLE_EVENTS:
                    continue
                if t.label == "Configure Trial":
                    if trace.count("Configure Trial") >= MAX_LOOP_ITERATIONS:
                        continue
                dfs(
                    new_marking,
                    trace + [t.label],
                    steps + 1,
                    seen_silent
                )

    dfs(dict(im), [], 0, {marking_key(dict(im))})
    return accepted


def accepts_visible_trace(net, im, fm, trace):
    """Check whether a single visible trace is accepted and reaches fm exactly."""
    trace = list(trace)
    seen = set()

    def dfs(marking, pos, steps):
        key = (marking_key(marking), pos)
        if key in seen:
            return False
        seen.add(key)

        if steps >= MAX_TOTAL_STEPS:
            return False

        if same_marking(marking, fm) and pos == len(trace):
            return True

        for t in pm4py.get_enabled_transitions(net, marking):
            new_marking = fire_transition(net, marking, t)
            if new_marking is None:
                continue

            if t.label is None:
                if dfs(new_marking, pos, steps + 1):
                    return True
            else:
                if pos < len(trace) and t.label == trace[pos]:
                    if dfs(new_marking, pos + 1, steps + 1):
                        return True

        return False

    return dfs(dict(im), 0, 0)

def rg_signature(rg):
    states = getattr(rg, "states", None)
    transitions = getattr(rg, "transitions", None)
    if states is not None:

    else:
        state_part = repr(rg)
    if transitions is not None:
        trans_part = len(transitions)
    else:
        trans_part = None
    return state_part, trans_part

# 6. Visual-Semantics Review

A BPMN picture can be behaviorally correct and still mislead stakeholders.

- **Parallel gateways:** the converter may use a parallel split after `Register Change` to show `Assess Risk` and `Prepare Evidence` as concurrent. This is close to the source semantics, but the layout may suggest that both branches must finish before anything else happens. In the source, `Legal Review` may already occur while `Prepare Evidence` is still running, because there is no edge from `Prepare Evidence` to `Legal Review`.
- **Exclusive gateways:** the optional legal branch is an XOR with a silent child. If BPMN shows an empty branch, readers may not realize that this is a real routing decision. If the empty branch is omitted, readers may think `Legal Review` is mandatory.
- **Loop visualization:** `LOOP(trial_body, correct_defect)` is a binary redo loop. BPMN has no standard binary redo-loop notation. A loop marker on a subprocess may imply a while-loop with a condition evaluated before the first iteration, which would allow zero iterations. That would violate the source requirement that at least one verification iteration occurs.
- **Silent routing nodes:** `skip_legal` is a tau transition. In BPMN, it may be drawn as an empty sequence flow, an event gateway, or nothing at all. The hidden skip is semantically important because `Release Change` must wait for the XOR branch to complete, even when no `Legal Review` is performed.
- **Apparent ordering of graph layout:** BPMN layout tools typically place activities left to right. If `Legal Review` is placed before the verification loop, a reader may infer a causal order that the POWL does not state. The invariant table must be checked, not the visual order.
- **Gateway or routing-node ambiguity:** the join before `Release Change` must be an AND join. If a layout collapses the silent skip branch into the incoming flow of `Release Change`, the result can look like an OR join or a plain sequence, hiding the synchronization with the optional-legal branch.

Suggested annotations/documentation:

- Add a BPMN annotation: “`Legal Review` and verification loop are unordered; both must complete before `Release Change`.”
- Add another annotation near the loop: “At least one verification iteration; repeated iterations are separated by `Correct Defect`.”
- Avoid editing BPMN layouts as if they were semantics. Store the invariant table next to the generated diagram.
- If the BPMN tool supports documentation, attach the accepted positive/negative witness traces to the diagram artifact.
- Keep the POWL model authoritative and label all BPMN exports as “non-authoritative view”.

# 7. Rejection Criteria

Reject or quarantine a conversion result if any of the following is true:

1. Any required conversion raises an exception or returns a `None` component instead of a valid net, marking, BPMN object, or process tree.
2. The accepting Petri net is not a workflow net according to `pm4py.check_is_workflow_net`.
3. The accepting Petri net is not sound according to `pm4py.check_soundness`.
4. Bounded visible-trace comparison shows missing required traces or extra traces relative to `net_from_powl`.
5. A negative witness trace is accepted, for example:
   - `Release Change` before verification,
   - `Release Change` before visible `Legal Review`,
   - `Inspect Trial` without a preceding `Configure Trial`,
   - two loop iterations without `Correct Defect` between them,
   - a complete trace that does not start with `Register Change`.
6. A positive witness trace is rejected, such as the `Prepare Evidence` before `Assess Risk` interleaving, legal-before-loop, legal-after-loop, or a two-iteration loop.
7. An exact-final-marking test fails: a trace that should finish reaches a non-final marking, or a candidate net reaches its final marking without consuming the full visible trace.
8. A process-tree conversion succeeds but the resulting tree changes the accepted trace set; even a successful tree conversion is not authoritative unless it passes the same invariant and trace checks.
9. BPMN visualization hides or collapses the silent skip branch in a way that makes the AND join before `Release Change` semantically unreadable.

# 8. Canonical Representation Recommendation

- **Authoritative storage model:** the source POWL model. It directly represents the strict partial order, optionality, loop structure, and synchronization requirements without relying on BPMN gateway encodings.
- **Stakeholder visualization model:** BPMN, generated from the accepting Petri net, not from hand editing. It is a view for discussion, not a source of truth.
- **Derivative artifacts:** `net_from_powl`, `bpmn_from_net`, any successful process tree, any round-tripped accepting Petri net, reachability graphs, footprints, and saved visualizations. Each derivative must be accompanied by:
  - source POWL version/hash,
  - PM4Py version and converter name,
  - generation timestamp,
  - validation report containing workflow-net/soundness results,
  - bounded trace comparison results,
  - invariant and witness trace checklist.
- **Process-tree intermediate:** not allowed as canonical. The source behavior is not block-structured enough to guarantee a faithful process tree. A process tree may be explored as a diagnostic artifact only if it passes every trace and invariant check; otherwise it must be rejected.
- **Avoiding round-trip drift:** edit the POWL, regenerate all derivatives, and rerun the validation suite. Never edit the BPMN and then convert it back to POWL as the normal workflow. If someone edits a BPMN diagram intentionally, treat that edited BPMN as a proposed new model, port the intended change back into POWL manually, regenerate, and validate. This prevents accumulated round-trip drift.

# 9. Residual Uncertainty

Even if every provided check passes, the following claims remain uncertain:

1. **Infinite behavior equivalence:** the bounded trace exploration is exhaustive only up to the chosen visible-length and step bounds. It does not prove equivalence for arbitrarily long traces.
2. **True concurrency vs. interleaving:** accepting the same visible traces does not prove that two models have the same true-concurrency or causal semantics.
3. **Converter correctness:** the validation assumes `pm4py.convert_to_petri_net(source_powl)` correctly compiles POWL semantics. The tests do not independently verify the internals of PM4Py.
4. **Hidden silent behavior:** after hiding tau transitions, visible traces cannot always distinguish “legal branch skipped” from “legal branch absent.” Thus some completeness properties of the optional legal branch are not fully observable.
5. **BPMN readability:** automated checks cannot prove that stakeholders will correctly interpret gateway layout, loop notation, or silent branches. A behaviorally correct diagram may still be visually misleading.
