BUSINESS INTENT
After a case is registered, an analysis and a review are started in parallel. The analysis is always mandatory. The review can be either a standard review or a formally recorded waiver. Authorization may occur only after the analysis and whichever review outcome was selected are both complete. The case should then reach the final marking with no remaining tokens.

GIVEN PM4PY CODE

```python
import pm4py
from pm4py.objects.petri_net.obj import PetriNet, Marking
from pm4py.objects.petri_net.utils import petri_utils

net = PetriNet("parallel_review_with_waiver")

source = PetriNet.Place("source")
p_ready = PetriNet.Place("p_ready")
p_analysis = PetriNet.Place("p_analysis")
p_review = PetriNet.Place("p_review")
p_analysis_done = PetriNet.Place("p_analysis_done")
p_review_done = PetriNet.Place("p_review_done")
p_waived = PetriNet.Place("p_waived")
p_decision = PetriNet.Place("p_decision")
sink = PetriNet.Place("sink")

for p in {
    source, p_ready, p_analysis, p_review, p_analysis_done,
    p_review_done, p_waived, p_decision, sink
}:
    net.places.add(p)

register = PetriNet.Transition("t_register", "Register Case")
split = PetriNet.Transition("t_split", None)
analyze = PetriNet.Transition("t_analyze", "Perform Analysis")
standard_review = PetriNet.Transition("t_standard_review", "Perform Standard Review")
waive_review = PetriNet.Transition("t_waive_review", "Record Review Waiver")
join_normal = PetriNet.Transition("t_join_normal", None)
join_waived = PetriNet.Transition("t_join_waived", None)
authorize = PetriNet.Transition("t_authorize", "Authorize Case")

for t in {
    register, split, analyze, standard_review, waive_review,
    join_normal, join_waived, authorize
}:
    net.transitions.add(t)

petri_utils.add_arc_from_to(source, register, net)
petri_utils.add_arc_from_to(register, p_ready, net)
petri_utils.add_arc_from_to(p_ready, split, net)
petri_utils.add_arc_from_to(split, p_analysis, net)
petri_utils.add_arc_from_to(split, p_review, net)

petri_utils.add_arc_from_to(p_analysis, analyze, net)
petri_utils.add_arc_from_to(analyze, p_analysis_done, net)

petri_utils.add_arc_from_to(p_review, standard_review, net)
petri_utils.add_arc_from_to(standard_review, p_review_done, net)

petri_utils.add_arc_from_to(p_review, waive_review, net)
petri_utils.add_arc_from_to(waive_review, p_waived, net)

petri_utils.add_arc_from_to(p_analysis_done, join_normal, net)
petri_utils.add_arc_from_to(p_review_done, join_normal, net)
petri_utils.add_arc_from_to(join_normal, p_decision, net)

# Waiver synchronization as implemented by the original modeler:
petri_utils.add_arc_from_to(p_waived, join_waived, net)
petri_utils.add_arc_from_to(join_waived, p_decision, net)

petri_utils.add_arc_from_to(p_decision, authorize, net)
petri_utils.add_arc_from_to(authorize, sink, net)

im = Marking()
im[source] = 1
fm = Marking()
fm[sink] = 1
```
